   339→function timeAgo(dateStr: string): string {
   340→  const seconds = Math.floor((Date.now() - new Date(dateStr).getTime()) / 1000)
   341→  if (seconds < 60) return 'just now'
   342→  const minutes = Math.floor(seconds / 60)
   343→  if (minutes < 60) return `${minutes}m ago`
   344→  const hours = Math.floor(minutes / 60)
   345→  if (hours < 24) return `${hours}h ago`
   346→  const days = Math.floor(hours / 24)
   347→  return `${days}d ago`
   348→}
   349→
   350→function formatDate(dateStr: string) {
   351→  try {
   352→    return new Date(dateStr).toLocaleDateString('en-US', {
   353→      month: 'short',
   354→      day: 'numeric',
   355→      year: 'numeric',
   356→      hour: '2-digit',
   357→      minute: '2-digit',
   358→    })
   359→  } catch {
   360→    return dateStr
   361→  }
   362→}

   364→function generateSimulatedBuildLog(project: ProjectDetail, deployment: Deployment): string {
   365→  const isFailed = deployment.status === 'failed'
   366→  const fw = project.framework === 'nextjs' ? 'next' : project.framework
   367→  const now = new Date(deployment.createdAt)
   368→  const pad = (n: number) => String(n).padStart(2, '0')
   369→  const ts = `${pad(now.getHours())}:${pad(now.getMinutes())}:${pad(now.getSeconds())}`
   370→
   371→  if (isFailed) {
   372→    return `${ts} > Cloning repository...
   373→${ts} > Installing dependencies...
   374→${ts} > npm install --production
   375→${ts} > Building application...
   376→${ts} > ${fw} build
   377→${ts} ✗ Type error: Property 'fetchData' does not exist on type 'never'.
   378→${ts} ✗ Build failed with 1 error
   379→${ts} > Error: Build process exited with code 1
   380→${ts} > Deployment failed!`
   381→  }
   382→
   383→  return `${ts} > Cloning repository...
   384→${ts} ✓ Repository cloned successfully
   385→${ts} > Installing dependencies...
   386→${ts} > npm install --production
   387→${ts} ✓ Dependencies installed (1.2s)
   388→${ts} > Building application...
   389→${ts} > ${fw} build
   390→${ts} ✓ Compiled successfully
   391→${ts} ✓ Linting and checking validity of types
   392→${ts} ✓ Collecting page data
   393→${ts} ✓ Generating static pages (8/8)
   394→${ts} ✓ Finalizing page optimization
   395→${ts} > Deploying to server...
   396→${ts} ✓ Deployment successful!
   397→${ts} > Live at https://${project.subdomain}`
   398→}

   400→function colorizeLogLine(line: string): React.ReactNode {
   401→  if (line.includes('✓') || line.includes('successful') || line.includes('Live at')) {
   402→    return <span className="text-emerald-400">{line}</span>
   403→  }
   404→  if (line.includes('✗') || line.includes('Error') || line.includes('failed')) {
   405→    return <span className="text-red-400">{line}</span>
   406→  }
   407→  if (line.includes('Warning') || line.includes('warn')) {
   408→    return <span className="text-yellow-400">{line}</span>
   409→  }
   410→  if (line.startsWith('>')) {
   411→    return <span className="text-zinc-400">{line}</span>
   412→  }
   413→  return <span className="text-zinc-200">{line}</span>
   414→}

   531→function CreateProjectDialog({
   532→  open,
   533→  onClose,
   534→  onCreated,
   535→}: {
   536→  open: boolean
   537→  onClose: () => void
   538→  onCreated: () => void
   539→}) {
   540→  const [name, setName] = useState('')
   541→  const [framework, setFramework] = useState('nextjs')
   542→  const [planId, setPlanId] = useState('')
   543→  const [plans, setPlans] = useState<Plan[]>([])
   544→  const [loading, setLoading] = useState(false)
   545→  const [plansLoading, setPlansLoading] = useState(false)
   546→
   547→  useEffect(() => {
   548→    if (open) {
   549→      setPlansLoading(true)
   550→      fetch('/api/billing/plans')
   551→        .then((r) => r.json())
   552→        .then((data) => {
   553→          setPlans(data.plans || [])
   554→          if (data.plans?.length > 0 && !planId) {
   555→            setPlanId(data.plans[0].id)
   556→          }
   557→        })
   558→        .catch(() => toast.error('Failed to load plans'))
   559→        .finally(() => setPlansLoading(false))
   560→    }
   561→  }, [open, planId])
   562→
   563→  const handleCreate = async () => {
   564→    if (!name.trim()) {
   565→      toast.error('Project name is required')
   566→      return
   567→    }
   568→    setLoading(true)
   569→    try {
   570→      const res = await fetch('/api/hosting/create', {
   571→        method: 'POST',
   572→        headers: getAuthHeaders(),
   573→        body: JSON.stringify({ name: name.trim(), framework, planId: planId || undefined }),
   574→      })
   575→      const data = await res.json()
   576→      if (!res.ok) {
   577→        toast.error(data.error || 'Failed to create project')
   578→        return
   579→      }
   580→      toast.success('Project created successfully!')
   581→      setName('')
   582→      setFramework('nextjs')
   583→      onCreated()
   584→      onClose()
   585→    } catch {
   586→      toast.error('Network error. Please try again.')
   587→    } finally {
   588→      setLoading(false)
   589→    }
   590→  }
   591→
   592→  return (
   593→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   594→      <DialogContent className="sm:max-w-lg">
   595→        <DialogHeader>
   596→          <DialogTitle>Create New Project</DialogTitle>
   597→          <DialogDescription>
   598→            Set up a new hosting project with your preferred framework and plan.
   599→          </DialogDescription>
   600→        </DialogHeader>
   601→
   602→        <div className="grid gap-4 py-4">
   603→          <div className="grid gap-2">
   604→            <Label htmlFor="project-name">Project Name</Label>
   605→            <Input
   606→              id="project-name"
   607→              placeholder="my-awesome-app"
   608→              value={name}
   609→              onChange={(e) => setName(e.target.value)}
   610→            />
   611→          </div>
   612→
   613→          <div className="grid gap-2">
   614→            <Label>Framework</Label>
   615→            <Select value={framework} onValueChange={setFramework}>
   616→              <SelectTrigger className="w-full">
   617→                <SelectValue placeholder="Select framework" />
   618→              </SelectTrigger>
   619→              <SelectContent>
   620→                <SelectItem value="nextjs">Next.js</SelectItem>
   621→                <SelectItem value="react">React</SelectItem>
   622→                <SelectItem value="node">Node.js</SelectItem>
   623→                <SelectItem value="static">Static</SelectItem>
   624→              </SelectContent>
   625→            </Select>
   626→          </div>
   627→
   628→          <div className="grid gap-2">
   629→            <Label>Plan</Label>
   630→            {plansLoading ? (
   631→              <div className="grid gap-2">
   632→                <Skeleton className="h-20 w-full" />
   633→                <Skeleton className="h-20 w-full" />
   634→              </div>
   635→            ) : (
   636→              <div className="grid gap-2">
   637→                {plans.map((plan) => (
   638→                  <button
   639→                    key={plan.id}
   640→                    type="button"
   641→                    onClick={() => setPlanId(plan.id)}
   642→                    className={`relative flex items-center gap-3 rounded-lg border p-3 text-left transition-all hover:border-emerald-300 dark:hover:border-emerald-700 ${
   643→                      planId === plan.id
   644→                        ? 'border-emerald-500 bg-emerald-50 dark:bg-emerald-900/20 ring-1 ring-emerald-500'
   645→                        : 'border-border'
   646→                    }`}
