    73→// ── Password Strength ──────────────────────────────────────────────────────
    74→function getPasswordStrength(password: string): { label: string; color: string; percent: number } {
    75→  if (!password) return { label: '', color: '', percent: 0 }
    76→  let score = 0
    77→  if (password.length >= 8) score++
    78→  if (password.length >= 12) score++
    79→  if (/[A-Z]/.test(password)) score++
    80→  if (/[0-9]/.test(password)) score++
    81→  if (/[^A-Za-z0-9]/.test(password)) score++
    82→
    83→  if (score <= 1) return { label: 'Weak', color: 'bg-red-500', percent: 20 }
    84→  if (score <= 2) return { label: 'Fair', color: 'bg-orange-500', percent: 40 }
    85→  if (score <= 3) return { label: 'Medium', color: 'bg-yellow-500', percent: 60 }
    86→  if (score <= 4) return { label: 'Strong', color: 'bg-emerald-400', percent: 80 }
    87→  return { label: 'Very Strong', color: 'bg-emerald-600', percent: 100 }
    88→}

   210→// ── Security Tab ───────────────────────────────────────────────────────────
   211→function SecurityTab() {
   212→  const [currentPassword, setCurrentPassword] = useState('')
   213→  const [newPassword, setNewPassword] = useState('')
   214→  const [confirmPassword, setConfirmPassword] = useState('')
   215→  const [showCurrent, setShowCurrent] = useState(false)
   216→  const [showNew, setShowNew] = useState(false)
   217→  const [saving, setSaving] = useState(false)
   218→
   219→  const strength = getPasswordStrength(newPassword)
   220→  const passwordsMatch = newPassword === confirmPassword
   221→  const canSubmit = currentPassword && newPassword.length >= 8 && passwordsMatch
   222→
   223→  const handleUpdatePassword = async () => {
   224→    if (!passwordsMatch) {
   225→      toast.error('Passwords do not match')
   226→      return
   227→    }
   228→    if (newPassword.length < 8) {
   229→      toast.error('Password must be at least 8 characters')
   230→      return
   231→    }
   232→    setSaving(true)
   233→    try {
   234→      await apiFetch('/user/password', {
   235→        method: 'PATCH',
   236→        body: JSON.stringify({ currentPassword, newPassword }),
   237→      })
   238→      toast.success('Password updated successfully')
   239→      setCurrentPassword('')
   240→      setNewPassword('')
   241→      setConfirmPassword('')
   242→    } catch (err) {
   243→      toast.error(err instanceof Error ? err.message : 'Failed to update password')
   244→    } finally {
   245→      setSaving(false)
   246→    }
   247→  }
   248→
   249→  return (
   250→    <div className="space-y-6">
   251→      <Card>
   252→        <CardHeader>
   253→          <CardTitle className="flex items-center gap-2">
   254→            <Shield className="size-5 text-emerald-600" />
   255→            Change Password
   256→          </CardTitle>
   257→          <CardDescription>
   258→            Update your password to keep your account secure.
   259→          </CardDescription>

   261→        <CardContent className="space-y-4">
   262→          {/* Current Password */}
   263→          <div className="space-y-2">
   264→            <Label htmlFor="current-password">Current Password</Label>
   265→            <div className="relative">
   266→              <Input
   267→                id="current-password"
   268→                type={showCurrent ? 'text' : 'password'}
   269→                value={currentPassword}
   270→                onChange={(e) => setCurrentPassword(e.target.value)}
   271→                placeholder="Enter current password"
   272→              />
   273→              <Button
   274→                type="button"
   275→                variant="ghost"
   276→                size="icon"
   277→                className="absolute right-1 top-1/2 -translate-y-1/2 size-7"
   278→                onClick={() => setShowCurrent(!showCurrent)}
   279→              >
   280→                {showCurrent ? <EyeOff className="size-3.5" /> : <Eye className="size-3.5" />}
   281→              </Button>
   282→            </div>
   283→          </div>
   284→
   285→          <Separator />
   286→
   287→          {/* New Password */}
   288→          <div className="space-y-2">
   289→            <Label htmlFor="new-password">New Password</Label>
   290→            <div className="relative">
   291→              <Input
   292→                id="new-password"
   293→                type={showNew ? 'text' : 'password'}
   294→                value={newPassword}
   295→                onChange={(e) => setNewPassword(e.target.value)}
   296→                placeholder="Enter new password"
   297→              />
   298→              <Button
   299→                type="button"
   300→                variant="ghost"
   301→                size="icon"
   302→                className="absolute right-1 top-1/2 -translate-y-1/2 size-7"
   303→                onClick={() => setShowNew(!showNew)}
   304→              >
   305→                {showNew ? <EyeOff className="size-3.5" /> : <Eye className="size-3.5" />}
   306→              </Button>
   307→            </div>
   308→            {/* Password Strength Indicator */}
   309→            {newPassword && (
   310→              <div className="space-y-1.5">
   311→                <div className="h-1.5 w-full bg-muted rounded-full overflow-hidden">
   312→                  <div
   313→                    className={`h-full rounded-full transition-all duration-300 ${strength.color}`}
   314→                    style={{ width: `${strength.percent}%` }}
   315→                  />
   316→                </div>
   317→                <p className={`text-xs font-medium ${
   318→                  strength.label === 'Weak' ? 'text-red-500' :
   319→                  strength.label === 'Fair' ? 'text-orange-500' :
   320→                  strength.label === 'Medium' ? 'text-yellow-600' :
   321→                  'text-emerald-600'
   322→                }`}>
   323→                  Password strength: {strength.label}
   324→                </p>
   325→              </div>
   326→            )}
   327→          </div>
   328→
   329→          {/* Confirm Password */}
   330→          <div className="space-y-2">
   331→            <Label htmlFor="confirm-password">Confirm New Password</Label>
   332→            <Input
   333→              id="confirm-password"
   334→              type="password"
   335→              value={confirmPassword}
   336→              onChange={(e) => setConfirmPassword(e.target.value)}
   337→              placeholder="Confirm new password"
   338→            />
   339→            {confirmPassword && !passwordsMatch && (
   340→              <p className="text-xs text-red-500">Passwords do not match</p>
   341→            )}
   342→            {confirmPassword && passwordsMatch && (
   343→              <p className="text-xs text-emerald-600 flex items-center gap-1">
   344→                <CheckCircle2 className="size-3" /> Passwords match
   345→              </p>
   346→            )}
   347→          </div>
   348→
   349→          <div className="flex justify-end pt-2">
   350→            <Button
   351→              onClick={handleUpdatePassword}
   352→              disabled={!canSubmit || saving}
   353→              className="bg-emerald-600 hover:bg-emerald-700 text-white"
   354→            >
   355→              {saving ? (
   356→                <Loader2 className="size-4 animate-spin mr-2" />
   357→              ) : (
   358→                <Shield className="size-4 mr-2" />
   359→              )}
   360→              Update Password
   361→            </Button>
   362→          </div>
   363→        </CardContent>
   364→      </Card>
   365→    </div>
   366→  )
   367→}

   369→// ── API Keys Tab ───────────────────────────────────────────────────────────
   370→function ApiKeysTab() {
   371→  const [apiKeys, setApiKeys] = useState<ApiKeyItem[]>([])
   372→  const [loading, setLoading] = useState(true)
   373→  const [createOpen, setCreateOpen] = useState(false)
   374→  const [newKeyName, setNewKeyName] = useState('')
   375→  const [creating, setCreating] = useState(false)
   376→  const [createdKey, setCreatedKey] = useState<{ id: string; name: string; key: string } | null>(null)
   377→  const [copied, setCopied] = useState(false)
   378→  const [deleteKey, setDeleteKey] = useState<ApiKeyItem | null>(null)
   379→  const [deleting, setDeleting] = useState(false)
   380→
   381→  const loadApiKeys = useCallback(async () => {
   382→    try {
   383→      const data = await apiFetch('/user/api-keys')
   384→      setApiKeys(data.apiKeys || [])
   385→    } catch {
   386→      toast.error('Failed to load API keys')
   387→    } finally {
   388→      setLoading(false)
   389→    }
   390→  }, [])
   391→
   392→  useEffect(() => {
   393→    loadApiKeys()
   394→  }, [loadApiKeys])
   395→
   396→  const handleCreate = async () => {
   397→    if (!newKeyName.trim()) {
   398→      toast.error('Please enter a name for the API key')
   399→      return
   400→    }
   401→    setCreating(true)
   402→    try {
   403→      const data = await apiFetch('/user/api-keys', {
   404→        method: 'POST',
   405→        body: JSON.stringify({ name: newKeyName.trim() }),
   406→      })
   407→      setCreatedKey(data.apiKey)
   408→      setNewKeyName('')
   409→      toast.success('API key created successfully')
   410→      loadApiKeys()
   411→    } catch (err) {
   412→      toast.error(err instanceof Error ? err.message : 'Failed to create API key')
   413→    } finally {
   414→      setCreating(false)
   415→    }
   416→  }
   417→
   418→  const handleCopy = async (key: string) => {
   419→    try {
   420→      await navigator.clipboard.writeText(key)
   421→      setCopied(true)
   422→      toast.success('API key copied to clipboard')
   423→      setTimeout(() => setCopied(false), 2000)
   424→    } catch {
   425→      toast.error('Failed to copy to clipboard')
   426→    }
   427→  }
   428→
   429→  const handleDelete = async () => {
   430→    if (!deleteKey) return
   431→    setDeleting(true)
   432→    try {
   433→      await apiFetch(`/user/api-keys/${deleteKey.id}`, {
   434→        method: 'DELETE',
   435→      })
   436→      toast.success('API key deleted')
   437→      setDeleteKey(null)
   438→      loadApiKeys()
   439→    } catch (err) {
   440→      toast.error(err instanceof Error ? err.message : 'Failed to delete API key')
   441→    } finally {
   442→      setDeleting(false)
   443→    }
   444→  }
   445→
   446→  const formatDate = (dateStr: string) => {
   447→    return new Date(dateStr).toLocaleDateString('en-US', {
   448→      month: 'short',
   449→      day: 'numeric',
   450→      year: 'numeric',
   451→    })
   452→  }
   453→
   454→  return (
   455→    <div className="space-y-6">
   456→      <Card>
   457→        <CardHeader>
   458→          <div className="flex items-center justify-between">
   459→            <div>
   460→              <CardTitle className="flex items-center gap-2">
   461→                <Key className="size-5 text-emerald-600" />
   462→                API Keys
   463→              </CardTitle>
   464→              <CardDescription className="mt-1.5">
   465→                Manage your API keys for programmatic access to DeployHub.
   466→              </CardDescription>
   467→            </div>
   468→            <Dialog open={createOpen} onOpenChange={(open) => {
   469→              setCreateOpen(open)
   470→              if (!open) setNewKeyName('')
   471→            }}>
   472→              <DialogTrigger asChild>
   473→                <Button className="bg-emerald-600 hover:bg-emerald-700 text-white">
   474→                  <Plus className="size-4 mr-2" />
   475→                  Create API Key
   476→                </Button>
   477→              </DialogTrigger>
   478→              <DialogContent>
   479→                <DialogHeader>
   480→                  <DialogTitle>Create New API Key</DialogTitle>
   481→                  <DialogDescription>
   482→                    Give your API key a descriptive name to help you identify it later.
   483→                  </DialogDescription>
   484→                </DialogHeader>
   485→                <div className="space-y-4 py-2">
   486→                  <div className="space-y-2">
   487→                    <Label htmlFor="key-name">Key Name</Label>
   488→                    <Input
   489→                      id="key-name"
   490→                      value={newKeyName}
   491→                      onChange={(e) => setNewKeyName(e.target.value)}
   492→                      placeholder="e.g., CI/CD Pipeline, Development"
   493→                      onKeyDown={(e) => {
   494→                        if (e.key === 'Enter') handleCreate()
   495→                      }}
   496→                    />
   497→                  </div>
   498→                </div>
   499→                <DialogFooter>
   500→                  <Button variant="outline" onClick={() => setCreateOpen(false)}>
   501→                    Cancel
   502→                  </Button>
   503→                  <Button
   504→                    onClick={handleCreate}
   505→                    disabled={creating || !newKeyName.trim()}
   506→                    className="bg-emerald-600 hover:bg-emerald-700 text-white"
   507→                  >
   508→                    {creating ? <Loader2 className="size-4 animate-spin mr-2" /> : <KeyRound className="size-4 mr-2" />}

   512→              </DialogContent>
   513→            </Dialog>
   514→          </div>
   515→        </CardHeader>
   516→        <CardContent>

   521→              <AlertDescription className="space-y-2">
   522→                <p className="font-medium text-emerald-800">
   523→                  API key created! Copy it now — it won&apos;t be shown again.
   524→                </p>
   525→                <div className="flex items-center gap-2">
   526→                  <code className="flex-1 bg-white border rounded px-3 py-2 text-sm font-mono break-all">

   528→                  </code>
   529→                  <Button
   530→                    size="sm"
   531→                    variant="outline"
   532→                    onClick={() => handleCopy(createdKey.key)}
   533→                    className="shrink-0"
   534→                  >
   535→                    {copied ? <CheckCircle2 className="size-4 text-emerald-600" /> : <Copy className="size-4" />}
   536→                  </Button>
   537→                </div>
   538→                <p className="text-xs text-emerald-700">
   539→                  Key name: {createdKey.name}
   540→                </p>
   541→                <Button
   542→                  size="sm"
   543→                  variant="ghost"
   544→                  className="text-emerald-700 hover:text-emerald-800"
   545→                  onClick={() => setCreatedKey(null)}
   546→                >
   547→                  Dismiss
   548→                </Button>
   549→              </AlertDescription>
   550→            </Alert>
   551→          )}
   552→
   553→          {loading ? (
   554→            <div className="flex items-center justify-center py-8">
   555→              <Loader2 className="size-6 animate-spin text-muted-foreground" />
   556→            </div>

   558→            <div className="text-center py-12">
   559→              <Key className="size-10 text-muted-foreground mx-auto mb-3" />
   560→              <p className="font-medium text-muted-foreground">No API keys yet</p>
   561→              <p className="text-sm text-muted-foreground mt-1">
   562→                Create an API key to access the DeployHub API programmatically.
   563→              </p>
   564→              <Button
   565→                onClick={() => setCreateOpen(true)}
   566→                className="mt-4 bg-emerald-600 hover:bg-emerald-700 text-white"
   567→              >
   568→                <Plus className="size-4 mr-2" />
   569→                Create Your First API Key
   570→              </Button>
   571→            </div>
   572→          ) : (
   573→            <div className="rounded-lg border overflow-hidden">
   574→              <Table>
   575→                <TableHeader>
   576→                  <TableRow>
   577→                    <TableHead>Name</TableHead>
   578→                    <TableHead>Key</TableHead>
   579→                    <TableHead className="hidden sm:table-cell">Created</TableHead>

   600→                        <AlertDialog open={deleteKey?.id === apiKey.id} onOpenChange={(open) => {
   601→                          if (!open) setDeleteKey(null)
   602→                        }}>
   603→                          <AlertDialogTrigger asChild>
   604→                            <Button
   605→                              variant="ghost"
   606→                              size="icon"
   607→                              className="size-8 text-muted-foreground hover:text-red-600"
   608→                              onClick={() => setDeleteKey(apiKey)}
   609→                            >
   610→                              <Trash2 className="size-4" />
   611→                            </Button>
   612→                          </AlertDialogTrigger>
   613→                          <AlertDialogContent>
   614→                            <AlertDialogHeader>
   615→                              <AlertDialogTitle>Delete API Key</AlertDialogTitle>
   616→                              <AlertDialogDescription>
   617→                                Are you sure you want to delete the API key &quot;{deleteKey?.name}&quot;?
   618→                                Any applications using this key will lose access immediately.
   619→                                This action cannot be undone.
   620→                              </AlertDialogDescription>
   621→                            </AlertDialogHeader>
   622→                            <AlertDialogFooter>
   623→                              <AlertDialogCancel>Cancel</AlertDialogCancel>
   624→                              <AlertDialogAction

   632→                            </AlertDialogFooter>
   633→                          </AlertDialogContent>
   634→                        </AlertDialog>
   635→                      </TableCell>
   636→                    </TableRow>
   637→                  ))}
   638→                </TableBody>
   639→              </Table>
   640→            </div>
   641→          )}
   642→        </CardContent>

   649→interface NotificationPrefs {

   652→  billing: boolean
   653→  security: boolean
   654→  emailDigest: boolean
   655→}

   657→const DEFAULT_PREFS: NotificationPrefs = {
   658→  deployment: true,
   659→  domain: true,
   660→  billing: true,
   661→  security: true,
   662→  emailDigest: false,
   663→}
   664→
   665→function NotificationsTab() {
   666→  const [prefs, setPrefs] = useState<NotificationPrefs>(DEFAULT_PREFS)
   667→  const [saving, setSaving] = useState(false)
   668→
   669→  useEffect(() => {
   670→    try {
   671→      const stored = localStorage.getItem('deployhub_notification_prefs')
   672→      if (stored) {
   673→        setPrefs({ ...DEFAULT_PREFS, ...JSON.parse(stored) })
   674→      }
   675→    } catch {
   676→      // Use defaults
   677→    }
   678→  }, [])
   679→
   680→  const handleToggle = (key: keyof NotificationPrefs) => {
   681→    const updated = { ...prefs, [key]: !prefs[key] }
   682→    setPrefs(updated)
   683→  }
   684→
   685→  const handleSave = async () => {
   686→    setSaving(true)
   687→    try {
   688→      localStorage.setItem('deployhub_notification_prefs', JSON.stringify(prefs))
   689→      // Simulate API delay
   690→      await new Promise((r) => setTimeout(r, 500))
   691→      toast.success('Notification preferences saved')
   692→    } catch {
   693→      toast.error('Failed to save preferences')
   694→    } finally {
   695→      setSaving(false)
   696→    }
   697→  }
   698→
   699→  const notificationItems = [
   700→    {
   701→      key: 'deployment' as const,
   702→      icon: Server,
   703→      title: 'Deployment Notifications',
   704→      description: 'Get notified when deployments succeed, fail, or complete.',

   706→    {
   707→      key: 'domain' as const,
   708→      icon: Globe,
   709→      title: 'Domain Notifications',
   710→      description: 'Receive alerts about domain registration, renewal, and DNS changes.',
   711→    },
   712→    {
   713→      key: 'billing' as const,
   714→      icon: CreditCard,
   715→      title: 'Billing Notifications',
   716→      description: 'Get notified about payments, invoices, and subscription changes.',
   717→    },
   718→    {
   719→      key: 'security' as const,
   720→      icon: Shield,
   721→      title: 'Security Alerts',
   722→      description: 'Receive important security alerts about your account and resources.',
   723→    },
   724→    {
   725→      key: 'emailDigest' as const,
   726→      icon: Mail,
   727→      title: 'Email Digest',
   728→      description: 'Receive a weekly email summary of all your activity.',
   729→    },
   730→  ]
   731→
   732→  return (
   733→    <div className="space-y-6">
   734→      <Card>
   735→        <CardHeader>
   736→          <CardTitle className="flex items-center gap-2">
   737→            <Bell className="size-5 text-emerald-600" />
   738→            Notification Preferences
   739→          </CardTitle>
   740→          <CardDescription>
   741→            Choose which notifications you want to receive.
   742→          </CardDescription>
   743→        </CardHeader>
   744→        <CardContent className="space-y-1">

   749→                <div className="flex items-center justify-between py-4">
   750→                  <div className="flex items-start gap-3">
   751→                    <div className="size-9 rounded-lg bg-emerald-50 flex items-center justify-center shrink-0 mt-0.5">
   752→                      <Icon className="size-4 text-emerald-600" />
   753→                    </div>
   754→                    <div>
   755→                      <p className="text-sm font-medium">{item.title}</p>
   756→                      <p className="text-xs text-muted-foreground mt-0.5">{item.description}</p>
   757→                    </div>
   758→                  </div>
   759→                  <Switch
   760→                    checked={prefs[item.key]}
   761→                    onCheckedChange={() => handleToggle(item.key)}
   762→                  />
   763→                </div>
   764→                {index < notificationItems.length - 1 && <Separator />}
   765→              </div>
   766→            )
   767→          })}
   768→
   769→          <div className="flex justify-end pt-4">
   770→            <Button
   771→              onClick={handleSave}
   772→              disabled={saving}
   773→              className="bg-emerald-600 hover:bg-emerald-700 text-white"
   774→            >
   775→              {saving ? (
   776→                <Loader2 className="size-4 animate-spin mr-2" />
   777→              ) : (
   778→                <Save className="size-4 mr-2" />
   779→              )}
   780→              Save Preferences
   781→            </Button>
   782→          </div>
   783→        </CardContent>
   784→      </Card>
   785→    </div>
   786→  )
