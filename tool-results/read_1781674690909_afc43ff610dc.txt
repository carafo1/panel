    78→// ── Password Strength ──────────────────────────────────────────────────────
    79→function getPasswordStrength(password: string): { label: string; color: string; percent: number } {
    80→  if (!password) return { label: '', color: '', percent: 0 }
    81→  let score = 0
    82→  if (password.length >= 8) score++
    83→  if (password.length >= 12) score++
    84→  if (/[A-Z]/.test(password)) score++
    85→  if (/[0-9]/.test(password)) score++
    86→  if (/[^A-Za-z0-9]/.test(password)) score++
    87→
    88→  if (score <= 1) return { label: 'Weak', color: 'bg-red-500', percent: 20 }
    89→  if (score <= 2) return { label: 'Fair', color: 'bg-orange-500', percent: 40 }
    90→  if (score <= 3) return { label: 'Medium', color: 'bg-yellow-500', percent: 60 }
    91→  if (score <= 4) return { label: 'Strong', color: 'bg-emerald-400', percent: 80 }
    92→  return { label: 'Very Strong', color: 'bg-emerald-600', percent: 100 }
    93→}

   103→    if (!name.trim()) {

   215→// ── Two-Factor Authentication Section ──────────────────────────────────────
   216→function TwoFactorSection() {
   217→  const { user } = useAuthStore()
   218→  const [enabled, setEnabled] = useState<boolean>(!!user?.twoFactorEnabled)
   219→
   220→  // Refresh from server on mount (the auth-store may have been hydrated
   221→  // before the 2FA field was added).
   222→  useEffect(() => {
   223→    let cancelled = false
   224→    ;(async () => {
   225→      try {
   226→        const data = await apiFetch('/auth/me')
   227→        if (!cancelled && typeof data?.user?.twoFactorEnabled === 'boolean') {
   228→          setEnabled(data.user.twoFactorEnabled)
   229→        }
   230→      } catch {
   231→        // ignore — fall back to the store value
   232→      }
   233→    })()
   234→    return () => {
   235→      cancelled = true
   236→    }
   237→  }, [])
   238→
   239→  return (
   240→    <TwoFactorCard enabled={enabled} onToggle={setEnabled} />
   241→  )
   242→}
   243→
   244→interface TwoFactorCardProps {
   245→  enabled: boolean
   246→  onToggle: (enabled: boolean) => void
   247→}
   248→
   249→function TwoFactorCard({ enabled, onToggle }: TwoFactorCardProps) {
   250→  // ── Enable flow state ───────────────────────────────────────────────
   251→  const [enableOpen, setEnableOpen] = useState(false)
   252→  const [setupData, setSetupData] = useState<{
   253→    secret: string
   254→    qrCode: string
   255→    manualEntry: string
   256→  } | null>(null)
   257→  const [verifyToken, setVerifyToken] = useState('')
   258→  const [enabling, setEnabling] = useState(false)
   259→  const [settingUp, setSettingUp] = useState(false)
   260→  // 'qr' = show QR + verify input, 'backup' = show backup codes
   261→  const [enableStep, setEnableStep] = useState<'qr' | 'backup'>('qr')
   262→  const [backupCodes, setBackupCodes] = useState<string[]>([])
   263→  const [copiedAll, setCopiedAll] = useState(false)
   264→
   265→  // ── Disable flow state ──────────────────────────────────────────────
   266→  const [disableOpen, setDisableOpen] = useState(false)
   267→  const [disableToken, setDisableToken] = useState('')
   268→  const [disabling, setDisabling] = useState(false)
   269→
   270→  const handleBeginEnable = async () => {
   271→    setEnableOpen(true)
   272→    setEnableStep('qr')
   273→    setVerifyToken('')
   274→    setBackupCodes([])
   275→    setSetupData(null)
   276→    setSettingUp(true)
   277→    try {
   278→      const data = await apiFetch('/auth/2fa/setup', { method: 'POST' })
   279→      setSetupData(data)
   280→    } catch (err) {
   281→      toast.error(err instanceof Error ? err.message : 'Failed to start 2FA setup')
   282→      setEnableOpen(false)
   283→    } finally {
   284→      setSettingUp(false)
   285→    }
   286→  }
   287→
   288→  const handleVerifyEnable = async () => {
   289→    if (!setupData) return
   290→    if (verifyToken.length !== 6) {
   291→      toast.error('Please enter the 6-digit code')
   292→      return
   293→    }
   294→    setEnabling(true)
   295→    try {
   296→      const data = await apiFetch('/auth/2fa/verify', {
   297→        method: 'POST',
   298→        body: JSON.stringify({ secret: setupData.secret, token: verifyToken }),
   299→      })
   300→      setBackupCodes(data.backupCodes)
   301→      setEnableStep('backup')
   302→      setEnabled(true)
   303→      onToggle(true)
   304→      toast.success('Two-factor authentication enabled!')
   305→    } catch (err) {
   306→      toast.error(err instanceof Error ? err.message : 'Verification failed')
   307→    } finally {
   308→      setEnabling(false)
   309→    }
   310→  }
   311→
   312→  const handleCloseEnable = () => {
   313→    setEnableOpen(false)
   314→    setSetupData(null)
   315→    setVerifyToken('')
   316→    setBackupCodes([])
   317→    setEnableStep('qr')
   318→    setCopiedAll(false)
   319→  }
   320→
   321→  const handleDownloadBackupCodes = () => {
   322→    if (!backupCodes.length) return
   323→    const text =
   324→      `DeployHub 2FA Backup Codes\n` +
   325→      `Generated: ${new Date().toLocaleString()}\n\n` +
   326→      backupCodes.map((c, i) => `${i + 1}.  ${c}`).join('\n') +
   327→      `\n\nEach code can be used only once. Store them somewhere safe.\n`
   328→    const blob = new Blob([text], { type: 'text/plain' })
   329→    const url = URL.createObjectURL(blob)
   330→    const a = document.createElement('a')
   331→    a.href = url
   332→    a.download = 'deployhub-backup-codes.txt'
   333→    document.body.appendChild(a)
   334→    a.click()
   335→    document.body.removeChild(a)
   336→    URL.revokeObjectURL(url)
   337→    toast.success('Backup codes downloaded')
   338→  }
   339→
   340→  const handleCopyAllBackupCodes = async () => {
   341→    if (!backupCodes.length) return
   342→    try {
   343→      await navigator.clipboard.writeText(backupCodes.join('\n'))
   344→      setCopiedAll(true)
   345→      toast.success('All backup codes copied')
   346→      setTimeout(() => setCopiedAll(false), 2000)
   347→    } catch {
   348→      toast.error('Failed to copy to clipboard')
   349→    }
   350→  }
   351→
   352→  const handleDisable = async () => {
   353→    if (disableToken.length !== 6) {
   354→      toast.error('Please enter your 6-digit verification code')
   355→      return
   356→    }
   357→    setDisabling(true)
   358→    try {
   359→      await apiFetch('/auth/2fa/disable', {
   360→        method: 'POST',
   361→        body: JSON.stringify({ token: disableToken }),
   362→      })
   363→      setEnabled(false)
   364→      onToggle(false)
   365→      setDisableOpen(false)
   366→      setDisableToken('')
   367→      toast.success('Two-factor authentication disabled')
   368→    } catch (err) {
   369→      toast.error(err instanceof Error ? err.message : 'Failed to disable 2FA')
   370→    } finally {
   371→      setDisabling(false)
   372→    }
   373→  }
   374→
   375→  const handleCloseDisable = () => {
   376→    setDisableOpen(false)
   377→    setDisableToken('')
   378→  }
   379→
   380→  return (
   381→    <Card>
   382→      <CardHeader>
   383→        <CardTitle className="flex items-center gap-2">
   384→          <ShieldCheck className="size-5 text-emerald-600" />
   385→          Two-Factor Authentication
   386→        </CardTitle>
   387→        <CardDescription>
   388→          Add an extra layer of security to your account using an authenticator app.
   389→          You&apos;ll need a 6-digit code every time you sign in.
   390→        </CardDescription>
   391→      </CardHeader>
   392→      <CardContent className="space-y-4">
   393→        {/* Status row */}
   394→        <div className="flex items-center justify-between rounded-lg border p-4">
   395→          <div className="flex items-start gap-3">
   396→            <div
   397→              className={`size-9 rounded-lg flex items-center justify-center shrink-0 ${
   398→                enabled
   399→                  ? 'bg-emerald-50 text-emerald-600'
   400→                  : 'bg-amber-50 text-amber-600'
   401→              }`}
   402→            >
   403→              {enabled ? <ShieldCheck className="size-4" /> : <Shield className="size-4" />}
   404→            </div>
   405→            <div>
   406→              <p className="text-sm font-medium">
   407→                {enabled ? 'Two-factor authentication is enabled' : 'Two-factor authentication is not enabled'}
   408→              </p>
   409→              <p className="text-xs text-muted-foreground mt-0.5">
   410→                {enabled
   411→                  ? 'Your account is protected with an authenticator app.'
   412→                  : 'Enable 2FA to protect your account with a second factor.'}
   413→              </p>
   414→            </div>
   415→          </div>
   416→          <Badge
   417→            className={`shrink-0 ${
   418→              enabled
   419→                ? 'bg-emerald-100 text-emerald-700 hover:bg-emerald-100'
   420→                : 'bg-amber-100 text-amber-700 hover:bg-amber-100'
   421→            }`}
   422→          >
   423→            {enabled ? 'Enabled' : 'Disabled'}
   424→          </Badge>
   425→        </div>
   426→
   427→        {/* Action buttons */}
   428→        <div className="flex flex-col sm:flex-row gap-2">
   429→          {!enabled ? (
   430→            <Button
   431→              onClick={handleBeginEnable}
   432→              className="bg-emerald-600 hover:bg-emerald-700 text-white"
   433→              disabled={settingUp}
   434→            >
   435→              {settingUp ? (
   436→                <Loader2 className="size-4 animate-spin mr-2" />
   437→              ) : (
   438→                <Shield className="size-4 mr-2" />
   439→              )}
   440→              Enable 2FA
   441→            </Button>
   442→          ) : (
   443→            <Button
   444→              onClick={() => setDisableOpen(true)}
   445→              variant="outline"
   446→              className="text-red-600 hover:text-red-700 hover:bg-red-50 border-red-200"

   448→              <Shield className="size-4 mr-2" />

   456→          if (!open) handleCloseEnable()
   457→        }}>
   458→          <DialogContent className="max-w-md">
   459→            <DialogHeader>
   460→              <DialogTitle className="flex items-center gap-2">

   465→                {enableStep === 'qr'
   466→                  ? 'Scan the QR code with your authenticator app, then enter the 6-digit code it shows.'

   468→              </DialogDescription>

   470→
   471→            {enableStep === 'qr' && (
   472→              <div className="space-y-4 py-2">
