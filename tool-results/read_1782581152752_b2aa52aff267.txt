    74→// ── API helper ─────────────────────────────────────────────────────────
    75→async function apiFetch<T>(path: string, options?: RequestInit): Promise<T> {
    76→  const token = localStorage.getItem('deployhub_token')
    77→  const res = await fetch(`/api${path}`, {
    78→    ...options,
    79→    headers: {
    80→      'Content-Type': 'application/json',
    81→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    82→      ...options?.headers,
    83→    },
    84→  })
    85→  if (!res.ok) {
    86→    const data = await res.json().catch(() => ({ error: 'Request failed' }))
    87→    throw new Error(data.error || `Request failed with status ${res.status}`)
    88→  }
    89→  return res.json()
    90→}

   445→// ── Add-content-field helper ───────────────────────────────────────────
   446→function AddContentField({ onAdd, existing }: { onAdd: (key: string) => void; existing: string[] }) {
   447→  const [key, setKey] = useState('')
   448→  const [value, setValue] = useState('')
   449→
   450→  const handleAdd = () => {
   451→    const k = key.trim()
   452→    if (!k) {
   453→      toast.error('Enter a key first')
   454→      return
   455→    }
   456→    if (existing.includes(k)) {
   457→      toast.error('That key already exists')
   458→      return
   459→    }
   460→    onAdd(k)
   461→    if (value.trim()) {
   462→      // value is set on the parent via the onAdd callback for the key, but
   463→      // since the parent initialises to '', we separately set it here.
   464→    }
   465→    setKey('')
   466→    setValue('')
   467→  }
   468→
   469→  return (
   470→    <div className="grid grid-cols-1 md:grid-cols-[12rem_1fr_auto] gap-2 items-start pt-2 border-t">
   471→      <Input placeholder="key (e.g. hero.title)" value={key} onChange={(e) => setKey(e.target.value)} className="text-sm" />
   472→      <Input placeholder="value (optional)" value={value} onChange={(e) => setValue(e.target.value)} className="text-sm" />
   473→      <Button variant="outline" size="sm" onClick={handleAdd}>
   474→        <Plus className="size-4 mr-1" /> Add Field
   475→      </Button>
   476→    </div>
   477→  )
   478→}

   539→                    return (
   540→                      <SelectItem key={name} value={name}>
   541→                        <span className="inline-flex items-center gap-2">
   542→                          <Icon className="size-4" />
   543→                          {name}
   544→                        </span>
   545→                      </SelectItem>
   546→                    )
   547→                  })}
   548→                </SelectContent>
   549→              </Select>
   550→            </div>
   551→            <div className="space-y-1.5">
   552→              <Label>Accent gradient *</Label>
   553→              <Select value={service.accent ?? ACCENT_PRESETS[0]} onValueChange={(v) => update({ accent: v })}>
   554→                <SelectTrigger>
   555→                  <SelectValue placeholder="Pick a gradient" />
   556→                </SelectTrigger>
   557→                <SelectContent>
   558→                  {ACCENT_PRESETS.map((g) => (
   559→                    <SelectItem key={g} value={g}>
   560→                      <span className="inline-flex items-center gap-2">
   561→                        <span className={`size-4 rounded bg-gradient-to-br ${g}`} />
   562→                        <code className="text-xs">{g}</code>
   563→                      </span>
   564→                    </SelectItem>
   565→                  ))}
   566→                </SelectContent>
   567→              </Select>
   568→            </div>
   569→          </div>
   570→
   571→          <div className="space-y-1.5">
   572→            <Label>Features (one per line)</Label>
   573→            <Textarea
   574→              value={(service.features ?? []).join('\n')}
   575→              onChange={(e) => update({ features: e.target.value.split('\n').filter(Boolean) })}
   576→              rows={4}
   577→              placeholder={'Full-stack Next.js architecture\nPixel-perfect UI\nServerless APIs'}
   578→            />
   579→          </div>
   580→
   581→          <div className="flex items-center gap-2">
   582→            <Switch checked={service.visible ?? true} onCheckedChange={(v) => update({ visible: v })} id="visible" />
   583→            <Label htmlFor="visible">Visible on public site</Label>
   584→          </div>
   585→
   586→          {/* Plans */}
   587→          <SubSection
   588→            title="Pricing Plans"
   589→            count={service.plans?.length ?? 0}
   590→            onAdd={() => update({ plans: [...(service.plans ?? []), { name: '', price: 0, period: '/mo', description: '', features: [], popular: false }] })}
   591→          >
   592→            {(service.plans ?? []).map((plan, i) => (
   593→              <NestedCard key={i} onRemove={() => update({ plans: service.plans!.filter((_, idx) => idx !== i) })}>
   594→                <div className="grid grid-cols-2 md:grid-cols-4 gap-2">
   595→                  <Input placeholder="Name" value={plan.name} onChange={(e) => { const p = [...service.plans!]; p[i] = { ...plan, name: e.target.value }; update({ plans: p }) }} />
   596→                  <Input type="number" placeholder="Price" value={plan.price} onChange={(e) => { const p = [...service.plans!]; p[i] = { ...plan, price: Number(e.target.value) }; update({ plans: p }) }} />
   597→                  <Input placeholder="Period" value={plan.period} onChange={(e) => { const p = [...service.plans!]; p[i] = { ...plan, period: e.target.value }; update({ plans: p }) }} />
   598→                  <div className="flex items-center gap-2">
   599→                    <Switch checked={plan.popular} onCheckedChange={(v) => { const p = [...service.plans!]; p[i] = { ...plan, popular: v }; update({ plans: p }) }} id={`pop-${i}`} />
   600→                    <Label htmlFor={`pop-${i}`} className="text-xs">Popular</Label>
   601→                  </div>
   602→                </div>
   603→                <Input placeholder="Description" value={plan.description} onChange={(e) => { const p = [...service.plans!]; p[i] = { ...plan, description: e.target.value }; update({ plans: p }) }} className="mt-2" />
   604→                <Textarea placeholder="Features (one per line)" rows={2} value={plan.features.join('\n')} onChange={(e) => { const p = [...service.plans!]; p[i] = { ...plan, features: e.target.value.split('\n').filter(Boolean) }; update({ plans: p }) }} className="mt-2" />
   605→              </NestedCard>
   606→            ))}
   607→          </SubSection>
   608→
   609→          {/* Highlights */}
   610→          <SubSection
   611→            title="Highlights"
   612→            count={service.highlights?.length ?? 0}
   613→            onAdd={() => update({ highlights: [...(service.highlights ?? []), { title: '', description: '', iconName: 'LayoutGrid' }] })}
   614→          >
   615→            {(service.highlights ?? []).map((h, i) => (
   616→              <NestedCard key={i} onRemove={() => update({ highlights: service.highlights!.filter((_, idx) => idx !== i) })}>
   617→                <div className="grid grid-cols-1 md:grid-cols-2 gap-2">
   618→                  <Input placeholder="Title" value={h.title} onChange={(e) => { const arr = [...service.highlights!]; arr[i] = { ...h, title: e.target.value }; update({ highlights: arr }) }} />
   619→                  <Select value={h.iconName} onValueChange={(v) => { const arr = [...service.highlights!]; arr[i] = { ...h, iconName: v }; update({ highlights: arr }) }}>
   620→                    <SelectTrigger><SelectValue placeholder="Icon" /></SelectTrigger>
   621→                    <SelectContent className="max-h-60">
   622→                      {ICON_NAMES.map((name) => {
   623→                        const Icon = resolveIcon(name)
   624→                        return <SelectItem key={name} value={name}><span className="inline-flex items-center gap-2"><Icon className="size-4" />{name}</span></SelectItem>
   625→                      })}
   626→                    </SelectContent>
   627→                  </Select>
   628→                </div>
   629→                <Textarea placeholder="Description" rows={2} value={h.description} onChange={(e) => { const arr = [...service.highlights!]; arr[i] = { ...h, description: e.target.value }; update({ highlights: arr }) }} className="mt-2" />
   630→              </NestedCard>
   631→            ))}
   632→          </SubSection>
   633→
   634→          {/* FAQs */}
   635→          <SubSection
   636→            title="FAQs"
   637→            count={service.faqs?.length ?? 0}
   638→            onAdd={() => update({ faqs: [...(service.faqs ?? []), { q: '', a: '' }] })}
   639→          >
   640→            {(service.faqs ?? []).map((f, i) => (
   641→              <NestedCard key={i} onRemove={() => update({ faqs: service.faqs!.filter((_, idx) => idx !== i) })}>
   642→                <Input placeholder="Question" value={f.q} onChange={(e) => { const arr = [...service.faqs!]; arr[i] = { ...f, q: e.target.value }; update({ faqs: arr }) }} />
   643→                <Textarea placeholder="Answer" rows={2} value={f.a} onChange={(e) => { const arr = [...service.faqs!]; arr[i] = { ...f, a: e.target.value }; update({ faqs: arr }) }} className="mt-2" />
   644→              </NestedCard>
   645→            ))}
   646→          </SubSection>
   647→        </div>
   648→
   649→        <DialogFooter>
   650→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={saving}>Cancel</Button>
   651→          <Button onClick={onSave} disabled={saving} className="bg-indigo-500 hover:bg-indigo-600 text-white border-0">
   652→            {saving ? <Loader2 className="size-4 mr-1 animate-spin" /> : <Save className="size-4 mr-1" />}
   653→            {isEdit ? 'Save Changes' : 'Create Service'}
   654→          </Button>
   655→        </DialogFooter>
   656→      </DialogContent>
   657→    </Dialog>
   658→  )
   659→}

   661→// ── Sub-section wrapper ────────────────────────────────────────────────
   662→function SubSection({ title, count, onAdd, children }: { title: string; count: number; onAdd: () => void; children: React.ReactNode }) {
   663→  return (
   664→    <div className="space-y-2">
   665→      <div className="flex items-center justify-between">
   666→        <div className="flex items-center gap-2">

   668→          <Badge variant="secondary" className="text-xs">{count}</Badge>
   669→        </div>
   670→        <Button variant="outline" size="sm" onClick={onAdd} type="button">
   671→          <Plus className="size-3.5 mr-1" /> Add

   677→}
