   204→   204→// Simulated SSL data per domain (deterministic based on domain name)
   205→   205→function getSimulatedSslData(domainName: string): SslCertificate {
   206→   206→  const hash = domainName.split('').reduce((acc, c) => acc + c.charCodeAt(0), 0)
   207→   207→  const statuses: SslCertificate['status'][] = ['active', 'active', 'active', 'expired', 'none']
   208→   208→  const status = statuses[hash % statuses.length]
   209→   209→
   210→   210→  if (status === 'none') {
   211→   211→    return {
   212→   212→      status: 'none',
   213→   213→      type: "Let's Encrypt",
   214→   214→      issuer: '-',
   215→   215→      validFrom: '-',
   216→   216→      validTo: '-',
   217→   217→      autoRenew: false,
   218→   218→      domain: domainName,
   219→   219→    }
   220→   220→  }
   221→   221→
   222→   222→  const now = new Date()
   223→   223→  const validFrom = new Date(now)
   224→   224→  validFrom.setMonth(validFrom.getMonth() - 3)
   225→   225→  const validTo = new Date(now)
   226→   226→  if (status === 'expired') {
   227→   227→    validTo.setMonth(validTo.getMonth() - 1)
   228→   228→  } else {
   229→   229→    validTo.setMonth(validTo.getMonth() + 9)
   230→   230→  }
   231→   231→
   232→   232→  return {
   233→   233→    status,
   234→   234→    type: hash % 3 === 0 ? 'Custom' : "Let's Encrypt",
   235→   235→    issuer: hash % 3 === 0 ? 'DigiCert Inc.' : "Let's Encrypt Authority X3",
   236→   236→    validFrom: validFrom.toISOString(),
   237→   237→    validTo: validTo.toISOString(),
   238→   238→    autoRenew: hash % 3 !== 0,
   239→   239→    domain: domainName,
   240→   240→    fingerprint: Array.from({ length: 8 }, (_, i) => ((hash * (i + 1)) % 256).toString(16).padStart(2, '0')).join(':'),
   241→   241→  }
   242→   242→}
