     1→/**
     2→ * Core Orchestration Engine
     3→ * =========================
     4→ *
     5→ * The heart of DeployHub's automated provisioning pipeline. Coordinates the
     6→ * full flow:
     7→ *
     8→ *     domain registration → DNS configuration → web hosting → email hosting → verification
     9→ *
    10→ * Each step is independently retryable via `withRetry()` with exponential
    11→ * backoff. Domain + DNS are CRITICAL steps — if either fails after retries,
    12→ * the pipeline stops and triggers `rollbackProvisioning()` in reverse order.
    13→ * Hosting + Email are NON-CRITICAL — if they fail but domain+DNS succeeded,
    14→ * the pipeline continues to verification and the overall status is `partial`.
    15→ *
    16→ * All external API calls (Dynadot, Coolify, Mailcow) have DEMO MODE fallbacks
    17→ * so the entire pipeline runs out-of-the-box without any real provider env
    18→ * vars configured.
    19→ *
    20→ * Used by:
    21→ *  - Task #19 Orchestration Engine — ties together Dynadot, DNS, Coolify, and
    22→ *    Mailcow into a single end-to-end provisioning flow.
    23→ */

   188→/** Resolve the hosting server IP — from Server table, or default for demo. */
   189→async function resolveHostingServerIp(serverId?: string): Promise<string> {
   190→  if (serverId) {
   191→    const server = await db.server.findUnique({ where: { id: serverId } })
   192→    if (server?.ip) return server.ip

   212→// ─── Retry helper ────────────────────────────────────────────────────────

   214→/**
   215→ * Retry a function up to `maxAttempts` times with exponential backoff
   216→ * (1000ms * 2^attempt). Each attempt increments the ProvisioningStep.attempts
   217→ * counter and updates lastAttemptAt.
   218→ */
   219→async function withRetry<T>(
   220→  fn: () => Promise<T>,
   221→  opts: { maxAttempts: number; stepName: string; jobId: string; stepId: string }
   222→): Promise<T> {
   223→  let lastError: unknown = null
   224→  for (let attempt = 1; attempt <= opts.maxAttempts; attempt++) {
   225→    try {
   226→      // Update attempt tracking on the step row.
   227→      await db.provisioningStep.update({
   228→        where: { id: opts.stepId },
   229→        data: {
   230→          attempts: attempt,
   231→          lastAttemptAt: new Date(),
   232→        },
   233→      })
   234→      console.log(
   235→        `[orchestration] job=${opts.jobId} step=${opts.stepName} attempt=${attempt}/${opts.maxAttempts}`
   236→      )
   237→      const result = await fn()
   238→      return result
   239→    } catch (err) {
   240→      lastError = err
   241→      const msg = err instanceof Error ? err.message : String(err)
   242→      console.warn(
   243→        `[orchestration] job=${opts.jobId} step=${opts.stepName} attempt=${attempt} failed: ${msg}`
   244→      )
   245→      if (attempt < opts.maxAttempts) {
   246→        const backoff = 1000 * Math.pow(2, attempt - 1)
   247→        await sleep(backoff)
   248→      }
   249→    }
   250→  }
   251→  throw lastError instanceof Error
   252→    ? lastError
   253→    : new Error(`Step ${opts.stepName} failed after ${opts.maxAttempts} attempts`)
   254→}

   256→// ─── startProvisioning ───────────────────────────────────────────────────
   257→
   258→/**
   259→ * Create a ProvisioningJob record with 5 ProvisioningStep rows, then kick off
   260→ * the pipeline in the background (NOT awaited). Returns immediately with the
   261→ * jobId so the caller can poll for status via getProvisioningStatus().
   262→ */
   263→export async function startProvisioning(
   264→  req: ProvisioningRequest
   265→): Promise<{ jobId: string; status: 'started' | 'failed'; error?: string }> {
   266→  const includeHosting = req.includeHosting ?? true
   267→  const includeEmail = req.includeEmail ?? true
   268→
   269→  try {
   270→    // 1. Create the ProvisioningJob record.
   271→    const job = await db.provisioningJob.create({
   272→      data: {
   273→        userId: req.userId,
   274→        domainName: req.domainName,
   275→        stack: req.stack,
   276→        includeHosting,
   277→        includeEmail,
   278→        status: 'pending',
   279→        currentStep: 'init',
   280→      },
   281→    })
   282→
   283→    // 2. Create the 5 ProvisioningStep records.
   284→    for (const def of STEP_DEFINITIONS) {
   285→      const skipped =
   286→        (def.name === 'hosting_provision' && !includeHosting) ||
   287→        (def.name === 'email_provision' && !includeEmail)
   288→      await db.provisioningStep.create({
   289→        data: {
   290→          jobId: job.id,
   291→          stepName: def.name,
   292→          stepOrder: def.order,
   293→          status: skipped ? 'skipped' : 'pending',
   294→          maxAttempts: 3,
   295→          input: JSON.stringify({
   296→            domainName: req.domainName,
   297→            stack: req.stack,
   298→            includeHosting,
   299→            includeEmail,
   300→            planId: req.planId,
   301→            serverId: req.serverId,
   302→          }),
   303→        },
   304→      })
   305→    }
   306→
   307→    await logAudit({
   308→      userId: req.userId,
   309→      action: 'provisioning.start',
   310→      resource: 'provisioning_job',
   311→      resourceId: job.id,
   312→      metadata: {
   313→        domainName: req.domainName,
   314→        stack: req.stack,
   315→        includeHosting,
   316→        includeEmail,
   317→      },
   318→    })
   319→
   320→    // 3. Kick off the pipeline in the background. Do NOT await.
   321→    runProvisioningPipeline(job.id).catch((err) => {
   322→      console.error(
   323→        `[orchestration] Background pipeline failed for job ${job.id}:`,
   324→        err instanceof Error ? err.message : err
   325→      )
   326→    })
   327→
   328→    return { jobId: job.id, status: 'started' }
   329→  } catch (err) {
   330→    const msg = err instanceof Error ? err.message : String(err)
   331→    console.error('[orchestration] startProvisioning failed:', msg)
   332→    return { jobId: '', status: 'failed', error: msg }
   333→  }
   334→}

   336→// ─── runProvisioningPipeline ─────────────────────────────────────────────
   337→
   338→/**
   339→ * Run the full provisioning pipeline for a job. Updates the job's status,
   340→ * runs each step in order with retries, builds the ProvisioningResult, and
   341→ * sends a notification to the user on completion.
   342→ *
   343→ * Called by startProvisioning() in the background (fire-and-forget) or by a
   344→ * worker process.
   345→ */
   346→export async function runProvisioningPipeline(
   347→  jobId: string
   348→): Promise<ProvisioningResult> {
   349→  const job = await db.provisioningJob.findUnique({
   350→    where: { id: jobId },
   351→    include: { steps: { orderBy: { stepOrder: 'asc' } } },
   352→  })
   353→  if (!job) {
   354→    throw new Error(`ProvisioningJob not found: ${jobId}`)
   355→  }
   356→  if (job.status === 'running') {
   357→    throw new Error(`Pipeline already running for job ${jobId}`)
   358→  }
   359→  if (job.status === 'success' || job.status === 'failed' || job.status === 'rolled_back') {
   360→    // Already completed — return the stored result if available.
   361→    if (job.result) {
   362→      try {
   363→        return JSON.parse(job.result) as ProvisioningResult
   364→      } catch {
   365→        // fall through and re-run
   366→      }
   367→    }
   368→  }
   369→
   370→  const startedAt = new Date()
   371→  await db.provisioningJob.update({
   372→    where: { id: jobId },
   373→    data: { status: 'running', startedAt, currentStep: 'domain' },
   374→  })
   375→
   376→  const req = JSON.parse(job.steps[0]?.input || '{}') as ProvisioningRequest
   377→  const ctx: PipelineContext = {
   378→    domainName: job.domainName,
   379→    provider: 'dynadot',
   380→    mailServerHostname: resolveMailServerHostname(),
   381→    hostingServerIp: await resolveHostingServerIp(req.serverId),
   382→    mailboxes: [],
   383→    dnsRecords: [],
   384→    errors: [],
   385→    hostingReachable: false,
   386→    emailVerified: false,
   387→    dnsPropagated: false,
   388→    nameservers: [],
   389→  }
   390→
   391→  let domainResult: { domainId: string; externalId?: string; provider: string } | undefined
   392→  let dnsResult:
   393→    | { records: Array<{ type: string; name: string; value: string; recordId: string }> }
   394→    | undefined
   395→  let hostingResult:
   396→    | { projectId: string; coolifyId: string; url: string }
   397→    | undefined
   398→  let emailResult:
   399→    | {
   400→        mailboxes: Array<{
   401→          address: string
   402→          password: string
   403→          type: string
   404→          mailboxId: string
   405→        }>
   406→        dkimPublicKey: string
   407→        webmailUrl: string
   408→      }
   409→    | undefined
   410→  let verifyResult:
   411→    | { hostingReachable: boolean; emailVerified: boolean; dnsPropagated: boolean }
   412→    | undefined
   413→
   414→  // Track whether non-critical steps failed (for 'partial' status).
   415→  let hostingFailed = false
   416→  let emailFailed = false
   417→  let criticalFailed = false
   418→
   419→  for (const step of job.steps) {
   420→    // Skip steps that are explicitly skipped (includeHosting/includeEmail = false)
   421→    if (step.status === 'skipped') continue
   422→    // Skip steps that already succeeded (idempotent re-runs)
   423→    if (step.status === 'success') {
   424→      // Re-hydrate context from stored output so subsequent steps have the data
   425→      try {
   426→        const out = step.output ? JSON.parse(step.output) : undefined
   427→        if (step.stepName === 'domain_register' && out) {
   428→          domainResult = out
   429→          ctx.domainId = out.domainId
   430→          ctx.externalId = out.externalId
   431→          ctx.provider = out.provider
   432→        }
   433→        if (step.stepName === 'dns_configure' && out) {
   434→          dnsResult = out
   435→          ctx.dnsRecords = out.records || []
   436→        }
   437→        if (step.stepName === 'hosting_provision' && out) {
   438→          hostingResult = out
   439→          ctx.projectId = out.projectId
   440→          ctx.coolifyId = out.coolifyId
   441→          ctx.hostingUrl = out.url
   442→        }
   443→        if (step.stepName === 'email_provision' && out) {
   444→          emailResult = out
   445→          ctx.mailboxes = out.mailboxes || []
   446→          ctx.dkimPublicKey = out.dkimPublicKey
   447→          ctx.webmailUrl = out.webmailUrl
   448→        }
   449→      } catch {
   450→        // ignore parse errors
   451→      }
   452→      continue
   453→    }
   454→
   455→    // Mark step running.
   456→    const stepStart = new Date()
   457→    await db.provisioningStep.update({
   458→      where: { id: step.id },
   459→      data: { status: 'running', startedAt: stepStart, attempts: 0 },
   460→    })
   461→    await db.provisioningJob.update({
   462→      where: { id: jobId },
   463→      data: { currentStep: step.stepName },
   464→    })
   465→
   466→    try {
   467→      let output: unknown
   468→      switch (step.stepName) {
   469→        case 'domain_register':
   470→          output = domainResult = await withRetry(
   471→            () => provisionDomain(job, step),
   472→            {
   473→              maxAttempts: step.maxAttempts,
   474→              stepName: step.stepName,
   475→              jobId,
   476→              stepId: step.id,
   477→            }
   478→          )
   479→          ctx.domainId = domainResult.domainId
   480→          ctx.externalId = domainResult.externalId
   481→          ctx.provider = domainResult.provider
   482→          break
   483→        case 'dns_configure':
   484→          if (!ctx.domainId) {
   485→            throw new Error('Cannot configure DNS — domain step did not produce a domainId')
   486→          }
   487→          output = dnsResult = await withRetry(
   488→            () =>
   489→              configureDns(
   490→                job,
   491→                step,
   492→                ctx.domainId!,
   493→                ctx.mailServerHostname,
   494→                ctx.hostingServerIp
   495→              ),
   496→            {
   497→              maxAttempts: step.maxAttempts,
   498→              stepName: step.stepName,
   499→              jobId,
   500→              stepId: step.id,
   501→            }
   502→          )
   503→          ctx.dnsRecords = dnsResult.records
   504→          break
   505→        case 'hosting_provision':
   506→          if (!ctx.domainId) {
   507→            throw new Error('Cannot provision hosting — no domainId in context')
   508→          }
   509→          output = hostingResult = await withRetry(
   510→            () => provisionHosting(job, step, ctx.domainId!, ctx.domainName),
   511→            {
   512→              maxAttempts: step.maxAttempts,
   513→              stepName: step.stepName,
   514→              jobId,
   515→              stepId: step.id,
   516→            }
   517→          )
   518→          ctx.projectId = hostingResult.projectId
   519→          ctx.coolifyId = hostingResult.coolifyId
   520→          ctx.hostingUrl = hostingResult.url
   521→          break
   522→        case 'email_provision':
   523→          if (!ctx.domainId) {
   524→            throw new Error('Cannot provision email — no domainId in context')
   525→          }
   526→          output = emailResult = await withRetry(
   527→            () => provisionEmail(job, step, ctx.domainId!, ctx.domainName),
   528→            {
   529→              maxAttempts: step.maxAttempts,
   530→              stepName: step.stepName,
   531→              jobId,
   532→              stepId: step.id,
   533→            }
   534→          )
   535→          ctx.mailboxes = emailResult.mailboxes
   536→          ctx.dkimPublicKey = emailResult.dkimPublicKey
   537→          ctx.webmailUrl = emailResult.webmailUrl
   538→          break
   539→        case 'verify':
   540→          output = verifyResult = await withRetry(
   541→            () => verifyProvisioning(job, step, ctx.domainName, ctx.projectId || ''),
   542→            {
   543→              maxAttempts: step.maxAttempts,
   544→              stepName: step.stepName,
   545→              jobId,
   546→              stepId: step.id,
   547→            }
   548→          )
   549→          ctx.hostingReachable = verifyResult.hostingReachable
   550→          ctx.emailVerified = verifyResult.emailVerified
   551→          ctx.dnsPropagated = verifyResult.dnsPropagated
   552→          break
   553→        default:
   554→          throw new Error(`Unknown step: ${step.stepName}`)
   555→      }
   556→
   557→      // Step succeeded — persist output + status.
   558→      const stepEnd = new Date()
   559→      await db.provisioningStep.update({
   560→        where: { id: step.id },
   561→        data: {
   562→          status: 'success',
   563→          output: JSON.stringify(output),
   564→          completedAt: stepEnd,
   565→          duration: stepEnd.getTime() - stepStart.getTime(),
   566→        },
   567→      })
   568→
   569→      await logAudit({
   570→        userId: job.userId,
   571→        action: `provisioning.step.${step.stepName}.success`,
   572→        resource: 'provisioning_step',
   573→        resourceId: step.id,
   574→        metadata: { jobId, stepName: step.stepName },
   575→      })
   576→    } catch (err) {
   577→      const msg = err instanceof Error ? err.message : String(err)
   578→      const stepEnd = new Date()
   579→      ctx.errors.push({
   580→        step: step.stepName,
   581→        error: msg,
   582→        timestamp: stepEnd.toISOString(),
   583→      })
   584→
   585→      await db.provisioningStep.update({
   586→        where: { id: step.id },
   587→        data: {
   588→          status: 'failed',
   589→          error: msg,
   590→          completedAt: stepEnd,
   591→          duration: stepEnd.getTime() - stepStart.getTime(),
   592→        },
   593→      })
   594→
   595→      await logAudit({
   596→        userId: job.userId,
   597→        action: `provisioning.step.${step.stepName}.failed`,
   598→        resource: 'provisioning_step',
   599→        resourceId: step.id,
   600→        metadata: { jobId, stepName: step.stepName, error: msg },
   601→        status: 'failure',
   602→      })
   603→
   604→      console.error(
   605→        `[orchestration] job=${jobId} step=${step.stepName} FAILED after retries: ${msg}`
   606→      )
   607→
   608→      // Critical step failure → stop pipeline + rollback.
   609→      if (CRITICAL_STEPS.has(step.stepName)) {
   610→        criticalFailed = true
   611→        break
   612→      }
   613→      // Non-critical step failure → mark partial + continue.
   614→      if (step.stepName === 'hosting_provision') hostingFailed = true
   615→      if (step.stepName === 'email_provision') emailFailed = true
   616→      // For verify step, just continue — we'll reflect status in the result.
   617→    }
   618→  }
   619→
   620→  // ─── Rollback on critical failure ────────────────────────────────────
   621→  if (criticalFailed) {
   622→    await rollbackProvisioning(jobId).catch((rbErr) => {
   623→      console.error(
   624→        `[orchestration] rollback failed for job ${jobId}:`,
   625→        rbErr instanceof Error ? rbErr.message : rbErr
   626→      )
   627→    })
   628→    const completedAt = new Date()
   629→    const duration = completedAt.getTime() - startedAt.getTime()
   630→    const result: ProvisioningResult = buildResult(
   631→      job.id,
   632→      ctx,
   633→      domainResult,
   634→      dnsResult,
   635→      hostingResult,
   636→      emailResult,
   637→      verifyResult,
   638→      hostingFailed,
   639→      emailFailed,
   640→      'failed',
   641→      duration
   642→    )
   643→    await db.provisioningJob.update({
   644→      where: { id: jobId },
   645→      data: {
   646→        status: 'rolled_back',
   647→        currentStep: 'rolled_back',
   648→        result: JSON.stringify(result),
   649→        completedAt,
   650→        error: 'Critical step failed — pipeline rolled back',
   651→      },
   652→    })
   653→    await createNotification({
   654→      userId: job.userId,
   655→      title: 'Provisioning failed',
   656→      message: `Provisioning of ${job.domainName} failed and was rolled back. See job ${jobId} for details.`,
   657→      type: 'error',
   658→      link: `/dashboard/provisioning/${jobId}`,
   659→    }).catch(() => {})
   660→    return result
   661→  }
   662→
   663→  // ─── Build final result ──────────────────────────────────────────────
   664→  const completedAt = new Date()
   665→  const duration = completedAt.getTime() - startedAt.getTime()
   666→  const overallStatus: 'success' | 'partial' | 'failed' =
   667→    hostingFailed || emailFailed ? (domainResult && dnsResult ? 'partial' : 'failed') : 'success'
   668→
   669→  const result = buildResult(
   670→    job.id,
   671→    ctx,
   672→    domainResult,
   673→    dnsResult,
   674→    hostingResult,
   675→    emailResult,
   676→    verifyResult,
   677→    hostingFailed,
   678→    emailFailed,
   679→    overallStatus,
   680→    duration
   681→  )
   682→
   683→  await db.provisioningJob.update({
   684→    where: { id: jobId },
   685→    data: {
   686→      status: overallStatus,
   687→      currentStep: 'done',
   688→      result: JSON.stringify(result),
   689→      completedAt,
   690→    },
   691→  })
   692→
   693→  // Send a notification to the user.
   694→  const notifType = overallStatus === 'success' ? 'success' : overallStatus === 'partial' ? 'warning' : 'error'
   695→  const notifMsg =
   696→    overallStatus === 'success'
   697→      ? `Your deployment ${job.domainName} is ready! Domain, DNS, hosting, and email are all configured.`
   698→      : overallStatus === 'partial'
   699→        ? `Provisioning of ${job.domainName} is partially complete. Some services may need attention.`
   700→        : `Provisioning of ${job.domainName} failed. Please check the job details.`
   701→  await createNotification({
   702→    userId: job.userId,
   703→    title:
   704→      overallStatus === 'success'
   705→        ? 'Provisioning complete'
   706→        : overallStatus === 'partial'
   707→          ? 'Provisioning partial'
   708→          : 'Provisioning failed',
   709→    message: notifMsg,
   710→    type: notifType,
   711→    link: `/dashboard/provisioning/${jobId}`,
   712→  }).catch(() => {})
   713→
   714→  await logAudit({
   715→    userId: job.userId,
   716→    action: `provisioning.complete.${overallStatus}`,
   717→    resource: 'provisioning_job',
   718→    resourceId: jobId,
   719→    metadata: { domainName: job.domainName, duration, overallStatus },
   720→  })
   721→
   722→  return result
   723→}
   724→
   725→// ─── Build the ProvisioningResult from the per-step outputs ──────────────
   726→
   727→function buildResult(
   728→  jobId: string,
   729→  ctx: PipelineContext,
   730→  domainResult: { domainId: string; externalId?: string; provider: string } | undefined,
   731→  dnsResult:
   732→    | { records: Array<{ type: string; name: string; value: string; recordId: string }> }
   733→    | undefined,
   734→  hostingResult: { projectId: string; coolifyId: string; url: string } | undefined,
   735→  emailResult:
   736→    | {
   737→        mailboxes: Array<{
   738→          address: string
   739→          password: string
   740→          type: string
   741→          mailboxId: string
   742→        }>
   743→        dkimPublicKey: string
   744→        webmailUrl: string
   745→      }
   746→    | undefined,
   747→  verifyResult:
   748→    | { hostingReachable: boolean; emailVerified: boolean; dnsPropagated: boolean }
   749→    | undefined,
   750→  hostingFailed: boolean,
   751→  emailFailed: boolean,
   752→  overallStatus: 'success' | 'partial' | 'failed',
   753→  duration: number
   754→): ProvisioningResult {
   755→  const domainStatus: ProvisioningResult['domainStatus'] = {
   756→    status: domainResult ? 'success' : 'fail',
   757→    domain: ctx.domainName,
   758→    provider: domainResult?.provider || ctx.provider,
   759→    externalId: domainResult?.externalId || ctx.externalId,
   760→    registered: !!domainResult,
   761→    details: domainResult
   762→      ? `Domain registered via ${domainResult.provider}.`
   763→      : 'Domain registration failed.',
   764→  }
   765→
   766→  const dnsStatus: ProvisioningResult['dnsStatus'] = {
   767→    status: dnsResult ? 'success' : 'fail',
   768→    records: (dnsResult?.records || ctx.dnsRecords || []).map((r) => ({
   769→      type: r.type,
   770→      name: r.name,
   771→      value: r.value,
   772→      status: 'created',
   773→    })),
   774→    propagationVerified: verifyResult?.dnsPropagated ?? false,
   775→    details: dnsResult
   776→      ? `${dnsResult.records.length} DNS records configured (A, MX, TXT for SPF/DKIM/DMARC).`
   777→      : 'DNS configuration failed.',
   778→  }
   779→
   780→  const hostingStatus: ProvisioningResult['hostingStatus'] = hostingResult
   781→    ? {
   782→        status: 'success',
   783→        projectId: hostingResult.projectId,
   784→        url: hostingResult.url,
   785→        coolifyId: hostingResult.coolifyId,
   786→        sslEnabled: true,
   787→        healthChecked: verifyResult?.hostingReachable ?? false,
   788→        details: `Application provisioned in Coolify (id=${hostingResult.coolifyId}). URL: ${hostingResult.url}`,
   789→      }
   790→    : hostingFailed
   791→      ? {
   792→          status: 'fail',
   793→          sslEnabled: false,
   794→          healthChecked: false,
   795→          details: 'Hosting provisioning failed after retries.',
   796→        }
   797→      : {
   798→          status: 'skipped',
   799→          sslEnabled: false,
   800→          healthChecked: false,
   801→          details: 'Hosting provisioning skipped (not requested).',
   802→        }
   803→
   804→  const emailStatus: ProvisioningResult['emailStatus'] = emailResult
   805→    ? {
   806→        status: 'success',
   807→        mailboxes: emailResult.mailboxes.map((m) => ({ address: m.address, type: m.type })),
   808→        dkimConfigured: !!emailResult.dkimPublicKey,
   809→        smtpEnabled: verifyResult?.emailVerified ?? true,
   810→        imapEnabled: verifyResult?.emailVerified ?? true,
   811→        webmailUrl: emailResult.webmailUrl,
   812→        details: `${emailResult.mailboxes.length} mailbox(es) created with DKIM signing enabled.`,
   813→      }
   814→    : emailFailed
   815→      ? {
   816→          status: 'fail',
   817→          mailboxes: [],
   818→          dkimConfigured: false,
   819→          smtpEnabled: false,
   820→          imapEnabled: false,
   821→          details: 'Email provisioning failed after retries.',
   822→        }
   823→      : {
   824→          status: 'skipped',
   825→          mailboxes: [],
   826→          dkimConfigured: false,
   827→          smtpEnabled: false,
   828→          imapEnabled: false,
   829→          details: 'Email provisioning skipped (not requested).',
   830→        }
   831→
   832→  // ─── Credentials ──────────────────────────────────────────────────
   833→  const credentials: ProvisioningResult['credentials'] = {
   834→    domain: {
   835→      registrar: domainResult?.provider === 'dynadot' ? 'Dynadot' : 'DeployHub',
   836→      managementUrl:
   837→        domainResult?.provider === 'dynadot'
   838→          ? 'https://dynadot.com/account/manage/domain.html'
   839→          : '/dashboard/domains',
   840→    },
   841→    dns: {
   842→      nameservers: ctx.nameservers.length
   843→        ? ctx.nameservers
   844→        : ['ns1.deployhub.io', 'ns2.deployhub.io'],
   845→      managementUrl: `/dashboard/domains/${ctx.domainId || ''}/dns`,
   846→    },
   847→  }
   848→
   849→  if (hostingResult) {
   850→    credentials.hosting = {
   851→      adminUrl: process.env.COOLIFY_API_URL || 'https://coolify.deployhub.io',
   852→      apiKey: isCoolifyDemoMode() ? undefined : process.env.COOLIFY_API_TOKEN,
   853→    }
   854→  }
   855→
   856→  if (emailResult) {
   857→    const info = ctx.mailServerInfo || {
   858→      imap: { host: ctx.mailServerHostname, port: 993, ssl: true },
   859→      smtp: { host: ctx.mailServerHostname, port: 587, tls: true },
   860→      webmail: emailResult.webmailUrl,
   861→    }
   862→    credentials.email = {
   863→      mailboxes: emailResult.mailboxes.map((m) => ({
   864→        address: m.address,
   865→        password: m.password, // plaintext — shown once
   866→        type: m.type,
   867→      })),
   868→      imap: info.imap,
   869→      smtp: info.smtp,
   870→      webmail: info.webmail,
   871→    }
   872→  }
   873→
   874→  return {
   875→    overallStatus,
   876→    domainStatus,
   877→    dnsStatus,
   878→    hostingStatus,
   879→    emailStatus,
   880→    credentials,
   881→    jobId,
   882→    duration,
   883→    errors: ctx.errors,
   884→  }
   885→}
   886→
   887→// ─── Step 1: provisionDomain ─────────────────────────────────────────────
   888→
   889→/**
   890→ * Step 1: Register the domain via Dynadot.
   891→ *
   892→ *  1. Search availability via Dynadot searchDomains().
   893→ *  2. If available (or in demo mode), registerDomain() via Dynadot.
   894→ *  3. Create a Domain record in DB with provider='dynadot', externalId,
   895→ *     status='active'.
   896→ */
   897→async function provisionDomain(
   898→  job: { id: string; userId: string; domainName: string },
   899→  step: { id: string }
   900→): Promise<{ domainId: string; externalId?: string; provider: string }> {
   901→  const domainName = job.domainName
   902→
   903→  // 1. Search availability.
   904→  const searchResults = await searchDomains([domainName])
   905→  const result = searchResults[0]
   906→  const available = result?.available === true
   907→
   908→  if (!available) {
   909→    // In demo mode, the search heuristic may falsely mark short domains as
   910→    // taken. Be forgiving: log a warning + proceed with registration anyway
   911→    // (Dynadot's registerDomain() will be the source of truth).
   912→    if (isDynadotDemoMode()) {
   913→      console.log(
   914→        `[orchestration] DEMO: domain ${domainName} reported unavailable by demo search, attempting registration anyway`
   915→      )
   916→    } else {
   917→      throw new Error(`Domain ${domainName} is not available for registration`)
   918→    }
   919→  }
   920→
   921→  // 2. Register the domain (1 year).
   922→  const registration = await registerDomain(domainName, 1)
   923→  if (!registration.success) {
   924→    throw new Error(
   925→      `Domain registration failed: ${registration.orderId ? 'order ' + registration.orderId : 'unknown error'}`
   926→    )
   927→  }
   928→
   929→  // 3. Fetch domain info for nameservers + expiry.
   930→  let info
   931→  try {
   932→    info = await getDomainInfo(domainName)
   933→  } catch (err) {
   934→    console.warn(
   935→      `[orchestration] getDomainInfo failed for ${domainName}:`,
   936→      err instanceof Error ? err.message : err
   937→    )
   938→    info = { nameservers: ['ns1.deployhub.io', 'ns2.deployhub.io'], expiry: undefined }
   939→  }
   940→
   941→  // 4. Create the Domain record in DB.
   942→  const domain = await db.domain.create({
   943→    data: {
   944→      userId: job.userId,
   945→      name: domainName,
   946→      status: 'active',
   947→      registrar: 'dynadot',
   948→      provider: 'dynadot',
   949→      externalId: registration.orderId,
   950→      registeredAt: new Date(),
   951→      expiresAt: info.expiry ? new Date(info.expiry) : undefined,
   952→      nameservers: JSON.stringify(info.nameservers || []),
   953→    },
   954→  })
   955→
   956→  console.log(
   957→    `[orchestration] step=domain_register domainId=${domain.id} externalId=${registration.orderId}`
   958→  )
   959→
   960→  return {
   961→    domainId: domain.id,
   962→    externalId: registration.orderId,
   963→    provider: 'dynadot',
   964→  }
   965→}
   966→
   967→// ─── Step 2: configureDns ────────────────────────────────────────────────
   968→
   969→/**
   970→ * Step 2: Configure DNS records via Dynadot's setDnsRecords() and persist
   971→ * each as a DnsRecord row.
   972→ *
   973→ * Records created:
   974→ *   - A     @            → hostingServerIp
   975→ *   - A     www          → hostingServerIp
   976→ *   - MX    @            → mailServerHostname (priority 10)
   977→ *   - TXT   @            → "v=spf1 mx a -all"  (SPF)
   978→ *   - TXT   mail._domainkey → DKIM public key (placeholder if Mailcow not yet run)
   979→ *   - TXT   _dmarc       → "v=DMARC1; p=quarantine; rua=mailto:admin@domain"
   980→ */
   981→async function configureDns(
   982→  job: { id: string; userId: string; domainName: string },
   983→  step: { id: string },
   984→  domainId: string,
   985→  mailServerHostname: string,
   986→  hostingServerIp: string
   987→): Promise<{
   988→  records: Array<{ type: string; name: string; value: string; recordId: string }>
   989→}> {
   990→  const domainName = job.domainName
   991→
   992→  // Build the full set of DNS records.
   993→  // NOTE: DKIM is generated by the email step which runs AFTER this one.
   994→  // We use a placeholder DKIM TXT record here; the email step will overwrite
   995→  // it with the real Mailcow-generated key when it runs (best-effort update).
   996→  const dkimPlaceholder = 'v=DKIM1; k=rsa; p=PLACEHOLDER_REPLACE_AFTER_EMAIL_PROVISIONING'
   997→
   998→  const recordsToCreate: Array<{
   999→    type: string
  1000→    name: string
  1001→    value: string
  1002→    ttl?: number
  1003→    priority?: number
  1004→  }> = [
  1005→    { type: 'A', name: '@', value: hostingServerIp, ttl: 3600 },
  1006→    { type: 'A', name: 'www', value: hostingServerIp, ttl: 3600 },
  1007→    { type: 'MX', name: '@', value: mailServerHostname, ttl: 3600, priority: 10 },
  1008→    { type: 'TXT', name: '@', value: 'v=spf1 mx a -all', ttl: 3600 },
  1009→    { type: 'TXT', name: 'mail._domainkey', value: dkimPlaceholder, ttl: 3600 },
  1010→    {
  1011→      type: 'TXT',
  1012→      name: '_dmarc',
  1013→      value: `v=DMARC1; p=quarantine; rua=mailto:admin@${domainName}`,
  1014→      ttl: 3600,
  1015→    },
  1016→  ]
  1017→
  1018→  // 1. Push to Dynadot (or demo).
  1019→  const dnsResult = await setDnsRecords(
  1020→    domainName,
  1021→    recordsToCreate.map((r) => ({
  1022→      type: r.type,
  1023→      name: r.name,
  1024→      value: r.value,
  1025→      ttl: r.ttl,
  1026→      priority: r.priority,
  1027→    }))
  1028→  )
  1029→  if (!dnsResult.success) {
  1030→    throw new Error(`Dynadot setDnsRecords failed for ${domainName}`)
  1031→  }
  1032→
  1033→  // 2. Persist each record to the DnsRecord table.
  1034→  const created: Array<{ type: string; name: string; value: string; recordId: string }> = []
  1035→  for (const r of recordsToCreate) {
  1036→    const rec = await db.dnsRecord.create({
  1037→      data: {
  1038→        domainId,
  1039→        type: r.type,
  1040→        name: r.name,
  1041→        value: r.value,
  1042→        ttl: r.ttl || 3600,
  1043→        priority: r.priority ?? null,
  1044→        proxied: false,
  1045→      },
  1046→    })
  1047→    created.push({
  1048→      type: r.type,
  1049→      name: r.name,
  1050→      value: r.value,
  1051→      recordId: rec.id,
  1052→    })
  1053→  }
  1054→
  1055→  console.log(
  1056→    `[orchestration] step=dns_configure domainId=${domainId} records=${created.length}`
  1057→  )
  1058→
  1059→  return { records: created }
  1060→}
  1061→
  1062→// ─── Step 3: provisionHosting ────────────────────────────────────────────
  1063→
  1064→/**
  1065→ * Step 3: Provision web hosting via Coolify.
  1066→ *
  1067→ *  1. Create a Project in DB with name=domainName, framework based on stack.
  1068→ *  2. Call Coolify createApplication() to create the app.
  1069→ *  3. Call Coolify deployApplication() to trigger an initial deploy.
  1070→ *  4. Create a Deployment record tracking this initial deploy.
  1071→ */
  1072→async function provisionHosting(
  1073→  job: { id: string; userId: string; domainName: string; stack: string },
  1074→  step: { id: string },
  1075→  domainId: string,
  1076→  domainName: string
  1077→): Promise<{ projectId: string; coolifyId: string; url: string }> {
  1078→  const framework = stackToFramework(job.stack)
  1079→  const slug = domainName
  1080→    .toLowerCase()
  1081→    .replace(/[^a-z0-9.-]+/g, '-')
  1082→    .replace(/^-|-$/g, '')
  1083→  const subdomain = `${slug.split('.')[0]}.deployhub.io`
  1084→
  1085→  // 1. Create the Project record.
  1086→  const project = await db.project.create({
  1087→    data: {
  1088→      userId: job.userId,
  1089→      name: domainName,
  1090→      slug,
  1091→      framework,
  1092→      stack: job.stack,
  1093→      status: 'creating',
  1094→      subdomain,
  1095→      customDomain: domainName,
  1096→      sslEnabled: true,
  1097→    },
  1098→  })
  1099→
  1100→  // 2. Link the Domain to the Project.
  1101→  await db.domain.update({
  1102→    where: { id: domainId },
  1103→    data: { projectId: project.id },
  1104→  })
  1105→
  1106→  // 3. Call Coolify createApplication().
  1107→  // In demo mode, serverId is synthetic — that's fine.
  1108→  const coolifyApp = await createApplication({
  1109→    name: domainName,
  1110→    framework,
  1111→    serverId: 'deployhub-default',
  1112→  })
  1113→
  1114→  // Persist the Coolify ID on the project.
  1115→  await db.project.update({
  1116→    where: { id: project.id },
  1117→    data: {
  1118→      coolifyId: coolifyApp.id,
  1119→      status: 'running',
  1120→    },
  1121→  })
  1122→
  1123→  // 4. Trigger an initial deploy.
  1124→  const deploy = await deployApplication(coolifyApp.id)
  1125→
  1126→  // 5. Create a Deployment record.
  1127→  await db.deployment.create({
  1128→    data: {
  1129→      projectId: project.id,
  1130→      userId: job.userId,
  1131→      branch: 'main',
  1132→      status: 'building',
  1133→      deployUrl: coolifyApp.url,
  1134→      triggeredBy: 'provisioning',
  1135→      coolifyDeploymentId: deploy.deploymentId,
  1136→    },
  1137→  })
  1138→
  1139→  console.log(
  1140→    `[orchestration] step=hosting_provision projectId=${project.id} coolifyId=${coolifyApp.id} url=${coolifyApp.url}`
  1141→  )
  1142→
  1143→  return {
  1144→    projectId: project.id,
  1145→    coolifyId: coolifyApp.id,
  1146→    url: coolifyApp.url,
  1147→  }
  1148→}
  1149→
  1150→// ─── Step 4: provisionEmail ──────────────────────────────────────────────
  1151→
  1152→/**
  1153→ * Step 4: Provision email hosting via Mailcow.
  1154→ *
  1155→ *  1. addDomain() in Mailcow.
  1156→ *  2. generateDkim() for the domain.
  1157→ *  3. Create 3 mailboxes: admin@, info@, support@ with random passwords.
  1158→ *     Each mailbox password is encrypted (AES-256-GCM) before DB storage.
  1159→ *  4. Update Domain.emailHosted=true, mailcowDomainId.
  1160→ *  5. Fetch mail server connection info (IMAP/SMTP/webmail).
  1161→ *
  1162→ * Returns the plaintext passwords alongside each mailbox so the caller can
  1163→ * include them in the one-time credentials payload.
  1164→ */
  1165→async function provisionEmail(
  1166→  job: { id: string; userId: string; domainName: string },
  1167→  step: { id: string },
  1168→  domainId: string,
  1169→  domainName: string
  1170→): Promise<{
  1171→  mailboxes: Array<{ address: string; password: string; type: string; mailboxId: string }>
  1172→  dkimPublicKey: string
  1173→  webmailUrl: string
  1174→}> {
  1175→  // 1. Add the domain to Mailcow.
  1176→  const addDomainResult = await addDomain({
  1177→    domain: domainName,
  1178→    description: `DeployHub provisioned domain for ${job.userId}`,
  1179→    mailboxes: 10,
  1180→    aliases: 20,
  1181→    quota: 1024,
  1182→  })
  1183→  if (!addDomainResult.success) {
  1184→    throw new Error(`Mailcow addDomain failed: ${addDomainResult.error}`)
  1185→  }
  1186→
  1187→  // 2. Generate DKIM.
  1188→  const dkimResult = await generateDkim(domainName, 'mail')
  1189→  if (!dkimResult.success || !dkimResult.publicKey) {
  1190→    throw new Error(`Mailcow generateDkim failed: ${dkimResult.error}`)
  1191→  }
  1192→
  1193→  // 3. Update the Domain with email hosting info.
  1194→  await db.domain.update({
  1195→    where: { id: domainId },
  1196→    data: {
  1197→      emailHosted: true,
  1198→      mailcowDomainId: addDomainResult.domainId || domainName,
  1199→    },
  1200→  })
  1201→
  1202→  // 4. Create 3 standard mailboxes: admin, info, support.
  1203→  const mailboxSpecs: Array<{ localPart: string; name: string; type: string; isDefault: boolean }> = [
  1204→    { localPart: 'admin', name: 'Administrator', type: 'admin', isDefault: true },
  1205→    { localPart: 'info', name: 'Info', type: 'info', isDefault: false },
  1206→    { localPart: 'support', name: 'Support', type: 'support', isDefault: false },
  1207→  ]
  1208→
  1209→  const mailboxes: Array<{
  1210→    address: string
  1211→    password: string
  1212→    type: string
  1213→    mailboxId: string
  1214→  }> = []
  1215→
  1216→  for (const spec of mailboxSpecs) {
  1217→    const password = generatePassword()
  1218→    const address = `${spec.localPart}@${domainName}`
  1219→
  1220→    const mbResult = await createMailbox({
  1221→      localPart: spec.localPart,
  1222→      domain: domainName,
  1223→      password,
  1224→      name: spec.name,
  1225→      quota: 1024,
  1226→    })
  1227→    if (!mbResult.success) {
  1228→      throw new Error(
  1229→        `Mailcow createMailbox failed for ${address}: ${mbResult.error}`
  1230→      )
  1231→    }
  1232→
  1233→    // Persist with encrypted password.
  1234→    await db.emailMailbox.create({
  1235→      data: {
  1236→        domainId,
  1237→        address,
  1238→        localPart: spec.localPart,
  1239→        password: encrypt(password),
  1240→        quota: 1073741824, // 1GB in bytes
  1241→        active: true,
  1242→        isDefault: spec.isDefault,
  1243→        mailcowId: mbResult.mailboxId,
  1244→        dkimEnabled: true,
  1245→      },
  1246→    })
  1247→
  1248→    mailboxes.push({
  1249→      address,
  1250→      password, // plaintext — kept in memory for the credentials payload
  1251→      type: spec.type,
  1252→      mailboxId: mbResult.mailboxId || address,
  1253→    })
  1254→  }
  1255→
  1256→  // 5. Fetch mail server connection info.
  1257→  const serverInfo = await getMailServerInfo(domainName)
  1258→
  1259→  console.log(
  1260→    `[orchestration] step=email_provision domainId=${domainId} mailboxes=${mailboxes.length} dkim=✓`
  1261→  )
  1262→
  1263→  return {
  1264→    mailboxes,
  1265→    dkimPublicKey: dkimResult.publicKey,
  1266→    webmailUrl: serverInfo.webmail,
  1267→  }
  1268→}
  1269→
  1270→// ─── Step 5: verifyProvisioning ──────────────────────────────────────────
  1271→
  1272→/**
  1273→ * Step 5: Verify that all provisioned services are reachable and properly
  1274→ * configured.
  1275→ *
  1276→ *  1. Hosting: poll Coolify getApplicationStatus() for the app.
  1277→ *  2. Email: call Mailcow verifyEmailSetup() for the domain.
  1278→ *  3. DNS: call dns.ts verifyTxtRecord() for the DKIM record (best-effort —
  1279→ *     DNS propagation can take minutes/hours so we don't fail the pipeline
  1280→ *     if it's not yet propagated).
  1281→ */
  1282→async function verifyProvisioning(
  1283→  job: { id: string; userId: string; domainName: string },
  1284→  step: { id: string },
  1285→  domainName: string,
  1286→  projectId: string
  1287→): Promise<{ hostingReachable: boolean; emailVerified: boolean; dnsPropagated: boolean }> {
  1288→  // 1. Hosting health check.
  1289→  let hostingReachable = false
  1290→  if (projectId) {
  1291→    try {
  1292→      const project = await db.project.findUnique({ where: { id: projectId } })
  1293→      if (project?.coolifyId) {
  1294→        const status = await getApplicationStatus(project.coolifyId)
  1295→        hostingReachable =
  1296→          status.status === 'running' ||
  1297→          status.status === 'healthy' ||
  1298→          status.status === 'ready'
  1299→      }
  1300→    } catch (err) {
  1301→      console.warn(
  1302→        `[orchestration] hosting health check failed:`,
  1303→        err instanceof Error ? err.message : err
  1304→      )
  1305→    }
  1306→  }
  1307→
  1308→  // 2. Email verification (Mailcow).
  1309→  let emailVerified = false
  1310→  try {
  1311→    const emailCheck = await verifyEmailSetup(domainName)
  1312→    emailVerified = emailCheck.success
  1313→  } catch (err) {
  1314→    console.warn(
  1315→      `[orchestration] email verification failed:`,
  1316→      err instanceof Error ? err.message : err
  1317→    )
  1318→  }
  1319→
  1320→  // 3. DNS propagation check (best-effort, never fails the pipeline).
  1321→  let dnsPropagated = false
  1322→  try {
  1323→    // Verify the SPF TXT record exists at the authoritative resolver. In demo
  1324→    // mode this will likely return false (the records aren't actually live),
  1325→    // so we treat it as informational.
  1326→    dnsPropagated = await verifyTxtRecord(domainName, 'v=spf1')
  1327→  } catch (err) {
  1328→    console.warn(
  1329→      `[orchestration] DNS propagation check failed:`,
  1330→      err instanceof Error ? err.message : err
  1331→    )
  1332→  }
  1333→
  1334→  console.log(
  1335→    `[orchestration] step=verify hosting=${hostingReachable} email=${emailVerified} dns=${dnsPropagated}`
  1336→  )
  1337→
  1338→  return { hostingReachable, emailVerified, dnsPropagated }
  1339→}

  1341→// ─── rollbackProvisioning ────────────────────────────────────────────────
  1342→
  1343→/**
  1344→ * Roll back all completed steps in REVERSE order. For each successful step:
  1345→ *
  1346→ *   email_provision      → Mailcow removeDomain() + delete EmailMailbox rows
  1347→ *   hosting_provision    → Coolify deleteApplication() + mark Project stopped
  1348→ *   dns_configure        → delete DnsRecord rows
  1349→ *   domain_register      → mark Domain status='cancelled' (can't unregister)
  1350→ *
  1351→ * Updates each step status to 'rolled_back' and sets job.status='rolled_back'
  1352→ * with rollbackStartedAt/rollbackCompletedAt timestamps.
  1353→ */
  1354→export async function rollbackProvisioning(
  1355→  jobId: string
  1356→): Promise<{
  1357→  rolledBack: string[]
  1358→  failed: Array<{ step: string; error: string }>
  1359→}> {
  1360→  const job = await db.provisioningJob.findUnique({
  1361→    where: { id: jobId },
  1362→    include: { steps: { orderBy: { stepOrder: 'desc' } } },
  1363→  })
  1364→  if (!job) {
  1365→    throw new Error(`ProvisioningJob not found: ${jobId}`)
  1366→  }
  1367→
  1368→  await db.provisioningJob.update({
  1369→    where: { id: jobId },
  1370→    data: { rollbackStartedAt: new Date(), status: 'rolled_back' },
  1371→  })
  1372→
  1373→  const rolledBack: string[] = []
  1374→  const failed: Array<{ step: string; error: string }> = []
  1375→
  1376→  for (const step of job.steps) {
  1377→    if (step.status !== 'success') continue
  1378→
  1379→    try {
  1380→      switch (step.stepName) {
  1381→        case 'email_provision': {
  1382→          // Delete mailboxes in Mailcow (best-effort), then remove the domain.
  1383→          const domain = await db.domain.findFirst({
  1384→            where: { name: job.domainName },
  1385→            include: { mailboxes: true },
  1386→          })
  1387→          if (domain) {
  1388→            for (const mb of domain.mailboxes) {
  1389→              try {
  1390→                await deleteMailbox(mb.address)
  1391→              } catch (err) {
  1392→                console.warn(
  1393→                  `[orchestration] rollback deleteMailbox ${mb.address} failed:`,
  1394→                  err instanceof Error ? err.message : err
  1395→                )
  1396→              }
  1397→            }
  1398→            try {
  1399→              await removeDomain(job.domainName)
  1400→            } catch (err) {
  1401→              console.warn(
  1402→                `[orchestration] rollback removeDomain ${job.domainName} failed:`,
  1403→                err instanceof Error ? err.message : err
  1404→              )
  1405→            }
  1406→            await db.emailMailbox.deleteMany({ where: { domainId: domain.id } })
  1407→            await db.domain.update({
  1408→              where: { id: domain.id },
  1409→              data: { emailHosted: false, mailcowDomainId: null },
  1410→            })
  1411→          }
  1412→          break
  1413→        }
  1414→        case 'hosting_provision': {
  1415→          const project = await db.project.findFirst({
  1416→            where: { userId: job.userId, name: job.domainName },
  1417→          })
  1418→          if (project?.coolifyId) {
  1419→            try {
  1420→              await deleteApplication(project.coolifyId)
  1421→            } catch (err) {
  1422→              console.warn(
  1423→                `[orchestration] rollback deleteApplication ${project.coolifyId} failed:`,
  1424→                err instanceof Error ? err.message : err
  1425→              )
  1426→            }
  1427→          }
  1428→          if (project) {
  1429→            await db.project.update({
  1430→              where: { id: project.id },
  1431→              data: { status: 'stopped' },
  1432→            })
  1433→          }
  1434→          break
  1435→        }
  1436→        case 'dns_configure': {
  1437→          const domain = await db.domain.findFirst({
  1438→            where: { name: job.domainName },
  1439→          })
  1440→          if (domain) {
  1441→            await db.dnsRecord.deleteMany({ where: { domainId: domain.id } })
  1442→          }
  1443→          break
  1444→        }
  1445→        case 'domain_register': {
  1446→          // Can't actually unregister a domain, but mark it cancelled in our DB.
  1447→          const domain = await db.domain.findFirst({
  1448→            where: { name: job.domainName },
  1449→          })
  1450→          if (domain) {
  1451→            await db.domain.update({
  1452→              where: { id: domain.id },
  1453→              data: { status: 'cancelled' },
  1454→            })
  1455→          }
  1456→          break
  1457→        }
  1458→        case 'verify': {
  1459→          // Nothing to roll back for the verify step.
  1460→          break
  1461→        }
  1462→        default:
  1463→          break
  1464→      }
  1465→
  1466→      await db.provisioningStep.update({
  1467→        where: { id: step.id },
  1468→        data: { status: 'rolled_back' },
  1469→      })
  1470→      rolledBack.push(step.stepName)
  1471→    } catch (err) {
  1472→      const msg = err instanceof Error ? err.message : String(err)
  1473→      failed.push({ step: step.stepName, error: msg })
  1474→      console.error(
  1475→        `[orchestration] rollback of step ${step.stepName} failed: ${msg}`
  1476→      )
  1477→    }
  1478→  }
  1479→
  1480→  await db.provisioningJob.update({
  1481→    where: { id: jobId },
  1482→    data: {
  1483→      rollbackCompletedAt: new Date(),
  1484→      rollbackSteps: JSON.stringify(rolledBack),
  1485→      status: 'rolled_back',
  1486→      currentStep: 'rolled_back',
  1487→    },
  1488→  })
  1489→
  1490→  await logAudit({
  1491→    userId: job.userId,
  1492→    action: 'provisioning.rollback',
  1493→    resource: 'provisioning_job',
  1494→    resourceId: jobId,
  1495→    metadata: { rolledBack, failed },
  1496→    status: failed.length > 0 ? 'failure' : 'success',
  1497→  })
  1498→
  1499→  console.log(
  1500→    `[orchestration] rollback complete for job=${jobId} rolledBack=${rolledBack.join(',')} failed=${failed.length}`
  1501→  )
  1502→
  1503→  return { rolledBack, failed }
  1504→}

  1506→// ─── getProvisioningStatus ───────────────────────────────────────────────

  1508→/**
  1509→ * Fetch the current status of a provisioning job: the job record, all steps,
  1510→ * and the parsed ProvisioningResult if the pipeline has completed.
  1511→ */
  1512→export async function getProvisioningStatus(jobId: string): Promise<{
  1513→  job: Record<string, unknown> | null
  1514→  steps: Array<Record<string, unknown>>
  1515→  result?: ProvisioningResult
  1516→}> {
  1517→  const job = await db.provisioningJob.findUnique({
  1518→    where: { id: jobId },
  1519→    include: { steps: { orderBy: { stepOrder: 'asc' } } },
  1520→  })
  1521→
  1522→  if (!job) {
  1523→    return { job: null, steps: [] }
  1524→  }
  1525→
  1526→  let result: ProvisioningResult | undefined
  1527→  if (job.result) {
  1528→    try {
  1529→      result = JSON.parse(job.result) as ProvisioningResult
  1530→    } catch {
  1531→      // ignore parse errors
  1532→    }
  1533→  }
  1534→
  1535→  // Strip the heavy `input`/`output` JSON blobs from the steps for the
  1536→  // response — callers can fetch a single step's detail separately if needed.
  1537→  const steps = job.steps.map((s) => ({
  1538→    id: s.id,
  1539→    stepName: s.stepName,
  1540→    stepOrder: s.stepOrder,
  1541→    status: s.status,
  1542→    attempts: s.attempts,
  1543→    maxAttempts: s.maxAttempts,
  1544→    error: s.error,
  1545→    startedAt: s.startedAt,
  1546→    completedAt: s.completedAt,
  1547→    duration: s.duration,
  1548→    lastAttemptAt: s.lastAttemptAt,
  1549→  }))
  1550→
  1551→  // Strip the huge `result` blob from the job — it's already in `result`.
  1552→  const { result: _r, steps: _s, ...jobClean } = job
  1553→  void _r
  1554→  void _s
  1555→
  1556→  return {
  1557→    job: jobClean as Record<string, unknown>,
  1558→    steps: steps as Array<Record<string, unknown>>,
  1559→    result,
  1560→  }
  1561→}
