   140→function demoIsAvailable(domain: string): boolean {
   141→  const lower = domain.toLowerCase()
   142→  if (lower.includes('taken') || lower.includes('unavailable')) return false
   143→  if (lower.includes('premium')) return true
   144→  // Deterministic pseudo-randomness based on string hash
   145→  const tld = '.' + lower.split('.').slice(1).join('.')
   146→  if (!DEMO_TLD_PRICES[tld]) return false
   147→  let hash = 0
   148→  for (let i = 0; i < lower.length; i++) {
   149→    hash = (hash << 5) - hash + lower.charCodeAt(i)
   150→    hash |= 0
   151→  }
   152→  // ~60% available
   153→  return Math.abs(hash) % 10 < 6
   154→}
   155→
   156→function demoIsPremium(domain: string): boolean {
   157→  const lower = domain.toLowerCase()
   158→  if (lower.includes('premium')) return true
   159→  // Short single-word .com / .ai treated as premium
   160→  const base = lower.split('.')[0]
   161→  return (base.length <= 4 && (lower.endsWith('.com') || lower.endsWith('.ai')))
   162→}

   166→class DynadotService {
   167→  private apiKey: string | null
   168→
   169→  constructor() {
   170→    this.apiKey = process.env.DYNADOT_API_KEY || null
   171→  }
   172→

   174→    return !this.apiKey

   177→  /**
   178→   * Low-level call to Dynadot with retry + backoff + rate-limit handling.
   179→   */
   180→  private async call(
   181→    params: Record<string, string>,
   182→    attempt = 0
   183→  ): Promise<Record<string, unknown>> {
   184→    if (!this.apiKey) {
   185→      throw new Error('DYNADOT_API_KEY not configured')
   186→    }
   187→    const url = new URL(API_BASE)
   188→    url.searchParams.set('key', this.apiKey)
   189→    for (const [k, v] of Object.entries(params)) {
   190→      url.searchParams.set(k, v)
   191→    }
   192→
   193→    try {
   194→      const res = await fetch(url.toString(), {
   195→        method: 'GET',
   196→        headers: { Accept: 'application/json' },
   197→      })
   198→
   199→      // Rate limited
   200→      if (res.status === 429) {
   201→        const retryAfter = parseInt(res.headers.get('retry-after') || '2', 10)
   202→        if (attempt < MAX_RETRIES) {
   203→          await this.sleep(retryAfter * 1000)
   204→          return this.call(params, attempt + 1)
   205→        }
   206→        throw new Error('Dynadot rate limit exceeded')
   207→      }
   208→
   209→      // 5xx → retry
   210→      if (res.status >= 500 && attempt < MAX_RETRIES) {
   211→        await this.sleep(BASE_DELAY_MS * Math.pow(2, attempt))
   212→        return this.call(params, attempt + 1)
   213→      }
   214→
   215→      if (!res.ok) {
   216→        throw new Error(`Dynadot API HTTP ${res.status}: ${await res.text()}`)
   217→      }
   218→
   219→      const json = (await res.json()) as Record<string, unknown>
   220→
   221→      // Dynadot returns HTTP 200 but embeds errors in the body.
   222→      // Errors can appear in ANY top-level *Response envelope, not just "Response"
   223→      // (e.g. domain_info → DomainInfoResponse, get_ns → GetNsResponse).
   224→      // Detect IP-authorization errors specifically — these are not
   225→      // retryable (retrying just hits another unauthorized IP).
   226→      const envelopes = [
   227→        json.Response,
   228→        json.DomainInfoResponse,
   229→        json.GetNsResponse,
   230→        json.SetNsResponse,
   231→        json.SearchResponse,
   232→      ]
   233→        .filter(Boolean) as Array<Record<string, unknown>>
   234→      for (const env of envelopes) {
   235→        if (env.ResponseCode === '-1' || env.ResponseCode === -1 || env.Status === 'error') {
   236→          const errMsg = String(env.Error || 'Unknown Dynadot error')
   237→          if (errMsg.includes('unauthorized ip address')) {
   238→            // Non-retryable — throw immediately, bypass the catch retry
   239→            const ipErr = new Error(`Dynadot IP not authorized: ${errMsg}`)
   240→            ;(ipErr as Error & { noRetry?: boolean }).noRetry = true
   241→            throw ipErr
   242→          }
   243→          // Search errors with "invalid characters" are also non-retryable
   244→          if (errMsg.includes('invalid characters')) {
   245→            const charErr = new Error(`Dynadot API error: ${errMsg}`)
   246→            ;(charErr as Error & { noRetry?: boolean }).noRetry = true
   247→            throw charErr
   248→          }
   249→          throw new Error(`Dynadot API error: ${errMsg}`)
   250→        }
   251→      }
   252→
   253→      return json
   254→    } catch (err) {
   255→      // Network failure → retry (but never retry IP-auth or non-retryable errors)
   256→      const noRetry = err instanceof Error && (err as Error & { noRetry?: boolean }).noRetry
   257→      if (noRetry) throw err
   258→      if (attempt < MAX_RETRIES && !(err instanceof Error && err.message.includes('rate limit'))) {
   259→        await this.sleep(BASE_DELAY_MS * Math.pow(2, attempt))
   260→        return this.call(params, attempt + 1)
   261→      }
   262→      throw err
   263→    }
   264→  }
   265→
   266→  private sleep(ms: number): Promise<void> {
   267→    return new Promise((r) => setTimeout(r, ms))
   268→  }
   269→
   270→  private tldOf(domain: string): string {
   271→    const parts = domain.toLowerCase().split('.')
   272→    if (parts.length >= 3 && ['co', 'org', 'com'].includes(parts[parts.length - 2])) {
   273→      return '.' + parts.slice(-2).join('.')
   274→    }
   275→    return '.' + parts[parts.length - 1]
   276→  }

   288→  /**
   289→   * Bulk search across many domains at once.
   290→   */
   291→  async bulkSearch(domains: string[]): Promise<DomainSearchResult[]> {
   292→    const cleaned = domains.map((d) => d.toLowerCase().trim()).filter(Boolean)
   293→
   294→    if (this.demoMode) {
   295→      return cleaned.map((domain) => {
   296→        const tld = this.tldOf(domain)
   297→        const prices = DEMO_TLD_PRICES[tld]
   298→        const available = prices ? demoIsAvailable(domain) : false
   299→        const premium = demoIsPremium(domain)
   300→        return {
   301→          domain,
   302→          available,
   303→          premium,
   304→          registrationPrice: available ? prices?.reg : undefined,
   305→          renewalPrice: available ? prices?.renew : undefined,
   306→          transferPrice: available ? prices?.transfer : undefined,
   307→          currency: 'USD',
   308→        }
   309→      })
   310→    }
   311→
   312→    // Dynadot's search API requires separate domain0, domain1, domain2... params.
   313→    // A single comma-separated domain0 value returns "invalid characters in domain".
   314→    const searchParams: Record<string, string> = { command: 'search' }
   315→    cleaned.forEach((d, i) => {
   316→      searchParams[`domain${i}`] = d
   317→    })
   318→    const json = await this.call(searchParams)
   319→    const out: DomainSearchResult[] = []
   320→
   321→    // Dynadot search API returns:
   322→    //   { SearchResponse: { ResponseCode: "0", SearchResults: [{ DomainName, Status, Available: "yes"|"no", Price?, Premium? }] } }
   323→    // (Older docs described a Response.domain0 format, so we support both.)
   324→    const searchResp = json.SearchResponse as
   325→      | { SearchResults?: Array<Record<string, unknown>> }
   326→      | undefined
   327→
   328→    if (searchResp?.SearchResults && Array.isArray(searchResp.SearchResults)) {
   329→      for (const item of searchResp.SearchResults) {
   330→        const name = String(item.DomainName || '')
   331→        // Dynadot returns Available as "yes"/"no" (string)
   332→        const available = String(item.Available || '').toLowerCase() === 'yes'
   333→        const premium = String(item.Premium || '').toLowerCase() === 'yes'
   334→        const priceStr = item.Price ? String(item.Price) : undefined
   335→        out.push({
   336→          domain: name,
   337→          available,
   338→          premium,
   339→          registrationPrice: priceStr ? parseFloat(priceStr) : undefined,
   340→          currency: 'USD',
   341→        })
   342→      }
   343→    } else {
   344→      // Fallback: old Dynadot format { Response: { domain0: {...}, domain1: {...} } }
   345→      const response = (json.Response as Record<string, unknown>) || {}
   346→      for (const [key, value] of Object.entries(response)) {
   347→        if (key.startsWith('domain') && typeof value === 'object' && value !== null) {
   348→          const dv = value as { Name?: string; Available?: string; Price?: string; Premium?: string }
   349→          const name = dv.Name || ''
   350→          out.push({
   351→            domain: name,
   352→            available: String(dv.Available).toLowerCase() === 'yes',
   353→            premium: String(dv.Premium).toLowerCase() === 'yes',
   354→            registrationPrice: dv.Price ? parseFloat(dv.Price) : undefined,
   355→            currency: 'USD',
   356→          })
   357→        }
   358→      }
   359→    }
   360→
   361→    // Ensure all requested domains have an entry
   362→    return cleaned.map((d) => {
   363→      const found = out.find((r) => r.domain.toLowerCase() === d)
   364→      return (
   365→        found || {
   366→          domain: d,
   367→          available: false,
   368→          premium: false,
   369→        }
   370→      )
   371→    })
   372→  }

   377→  async suggestTlds(baseName: string, tlds?: string[]): Promise<DomainSearchResult[]> {

   479→  async getDomainInfo(domain: string): Promise<DomainInfo> {
   480→    if (this.demoMode) {
   481→      const expiry = new Date()
   482→      expiry.setFullYear(expiry.getFullYear() + 1)
   483→      return {
   484→        domain,
   485→        status: 'active',
   486→        expiry: expiry.toISOString(),
   487→        registeredAt: new Date(Date.now() - 30 * 24 * 60 * 60 * 1000).toISOString(),
   488→        nameservers: ['ns1.deployhub.io', 'ns2.deployhub.io'],
   489→        locked: true,
   490→        whoisPrivacy: true,
   491→        autoRenew: true,
   492→        premium: demoIsPremium(domain),
   493→      }
   494→    }
   495→
   496→    const json = await this.call({ command: 'domain_info', domain })
   497→    const resp = (json.DomainInfoResponse as Record<string, unknown>) || {}
   498→    const info = (resp.DomainInfo || {}) as Record<string, unknown>
   499→
   500→    // Expiration / Registration are Unix timestamps in MILLISECONDS
   501→    const expiryRaw = info.Expiration ? String(info.Expiration) : undefined
   502→    const expiry = expiryRaw && !isNaN(Number(expiryRaw))
   503→      ? new Date(Number(expiryRaw)).toISOString()
   504→      : undefined
   505→    const regRaw = info.Registration ? String(info.Registration) : undefined
   506→    const registeredAt = regRaw && !isNaN(Number(regRaw))
   507→      ? new Date(Number(regRaw)).toISOString()
   508→      : undefined
   509→
   510→    // domain_info returns NameServerSettings (a mode object), NOT hostnames.
   511→    // Fetch the real hostnames via the separate get_ns command.
   512→    let nameservers: string[] = []
   513→    try {
   514→      const nsJson = await this.call({ command: 'get_ns', domain })
   515→      const nsResp = (nsJson.GetNsResponse as Record<string, unknown>) || {}
   516→      const nsHosts = nsResp.NameServer || nsResp.Ns
   517→      if (Array.isArray(nsHosts)) {
   518→        nameservers = nsHosts.map((s) => String(s).trim()).filter(Boolean)
   519→      } else if (nsHosts) {
   520→        nameservers = String(nsHosts).split(',').map((s) => s.trim()).filter(Boolean)
   521→      }
   522→    } catch {
   523→      // best-effort — other fields are still useful
   524→    }
   525→
   526→    const renewOption = String(info.RenewOption || '').toLowerCase()
   527→    return {
   528→      domain,
   529→      status: String(info.Status || 'unknown'),
   530→      expiry,
   531→      registeredAt,
   532→      nameservers,
   533→      locked: String(info.Locked || 'no').toLowerCase() === 'yes',
   534→      whoisPrivacy: String(info.Privacy || 'none').toLowerCase() !== 'none',
   535→      autoRenew: renewOption === 'auto' || renewOption === 'renew',
   536→      premium: String(info.isForSale || 'no').toLowerCase() === 'yes',
   537→    }
   538→  }

   698→
   699→  async getExpiration(domain: string): Promise<{ expiry?: string; daysUntilExpiry?: number }> {
   700→    const info = await this.getDomainInfo(domain)
   701→    if (!info.expiry) return {}
   702→    const expiryDate = new Date(info.expiry)
   703→    const days = Math.ceil((expiryDate.getTime() - Date.now()) / (24 * 60 * 60 * 1000))
   704→    return { expiry: info.expiry, daysUntilExpiry: days }
   705→  }
