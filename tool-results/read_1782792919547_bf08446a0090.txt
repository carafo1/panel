   117→function statusColor(status: string): string {
   118→  switch (status) {
   119→    case 'active': return 'bg-indigo-500/15 text-indigo-600 dark:text-indigo-400 border-indigo-500/30'
   120→    case 'pending': case 'pending_verification': return 'bg-amber-500/15 text-amber-600 dark:text-amber-400 border-amber-500/30'
   121→    case 'transferring': return 'bg-indigo-500/15 text-indigo-600 dark:text-indigo-400 border-indigo-500/30'
   122→    case 'expired': case 'suspended': case 'deleted': return 'bg-red-500/15 text-red-600 dark:text-red-400 border-red-500/30'
   123→    default: return 'bg-gray-500/15 text-gray-600 dark:text-gray-400 border-gray-500/30'
   124→  }
   125→}
   126→
   127→function daysUntil(dateStr?: string): number | null {
   128→  if (!dateStr) return null
   129→  const diff = new Date(dateStr).getTime() - Date.now()
   130→  return Math.ceil(diff / (24 * 60 * 60 * 1000))
   131→}

   242→function SearchTab({ onRegistered }: { onRegistered: () => void }) {
   243→  const [query, setQuery] = useState('')
   244→  const [loading, setLoading] = useState(false)
   245→  const [result, setResult] = useState<SearchResult | null>(null)
   246→  const [bulkMode, setBulkMode] = useState(false)
   247→  const [bulkQuery, setBulkQuery] = useState('')
   248→  const [bulkResults, setBulkResults] = useState<SearchResult[]>([])
   249→  const [registering, setRegistering] = useState<string | null>(null)
   250→  const [registerDialog, setRegisterDialog] = useState<{ domain: string; price: number } | null>(null)
   251→
   252→  const handleSearch = useCallback(async () => {
   253→    if (!query.trim()) return
   254→    setLoading(true)
   255→    setResult(null)
   256→    try {
   257→      const data = await apiFetch(`/domains/search?domain=${encodeURIComponent(query.trim())}`)
   258→      setResult(data.data)
   259→    } catch (err) {
   260→      toast.error('Search failed', { description: (err as Error).message })
   261→    } finally {
   262→      setLoading(false)
   263→    }
   264→  }, [query])
   265→
   266→  const handleBulkSearch = useCallback(async () => {
   267→    const domains = bulkQuery.split('\n').map((d) => d.trim()).filter(Boolean)
   268→    if (domains.length === 0) return
   269→    setLoading(true)
   270→    try {
   271→      const data = await apiFetch('/domains/search', {
   272→        method: 'POST',
   273→        body: JSON.stringify({ domains }),
   274→      })
   275→      setBulkResults(data.data || [])
   276→    } catch (err) {
   277→      toast.error('Bulk search failed', { description: (err as Error).message })
   278→    } finally {
   279→      setLoading(false)
   280→    }
   281→  }, [bulkQuery])
   282→
   283→  const handleRegister = async (domain: string, duration: number) => {
   284→    setRegistering(domain)
   285→    try {
   286→      await apiFetch('/domains/register', {
   287→        method: 'POST',
   288→        body: JSON.stringify({ name: domain, duration, provider: 'dynadot' }),
   289→      })
   290→      toast.success('Domain registered!', {
   291→        description: `${domain} is now active. Invoice generated.`,
   292→      })
   293→      setRegisterDialog(null)
   294→      onRegistered()
   295→    } catch (err) {
   296→      toast.error('Registration failed', { description: (err as Error).message })
   297→    } finally {
   298→      setRegistering(null)
   299→    }
   300→  }
   301→
   302→  return (
   303→    <div className="space-y-6">
   304→      <Card>
   305→        <CardHeader>
   306→          <div className="flex items-center justify-between">
   307→            <div>
   308→              <CardTitle className="flex items-center gap-2">
   309→                <Search className="h-5 w-5 text-indigo-500" />
   310→                Domain Search
   311→              </CardTitle>
   312→              <CardDescription>Instant availability with registration, renewal &amp; transfer pricing</CardDescription>
   313→            </div>
   314→            <Button
   315→              variant={bulkMode ? 'default' : 'outline'}
   316→              size="sm"
   317→              onClick={() => { setBulkMode(!bulkMode); setResult(null); setBulkResults([]) }}
   318→            >
   319→              {bulkMode ? 'Single Search' : 'Bulk Search'}
   320→            </Button>
   321→          </div>
   322→        </CardHeader>
   323→        <CardContent className="space-y-4">
   324→          {!bulkMode ? (
   325→            <>
   326→              <div className="flex gap-2">
   327→                <Input
   328→                  placeholder="example.com"
   329→                  value={query}
   330→                  onChange={(e) => setQuery(e.target.value)}
   331→                  onKeyDown={(e) => e.key === 'Enter' && handleSearch()}
   332→                  className="text-base"
   333→                />
   334→                <Button onClick={handleSearch} disabled={loading || !query.trim()}>
   335→                  {loading ? <Loader2 className="h-4 w-4 animate-spin" /> : <Search className="h-4 w-4" />}
   336→                  Search
   337→                </Button>
   338→              </div>
   339→
   340→              {loading && <SearchSkeleton />}
   341→
   342→              {result && (
   343→                <SearchResultCard
   344→                  result={result}
   345→                  onRegister={(domain, price) => setRegisterDialog({ domain, price })}
   346→                  registering={registering}
   347→                />
   348→              )}
   349→            </>
   350→          ) : (
   351→            <>
   352→              <div className="space-y-2">
   353→                <Label>Enter one domain per line (max 50)</Label>
   354→                <Textarea
   355→                  placeholder={'example.com\nmybrand.net\nawesome.io'}
   356→                  value={bulkQuery}
   357→                  onChange={(e) => setBulkQuery(e.target.value)}
   358→                  rows={6}
   359→                  className="font-mono text-sm"
   360→                />
   361→                <Button onClick={handleBulkSearch} disabled={loading || !bulkQuery.trim()}>
   362→                  {loading ? <Loader2 className="h-4 w-4 animate-spin" /> : <Search className="h-4 w-4" />}
   363→                  Search {bulkQuery.split('\n').filter(Boolean).length} Domains
   364→                </Button>
   365→              </div>
   366→
   367→              {bulkResults.length > 0 && (
   368→                <div className="rounded-md border">
   369→                  <Table>
   370→                    <TableHeader>
   371→                      <TableRow>
   372→                        <TableHead>Domain</TableHead>
   373→                        <TableHead>Status</TableHead>
   374→                        <TableHead className="text-right">Register</TableHead>
   375→                        <TableHead className="text-right">Renew</TableHead>
   376→                        <TableHead className="text-right">Transfer</TableHead>
   377→                        <TableHead></TableHead>
   378→                      </TableRow>
   379→                    </TableHeader>
   380→                    <TableBody>
   381→                      {bulkResults.map((r) => (
   382→                        <TableRow key={r.domain}>
   383→                          <TableCell className="font-medium flex items-center gap-2">
   384→                            {r.premium && <Sparkles className="h-3 w-3 text-amber-500" />}
   385→                            {r.domain}
   386→                          </TableCell>
   387→                          <TableCell>
   388→                            {r.available ? (
   389→                              <Badge className="bg-indigo-500/15 text-indigo-600 border-indigo-500/30">Available</Badge>
   390→                            ) : (
   391→                              <Badge variant="secondary">Taken</Badge>
   392→                            )}
   393→                          </TableCell>
   394→                          <TableCell className="text-right font-mono">
   395→                            {r.registrationPrice ? `$${r.registrationPrice}` : '—'}
   396→                          </TableCell>
   397→                          <TableCell className="text-right font-mono">
   398→                            {r.renewalPrice ? `$${r.renewalPrice}` : '—'}
   399→                          </TableCell>
   400→                          <TableCell className="text-right font-mono">
   401→                            {r.transferPrice ? `$${r.transferPrice}` : '—'}
   402→                          </TableCell>
   403→                          <TableCell>
   404→                            {r.available && r.registrationPrice && (
   405→                              <Button
   406→                                size="sm"
   407→                                variant="outline"
   408→                                onClick={() => setRegisterDialog({ domain: r.domain, price: r.registrationPrice! })}
   409→                              >
   410→                                <ShoppingBag className="h-3 w-3 mr-1" /> Buy
   411→                              </Button>
   412→                            )}
   413→                          </TableCell>
   414→                        </TableRow>
   415→                      ))}
   416→                    </TableBody>
   417→                  </Table>
   418→                </div>
   419→              )}
   420→            </>
   421→          )}
   422→        </CardContent>
   423→      </Card>
   424→
   425→      {registerDialog && (
   426→        <RegisterDialog
   427→          domain={registerDialog.domain}
   428→          price={registerDialog.price}
   429→          loading={registering === registerDialog.domain}
   430→          onClose={() => setRegisterDialog(null)}
   431→          onConfirm={(duration) => handleRegister(registerDialog.domain, duration)}
   432→        />
   433→      )}
   434→    </div>
   435→  )
   436→}

  1034→function NameserversSection({ domain, onUpdated }: { domain: Domain; onUpdated: () => void }) {
  1035→  const [nameservers, setNameservers] = useState<string[]>(['', ''])
  1036→  const [saving, setSaving] = useState(false)
  1037→
  1038→  useEffect(() => {
  1039→    try {
  1040→      const ns = domain.nameservers ? JSON.parse(domain.nameservers) : []
  1041→      setNameservers(ns.length >= 2 ? ns : [...ns, '', ''].slice(0, 2))
  1042→    } catch {
  1043→      setNameservers(['', ''])
  1044→    }
  1045→  }, [domain.nameservers])
  1046→
  1047→  const update = (i: number, val: string) => {
  1048→    const next = [...nameservers]
  1049→    next[i] = val
  1050→    setNameservers(next)
  1051→  }
  1052→
  1053→  const add = () => setNameservers([...nameservers, ''])
  1054→  const remove = (i: number) => nameservers.length > 2 && setNameservers(nameservers.filter((_, idx) => idx !== i))
  1055→
  1056→  const save = async () => {
  1057→    const clean = nameservers.map((n) => n.trim()).filter(Boolean)
  1058→    if (clean.length < 2) { toast.error('At least 2 nameservers required'); return }
  1059→    setSaving(true)
  1060→    try {
  1061→      await apiFetch(`/domains/${domain.id}/nameservers`, {
  1062→        method: 'PUT',
  1063→        body: JSON.stringify({ nameservers: clean }),
  1064→      })
  1065→      toast.success('Nameservers updated', {
  1066→        description: `${clean.length} nameservers configured via Apsamo`,
  1067→      })
  1068→      onUpdated()
  1069→    } catch (err) {
  1070→      toast.error('Failed to update nameservers', { description: (err as Error).message })
  1071→    } finally {
  1072→      setSaving(false)
  1073→    }
  1074→  }
  1075→
  1076→  return (
