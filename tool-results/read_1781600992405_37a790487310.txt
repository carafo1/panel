   140→function frameworkLabel(fw: string) {
   141→  switch (fw) {
   142→    case 'nextjs':
   143→      return 'Next.js'
   144→    case 'react':
   145→      return 'React'
   146→    case 'node':
   147→      return 'Node.js'
   148→    case 'static':
   149→      return 'Static'
   150→    default:
   151→      return fw
   152→  }
   153→}

   189→function timeAgo(dateStr: string): string {
   190→  const seconds = Math.floor((Date.now() - new Date(dateStr).getTime()) / 1000)
   191→  if (seconds < 60) return 'just now'
   192→  const minutes = Math.floor(seconds / 60)
   193→  if (minutes < 60) return `${minutes}m ago`
   194→  const hours = Math.floor(minutes / 60)
   195→  if (hours < 24) return `${hours}h ago`
   196→  const days = Math.floor(hours / 24)
   197→  return `${days}d ago`
   198→}
   199→
   200→function formatDate(dateStr: string) {
   201→  try {
   202→    return new Date(dateStr).toLocaleDateString('en-US', {
   203→      month: 'short',
   204→      day: 'numeric',
   205→      year: 'numeric',
   206→      hour: '2-digit',
   207→      minute: '2-digit',
   208→    })
   209→  } catch {
   210→    return dateStr
   211→  }
   212→}

   268→function ConnectRepositoryDialog({
   269→  open,
   270→  onClose,
   271→  projects,
   272→  onConnected,
   273→}: {
   274→  open: boolean
   275→  onClose: () => void
   276→  projects: ProjectWithIntegration[]
   277→  onConnected: () => void
   278→}) {
   279→  const [projectId, setProjectId] = useState('')
   280→  const [repoUrl, setRepoUrl] = useState('')
   281→  const [branch, setBranch] = useState('main')
   282→  const [connecting, setConnecting] = useState(false)
   283→
   284→  // Filter out projects that already have GitHub integration
   285→  const availableProjects = projects.filter((p) => !p.githubIntegration)
   286→
   287→  useEffect(() => {
   288→    if (open) {
   289→      setProjectId('')
   290→      setRepoUrl('')
   291→      setBranch('main')
   292→    }
   293→  }, [open])
   294→
   295→  const handleConnect = async () => {
   296→    if (!projectId) {
   297→      toast.error('Please select a project')
   298→      return
   299→    }
   300→    if (!repoUrl.trim()) {
   301→      toast.error('Repository URL is required')
   302→      return
   303→    }
   304→
   305→    setConnecting(true)
   306→    try {
   307→      await apiFetch('/github/connect', {
   308→        method: 'POST',
   309→        body: JSON.stringify({
   310→          projectId,
   311→          repoUrl: repoUrl.trim(),
   312→          branch: branch || 'main',
   313→        }),
   314→      })
   315→      toast.success('Repository connected successfully!')
   316→      onConnected()
   317→      onClose()
   318→    } catch (err) {
   319→      toast.error(err instanceof Error ? err.message : 'Failed to connect repository')
   320→    } finally {
   321→      setConnecting(false)
   322→    }
   323→  }
   324→
   325→  return (
   326→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   327→      <DialogContent className="sm:max-w-lg">
   328→        <DialogHeader>
   329→          <DialogTitle className="flex items-center gap-2">
   330→            <Github className="size-5 text-emerald-600" />
   331→            Connect Repository
   332→          </DialogTitle>
   333→          <DialogDescription>
   334→            Link a GitHub repository to your project for automatic deployments.
   335→          </DialogDescription>
   336→        </DialogHeader>
   337→
   338→        <div className="grid gap-4 py-4">
   339→          {/* Project Selector */}
   340→          <div className="grid gap-2">
   341→            <Label>Project</Label>
   342→            {availableProjects.length === 0 ? (
   343→              <p className="text-sm text-muted-foreground py-2">
   344→                All projects already have a GitHub integration.
   345→              </p>
   346→            ) : (
   347→              <Select value={projectId} onValueChange={setProjectId}>
   348→                <SelectTrigger className="w-full">
   349→                  <SelectValue placeholder="Select a project" />
   350→                </SelectTrigger>
   351→                <SelectContent>
   352→                  {availableProjects.map((project) => (
   353→                    <SelectItem key={project.id} value={project.id}>
   354→                      <span className="flex items-center gap-2">
   355→                        {project.name}
   356→                        <Badge variant="outline" className={`text-[10px] ${frameworkColor(project.framework)}`}>
   357→                          {frameworkLabel(project.framework)}
   358→                        </Badge>
   359→                      </span>
   360→                    </SelectItem>
   361→                  ))}
   362→                </SelectContent>
   363→              </Select>
   364→            )}
   365→          </div>
   366→
   367→          {/* Repository URL */}
   368→          <div className="grid gap-2">
   369→            <Label htmlFor="repo-url">GitHub Repository URL</Label>
   370→            <Input
   371→              id="repo-url"
   372→              placeholder="https://github.com/owner/repo"
   373→              value={repoUrl}
   374→              onChange={(e) => setRepoUrl(e.target.value)}
   375→              className="font-mono text-sm"
   376→            />
   377→            <p className="text-xs text-muted-foreground">
   378→              Supports https://github.com/owner/repo or owner/repo format
   379→            </p>
   380→          </div>
   381→
   382→          {/* Branch Selector */}
   383→          <div className="grid gap-2">
   384→            <Label>Branch</Label>
   385→            <Select value={branch} onValueChange={setBranch}>
   386→              <SelectTrigger className="w-full">
   387→                <SelectValue placeholder="Select branch" />
   388→              </SelectTrigger>
   389→              <SelectContent>
   390→                <SelectItem value="main">main</SelectItem>
   391→                <SelectItem value="master">master</SelectItem>
   392→                <SelectItem value="develop">develop</SelectItem>
   393→                <SelectItem value="staging">staging</SelectItem>
   394→                <SelectItem value="production">production</SelectItem>
   395→              </SelectContent>
   396→            </Select>
   397→          </div>
   398→        </div>
   399→
   400→        <DialogFooter>
   401→          <Button variant="outline" onClick={onClose} disabled={connecting}>
   402→            Cancel
   403→          </Button>
   404→          <Button
   405→            onClick={handleConnect}
   406→            disabled={connecting || !projectId || !repoUrl.trim()}
   407→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   408→          >
   409→            {connecting ? (
   410→              <>
   411→                <Loader2 className="size-4 mr-2 animate-spin" />
   412→                Connecting...
   413→              </>
   414→            ) : (
   415→              <>
   416→                <Github className="size-4 mr-2" />
   417→                Connect
   418→              </>
   419→            )}
   420→          </Button>
   421→        </DialogFooter>
   422→      </DialogContent>
   423→    </Dialog>
   424→  )
   425→}

   429→function IntegrationDetailView({
   430→  project,
   431→  integration,
   432→  onBack,
   433→  onRefresh,
   434→}: {
   435→  project: ProjectWithIntegration
   436→  integration: GithubIntegration
   437→  onBack: () => void
   438→  onRefresh: () => void
   439→}) {
   440→  const [commits, setCommits] = useState<CommitData[]>([])
   441→  const [commitsLoading, setCommitsLoading] = useState(true)
   442→  const [repos, setRepos] = useState<RepoData[]>([])
   443→  const [reposLoading, setReposLoading] = useState(true)
   444→  const [activeTab, setActiveTab] = useState('overview')
   445→
   446→  // Settings
   447→  const [autoDeploy, setAutoDeploy] = useState(integration.autoDeploy)
   448→  const [branch, setBranch] = useState(integration.branch)
   449→  const [savingSettings, setSavingSettings] = useState(false)
   450→
   451→  // Disconnect
   452→  const [disconnectDialogOpen, setDisconnectDialogOpen] = useState(false)
   453→  const [disconnecting, setDisconnecting] = useState(false)
   454→
   455→  const fetchCommits = useCallback(async () => {
   456→    setCommitsLoading(true)
   457→    try {
   458→      const data = await apiFetch(`/github/${integration.id}/commits`)
   459→      setCommits(data.commits || [])
   460→    } catch {
   461→      toast.error('Failed to load commits')
   462→    } finally {
   463→      setCommitsLoading(false)
   464→    }
   465→  }, [integration.id])
   466→
   467→  const fetchRepos = useCallback(async () => {
   468→    setReposLoading(true)
   469→    try {
   470→      const data = await apiFetch(`/github/${integration.id}/repos`)
   471→      setRepos(data.repos || [])
   472→    } catch {
   473→      toast.error('Failed to load repositories')
   474→    } finally {
   475→      setReposLoading(false)
   476→    }
   477→  }, [integration.id])
   478→
   479→  useEffect(() => {
   480→    fetchCommits()
   481→    fetchRepos()
   482→  }, [fetchCommits, fetchRepos])
   483→
   484→  const handleDisconnect = async () => {
   485→    setDisconnecting(true)
   486→    try {
   487→      await apiFetch('/github/disconnect', {
   488→        method: 'POST',
   489→        body: JSON.stringify({ projectId: project.id }),
   490→      })
   491→      toast.success('GitHub integration disconnected')
   492→      onRefresh()
   493→      onBack()
   494→    } catch (err) {
   495→      toast.error(err instanceof Error ? err.message : 'Failed to disconnect')
   496→    } finally {
   497→      setDisconnecting(false)
   498→      setDisconnectDialogOpen(false)
   499→    }
   500→  }
   501→
   502→  const handleRefreshData = () => {
   503→    fetchCommits()
   504→    fetchRepos()
   505→    toast.success('Data refreshed')
   506→  }
   507→
   508→  return (
   509→    <div className="space-y-6">
   510→      {/* Back Button + Title */}
   511→      <div className="flex items-center gap-3">
   512→        <Button variant="ghost" size="icon" onClick={onBack} className="shrink-0">
   513→          <ArrowLeft className="size-4" />
   514→        </Button>
   515→        <div className="flex-1 min-w-0">
   516→          <h2 className="text-xl font-bold truncate">{project.name}</h2>
   517→          <p className="text-sm text-muted-foreground">GitHub Integration Details</p>
   518→        </div>
   519→        <Button variant="outline" size="sm" onClick={handleRefreshData}>
   520→          <RefreshCw className="size-4 mr-1.5" />
   521→          Refresh
   522→        </Button>
   523→      </div>
   524→
   525→      {/* Repository Info Card */}
   526→      <Card className="rounded-xl">
   527→        <CardHeader className="pb-3">
   528→          <div className="flex items-start justify-between">
   529→            <div className="flex items-center gap-3">
   530→              <div className="flex size-10 items-center justify-center rounded-lg bg-emerald-100 dark:bg-emerald-900/30">
   531→                <Github className="size-5 text-emerald-600 dark:text-emerald-400" />
   532→              </div>
   533→              <div>
   534→                <CardTitle className="text-base">{integration.repoFullName}</CardTitle>
   535→                <CardDescription className="font-mono text-xs mt-0.5">
   536→                  {integration.repoUrl}
   537→                </CardDescription>
   538→              </div>
   539→            </div>
   540→            <Badge variant="outline" className="bg-emerald-50 text-emerald-600 dark:bg-emerald-900/20 dark:text-emerald-400 shrink-0">
   541→              <CheckCircle2 className="size-3 mr-1" />
   542→              Connected
   543→            </Badge>
   544→          </div>
   545→        </CardHeader>
   546→        <CardContent className="space-y-4">
   547→          <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
   548→            <div className="space-y-1">
   549→              <p className="text-xs text-muted-foreground">Branch</p>

   849→// ─── Activity Feed ─────────────────────────────────────────────────
   850→
   851→function ActivityFeed({ projects }: { projects: ProjectWithIntegration[] }) {
   852→  // Generate activity events from integration data
   853→  const events: ActivityEvent[] = []
   854→
   855→  for (const project of projects) {
   856→    if (project.githubIntegration) {
   857→      const integration = project.githubIntegration
   858→
   859→      // Connection event
   860→      events.push({
   861→        id: `conn-${integration.id}`,
   862→        type: 'integration_connected',
   863→        projectName: project.name,
   864→        message: `Connected ${integration.repoFullName}`,
   865→        timestamp: integration.createdAt,
   866→        detail: `Branch: ${integration.branch}`,
   867→      })
   868→
   869→      // Auto-deploy events (simulated)
   870→      if (integration.autoDeploy) {
   871→        events.push({
   872→          id: `deploy-${integration.id}`,
   873→          type: 'auto_deploy_triggered',
   874→          projectName: project.name,
   875→          message: `Auto-deploy enabled for ${integration.branch}`,
   876→          timestamp: integration.updatedAt,
   877→          detail: `${integration.repoFullName} → ${project.subdomain}`,
   878→        })
   879→      }
   880→
   881→      // Simulated commit push
   882→      events.push({
   883→        id: `push-${integration.id}`,
   884→        type: 'commit_pushed',
   885→        projectName: project.name,
   886→        message: `New push to ${integration.branch}`,
   887→        timestamp: new Date(Date.now() - Math.random() * 86400000 * 3).toISOString(),
   888→        detail: integration.repoFullName,
   889→      })
   890→    }
   891→  }
   892→
   893→  // Sort by timestamp descending
   894→  events.sort((a, b) => new Date(b.timestamp).getTime() - new Date(a.timestamp).getTime())
   895→
   896→  function eventIcon(type: ActivityEvent['type']) {
   897→    switch (type) {
   898→      case 'commit_pushed':
   899→        return <GitCommit className="size-3.5 text-emerald-600 dark:text-emerald-400" />
   900→      case 'auto_deploy_triggered':
   901→        return <Zap className="size-3.5 text-amber-600 dark:text-amber-400" />
   902→      case 'integration_connected':
   903→        return <CheckCircle2 className="size-3.5 text-emerald-600 dark:text-emerald-400" />
   904→      case 'integration_disconnected':
   905→        return <XCircle className="size-3.5 text-red-500" />
   906→    }
   907→  }
   908→
   909→  function eventBgColor(type: ActivityEvent['type']) {
   910→    switch (type) {
   911→      case 'commit_pushed':
   912→        return 'bg-emerald-100 dark:bg-emerald-900/30'
   913→      case 'auto_deploy_triggered':
   914→        return 'bg-amber-100 dark:bg-amber-900/30'
   915→      case 'integration_connected':
   916→        return 'bg-emerald-100 dark:bg-emerald-900/30'
   917→      case 'integration_disconnected':
   918→        return 'bg-red-100 dark:bg-red-900/30'
   919→    }
   920→  }
   921→
   922→  if (events.length === 0) {
   923→    return (
   924→      <Card className="rounded-xl border-dashed">
   925→        <CardContent className="flex flex-col items-center justify-center py-8">
   926→          <Clock className="size-8 text-muted-foreground/40 mb-2" />
   927→          <p className="text-sm text-muted-foreground">No activity yet</p>
   928→          <p className="text-xs text-muted-foreground mt-1">
   929→            Connect a repository to see GitHub activity here
   930→          </p>
   931→        </CardContent>
   932→      </Card>
   933→    )
   934→  }
   935→
   936→  return (
   937→    <Card className="rounded-xl">
   938→      <CardHeader className="pb-3">
   939→        <CardTitle className="text-base flex items-center gap-2">
   940→          <Clock className="size-4 text-emerald-600" />
   941→          Activity Feed
   942→        </CardTitle>
   943→        <CardDescription>Recent GitHub events across all integrations</CardDescription>
   944→      </CardHeader>
   945→      <CardContent className="p-0">
   946→        <ScrollArea className="max-h-64">
   947→          <div className="divide-y">
   948→            {events.slice(0, 10).map((event) => (
   949→              <div
   950→                key={event.id}
   951→                className="flex items-start gap-3 px-6 py-3 hover:bg-muted/50 transition-colors"
   952→              >
   953→                <div
   954→                  className={`flex size-7 items-center justify-center rounded-full shrink-0 mt-0.5 ${eventBgColor(event.type)}`}
   955→                >
   956→                  {eventIcon(event.type)}
   957→                </div>
   958→                <div className="flex-1 min-w-0">
   959→                  <p className="text-sm truncate">{event.message}</p>
   960→                  <div className="flex items-center gap-2 mt-0.5">
   961→                    <Badge variant="outline" className="text-[10px]">
   962→                      {event.projectName}
   963→                    </Badge>
   964→                    {event.detail && (
   965→                      <span className="text-xs text-muted-foreground truncate">{event.detail}</span>
   966→                    )}
   967→                  </div>
   968→                </div>
   969→                <span className="text-xs text-muted-foreground shrink-0 whitespace-nowrap">
   970→                  {timeAgo(event.timestamp)}
   971→                </span>
   972→              </div>
   973→            ))}
   974→          </div>
   975→        </ScrollArea>
   976→      </CardContent>
   977→    </Card>
   978→  )
