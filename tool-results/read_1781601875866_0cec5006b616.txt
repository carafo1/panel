    29→/* ─── Counter animation hook ─── */
    30→function useCountUp(end: number, duration = 2000, suffix = '') {
    31→  const [count, setCount] = useState(0)
    32→  const ref = useRef<HTMLDivElement>(null)
    33→  const hasAnimated = useRef(false)
    34→
    35→  useEffect(() => {
    36→    const observer = new IntersectionObserver(
    37→      ([entry]) => {
    38→        if (entry.isIntersecting && !hasAnimated.current) {
    39→          hasAnimated.current = true
    40→          const startTime = performance.now()
    41→          const animate = (currentTime: number) => {
    42→            const elapsed = currentTime - startTime
    43→            const progress = Math.min(elapsed / duration, 1)
    44→            // Ease out cubic
    45→            const eased = 1 - Math.pow(1 - progress, 3)
    46→            setCount(Math.round(eased * end))
    47→            if (progress < 1) {
    48→              requestAnimationFrame(animate)
    49→            }
    50→          }
    51→          requestAnimationFrame(animate)
    52→        }
    53→      },
    54→      { threshold: 0.3 }
    55→    )
    56→    if (ref.current) observer.observe(ref.current)
    57→    return () => observer.disconnect()
    58→  }, [end, duration])
    59→
    60→  return { count: count + suffix, ref }
    61→}
