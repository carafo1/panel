     1→/**
     2→ * Container Deployer
     3→ * ==================
     4→ *
     5→ * Real Docker container deployment on a provisioned VPS:
     6→ *
     7→ *   1. Generate a Dockerfile + docker-compose.yml from the project's stack
     8→ *      (nodejs, php, python, docker, wordpress) and any custom repo URL.
     9→ *   2. SFTP-upload the project files to /opt/deployhub/projects/<slug>/
    10→ *   3. Pull the repo (if GitHub-connected) OR write the generated files.
    11→ *   4. Run `docker compose up -d --build` and capture logs.
    12→ *   5. Register the container on the deployhub docker network (so Caddy can
    13→ *      proxy to it via the container's service name).
    14→ *   6. Persist containerId, containerName, containerPort, externalPort,
    15→ *      deployPath, lastDeployedAt back to the Project row.
    16→ *   7. Trigger the reverse-proxy manager to add the Caddyfile entry.
    17→ *
    18→ * In DEMO MODE (VPS_DEMO_MODE=1 or no server.sshHost), the deployer returns
    19→ * a realistic simulated success — the Project row is updated with a fake
    20→ * containerId + simulated build logs, so the dashboard UX is intact.
    21→ */
    22→import { db } from '@/lib/db'
    23→import { STACK_DEFINITIONS, getStackDefinition } from '@/lib/stacks'
    24→import {
    25→  withConnection,
    26→  exec,
    27→  execStream,
    28→  putFile,
    29→  isVpsDemoMode,
    30→  isServerInDemoMode,
    31→  type ExecResult,
    32→  type StreamChunk,
    33→} from './ssh'
    34→import { getServerCredentials } from './provisioner'
    35→import { configureProjectProxy, removeProjectProxy } from './proxy'

    39→export interface DeployRequest {
    40→  projectId: string
    41→  deploymentId: string
    42→  onLog?: (msg: string) => void
    43→  onStreamChunk?: (chunk: StreamChunk) => void
    44→}
    45→
    46→export interface DeployResult {
    47→  ok: boolean
    48→  projectId: string
    49→  deploymentId: string
    50→  containerId?: string

    53→  externalPort?: number
    54→  deployPath?: string
    55→  buildLog: string
    56→  durationMs: number
    57→  error?: string
    58→}

    67→  createdAt: string

    70→// ─── Main deploy function ────────────────────────────────────────────────
    71→
    72→export async function deployProject(req: DeployRequest): Promise<DeployResult> {
    73→  const start = Date.now()
    74→  const { projectId, deploymentId } = req
    75→  const log = (m: string) => req.onLog?.(m)
    76→
    77→  const project = await db.project.findUnique({
    78→    where: { id: projectId },
    79→    include: { server: true },
    80→  })
    81→  if (!project) throw new Error(`Project ${projectId} not found`)
    82→  if (!project.serverId || !project.server) {
    83→    throw new Error('Project has no server assigned')
    84→  }
    85→
    86→  const stack = getStackDefinition(project.stack)
    87→  if (!stack) throw new Error(`Unknown stack: ${project.stack}`)
    88→
    89→  const deployPath = project.deployPath || `/opt/deployhub/projects/${project.slug}`
    90→  const containerName = project.containerName || `deployhub-${project.slug}`
    91→  const containerPort = project.containerPort || stack.port
    92→  const externalPort = project.externalPort || await pickExternalPort(project.serverId)
    93→
    94→  log(`Deploying ${project.name} (${project.stack}) to ${project.server.name}`)
    95→  log(`  Path: ${deployPath}`)
    96→  log(`  Container: ${containerName} (internal ${containerPort} → host ${externalPort})`)
    97→
    98→  // ── DEMO MODE: simulated deploy with realistic build log ────────────
    99→  // Demo mode is active when: VPS_DEMO_MODE env is set, OR the server has
   100→  // no sshHost, OR the server has no SSH credentials (no key AND no password).
   101→  if (isServerInDemoMode(project.server)) {
   102→    const buildLog = generateDemoBuildLog(project, stack.name, containerName)
   103→    log('[demo mode] Simulating deploy…')
   104→    for (const line of buildLog.split('\n')) {
   105→      req.onStreamChunk?.({ stream: 'stdout', data: line + '\n' })
   106→      await sleep(150)
   107→    }
   108→    const fakeContainerId = generateFakeContainerId()
   109→
   110→    await db.project.update({
   111→      where: { id: projectId },
   112→      data: {
   113→        status: 'running',
   114→        containerId: fakeContainerId,
   115→        containerName,
   116→        containerPort,
   117→        externalPort,
   118→        deployPath,
   119→        lastDeployedAt: new Date(),
   120→        sslCertStatus: project.sslEnabled ? 'issued' : 'none',
   121→        sslCertExpiry: project.sslEnabled
   122→          ? new Date(Date.now() + 90 * 24 * 60 * 60 * 1000)
   123→          : null,
   124→      },
   125→    })
   126→
   127→    await db.deployment.update({
   128→      where: { id: deploymentId },
   129→      data: {
   130→        status: 'live',
   131→        buildLog,
   132→        deployUrl: `https://${project.customDomain || project.subdomain}`,
   133→        duration: Math.round((Date.now() - start) / 1000),
   134→      },
   135→    })
   136→
   137→    log('[demo mode] Deploy simulated successfully')
   138→    return {
   139→      ok: true,
   140→      projectId,
   141→      deploymentId,
   142→      containerId: fakeContainerId,
   143→      containerName,
   144→      containerPort,
   145→      externalPort,
   146→      deployPath,
   147→      buildLog,
   148→      durationMs: Date.now() - start,
   149→    }
   150→  }
   151→
   152→  // ── REAL deploy ─────────────────────────────────────────────────────
   153→  const creds = await getServerCredentials(project.serverId)
   154→  let buildLog = ''
   155→  let containerId: string | undefined
   156→
   157→  try {
   158→    await withConnection(creds, async (client) => {
   159→      // 1. Create project directory
   160→      log('Creating project directory…')
   161→      await exec(
   162→        client,
   163→        `mkdir -p ${deployPath}`,
   164→        { timeoutMs: 10000 }
   165→      )
   166→
   167→      // 2. Either clone the repo OR write a placeholder app
   168→      if (project.repoUrl) {
   169→        log(`Cloning ${project.repoUrl}…`)
   170→        const clone = await exec(
   171→          client,
   172→          `if [ -d ${deployPath}/.git ]; then cd ${deployPath} && git pull; else rm -rf ${deployPath}/* && git clone ${shellQuote(project.repoUrl)} ${deployPath}; fi`,
   173→          { timeoutMs: 120000 }
   174→        )
   175→        buildLog += `[git] ${clone.stdout}\n`
   176→        if (clone.exitCode !== 0) {
   177→          throw new Error(`git clone failed: ${clone.stderr}`)
   178→        }
   179→      } else {
   180→        log('No repo URL — writing placeholder app from stack template…')
   181→        const placeholder = generatePlaceholderApp(project.stack, project.name)
   182→        for (const [filename, content] of Object.entries(placeholder)) {
   183→          await putFile(client, `${deployPath}/${filename}`, content, { mode: 0o644 })
   184→        }
   185→        buildLog += `[placeholder] Wrote ${Object.keys(placeholder).length} files\n`
   186→      }
   187→
   188→      // 3. Write Dockerfile if stack defines one
   189→      if (stack.dockerfile && project.stack !== 'docker') {
   190→        log('Writing Dockerfile…')
   191→        await putFile(client, `${deployPath}/Dockerfile`, stack.dockerfile, {
   192→          mode: 0o644,
   193→        })
   194→        buildLog += `[dockerfile] Written (${stack.dockerfile.length} bytes)\n`
   195→      }
   196→
   197→      // 4. Write docker-compose.yml
   198→      log('Writing docker-compose.yml…')
   199→      const composeYml = generateDockerCompose({
   200→        projectName: project.slug,
   201→        containerName,
   202→        deployPath,
   203→        containerPort,
   204→        externalPort,
   205→        stack: project.stack,
   206→        envVars: await getProjectEnvVars(projectId),
   207→      })
   208→      await putFile(client, `${deployPath}/docker-compose.yml`, composeYml, {
   209→        mode: 0o644,
   210→      })
   211→      buildLog += `[compose] Written docker-compose.yml\n`
   212→
   213→      // 5. docker compose build + up (stream logs live)
   214→      log('Running docker compose build…')
   215→      const buildGen = execStream(
   216→        client,
   217→        `cd ${deployPath} && docker compose build --progress=plain 2>&1`,
   218→        { timeoutMs: 600000 }
   219→      )
   220→      for await (const chunk of buildGen) {
   221→        buildLog += chunk.data
   222→        req.onStreamChunk?.(chunk)
   223→      }
   224→      const buildResult = await buildGen.next().then(() => buildGen).catch(() => null)
   225→      log(`Build finished (exit code from stream closed)`)
   226→
   227→      log('Starting container with docker compose up -d…')
   228→      const upGen = execStream(
   229→        client,
   230→        `cd ${deployPath} && docker compose up -d 2>&1`,
   231→        { timeoutMs: 120000 }
   232→      )
   233→      for await (const chunk of upGen) {
   234→        buildLog += chunk.data
   235→        req.onStreamChunk?.(chunk)
   236→      }
   237→
   238→      // 6. Connect container to the deployhub network so Caddy can reach it
   239→      log('Connecting container to deployhub network…')
   240→      const net = await exec(
   241→        client,
   242→        `docker network connect deployhub ${containerName} 2>/dev/null || true`,
   243→        { timeoutMs: 15000 }
   244→      )
   245→      buildLog += `[network] ${net.stdout || 'already connected'}\n`
   246→
   247→      // 7. Get container ID + verify it's running
   248→      log('Verifying container is running…')
   249→      const inspect = await exec(
   250→        client,
   251→        `docker inspect --format '{{.Id}} {{.State.Status}}' ${containerName}`,
   252→        { timeoutMs: 15000 }
   253→      )
   254→      const [cid, state] = inspect.stdout.trim().split(' ')
   255→      if (!cid || state !== 'running') {
   256→        throw new Error(
   257→          `Container did not start properly (state=${state}). Logs:\n` +
   258→            (await exec(client, `docker logs ${containerName} --tail 50 2>&1`, {
   259→              timeoutMs: 15000,
   260→            })).stdout
   261→        )
   262→      }
   263→      containerId = cid
   264→      log(`Container running: ${cid.slice(0, 12)}`)
   265→
   266→      // 8. Update Caddy reverse proxy config (auto-HTTPS)
   267→      log('Configuring Caddy reverse proxy…')
   268→      await configureProjectProxy(project.id, { onLog: log })
   269→      log('Caddy reloaded with new site config')
   270→    })
   271→
   272→    // 9. Persist project + deployment state
   273→    await db.project.update({
   274→      where: { id: projectId },
   275→      data: {
   276→        status: 'running',
   277→        containerId,
   278→        containerName,
   279→        containerPort,
   280→        externalPort,
   281→        deployPath,
   282→        lastDeployedAt: new Date(),
   283→        sslCertStatus: project.sslEnabled ? 'issued' : 'none',
   284→        sslCertExpiry: project.sslEnabled
   285→          ? new Date(Date.now() + 90 * 24 * 60 * 60 * 1000)
   286→          : null,
   287→      },
   288→    })
   289→
   290→    await db.deployment.update({
   291→      where: { id: deploymentId },
   292→      data: {
   293→        status: 'live',
   294→        buildLog,
   295→        deployUrl: `https://${project.customDomain || project.subdomain}`,
   296→        duration: Math.round((Date.now() - start) / 1000),
   297→      },
   298→    })
   299→
   300→    log('Deploy complete!')
   301→    return {
   302→      ok: true,
   303→      projectId,
   304→      deploymentId,
   305→      containerId,
   306→      containerName,
   307→      containerPort,
   308→      externalPort,
   309→      deployPath,
   310→      buildLog,
   311→      durationMs: Date.now() - start,
   312→    }
   313→  } catch (err) {
   314→    const message = err instanceof Error ? err.message : String(err)
   315→    log(`DEPLOY FAILED: ${message}`)
   316→    buildLog += `\n[FATAL] ${message}\n`
   317→
   318→    await db.project.update({
   319→      where: { id: projectId },
   320→      data: { status: 'error' },
   321→    })
   322→    await db.deployment.update({
   323→      where: { id: deploymentId },
   324→      data: { status: 'failed', buildLog, duration: Math.round((Date.now() - start) / 1000) },
   325→    })
   326→
   327→    return {
   328→      ok: false,
   329→      projectId,
   330→      deploymentId,
   331→      containerId,
   332→      containerName,
   333→      containerPort,
   334→      externalPort,
   335→      deployPath,
   336→      buildLog,
   337→      durationMs: Date.now() - start,
   338→      error: message,
   339→    }
   340→  }
   341→}

   345→export async function stopProject(projectId: string): Promise<{ ok: boolean; message: string }> {
   346→  const project = await db.project.findUnique({ where: { id: projectId }, include: { server: true } })
   347→  if (!project || !project.containerName) {
   348→    return { ok: false, message: 'Project has no running container' }
   349→  }
   350→  if (isServerInDemoMode(project.server)) {
   351→    await db.project.update({ where: { id: projectId }, data: { status: 'stopped' } })
   352→    return { ok: true, message: 'Demo mode: container stopped (simulated)' }
   353→  }
   354→  try {
   355→    const creds = await getServerCredentials(project.serverId!)
   356→    await withConnection(creds, async (client) => {
   357→      await exec(client, `cd ${project.deployPath} && docker compose down 2>&1`, {
   358→        timeoutMs: 60000,
   359→      })
   360→    })
   361→    await db.project.update({ where: { id: projectId }, data: { status: 'stopped' } })
   362→    return { ok: true, message: 'Container stopped' }
   363→  } catch (err) {
   364→    return {
   365→      ok: false,
   366→      message: err instanceof Error ? err.message : String(err),
   367→    }
   368→  }
   369→}

   395→export async function removeProjectContainer(projectId: string): Promise<{ ok: boolean; message: string }> {
   396→  const project = await db.project.findUnique({ where: { id: projectId }, include: { server: true } })
   397→  if (!project) return { ok: false, message: 'Project not found' }
   398→
   399→  // Remove Caddy config first
   400→  await removeProjectProxy(projectId).catch(() => {})
   401→
   402→  if (!isServerInDemoMode(project.server) && project.containerName) {
   403→    try {
   404→      const creds = await getServerCredentials(project.serverId!)
   405→      await withConnection(creds, async (client) => {
   406→        await exec(
   407→          client,
   408→          `cd ${project.deployPath} && docker compose down --rmi local -v 2>&1 || true`,
   409→          { timeoutMs: 60000 }
   410→        )
   411→        await exec(client, `rm -rf ${project.deployPath}`, { timeoutMs: 30000 })
   412→      })
   413→    } catch (err) {
   414→      console.error('[vps-deploy] remove container error:', err)
   415→    }
   416→  }
   417→
   418→  await db.project.update({
   419→    where: { id: projectId },
   420→    data: {
   421→      status: 'stopped',
   422→      containerId: null,
   423→      containerName: null,
   424→      containerPort: null,
   425→      externalPort: null,
   426→      deployPath: null,
   427→      sslCertStatus: 'none',
   428→      sslCertExpiry: null,
   429→    },
   430→  })
   431→  return { ok: true, message: 'Container removed' }
   432→}

   436→export async function getContainerStatus(projectId: string): Promise<ContainerStatus | null> {
   437→  const project = await db.project.findUnique({ where: { id: projectId }, include: { server: true } })
   438→  if (!project || !project.containerName) return null
   439→
   440→  if (isServerInDemoMode(project.server)) {
   441→    return {
   442→      containerId: project.containerId || 'demo',
   443→      name: project.containerName,
   444→      state: project.status === 'running' ? 'running' : 'exited',
   445→      status: project.status === 'running' ? 'Up (simulated)' : 'Exited (simulated)',
   446→      image: `deployhub/${project.stack}:latest`,
   447→      ports: [`${project.externalPort}->${project.containerPort}`],
   448→      createdAt: project.lastDeployedAt?.toISOString() || new Date().toISOString(),
   449→    }
   450→  }
   451→
   452→  try {
   453→    const creds = await getServerCredentials(project.serverId!)
   454→    return await withConnection(creds, async (client) => {
   455→      const r = await exec(
   456→        client,
   457→        `docker inspect --format '{{.Id}}|{{.Name}}|{{.State.Status}}|{{.State.Health.Status}}|{{.Config.Image}}|{{range $p, $conf := .NetworkSettings.Ports}}{{$p}}->{{(index $conf 0).HostPort}} {{end}}|{{.Created}}' ${project.containerName} 2>/dev/null`,
   458→        { timeoutMs: 10000 }
   459→      )
   460→      if (r.exitCode !== 0) {
   461→        return {
   462→          containerId: project.containerId || '',
   463→          name: project.containerName,
   464→          state: 'not_found' as const,
   465→          status: 'Container not found on server',
   466→          image: '',
   467→          ports: [],
   468→          createdAt: '',
   469→        }
   470→      }
   471→      const parts = r.stdout.trim().split('|')
   472→      const [id, name, state, health, image, ports, created] = parts
   473→      return {
   474→        containerId: id,
   475→        name: name.replace(/^\//, ''),
   476→        state: state as ContainerStatus['state'],
   477→        status: health && health !== '<no value>' ? `${state} (${health})` : state,
   478→        image,
   479→        ports: ports ? ports.trim().split(' ').filter(Boolean) : [],
   480→        createdAt: created || '',
   481→      }
   482→    })
   483→  } catch (err) {
   484→    console.error('[vps-deploy] container status error:', err)
   485→    return null
   486→  }
   487→}

   491→export async function getContainerLogs(
   492→  projectId: string,
   493→  opts: { tail?: number; since?: string } = {}
   494→): Promise<string> {
   495→  const project = await db.project.findUnique({ where: { id: projectId }, include: { server: true } })
   496→  if (!project || !project.containerName) return 'No container running'
   497→
   498→  if (isServerInDemoMode(project.server)) {
   499→    return generateDemoRuntimeLog(project)
   500→  }
   501→
   502→  try {
   503→    const creds = await getServerCredentials(project.serverId!)
   504→    return await withConnection(creds, async (client) => {
   505→      const tail = opts.tail ?? 200
   506→      const sinceFlag = opts.since ? `--since ${shellQuote(opts.since)}` : ''
   507→      const r = await exec(
   508→        client,
   509→        `docker logs ${project.containerName} --tail ${tail} ${sinceFlag} 2>&1`,
   510→        { timeoutMs: 30000 }
   511→      )
   512→      return r.stdout || r.stderr
   513→    })
   514→  } catch (err) {
   515→    return `Failed to fetch logs: ${err instanceof Error ? err.message : err}`
   516→  }
   517→}

   521→async function pickExternalPort(serverId: string): Promise<number> {
   522→  // Find the highest externalPort in use on this server + 1, starting from 30000.
   523→  const projects = await db.project.findMany({
   524→    where: { serverId, externalPort: { not: null } },
   525→    select: { externalPort: true },
   526→  })
   527→  const usedPorts = new Set(projects.map((p) => p.externalPort!))
   528→  for (let port = 30000; port < 40000; port++) {
   529→    if (!usedPorts.has(port)) return port
   530→  }
   531→  return 30000 + Math.floor(Math.random() * 9999)
   532→}

   534→async function getProjectEnvVars(projectId: string): Promise<Record<string, string>> {
   535→  const envVars = await db.environmentVariable.findMany({
   536→    where: { projectId },
   537→    select: { key: true, value: true, encrypted: true },
   538→  })
   539→  // Note: env var values are encrypted at rest; decrypt here. The actual
   540→  // decryption is handled by the env var API; for compose we re-read the
   541→  // raw value via the encryption module.
   542→  const { decrypt } = await import('@/lib/encryption')
   543→  const out: Record<string, string> = {}
   544→  for (const v of envVars) {
   545→    try {
   546→      out[v.key] = v.encrypted ? decrypt(v.value) : v.value
   547→    } catch {
   548→      out[v.key] = v.value
   549→    }
   550→  }
   551→  return out
   552→}

   554→function generateDockerCompose(opts: {
   555→  projectName: string
   556→  containerName: string
   557→  deployPath: string
   558→  containerPort: number
   559→  externalPort: number
   560→  stack: string
   561→  envVars: Record<string, string>
   562→}): string {
   563→  const envLines = Object.entries(opts.envVars)
   564→    .map(([k, v]) => `      ${k}: "${v.replace(/"/g, '\\"')}"`)
   565→    .join('\n')
   566→
   567→  // For WordPress we add a MariaDB service.
   568→  if (opts.stack === 'wordpress') {
   569→    return `# Auto-generated by Apsamo Panel
   570→services:
   571→  wordpress:
   572→    image: wordpress:php8.2-apache
   573→    container_name: ${opts.containerName}
   574→    restart: unless-stopped
   575→    ports:
   576→      - "${opts.externalPort}:80"
   577→    environment:
   578→      WORDPRESS_DB_HOST: db
   579→      WORDPRESS_DB_USER: wp
   580→      WORDPRESS_DB_PASSWORD: wpdbpass_${randomHex(8)}
   581→      WORDPRESS_DB_NAME: wp
   582→${envLines ? envLines + '\n' : ''}    volumes:
   583→      - ./wp-content:/var/www/html/wp-content

   810→function shellQuote(s: string): string {
   811→  if (s === '') return "''"
   812→  if (/^[A-Za-z0-9_\-=/.,:@+]+$/.test(s)) return s
   813→  return "'" + s.replace(/'/g, "'\\''") + "'"
   814→}
