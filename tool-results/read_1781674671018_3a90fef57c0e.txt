   105→// ── API helper ─────────────────────────────────────────────────────────
   106→async function apiFetch<T>(path: string, options?: RequestInit): Promise<T> {
   107→  const token = localStorage.getItem('deployhub_token')
   108→  const res = await fetch(`/api${path}`, {
   109→    ...options,
   110→    headers: {
   111→      'Content-Type': 'application/json',
   112→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
   113→      ...options?.headers,
   114→    },
   115→  })
   116→  if (!res.ok) {
   117→    const data = await res.json().catch(() => ({ error: 'Request failed' }))
   118→    throw new Error(data.error || `Request failed with status ${res.status}`)
   119→  }
   120→  return res.json()
   121→}

   164→function PaymentStatusBadge({ status }: { status: string }) {
   165→  const map: Record<string, { className: string; label: string }> = {
   166→    completed: { className: 'bg-emerald-100 text-emerald-700 border-emerald-200', label: 'Completed' },
   167→    pending: { className: 'bg-yellow-100 text-yellow-700 border-yellow-200', label: 'Pending' },
   168→    failed: { className: 'bg-red-100 text-red-700 border-red-200', label: 'Failed' },
   169→    refunded: { className: 'bg-gray-100 text-gray-700 border-gray-200', label: 'Refunded' },
   170→  }
   171→  const cfg = map[status] || map.pending
   172→  return (
   173→    <Badge variant="outline" className={cfg.className}>
   174→      {cfg.label}
   175→    </Badge>
   176→  )

   584→                <p className="text-sm text-emerald-700">
   585→                  ${subscription.plan?.price || '—'}/month

   587→                <p className="text-xs text-emerald-600">
   588→                  Renews {new Date(subscription.currentPeriodEnd).toLocaleDateString()}
   589→                </p>
   590→              </div>
   591→              <div className="flex items-center gap-2 text-emerald-700">
   592→                <Check className="size-4" />
   593→                <span className="text-sm">All features included</span>
   594→              </div>
   595→            </div>
   596→          ) : (
   597→            <div className="flex flex-col items-center justify-center py-8 text-center">

   607→                className="bg-emerald-600 hover:bg-emerald-700 text-white"

   614→        </CardContent>

   617→      {/* ── Plan Comparison ───────────────────────────────────────── */}
   618→      <div id="plan-comparison">
   619→        <h2 className="text-xl font-semibold mb-4">Plan Comparison</h2>
   620→        <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
   621→          {PLAN_DEFINITIONS.map((def) => {

   623→              (p) => p.slug === def.slug || p.name.toLowerCase() === def.slug

   684→                      }`}

   729→      <Card>
   730→        <CardHeader>
   731→          <CardTitle className="flex items-center gap-2">
   732→            <CreditCard className="size-5 text-emerald-600" />
   733→            Payment History
   734→          </CardTitle>
   735→          <CardDescription>Your recent transactions</CardDescription>
   736→        </CardHeader>
   737→        <CardContent>
   738→          {payments.length === 0 ? (
   739→            <div className="flex flex-col items-center justify-center py-8 text-center">
   740→              <p className="text-muted-foreground">No payments yet</p>

   743→            <div className="max-h-96 overflow-y-auto [&::-webkit-scrollbar]:w-2 [&::-webkit-scrollbar-track]:bg-transparent [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-muted-foreground/20">
   744→              <Table>
   745→                <TableHeader>
   746→                  <TableRow>
   747→                    <TableHead>Date</TableHead>
   748→                    <TableHead>Description</TableHead>
   749→                    <TableHead>Amount</TableHead>

   753→                <TableBody>
   754→                  {payments.map((p) => (

   756→                      <TableCell className="text-muted-foreground">
   757→                        {new Date(p.createdAt).toLocaleDateString()}
   758→                      </TableCell>

   773→                      </TableCell>
   774→                    </TableRow>
   775→                  ))}
   776→                </TableBody>
   777→              </Table>
   778→            </div>
   779→          )}
   780→        </CardContent>
   781→      </Card>
   782→
   783→      {/* ── Checkout Dialog ───────────────────────────────────────── */}
   784→      {selectedPlan && selectedPlanDef && (
   785→        <CheckoutDialog
   786→          open={checkoutOpen}
   787→          onOpenChange={setCheckoutOpen}
   788→          plan={selectedPlan}
   789→          planDefinition={selectedPlanDef}
   790→          onSuccess={handleCheckoutSuccess}
   791→        />
   792→      )}
   793→    </div>
