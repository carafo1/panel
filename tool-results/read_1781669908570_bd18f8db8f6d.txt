    74→    74→function getPasswordStrength(password: string): { label: string; color: string; percent: number } {
    75→    75→  if (!password) return { label: '', color: '', percent: 0 }
    76→    76→  let score = 0
    77→    77→  if (password.length >= 8) score++
    78→    78→  if (password.length >= 12) score++
    79→    79→  if (/[A-Z]/.test(password)) score++
    80→    80→  if (/[0-9]/.test(password)) score++
    81→    81→  if (/[^A-Za-z0-9]/.test(password)) score++
    82→    82→
    83→    83→  if (score <= 1) return { label: 'Weak', color: 'bg-red-500', percent: 20 }
    84→    84→  if (score <= 2) return { label: 'Fair', color: 'bg-orange-500', percent: 40 }
    85→    85→  if (score <= 3) return { label: 'Medium', color: 'bg-yellow-500', percent: 60 }
    86→    86→  if (score <= 4) return { label: 'Strong', color: 'bg-emerald-400', percent: 80 }
    87→    87→  return { label: 'Very Strong', color: 'bg-emerald-600', percent: 100 }
    88→    88→}
