    69→// ── API helper ─────────────────────────────────────────────────────────
    70→async function apiFetch<T>(path: string, options?: RequestInit): Promise<T> {
    71→  const token = localStorage.getItem('deployhub_token')
    72→  const res = await fetch(`/api${path}`, {
    73→    ...options,
    74→    headers: {
    75→      'Content-Type': 'application/json',
    76→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    77→      ...options?.headers,
    78→    },
    79→  })
    80→  if (!res.ok) {
    81→    const data = await res.json().catch(() => ({ error: 'Request failed' }))
    82→    throw new Error(data.error || `Request failed with status ${res.status}`)
    83→  }
    84→  return res.json()
    85→}

   108→// ── Time helpers ───────────────────────────────────────────────────────
   109→function timeAgo(date: string): string {
   110→  const seconds = Math.floor((Date.now() - new Date(date).getTime()) / 1000)
   111→  if (seconds < 60) return 'just now'
   112→  const minutes = Math.floor(seconds / 60)
   113→  if (minutes < 60) return `${minutes}m ago`
   114→  const hours = Math.floor(minutes / 60)
   115→  if (hours < 24) return `${hours}h ago`
   116→  const days = Math.floor(hours / 24)
   117→  return `${days}d ago`
   118→}
   119→
   120→function formatDuration(seconds: number | null): string {
   121→  if (!seconds) return '—'
   122→  if (seconds < 60) return `${seconds}s`
   123→  const m = Math.floor(seconds / 60)
   124→  const s = seconds % 60
   125→  return `${m}m ${s}s`
   126→}

   141→// ── Simulated build logs ───────────────────────────────────────────────
   142→function generateSimulatedBuildLog(deployment: DeploymentDetail): string {
   143→  const isFailed = deployment.status === 'failed'
   144→  const isBuilding = deployment.status === 'building' || deployment.status === 'deploying'
   145→  const fw = deployment.project?.framework === 'nextjs' ? 'next' : (deployment.project?.framework || 'node')
   146→  const now = new Date(deployment.createdAt)
   147→  const pad = (n: number) => String(n).padStart(2, '0')
   148→  const ts = `${pad(now.getHours())}:${pad(now.getMinutes())}:${pad(now.getSeconds())}`
   149→
   150→  if (isFailed) {
   151→    return `${ts} > Cloning repository...
   152→${ts} ✓ Repository cloned successfully
   153→${ts} > Installing dependencies...
   154→${ts} > npm install --production
   155→${ts} ✓ Dependencies installed (1.2s)
   156→${ts} > Building application...
   157→${ts} > ${fw} build
   158→${ts} ✗ Type error: Property 'fetchData' does not exist on type 'never'.
   159→${ts} ✗ Build failed with 1 error
   160→${ts} > Error: Build process exited with code 1
   161→${ts} > Deployment failed!`
   162→  }
   163→
   164→  if (isBuilding) {
   165→    return `${ts} > Cloning repository...
   166→${ts} ✓ Repository cloned successfully
   167→${ts} > Installing dependencies...
   168→${ts} > npm install --production
   169→${ts} ✓ Dependencies installed (1.2s)
   170→${ts} > Building application...
   171→${ts} > ${fw} build
   172→${ts} ⠹ Compiling...`
   173→  }
   174→
   175→  return `${ts} > Cloning repository...
   176→${ts} ✓ Repository cloned successfully
   177→${ts} > Installing dependencies...
   178→${ts} > npm install --production
   179→${ts} ✓ Dependencies installed (1.2s)
   180→${ts} > Building application...
   181→${ts} > ${fw} build
   182→${ts} ✓ Compiled successfully
   183→${ts} ✓ Linting and checking validity of types
   184→${ts} ✓ Collecting page data
   185→${ts} ✓ Generating static pages
   186→${ts} ✓ Finalizing page optimization
   187→${ts} > Deploying to server...
   188→${ts} ✓ Deployment successful!
   189→${ts} > Live at https://${deployment.project?.subdomain || 'myapp'}.deployhub.io`
   190→}

   192→function colorizeLogLine(line: string): React.ReactNode {
   193→  if (line.includes('✓') || line.includes('successful') || line.includes('Live at')) {
   194→    return <span className="text-emerald-400">{line}</span>
   195→  }
   196→  if (line.includes('✗') || line.includes('Error') || line.includes('failed')) {
   197→    return <span className="text-red-400">{line}</span>
   198→  }
   199→  if (line.includes('Warning') || line.includes('warn') || line.includes('⠹') || line.includes('⠋') || line.includes('⠼')) {
   200→    return <span className="text-yellow-400">{line}</span>
   201→  }
   202→  if (line.startsWith('>')) {
   203→    return <span className="text-zinc-400">{line}</span>
   204→  }
   205→  return <span className="text-zinc-200">{line}</span>
   206→}
