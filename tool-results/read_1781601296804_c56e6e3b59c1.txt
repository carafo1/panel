   226→   226→
   227→   227→function formatNumber(n: number): string {
   228→   228→  if (n >= 1000) {
   229→   229→    return `${(n / 1000).toFixed(1).replace(/\.0$/, '')}k`
   230→   230→  }
   231→   231→  return n.toString()
   232→   232→}
   233→   233→
   234→   234→function slugify(text: string): string {
   235→   235→  return text
   236→   236→    .toLowerCase()
   237→   237→    .replace(/[^a-z0-9]+/g, '-')
   238→   238→    .replace(/^-|-$/g, '')
   239→   239→}
