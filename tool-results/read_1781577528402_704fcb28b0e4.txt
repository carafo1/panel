   260→   260→function timeAgo(dateStr: string): string {
   261→   261→  const seconds = Math.floor((Date.now() - new Date(dateStr).getTime()) / 1000)
   262→   262→  if (seconds < 60) return 'just now'
   263→   263→  const minutes = Math.floor(seconds / 60)
   264→   264→  if (minutes < 60) return `${minutes}m ago`
   265→   265→  const hours = Math.floor(minutes / 60)
   266→   266→  if (hours < 24) return `${hours}h ago`
   267→   267→  const days = Math.floor(hours / 24)
   268→   268→  return `${days}d ago`
   269→   269→}
   270→   270→
   271→   271→function formatDate(dateStr: string) {
   272→   272→  try {
   273→   273→    return new Date(dateStr).toLocaleDateString('en-US', {
   274→   274→      month: 'short',
   275→   275→      day: 'numeric',
   276→   276→      year: 'numeric',
   277→   277→      hour: '2-digit',
   278→   278→      minute: '2-digit',
   279→   279→    })
   280→   280→  } catch {
   281→   281→    return dateStr
   282→   282→  }
   283→   283→}

   285→   285→function generateSimulatedBuildLog(project: ProjectDetail, deployment: Deployment): string {

   287→   287→  const fw = project.framework === 'nextjs' ? 'next' : project.framework
   288→   288→  const now = new Date(deployment.createdAt)
   289→   289→  const pad = (n: number) => String(n).padStart(2, '0')
   290→   290→  const ts = `${pad(now.getHours())}:${pad(now.getMinutes())}:${pad(now.getSeconds())}`
   291→   291→
   292→   292→  if (isFailed) {
   293→   293→    return `${ts} > Cloning repository...

   321→   321→function colorizeLogLine(line: string): React.ReactNode {
   322→   322→  if (line.includes('✓') || line.includes('successful') || line.includes('Live at')) {
   323→   323→    return <span className="text-emerald-400">{line}</span>
   324→   324→  }
   325→   325→  if (line.includes('✗') || line.includes('Error') || line.includes('failed')) {
   326→   326→    return <span className="text-red-400">{line}</span>
   327→   327→  }
   328→   328→  if (line.includes('Warning') || line.includes('warn')) {
   329→   329→    return <span className="text-yellow-400">{line}</span>
   330→   330→  }
   331→   331→  if (line.startsWith('>')) {
   332→   332→    return <span className="text-zinc-400">{line}</span>
   333→   333→  }
   334→   334→  return <span className="text-zinc-200">{line}</span>
   335→   335→}

   413→   413→function CreateProjectDialog({
   414→   414→  open,
   415→   415→  onClose,
   416→   416→  onCreated,
   417→   417→}: {
   418→   418→  open: boolean
   419→   419→  onClose: () => void
   420→   420→  onCreated: () => void
   421→   421→}) {
   422→   422→  const [name, setName] = useState('')
   423→   423→  const [framework, setFramework] = useState('nextjs')
   424→   424→  const [planId, setPlanId] = useState('')
   425→   425→  const [plans, setPlans] = useState<Plan[]>([])
   426→   426→  const [loading, setLoading] = useState(false)
   427→   427→  const [plansLoading, setPlansLoading] = useState(false)
   428→   428→
   429→   429→  useEffect(() => {
   430→   430→    if (open) {
   431→   431→      setPlansLoading(true)
   432→   432→      fetch('/api/billing/plans')
   433→   433→        .then((r) => r.json())
   434→   434→        .then((data) => {
   435→   435→          setPlans(data.plans || [])
   436→   436→          if (data.plans?.length > 0 && !planId) {
   437→   437→            setPlanId(data.plans[0].id)
   438→   438→          }
   439→   439→        })
   440→   440→        .catch(() => toast.error('Failed to load plans'))
   441→   441→        .finally(() => setPlansLoading(false))
   442→   442→    }
   443→   443→  }, [open, planId])
   444→   444→
   445→   445→  const handleCreate = async () => {
   446→   446→    if (!name.trim()) {
   447→   447→      toast.error('Project name is required')
   448→   448→      return
   449→   449→    }
   450→   450→    setLoading(true)
   451→   451→    try {
   452→   452→      const res = await fetch('/api/hosting/create', {
   453→   453→        method: 'POST',
   454→   454→        headers: getAuthHeaders(),
   455→   455→        body: JSON.stringify({ name: name.trim(), framework, planId: planId || undefined }),
   456→   456→      })
   457→   457→      const data = await res.json()
   458→   458→      if (!res.ok) {
   459→   459→        toast.error(data.error || 'Failed to create project')
   460→   460→        return
   461→   461→      }
   462→   462→      toast.success('Project created successfully!')
   463→   463→      setName('')
   464→   464→      setFramework('nextjs')
   465→   465→      onCreated()
   466→   466→      onClose()
   467→   467→    } catch {
   468→   468→      toast.error('Network error. Please try again.')
   469→   469→    } finally {
   470→   470→      setLoading(false)
   471→   471→    }
   472→   472→  }
   473→   473→
   474→   474→  return (
   475→   475→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   476→   476→      <DialogContent className="sm:max-w-lg">
   477→   477→        <DialogHeader>
   478→   478→          <DialogTitle>Create New Project</DialogTitle>
   479→   479→          <DialogDescription>
   480→   480→            Set up a new hosting project with your preferred framework and plan.
   481→   481→          </DialogDescription>
   482→   482→        </DialogHeader>
   483→   483→
   484→   484→        <div className="grid gap-4 py-4">
   485→   485→          <div className="grid gap-2">
   486→   486→            <Label htmlFor="project-name">Project Name</Label>
   487→   487→            <Input
   488→   488→              id="project-name"
   489→   489→              placeholder="my-awesome-app"
   490→   490→              value={name}
   491→   491→              onChange={(e) => setName(e.target.value)}
   492→   492→            />
   493→   493→          </div>
   494→   494→
   495→   495→          <div className="grid gap-2">
   496→   496→            <Label>Framework</Label>
   497→   497→            <Select value={framework} onValueChange={setFramework}>
   498→   498→              <SelectTrigger className="w-full">
   499→   499→                <SelectValue placeholder="Select framework" />
   500→   500→              </SelectTrigger>
   501→   501→              <SelectContent>
   502→   502→                <SelectItem value="nextjs">Next.js</SelectItem>
   503→   503→                <SelectItem value="react">React</SelectItem>
   504→   504→                <SelectItem value="node">Node.js</SelectItem>
   505→   505→                <SelectItem value="static">Static</SelectItem>
   506→   506→              </SelectContent>
   507→   507→            </Select>
   508→   508→          </div>
   509→   509→
   510→   510→          <div className="grid gap-2">
   511→   511→            <Label>Plan</Label>
   512→   512→            {plansLoading ? (
   513→   513→              <div className="grid gap-2">
   514→   514→                <Skeleton className="h-20 w-full" />
   515→   515→                <Skeleton className="h-20 w-full" />
