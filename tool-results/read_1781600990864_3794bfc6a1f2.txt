   192→function formatDate(dateStr?: string) {
   193→  if (!dateStr) return '-'
   194→  try {
   195→    return new Date(dateStr).toLocaleDateString('en-US', {
   196→      month: 'short',
   197→      day: 'numeric',
   198→      year: 'numeric',
   199→    })
   200→  } catch {
   201→    return dateStr
   202→  }
   203→}

   205→function formatTTL(ttl: number) {
   206→  if (ttl >= 3600) return `${ttl / 3600}h`
   207→  if (ttl >= 60) return `${ttl / 60}m`
   208→  return `${ttl}s`
   209→}

   223→function PropagationRing({ percentage, size = 80 }: { percentage: number; size?: number }) {
   224→  const strokeWidth = 6
   225→  const radius = (size - strokeWidth) / 2
   226→  const circumference = radius * 2 * Math.PI
   227→  const offset = circumference - (percentage / 100) * circumference
   228→
   229→  const color =
   230→    percentage >= 80
   231→      ? 'text-emerald-500'
   232→      : percentage >= 50
   233→      ? 'text-yellow-500'
   234→      : 'text-red-500'
   235→
   236→  const strokeColor =
   237→    percentage >= 80
   238→      ? 'stroke-emerald-500'
   239→      : percentage >= 50
   240→      ? 'stroke-yellow-500'
   241→      : 'stroke-red-500'
   242→
   243→  return (
   244→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   245→      <svg className="transform -rotate-90" width={size} height={size}>
   246→        <circle
   247→          className="stroke-muted"
   248→          strokeWidth={strokeWidth}
   249→          fill="transparent"
   250→          r={radius}
   251→          cx={size / 2}
   252→          cy={size / 2}
   253→        />
   254→        <circle
   255→          className={strokeColor}
   256→          strokeWidth={strokeWidth}
   257→          strokeDasharray={circumference}
   258→          strokeDashoffset={offset}
   259→          strokeLinecap="round"
   260→          fill="transparent"
   261→          r={radius}
   262→          cx={size / 2}
   263→          cy={size / 2}
   264→          style={{ transition: 'stroke-dashoffset 0.8s ease-in-out' }}
   265→        />
   266→      </svg>
   267→      <div className={`absolute text-sm font-bold ${color}`}>
   268→        {percentage}%
   269→      </div>
   270→    </div>
   271→  )
   272→}

   298→function AddDnsRecordDialog({
   299→  open,

   312→  const [ttl, setTtl] = useState('3600')

   315→  const handleAdd = async () => {
   316→    if (!name.trim() || !value.trim()) {
   317→      toast.error('Name and value are required')

   423→function LinkProjectDialog({
   424→  open,
   425→  onClose,
   426→  domainId,
   427→  onLinked,
   428→}: {
   429→  open: boolean
   430→  onClose: () => void
   431→  domainId: string
   432→  onLinked: () => void
   433→}) {
   434→  const [projects, setProjects] = useState<Project[]>([])
   435→  const [selectedProjectId, setSelectedProjectId] = useState('')
   436→  const [loading, setLoading] = useState(false)
   437→
   438→  useEffect(() => {
   439→    if (open) {
   440→      fetch('/api/hosting', { headers: getAuthHeaders() })
   441→        .then((r) => r.json())
   442→        .then((data) => setProjects(data.projects || []))
   443→        .catch(() => toast.error('Failed to load projects'))
   444→    }
   445→  }, [open])
   446→
   447→  const handleLink = async () => {
   448→    if (!selectedProjectId) {
   449→      toast.error('Select a project')
   450→      return
   451→    }
   452→    setLoading(true)
   453→    try {
   454→      await fetch(`/api/domains/${domainId}`, {
   455→        method: 'PUT',
   456→        headers: getAuthHeaders(),
   457→        body: JSON.stringify({ projectId: selectedProjectId }),
   458→      })
   459→      toast.success('Domain linked to project')
   460→      onLinked()
   461→      onClose()
   462→    } catch {
   463→      toast.error('Network error')
   464→    } finally {
   465→      setLoading(false)
   466→    }
   467→  }
   468→
   469→  return (
   470→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   471→      <DialogContent className="sm:max-w-md">
   472→        <DialogHeader>
   473→          <DialogTitle>Link to Project</DialogTitle>
   474→          <DialogDescription>
   475→            Connect this domain to one of your hosting projects.
   476→          </DialogDescription>
   477→        </DialogHeader>
   478→        <div className="grid gap-4 py-4">
   479→          <div className="grid gap-2">

   481→            <Select value={selectedProjectId} onValueChange={setSelectedProjectId}>

   483→                <SelectValue placeholder="Choose a project" />

   485→              <SelectContent>

   487→                  <SelectItem key={p.id} value={p.id}>
   488→                    {p.name} ({p.subdomain})
   489→                  </SelectItem>
   490→                ))}
   491→              </SelectContent>
   492→            </Select>
   493→          </div>
   494→        </div>
   495→        <DialogFooter>
   496→          <Button variant="outline" onClick={onClose} disabled={loading}>
   497→            Cancel
   498→          </Button>
   499→          <Button
   500→            onClick={handleLink}

   505→            Link Project

   507→        </DialogFooter>
   508→      </DialogContent>
   509→    </Dialog>
   510→  )
   511→}

   515→function DomainDetailView({
   516→  domainId,
   517→  onBack,
   518→  onRefresh,
   519→}: {
   520→  domainId: string
   521→  onBack: () => void
   522→  onRefresh: () => void
   523→}) {
   524→  const [domain, setDomain] = useState<Domain | null>(null)
   525→  const [loading, setLoading] = useState(true)
   526→  const [showAddDns, setShowAddDns] = useState(false)
   527→  const [showLinkProject, setShowLinkProject] = useState(false)
   528→  const [detailTab, setDetailTab] = useState('dns')
   529→
   530→  // SSL state
   531→  const [sslData, setSslData] = useState<SslCertificateApi | null>(null)
   532→  const [sslLoading, setSslLoading] = useState(true)
   533→  const [renewingSsl, setRenewingSsl] = useState(false)
   534→  const [showUploadCert, setShowUploadCert] = useState(false)
   535→  const [certPem, setCertPem] = useState('')
   536→  const [certKey, setCertKey] = useState('')
   537→  const [uploadingCert, setUploadingCert] = useState(false)
   538→
   539→  // DNS Propagation state
   540→  const [propagationData, setPropagationData] = useState<PropagationResponse | null>(null)
   541→  const [propagationLoading, setPropagationLoading] = useState(false)
   542→  const [expandedRecords, setExpandedRecords] = useState<Set<string>>(new Set())
   543→
   544→  const fetchDomain = useCallback(async () => {
   545→    try {
   546→      const res = await fetch(`/api/domains/${domainId}`, {
   547→        headers: getAuthHeaders(),
   548→      })
   549→      const data = await res.json()
   550→      if (!res.ok) {
   551→        toast.error(data.error || 'Failed to load domain')
   552→        onBack()
   553→        return
   554→      }
   555→      setDomain(data.domain)
   556→    } catch {
   557→      toast.error('Network error')
   558→      onBack()
   559→    } finally {
   560→      setLoading(false)
   561→    }
   562→  }, [domainId, onBack])
   563→
   564→  const fetchSslData = useCallback(async () => {
   565→    setSslLoading(true)
   566→    try {
   567→      const res = await fetch(`/api/domains/${domainId}/ssl`, {
   568→        headers: getAuthHeaders(),
   569→      })
   570→      const data = await res.json()
   571→      if (res.ok) {
   572→        setSslData(data.certificate)
   573→      }
   574→    } catch {
   575→      // Silently fail - will show fallback
   576→    } finally {
   577→      setSslLoading(false)
   578→    }
   579→  }, [domainId])
   580→
   581→  const fetchPropagationData = useCallback(async () => {
   582→    setPropagationLoading(true)
   583→    try {
   584→      const res = await fetch(`/api/domains/${domainId}/propagation`, {
   585→        headers: getAuthHeaders(),
   586→      })
   587→      const data = await res.json()
   588→      if (res.ok) {
   589→        setPropagationData(data)
   590→      }
   591→    } catch {
   592→      toast.error('Failed to check propagation')
   593→    } finally {
   594→      setPropagationLoading(false)
   595→    }
   596→  }, [domainId])
   597→
   598→  useEffect(() => {
   599→    setLoading(true)
   600→    fetchDomain()
   601→  }, [fetchDomain])
   602→
   603→  useEffect(() => {
   604→    if (domainId) {
   605→      fetchSslData()
   606→    }
   607→  }, [domainId, fetchSslData])
   608→
   609→  // SSL Renew handler
   610→  const handleRenewSsl = async () => {
   611→    setRenewingSsl(true)
   612→    try {
   613→      const res = await fetch(`/api/domains/${domainId}/ssl`, {
   614→        method: 'POST',
   615→        headers: getAuthHeaders(),
   616→        body: JSON.stringify({ action: 'renew' }),
   617→      })
   618→      const data = await res.json()
   619→      if (!res.ok) {
   620→        toast.error(data.error || 'Failed to renew certificate')
   621→        return
   622→      }
   623→      toast.success('Certificate renewal initiated')
   624→      // Poll for completion after 2 seconds
   625→      setTimeout(() => {
   626→        fetchSslData()
   627→      }, 2500)
   628→    } catch {
   629→      toast.error('Network error')
   630→    } finally {
   631→      setRenewingSsl(false)
   632→    }
   633→  }
   634→
   635→  // SSL Upload handler
   636→  const handleUploadCert = async () => {
   637→    if (!certPem.trim() || !certKey.trim()) {
   638→      toast.error('Certificate and private key are required')
   639→      return
   640→    }
   641→    setUploadingCert(true)
   642→    try {
   643→      const res = await fetch(`/api/domains/${domainId}/ssl`, {
   644→        method: 'POST',
   645→        headers: getAuthHeaders(),
   646→        body: JSON.stringify({
   647→          action: 'upload',
   648→          certificate: certPem,
   649→          privateKey: certKey,
   650→        }),
   651→      })
   652→      const data = await res.json()
   653→      if (!res.ok) {
   654→        toast.error(data.error || 'Failed to upload certificate')
   655→        return
   656→      }
   657→      toast.success('Custom certificate uploaded successfully')
   658→      setShowUploadCert(false)
   659→      setCertPem('')
   660→      setCertKey('')
   661→      fetchSslData()
   662→    } catch {
   663→      toast.error('Network error')
   664→    } finally {
   665→      setUploadingCert(false)
   666→    }
   667→  }
   668→
   669→  const handleDeleteDns = async (recordId: string) => {
   670→    if (!confirm('Delete this DNS record?')) return
   671→    try {
   672→      const res = await fetch(`/api/domains/${domainId}/dns/${recordId}`, {
   673→        method: 'DELETE',
