   171→function formatDate(dateStr: string): string {
   172→  const d = new Date(dateStr)
   173→  return d.toLocaleDateString('en-US', {
   174→    month: 'short',
   175→    day: 'numeric',
   176→    year: 'numeric',
   177→  })
   178→}
   179→
   180→function formatTime(dateStr: string): string {
   181→  const d = new Date(dateStr)
   182→  return d.toLocaleTimeString('en-US', {
   183→    hour: '2-digit',
   184→    minute: '2-digit',
   185→  })
   186→}
   187→
   188→function formatDuration(seconds: number | null | undefined): string {
   189→  if (seconds == null) return '—'
   190→  if (seconds < 60) return `${seconds}s`
   191→  const mins = Math.floor(seconds / 60)
   192→  const secs = seconds % 60
   193→  return `${mins}m ${secs}s`
   194→}

   196→function getStatusColor(status: string): string {
   197→  switch (status) {
   198→    case 'success':
   199→      return 'bg-emerald-500'
   200→    case 'failed':
   201→      return 'bg-red-500'
   202→    case 'running':
   203→      return 'bg-amber-500'
   204→    case 'cancelled':
   205→      return 'bg-gray-400'
   206→    case 'active':
   207→      return 'bg-emerald-500'
   208→    case 'paused':
   209→      return 'bg-amber-500'
   210→    case 'draft':
   211→      return 'bg-gray-400'
   212→    default:
   213→      return 'bg-gray-400'
   214→  }
   215→}

   321→function VisualStageEditor({
   322→  stages,
   323→  onUpdateStages,
   324→  readOnly,
   325→}: {
   326→  stages: PipelineStage[]
   327→  onUpdateStages: (stages: PipelineStage[]) => void
   328→  readOnly?: boolean
   329→}) {
   330→  const addStage = () => {
   331→    const newStage: PipelineStage = {
   332→      name: `Stage ${stages.length + 1}`,
   333→      type: 'custom',
   334→      config: { command: 'echo "Hello"' },
   335→      order: stages.length,
   336→    }
   337→    onUpdateStages([...stages, newStage])
   338→  }
   339→
   340→  const removeStage = (index: number) => {
   341→    const updated = stages.filter((_, i) => i !== index).map((s, i) => ({ ...s, order: i }))
   342→    onUpdateStages(updated)
   343→  }
   344→
   345→  const updateStage = (index: number, updates: Partial<PipelineStage>) => {
   346→    const updated = stages.map((s, i) => (i === index ? { ...s, ...updates } : s))
   347→    onUpdateStages(updated)
   348→  }
   349→
   350→  const moveStage = (index: number, direction: 'left' | 'right') => {
   351→    const newIndex = direction === 'left' ? index - 1 : index + 1
   352→    if (newIndex < 0 || newIndex >= stages.length) return
   353→    const updated = [...stages]
   354→    const temp = updated[index]
   355→    updated[index] = { ...updated[newIndex], order: index }
   356→    updated[newIndex] = { ...temp, order: newIndex }
   357→    onUpdateStages(updated)
   358→  }
   359→
   360→  const stageTypeColors: Record<string, string> = {
   361→    build: 'border-amber-300 bg-amber-50 dark:border-amber-700 dark:bg-amber-950/30',
   362→    test: 'border-blue-300 bg-blue-50 dark:border-blue-700 dark:bg-blue-950/30',
   363→    deploy: 'border-emerald-300 bg-emerald-50 dark:border-emerald-700 dark:bg-emerald-950/30',
   364→    custom: 'border-purple-300 bg-purple-50 dark:border-purple-700 dark:bg-purple-950/30',
   365→  }
   366→
   367→  const stageTypeAccents: Record<string, string> = {
   368→    build: 'text-amber-600 dark:text-amber-400',
   369→    test: 'text-blue-600 dark:text-blue-400',
   370→    deploy: 'text-emerald-600 dark:text-emerald-400',
   371→    custom: 'text-purple-600 dark:text-purple-400',
   372→  }
   373→
   374→  return (
   375→    <div className="space-y-4">
   376→      <div className="flex items-center justify-between">
   377→        <Label className="text-sm font-medium">Pipeline Stages</Label>
   378→        {!readOnly && (
   379→          <Button variant="outline" size="sm" onClick={addStage} className="h-7 text-xs">
   380→            <Plus className="size-3 mr-1" />
   381→            Add Stage
   382→          </Button>
   383→        )}
   384→      </div>
   385→
   386→      {stages.length === 0 ? (
   387→        <div className="flex flex-col items-center justify-center py-12 border-2 border-dashed rounded-lg">
   388→          <GitMerge className="size-10 text-muted-foreground/40 mb-3" />
   389→          <p className="text-sm text-muted-foreground mb-1">No stages configured</p>
   390→          <p className="text-xs text-muted-foreground/60 mb-4">
   391→            Add stages to define your pipeline workflow
   392→          </p>
   393→          {!readOnly && (
   394→            <Button variant="outline" size="sm" onClick={addStage}>
   395→              <Plus className="size-3.5 mr-1" />
   396→              Add First Stage
   397→            </Button>
   398→          )}
   399→        </div>
   400→      ) : (
   401→        <div className="overflow-x-auto pb-2">
   402→          <div className="flex items-start gap-0 min-w-max">
   403→            {stages.map((stage, index) => (
   404→              <div key={index} className="flex items-start">
   405→                {/* Stage Card */}
   406→                <div className={`relative w-56 rounded-lg border-2 p-4 transition-all hover:shadow-md ${stageTypeColors[stage.type] || stageTypeColors.custom}`}>
   407→                  {/* Stage header */}
   408→                  <div className="flex items-center justify-between mb-2">
   409→                    <div className="flex items-center gap-1.5">
   410→                      <StageTypeIcon type={stage.type} className={`size-4 ${stageTypeAccents[stage.type] || stageTypeAccents.custom}`} />
   411→                      {readOnly ? (
   412→                        <span className="text-sm font-semibold truncate max-w-[120px]">{stage.name}</span>
   413→                      ) : (
   414→                        <Input
   415→                          value={stage.name}
   416→                          onChange={(e) => updateStage(index, { name: e.target.value })}
   417→                          className="h-6 text-sm font-semibold border-0 bg-transparent p-0 focus-visible:ring-0 focus-visible:ring-offset-0 truncate max-w-[120px]"
   418→                        />
   419→                      )}
   420→                    </div>
   421→                    {!readOnly && (
   422→                      <div className="flex items-center gap-0.5">
   423→                        <Button
   424→                          variant="ghost"
   425→                          size="icon"
   426→                          className="size-6"
   427→                          onClick={() => moveStage(index, 'left')}
   428→                          disabled={index === 0}
   429→                        >
   430→                          <ChevronRight className="size-3 rotate-180" />
   431→                        </Button>
   432→                        <Button
   433→                          variant="ghost"
   434→                          size="icon"
   435→                          className="size-6"
   436→                          onClick={() => moveStage(index, 'right')}
   437→                          disabled={index === stages.length - 1}
   438→                        >
   439→                          <ChevronRight className="size-3" />
   440→                        </Button>
   441→                        <Button
   442→                          variant="ghost"
   443→                          size="icon"
   444→                          className="size-6 text-red-500 hover:text-red-600"
   445→                          onClick={() => removeStage(index)}
   446→                        >
   447→                          <Trash2 className="size-3" />
   448→                        </Button>
   449→                      </div>
   450→                    )}
   451→                  </div>
   452→
   453→                  {/* Stage type selector */}
   454→                  {!readOnly ? (
   455→                    <Select
   456→                      value={stage.type}
   457→                      onValueChange={(val) => updateStage(index, { type: val as PipelineStage['type'] })}
   458→                    >
   459→                      <SelectTrigger className="h-6 text-xs border-0 bg-white/60 dark:bg-black/20 w-full">
   460→                        <SelectValue />
   461→                      </SelectTrigger>
   462→                      <SelectContent>
   463→                        <SelectItem value="build">
   464→                          <span className="flex items-center gap-1.5">
   465→                            <Hammer className="size-3" /> Build
   466→                          </span>
   467→                        </SelectItem>
   468→                        <SelectItem value="test">
   469→                          <span className="flex items-center gap-1.5">
   470→                            <TestTube className="size-3" /> Test
   471→                          </span>
   472→                        </SelectItem>
   473→                        <SelectItem value="deploy">
   474→                          <span className="flex items-center gap-1.5">
   475→                            <Rocket className="size-3" /> Deploy
   476→                          </span>
   477→                        </SelectItem>
   478→                        <SelectItem value="custom">
   479→                          <span className="flex items-center gap-1.5">
   480→                            <Puzzle className="size-3" /> Custom
   481→                          </span>
   482→                        </SelectItem>
   483→                      </SelectContent>
   484→                    </Select>
   485→                  ) : (
   486→                    <Badge variant="outline" className="text-[10px] h-5">
   487→                      <StageTypeIcon type={stage.type} className="size-2.5 mr-1" />
   488→                      {stage.type}
   489→                    </Badge>
   490→                  )}
   491→
   492→                  {/* Config summary */}
   493→                  <div className="mt-2 text-[11px] text-muted-foreground font-mono truncate">
   494→                    {stage.config?.command
   495→                      ? String(stage.config.command)
   496→                      : stage.config?.script
   497→                        ? String(stage.config.script)
   498→                        : 'No command configured'}
   499→                  </div>
   500→
   501→                  {/* Command editor */}
   502→                  {!readOnly && (
   503→                    <div className="mt-2">
   504→                      <Input
   505→                        value={String(stage.config?.command || '')}
   506→                        onChange={(e) =>
   507→                          updateStage(index, {
   508→                            config: { ...stage.config, command: e.target.value },
   509→                          })
   510→                        }
   511→                        placeholder="Command to run"
   512→                        className="h-6 text-xs font-mono border-white/40 dark:border-white/10 bg-white/40 dark:bg-black/10"
   513→                      />
   514→                    </div>
   515→                  )}
   516→
   517→                  {/* Order badge */}
   518→                  <div className="absolute -top-2 -left-2 size-5 rounded-full bg-background border-2 flex items-center justify-center text-[10px] font-bold text-muted-foreground" style={{ borderColor: 'var(--border)' }}>
   519→                    {index + 1}
   520→                  </div>
   521→                </div>
   522→
   523→                {/* Connector Arrow */}
   524→                {index < stages.length - 1 && (
   525→                  <div className="flex items-center h-32 px-1">
   526→                    <div className="flex items-center">
   527→                      <div className="w-6 h-0.5 bg-border" />
   528→                      <ChevronRight className="size-4 text-muted-foreground -ml-1" />
   529→                    </div>
   530→                  </div>
   531→                )}
   532→              </div>
   533→            ))}
   534→
   535→            {/* Add stage button at the end */}
   536→            {!readOnly && stages.length > 0 && (
   537→              <div className="flex items-center h-32 pl-2">
   538→                <Button
   539→                  variant="outline"
   540→                  size="icon"
   541→                  className="size-8 rounded-full border-dashed"
   542→                  onClick={addStage}
   543→                >
   544→                  <Plus className="size-4" />
   545→                </Button>
   546→              </div>
   547→            )}
   548→          </div>
   549→        </div>
   550→      )}
   551→    </div>
   552→  )
   553→}

   555→// ── Run Timeline ─────────────────────────────────────────────────────
   556→
   557→function RunTimeline({ stageResults }: { stageResults: StageResult[] }) {
   558→  if (stageResults.length === 0) {
   559→    return (
   560→      <div className="text-sm text-muted-foreground text-center py-8">
   561→        No stage results available
   562→      </div>
   563→    )
   564→  }
   565→
   566→  return (
   567→    <div className="space-y-0">
   568→      {stageResults.map((stage, index) => {
   569→        const statusIcon =
   570→          stage.status === 'success' ? (
   571→            <CheckCircle2 className="size-5 text-emerald-500" />
   572→          ) : stage.status === 'failed' ? (
   573→            <XCircle className="size-5 text-red-500" />
   574→          ) : (
   575→            <Clock className="size-5 text-gray-400" />
   576→          )
   577→
   578→        const statusBg =
   579→          stage.status === 'success'
   580→            ? 'bg-emerald-50 dark:bg-emerald-950/20 border-emerald-200 dark:border-emerald-800'
   581→            : stage.status === 'failed'
   582→              ? 'bg-red-50 dark:bg-red-950/20 border-red-200 dark:border-red-800'
   583→              : 'bg-gray-50 dark:bg-gray-900/20 border-gray-200 dark:border-gray-800'
   584→
   585→        return (
   586→          <div key={index} className="flex gap-4">
   587→            {/* Timeline line + dot */}
   588→            <div className="flex flex-col items-center">
   589→              <div className="shrink-0 mt-1">{statusIcon}</div>
   590→              {index < stageResults.length - 1 && (
   591→                <div className="w-px flex-1 min-h-[24px] bg-border my-1" />
   592→              )}
   593→            </div>
   594→
   595→            {/* Stage content */}
   596→            <div className={`flex-1 rounded-lg border p-3 mb-2 ${statusBg}`}>
   597→              <div className="flex items-center justify-between mb-1">
   598→                <div className="flex items-center gap-2">
   599→                  <StageTypeIcon type={stage.type} className="size-3.5 text-muted-foreground" />
   600→                  <span className="text-sm font-medium">{stage.name}</span>
   601→                  <Badge variant="outline" className="text-[10px] h-4 px-1.5">
   602→                    {stage.type}
   603→                  </Badge>
   604→                </div>
   605→                <span className="text-xs text-muted-foreground font-mono">
   606→                  {formatDuration(stage.duration)}
   607→                </span>
   608→              </div>
   609→              {stage.output && (
