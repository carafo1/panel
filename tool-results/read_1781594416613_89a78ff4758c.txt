   171→   171→function formatDate(dateStr: string): string {
   172→   172→  const d = new Date(dateStr)
   173→   173→  return d.toLocaleDateString('en-US', {
   174→   174→    month: 'short',
   175→   175→    day: 'numeric',
   176→   176→    year: 'numeric',
   177→   177→  })
   178→   178→}
   179→   179→
   180→   180→function formatTime(dateStr: string): string {
   181→   181→  const d = new Date(dateStr)
   182→   182→  return d.toLocaleTimeString('en-US', {
   183→   183→    hour: '2-digit',
   184→   184→    minute: '2-digit',
   185→   185→  })
   186→   186→}
   187→   187→
   188→   188→function formatDuration(seconds: number | null | undefined): string {
   189→   189→  if (seconds == null) return '—'
   190→   190→  if (seconds < 60) return `${seconds}s`
   191→   191→  const mins = Math.floor(seconds / 60)
   192→   192→  const secs = seconds % 60
   193→   193→  return `${mins}m ${secs}s`
   194→   194→}

   321→   321→function VisualStageEditor({
   322→   322→  stages,
   323→   323→  onUpdateStages,
   324→   324→  readOnly,
   325→   325→}: {
   326→   326→  stages: PipelineStage[]
   327→   327→  onUpdateStages: (stages: PipelineStage[]) => void
   328→   328→  readOnly?: boolean
   329→   329→}) {
   330→   330→  const addStage = () => {
   331→   331→    const newStage: PipelineStage = {
   332→   332→      name: `Stage ${stages.length + 1}`,
   333→   333→      type: 'custom',
   334→   334→      config: { command: 'echo "Hello"' },
   335→   335→      order: stages.length,
   336→   336→    }
   337→   337→    onUpdateStages([...stages, newStage])
   338→   338→  }
   339→   339→
   340→   340→  const removeStage = (index: number) => {
   341→   341→    const updated = stages.filter((_, i) => i !== index).map((s, i) => ({ ...s, order: i }))
   342→   342→    onUpdateStages(updated)
   343→   343→  }
   344→   344→
   345→   345→  const updateStage = (index: number, updates: Partial<PipelineStage>) => {
   346→   346→    const updated = stages.map((s, i) => (i === index ? { ...s, ...updates } : s))
   347→   347→    onUpdateStages(updated)
   348→   348→  }
   349→   349→
   350→   350→  const moveStage = (index: number, direction: 'left' | 'right') => {
   351→   351→    const newIndex = direction === 'left' ? index - 1 : index + 1
   352→   352→    if (newIndex < 0 || newIndex >= stages.length) return
   353→   353→    const updated = [...stages]
   354→   354→    const temp = updated[index]
   355→   355→    updated[index] = { ...updated[newIndex], order: index }
   356→   356→    updated[newIndex] = { ...temp, order: newIndex }
   357→   357→    onUpdateStages(updated)
   358→   358→  }
   359→   359→
   360→   360→  const stageTypeColors: Record<string, string> = {
   361→   361→    build: 'border-amber-300 bg-amber-50 dark:border-amber-700 dark:bg-amber-950/30',
   362→   362→    test: 'border-blue-300 bg-blue-50 dark:border-blue-700 dark:bg-blue-950/30',
   363→   363→    deploy: 'border-emerald-300 bg-emerald-50 dark:border-emerald-700 dark:bg-emerald-950/30',
   364→   364→    custom: 'border-purple-300 bg-purple-50 dark:border-purple-700 dark:bg-purple-950/30',
   365→   365→  }
   366→   366→
   367→   367→  const stageTypeAccents: Record<string, string> = {
   368→   368→    build: 'text-amber-600 dark:text-amber-400',
   369→   369→    test: 'text-blue-600 dark:text-blue-400',
   370→   370→    deploy: 'text-emerald-600 dark:text-emerald-400',
   371→   371→    custom: 'text-purple-600 dark:text-purple-400',
   372→   372→  }
   373→   373→
   374→   374→  return (
   375→   375→    <div className="space-y-4">
   376→   376→      <div className="flex items-center justify-between">
   377→   377→        <Label className="text-sm font-medium">Pipeline Stages</Label>
   378→   378→        {!readOnly && (
   379→   379→          <Button variant="outline" size="sm" onClick={addStage} className="h-7 text-xs">
   380→   380→            <Plus className="size-3 mr-1" />
   381→   381→            Add Stage
   382→   382→          </Button>
   383→   383→        )}
   384→   384→      </div>
   385→   385→
   386→   386→      {stages.length === 0 ? (
   387→   387→        <div className="flex flex-col items-center justify-center py-12 border-2 border-dashed rounded-lg">
   388→   388→          <GitMerge className="size-10 text-muted-foreground/40 mb-3" />
   389→   389→          <p className="text-sm text-muted-foreground mb-1">No stages configured</p>
   390→   390→          <p className="text-xs text-muted-foreground/60 mb-4">
   391→   391→            Add stages to define your pipeline workflow
   392→   392→          </p>
   393→   393→          {!readOnly && (
   394→   394→            <Button variant="outline" size="sm" onClick={addStage}>
   395→   395→              <Plus className="size-3.5 mr-1" />
   396→   396→              Add First Stage
   397→   397→            </Button>
   398→   398→          )}
   399→   399→        </div>
   400→   400→      ) : (
   401→   401→        <div className="overflow-x-auto pb-2">
   402→   402→          <div className="flex items-start gap-0 min-w-max">
   403→   403→            {stages.map((stage, index) => (
   404→   404→              <div key={index} className="flex items-start">
   405→   405→                {/* Stage Card */}
   406→   406→                <div className={`relative w-56 rounded-lg border-2 p-4 transition-all hover:shadow-md ${stageTypeColors[stage.type] || stageTypeColors.custom}`}>
   407→   407→                  {/* Stage header */}
   408→   408→                  <div className="flex items-center justify-between mb-2">
   409→   409→                    <div className="flex items-center gap-1.5">
   410→   410→                      <StageTypeIcon type={stage.type} className={`size-4 ${stageTypeAccents[stage.type] || stageTypeAccents.custom}`} />
   411→   411→                      {readOnly ? (
   412→   412→                        <span className="text-sm font-semibold truncate max-w-[120px]">{stage.name}</span>
   413→   413→                      ) : (
   414→   414→                        <Input
   415→   415→                          value={stage.name}
   416→   416→                          onChange={(e) => updateStage(index, { name: e.target.value })}
   417→   417→                          className="h-6 text-sm font-semibold border-0 bg-transparent p-0 focus-visible:ring-0 focus-visible:ring-offset-0 truncate max-w-[120px]"
   418→   418→                        />
   419→   419→                      )}
   420→   420→                    </div>
   421→   421→                    {!readOnly && (
   422→   422→                      <div className="flex items-center gap-0.5">
   423→   423→                        <Button
   424→   424→                          variant="ghost"
   425→   425→                          size="icon"
   426→   426→                          className="size-6"
   427→   427→                          onClick={() => moveStage(index, 'left')}
   428→   428→                          disabled={index === 0}
   429→   429→                        >
   430→   430→                          <ChevronRight className="size-3 rotate-180" />
   431→   431→                        </Button>
   432→   432→                        <Button
   433→   433→                          variant="ghost"
   434→   434→                          size="icon"
   435→   435→                          className="size-6"
   436→   436→                          onClick={() => moveStage(index, 'right')}
   437→   437→                          disabled={index === stages.length - 1}
   438→   438→                        >
   439→   439→                          <ChevronRight className="size-3" />
   440→   440→                        </Button>
   441→   441→                        <Button
   442→   442→                          variant="ghost"
   443→   443→                          size="icon"
   444→   444→                          className="size-6 text-red-500 hover:text-red-600"
   445→   445→                          onClick={() => removeStage(index)}
   446→   446→                        >
   447→   447→                          <Trash2 className="size-3" />
   448→   448→                        </Button>
   449→   449→                      </div>
   450→   450→                    )}
   451→   451→                  </div>
   452→   452→
   453→   453→                  {/* Stage type selector */}
   454→   454→                  {!readOnly ? (
   455→   455→                    <Select
   456→   456→                      value={stage.type}
   457→   457→                      onValueChange={(val) => updateStage(index, { type: val as PipelineStage['type'] })}
   458→   458→                    >
   459→   459→                      <SelectTrigger className="h-6 text-xs border-0 bg-white/60 dark:bg-black/20 w-full">
   460→   460→                        <SelectValue />
   461→   461→                      </SelectTrigger>
   462→   462→                      <SelectContent>
   463→   463→                        <SelectItem value="build">
   464→   464→                          <span className="flex items-center gap-1.5">
   465→   465→                            <Hammer className="size-3" /> Build
   466→   466→                          </span>
   467→   467→                        </SelectItem>
   468→   468→                        <SelectItem value="test">
   469→   469→                          <span className="flex items-center gap-1.5">
   470→   470→                            <TestTube className="size-3" /> Test
   471→   471→                          </span>
   472→   472→                        </SelectItem>
   473→   473→                        <SelectItem value="deploy">
   474→   474→                          <span className="flex items-center gap-1.5">
   475→   475→                            <Rocket className="size-3" /> Deploy
   476→   476→                          </span>
   477→   477→                        </SelectItem>
   478→   478→                        <SelectItem value="custom">
   479→   479→                          <span className="flex items-center gap-1.5">
   480→   480→                            <Puzzle className="size-3" /> Custom
   481→   481→                          </span>
   482→   482→                        </SelectItem>
   483→   483→                      </SelectContent>
   484→   484→                    </Select>
   485→   485→                  ) : (
   486→   486→                    <Badge variant="outline" className="text-[10px] h-5">
   487→   487→                      <StageTypeIcon type={stage.type} className="size-2.5 mr-1" />
   488→   488→                      {stage.type}
   489→   489→                    </Badge>
   490→   490→                  )}
   491→   491→
   492→   492→                  {/* Config summary */}
   493→   493→                  <div className="mt-2 text-[11px] text-muted-foreground font-mono truncate">
   494→   494→                    {stage.config?.command
   495→   495→                      ? String(stage.config.command)
   496→   496→                      : stage.config?.script
   497→   497→                        ? String(stage.config.script)
   498→   498→                        : 'No command configured'}
   499→   499→                  </div>
   500→   500→
   501→   501→                  {/* Command editor */}
   502→   502→                  {!readOnly && (
   503→   503→                    <div className="mt-2">
   504→   504→                      <Input
   505→   505→                        value={String(stage.config?.command || '')}
