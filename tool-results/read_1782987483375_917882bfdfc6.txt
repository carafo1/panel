     1→     1→/**
     2→     2→ * Container Deployer
     3→     3→ * ==================
     4→     4→ *
     5→     5→ * Real Docker container deployment on a provisioned VPS:
     6→     6→ *
     7→     7→ *   1. Generate a Dockerfile + docker-compose.yml from the project's stack
     8→     8→ *      (nodejs, php, python, docker, wordpress) and any custom repo URL.
     9→     9→ *   2. SFTP-upload the project files to /opt/deployhub/projects/<slug>/
    10→    10→ *   3. Pull the repo (if GitHub-connected) OR write the generated files.
    11→    11→ *   4. Run `docker compose up -d --build` and capture logs.
    12→    12→ *   5. Register the container on the deployhub docker network (so Caddy can
    13→    13→ *      proxy to it via the container's service name).
    14→    14→ *   6. Persist containerId, containerName, containerPort, externalPort,
    15→    15→ *      deployPath, lastDeployedAt back to the Project row.
    16→    16→ *   7. Trigger the reverse-proxy manager to add the Caddyfile entry.
    17→    17→ *
    18→    18→ * In DEMO MODE (VPS_DEMO_MODE=1 or no server.sshHost), the deployer returns
    19→    19→ * a realistic simulated success — the Project row is updated with a fake
    20→    20→ * containerId + simulated build logs, so the dashboard UX is intact.
    21→    21→ */
    22→    22→import { db } from '@/lib/db'
    23→    23→import { STACK_DEFINITIONS, getStackDefinition } from '@/lib/stacks'

    26→    26→  exec,
    27→    27→  execStream,
    28→    28→  putFile,
    29→    29→  isVpsDemoMode,
    30→    30→  isServerInDemoMode,
    31→    31→  type ExecResult,

    33→    33→} from './ssh'
    34→    34→import { getServerCredentials } from './provisioner'
    35→    35→import { configureProjectProxy, removeProjectProxy } from './proxy'

    37→    37→// ─── Types ───────────────────────────────────────────────────────────────
    38→    38→
    39→    39→export interface DeployRequest {
    40→    40→  projectId: string
    41→    41→  deploymentId: string
    42→    42→  onLog?: (msg: string) => void
    43→    43→  onStreamChunk?: (chunk: StreamChunk) => void
    44→    44→}

    48→    48→  projectId: string
    49→    49→  deploymentId: string

    51→    51→  containerName?: string
    52→    52→  containerPort?: number
    53→    53→  externalPort?: number
    54→    54→  deployPath?: string
    55→    55→  buildLog: string
    56→    56→  durationMs: number
    57→    57→  error?: string
    58→    58→}

    60→    60→export interface ContainerStatus {

    62→    62→  name: string
    63→    63→  state: 'running' | 'exited' | 'paused' | 'restarting' | 'not_found'
    64→    64→  status: string
    65→    65→  image: string
    66→    66→  ports: string[]
    67→    67→  createdAt: string

    70→    70→// ─── Main deploy function ────────────────────────────────────────────────
    71→    71→
    72→    72→export async function deployProject(req: DeployRequest): Promise<DeployResult> {
    73→    73→  const start = Date.now()
    74→    74→  const { projectId, deploymentId } = req
    75→    75→  const log = (m: string) => req.onLog?.(m)
    76→    76→
    77→    77→  const project = await db.project.findUnique({
    78→    78→    where: { id: projectId },
    79→    79→    include: { server: true },
    80→    80→  })
    81→    81→  if (!project) throw new Error(`Project ${projectId} not found`)
    82→    82→  if (!project.serverId || !project.server) {
    83→    83→    throw new Error('Project has no server assigned')
    84→    84→  }
    85→    85→
    86→    86→  const stack = getStackDefinition(project.stack)
    87→    87→  if (!stack) throw new Error(`Unknown stack: ${project.stack}`)
    88→    88→
    89→    89→  const deployPath = project.deployPath || `/opt/deployhub/projects/${project.slug}`
    90→    90→  const containerName = project.containerName || `deployhub-${project.slug}`
    91→    91→  const containerPort = project.containerPort || stack.port
    92→    92→  const externalPort = project.externalPort || await pickExternalPort(project.serverId)
    93→    93→
    94→    94→  log(`Deploying ${project.name} (${project.stack}) to ${project.server.name}`)
    95→    95→  log(`  Path: ${deployPath}`)
    96→    96→  log(`  Container: ${containerName} (internal ${containerPort} → host ${externalPort})`)
    97→    97→
    98→    98→  // ── DEMO MODE: simulated deploy with realistic build log ────────────
    99→    99→  // Demo mode is active when: VPS_DEMO_MODE env is set, OR the server has
   100→   100→  // no sshHost, OR the server has no SSH credentials (no key AND no password).
   101→   101→  if (isServerInDemoMode(project.server)) {
   102→   102→    const buildLog = generateDemoBuildLog(project, stack.name, containerName)
   103→   103→    log('[demo mode] Simulating deploy…')
   104→   104→    for (const line of buildLog.split('\n')) {
   105→   105→      req.onStreamChunk?.({ stream: 'stdout', data: line + '\n' })
   106→   106→      await sleep(150)
   107→   107→    }
   108→   108→    const fakeContainerId = generateFakeContainerId()
   109→   109→
   110→   110→    await db.project.update({
   111→   111→      where: { id: projectId },
   112→   112→      data: {
   113→   113→        status: 'running',
   114→   114→        containerId: fakeContainerId,
   115→   115→        containerName,
   116→   116→        containerPort,
   117→   117→        externalPort,
   118→   118→        deployPath,
   119→   119→        lastDeployedAt: new Date(),
   120→   120→        sslCertStatus: project.sslEnabled ? 'issued' : 'none',
   121→   121→        sslCertExpiry: project.sslEnabled
   122→   122→          ? new Date(Date.now() + 90 * 24 * 60 * 60 * 1000)
   123→   123→          : null,
   124→   124→      },
   125→   125→    })
   126→   126→
   127→   127→    await db.deployment.update({
   128→   128→      where: { id: deploymentId },
   129→   129→      data: {
   130→   130→        status: 'live',
   131→   131→        buildLog,
   132→   132→        deployUrl: `https://${project.customDomain || project.subdomain}`,
   133→   133→        duration: Math.round((Date.now() - start) / 1000),
   134→   134→      },
   135→   135→    })
   136→   136→
   137→   137→    log('[demo mode] Deploy simulated successfully')
   138→   138→    return {
   139→   139→      ok: true,
   140→   140→      projectId,
   141→   141→      deploymentId,
   142→   142→      containerId: fakeContainerId,
   143→   143→      containerName,
   144→   144→      containerPort,
   145→   145→      externalPort,
   146→   146→      deployPath,
   147→   147→      buildLog,
   148→   148→      durationMs: Date.now() - start,
   149→   149→    }
   150→   150→  }
   151→   151→
   152→   152→  // ── REAL deploy ─────────────────────────────────────────────────────
   153→   153→  const creds = await getServerCredentials(project.serverId)
   154→   154→  let buildLog = ''
   155→   155→  let containerId: string | undefined
   156→   156→
   157→   157→  try {
   158→   158→    await withConnection(creds, async (client) => {
   159→   159→      // 1. Create project directory
   160→   160→      log('Creating project directory…')
   161→   161→      await exec(
   162→   162→        client,
   163→   163→        `mkdir -p ${deployPath}`,
   164→   164→        { timeoutMs: 10000 }
   165→   165→      )
   166→   166→
   167→   167→      // 2. Either clone the repo OR write a placeholder app
   168→   168→      if (project.repoUrl) {
   169→   169→        log(`Cloning ${project.repoUrl}…`)
   170→   170→        const clone = await exec(
   171→   171→          client,
   172→   172→          `if [ -d ${deployPath}/.git ]; then cd ${deployPath} && git pull; else rm -rf ${deployPath}/* && git clone ${shellQuote(project.repoUrl)} ${deployPath}; fi`,
   173→   173→          { timeoutMs: 120000 }
   174→   174→        )
   175→   175→        buildLog += `[git] ${clone.stdout}\n`
   176→   176→        if (clone.exitCode !== 0) {
   177→   177→          throw new Error(`git clone failed: ${clone.stderr}`)
   178→   178→        }
   179→   179→      } else {
   180→   180→        log('No repo URL — writing placeholder app from stack template…')
   181→   181→        const placeholder = generatePlaceholderApp(project.stack, project.name)
   182→   182→        for (const [filename, content] of Object.entries(placeholder)) {
   183→   183→          await putFile(client, `${deployPath}/${filename}`, content, { mode: 0o644 })
   184→   184→        }
   185→   185→        buildLog += `[placeholder] Wrote ${Object.keys(placeholder).length} files\n`
   186→   186→      }
   187→   187→
   188→   188→      // 3. Write Dockerfile if stack defines one
   189→   189→      if (stack.dockerfile && project.stack !== 'docker') {
   190→   190→        log('Writing Dockerfile…')
   191→   191→        await putFile(client, `${deployPath}/Dockerfile`, stack.dockerfile, {
   192→   192→          mode: 0o644,
   193→   193→        })
   194→   194→        buildLog += `[dockerfile] Written (${stack.dockerfile.length} bytes)\n`
   195→   195→      }
   196→   196→
   197→   197→      // 4. Write docker-compose.yml
   198→   198→      log('Writing docker-compose.yml…')
   199→   199→      const composeYml = generateDockerCompose({
   200→   200→        projectName: project.slug,
   201→   201→        containerName,
   202→   202→        deployPath,
   203→   203→        containerPort,
   204→   204→        externalPort,
   205→   205→        stack: project.stack,
   206→   206→        envVars: await getProjectEnvVars(projectId),
   207→   207→      })
   208→   208→      await putFile(client, `${deployPath}/docker-compose.yml`, composeYml, {
   209→   209→        mode: 0o644,
   210→   210→      })
   211→   211→      buildLog += `[compose] Written docker-compose.yml\n`
   212→   212→
   213→   213→      // 5. docker compose build + up (stream logs live)
   214→   214→      log('Running docker compose build…')
   215→   215→      const buildGen = execStream(
   216→   216→        client,
   217→   217→        `cd ${deployPath} && docker compose build --progress=plain 2>&1`,
   218→   218→        { timeoutMs: 600000 }
   219→   219→      )
   220→   220→      for await (const chunk of buildGen) {
   221→   221→        buildLog += chunk.data
   222→   222→        req.onStreamChunk?.(chunk)
   223→   223→      }
   224→   224→      const buildResult = await buildGen.next().then(() => buildGen).catch(() => null)
   225→   225→      log(`Build finished (exit code from stream closed)`)
   226→   226→
   227→   227→      log('Starting container with docker compose up -d…')
   228→   228→      const upGen = execStream(
   229→   229→        client,
   230→   230→        `cd ${deployPath} && docker compose up -d 2>&1`,
   231→   231→        { timeoutMs: 120000 }
   232→   232→      )
   233→   233→      for await (const chunk of upGen) {
   234→   234→        buildLog += chunk.data
   235→   235→        req.onStreamChunk?.(chunk)
   236→   236→      }
   237→   237→
   238→   238→      // 6. Connect container to the deployhub network so Caddy can reach it
   239→   239→      log('Connecting container to deployhub network…')
   240→   240→      const net = await exec(
   241→   241→        client,
   242→   242→        `docker network connect deployhub ${containerName} 2>/dev/null || true`,
   243→   243→        { timeoutMs: 15000 }
   244→   244→      )
   245→   245→      buildLog += `[network] ${net.stdout || 'already connected'}\n`
   246→   246→
   247→   247→      // 7. Get container ID + verify it's running
   248→   248→      log('Verifying container is running…')
   249→   249→      const inspect = await exec(
   250→   250→        client,
   251→   251→        `docker inspect --format '{{.Id}} {{.State.Status}}' ${containerName}`,
   252→   252→        { timeoutMs: 15000 }
   253→   253→      )
   254→   254→      const [cid, state] = inspect.stdout.trim().split(' ')
   255→   255→      if (!cid || state !== 'running') {
   256→   256→        throw new Error(
   257→   257→          `Container did not start properly (state=${state}). Logs:\n` +
   258→   258→            (await exec(client, `docker logs ${containerName} --tail 50 2>&1`, {
   259→   259→              timeoutMs: 15000,
   260→   260→            })).stdout
   261→   261→        )
   262→   262→      }
   263→   263→      containerId = cid
   264→   264→      log(`Container running: ${cid.slice(0, 12)}`)
   265→   265→
   266→   266→      // 8. Update Caddy reverse proxy config (auto-HTTPS)
   267→   267→      log('Configuring Caddy reverse proxy…')
   268→   268→      await configureProjectProxy(project.id, { onLog: log })
   269→   269→      log('Caddy reloaded with new site config')
   270→   270→    })
   271→   271→
   272→   272→    // 9. Persist project + deployment state
   273→   273→    await db.project.update({
   274→   274→      where: { id: projectId },
   275→   275→      data: {
   276→   276→        status: 'running',
   277→   277→        containerId,
   278→   278→        containerName,
   279→   279→        containerPort,
   280→   280→        externalPort,
   281→   281→        deployPath,
   282→   282→        lastDeployedAt: new Date(),
   283→   283→        sslCertStatus: project.sslEnabled ? 'issued' : 'none',
   284→   284→        sslCertExpiry: project.sslEnabled
   285→   285→          ? new Date(Date.now() + 90 * 24 * 60 * 60 * 1000)
   286→   286→          : null,
   287→   287→      },
   288→   288→    })
   289→   289→
   290→   290→    await db.deployment.update({
   291→   291→      where: { id: deploymentId },
   292→   292→      data: {
   293→   293→        status: 'live',
   294→   294→        buildLog,
   295→   295→        deployUrl: `https://${project.customDomain || project.subdomain}`,
   296→   296→        duration: Math.round((Date.now() - start) / 1000),
   297→   297→      },
   298→   298→    })
   299→   299→
   300→   300→    log('Deploy complete!')
   301→   301→    return {
   302→   302→      ok: true,
   303→   303→      projectId,
   304→   304→      deploymentId,
   305→   305→      containerId,
   306→   306→      containerName,
   307→   307→      containerPort,
   308→   308→      externalPort,
   309→   309→      deployPath,
   310→   310→      buildLog,
   311→   311→      durationMs: Date.now() - start,
   312→   312→    }
   313→   313→  } catch (err) {
   314→   314→    const message = err instanceof Error ? err.message : String(err)
   315→   315→    log(`DEPLOY FAILED: ${message}`)
   316→   316→    buildLog += `\n[FATAL] ${message}\n`
   317→   317→
   318→   318→    await db.project.update({
   319→   319→      where: { id: projectId },
   320→   320→      data: { status: 'error' },
   321→   321→    })
   322→   322→    await db.deployment.update({
   323→   323→      where: { id: deploymentId },
   324→   324→      data: { status: 'failed', buildLog, duration: Math.round((Date.now() - start) / 1000) },
   325→   325→    })
   326→   326→
   327→   327→    return {
   328→   328→      ok: false,
   329→   329→      projectId,
   330→   330→      deploymentId,
   331→   331→      containerId,
   332→   332→      containerName,
   333→   333→      containerPort,
   334→   334→      externalPort,
   335→   335→      deployPath,
   336→   336→      buildLog,
   337→   337→      durationMs: Date.now() - start,
   338→   338→      error: message,
   339→   339→    }
   340→   340→  }
   341→   341→}

   345→   345→export async function stopProject(projectId: string): Promise<{ ok: boolean; message: string }> {
   346→   346→  const project = await db.project.findUnique({ where: { id: projectId }, include: { server: true } })
   347→   347→  if (!project || !project.containerName) {
   348→   348→    return { ok: false, message: 'Project has no running container' }
   349→   349→  }
   350→   350→  if (isServerInDemoMode(project.server)) {
   351→   351→    await db.project.update({ where: { id: projectId }, data: { status: 'stopped' } })
   352→   352→    return { ok: true, message: 'Demo mode: container stopped (simulated)' }
   353→   353→  }
   354→   354→  try {
   355→   355→    const creds = await getServerCredentials(project.serverId!)
   356→   356→    await withConnection(creds, async (client) => {
   357→   357→      await exec(client, `cd ${project.deployPath} && docker compose down 2>&1`, {
   358→   358→        timeoutMs: 60000,
   359→   359→      })
   360→   360→    })
   361→   361→    await db.project.update({ where: { id: projectId }, data: { status: 'stopped' } })
   362→   362→    return { ok: true, message: 'Container stopped' }
   363→   363→  } catch (err) {
   364→   364→    return {
   365→   365→      ok: false,
   366→   366→      message: err instanceof Error ? err.message : String(err),
   367→   367→    }
   368→   368→  }
   369→   369→}

   395→   395→export async function removeProjectContainer(projectId: string): Promise<{ ok: boolean; message: string }> {
   396→   396→  const project = await db.project.findUnique({ where: { id: projectId }, include: { server: true } })
   397→   397→  if (!project) return { ok: false, message: 'Project not found' }
   398→   398→
   399→   399→  // Remove Caddy config first
   400→   400→  await removeProjectProxy(projectId).catch(() => {})
   401→   401→
   402→   402→  if (!isServerInDemoMode(project.server) && project.containerName) {
   403→   403→    try {
   404→   404→      const creds = await getServerCredentials(project.serverId!)
   405→   405→      await withConnection(creds, async (client) => {
   406→   406→        await exec(
   407→   407→          client,
   408→   408→          `cd ${project.deployPath} && docker compose down --rmi local -v 2>&1 || true`,
   409→   409→          { timeoutMs: 60000 }
   410→   410→        )
   411→   411→        await exec(client, `rm -rf ${project.deployPath}`, { timeoutMs: 30000 })
   412→   412→      })
   413→   413→    } catch (err) {
   414→   414→      console.error('[vps-deploy] remove container error:', err)
   415→   415→    }
   416→   416→  }
   417→   417→
   418→   418→  await db.project.update({
   419→   419→    where: { id: projectId },
   420→   420→    data: {
   421→   421→      status: 'stopped',
   422→   422→      containerId: null,
   423→   423→      containerName: null,
   424→   424→      containerPort: null,
   425→   425→      externalPort: null,
   426→   426→      deployPath: null,
   427→   427→      sslCertStatus: 'none',
   428→   428→      sslCertExpiry: null,
   429→   429→    },
   430→   430→  })
   431→   431→  return { ok: true, message: 'Container removed' }
   432→   432→}

   436→   436→export async function getContainerStatus(projectId: string): Promise<ContainerStatus | null> {
   437→   437→  const project = await db.project.findUnique({ where: { id: projectId }, include: { server: true } })
   438→   438→  if (!project || !project.containerName) return null
   439→   439→
   440→   440→  if (isServerInDemoMode(project.server)) {
   441→   441→    return {
   442→   442→      containerId: project.containerId || 'demo',
   443→   443→      name: project.containerName,
   444→   444→      state: project.status === 'running' ? 'running' : 'exited',
   445→   445→      status: project.status === 'running' ? 'Up (simulated)' : 'Exited (simulated)',
   446→   446→      image: `deployhub/${project.stack}:latest`,
   447→   447→      ports: [`${project.externalPort}->${project.containerPort}`],
   448→   448→      createdAt: project.lastDeployedAt?.toISOString() || new Date().toISOString(),
   449→   449→    }
   450→   450→  }
   451→   451→
   452→   452→  try {
   453→   453→    const creds = await getServerCredentials(project.serverId!)
   454→   454→    return await withConnection(creds, async (client) => {
   455→   455→      const r = await exec(
   456→   456→        client,
   457→   457→        `docker inspect --format '{{.Id}}|{{.Name}}|{{.State.Status}}|{{.State.Health.Status}}|{{.Config.Image}}|{{range $p, $conf := .NetworkSettings.Ports}}{{$p}}->{{(index $conf 0).HostPort}} {{end}}|{{.Created}}' ${project.containerName} 2>/dev/null`,
   458→   458→        { timeoutMs: 10000 }
   459→   459→      )
   460→   460→      if (r.exitCode !== 0) {
   461→   461→        return {
   462→   462→          containerId: project.containerId || '',
   463→   463→          name: project.containerName,
   464→   464→          state: 'not_found' as const,
   465→   465→          status: 'Container not found on server',
   466→   466→          image: '',
   467→   467→          ports: [],
   468→   468→          createdAt: '',
   469→   469→        }
   470→   470→      }
   471→   471→      const parts = r.stdout.trim().split('|')
   472→   472→      const [id, name, state, health, image, ports, created] = parts
   473→   473→      return {
   474→   474→        containerId: id,
   475→   475→        name: name.replace(/^\//, ''),
   476→   476→        state: state as ContainerStatus['state'],
   477→   477→        status: health && health !== '<no value>' ? `${state} (${health})` : state,
   478→   478→        image,
   479→   479→        ports: ports ? ports.trim().split(' ').filter(Boolean) : [],
   480→   480→        createdAt: created || '',
   481→   481→      }
   482→   482→    })
   483→   483→  } catch (err) {

   491→   491→export async function getContainerLogs(
   492→   492→  projectId: string,
   493→   493→  opts: { tail?: number; since?: string } = {}
   494→   494→): Promise<string> {
   495→   495→  const project = await db.project.findUnique({ where: { id: projectId }, include: { server: true } })
   496→   496→  if (!project || !project.containerName) return 'No container running'
   497→   497→
   498→   498→  if (isServerInDemoMode(project.server)) {
   499→   499→    return generateDemoRuntimeLog(project)
   500→   500→  }
   501→   501→
   502→   502→  try {
   503→   503→    const creds = await getServerCredentials(project.serverId!)
   504→   504→    return await withConnection(creds, async (client) => {
   505→   505→      const tail = opts.tail ?? 200
   506→   506→      const sinceFlag = opts.since ? `--since ${shellQuote(opts.since)}` : ''
   507→   507→      const r = await exec(
   508→   508→        client,
   509→   509→        `docker logs ${project.containerName} --tail ${tail} ${sinceFlag} 2>&1`,
   510→   510→        { timeoutMs: 30000 }
   511→   511→      )
   512→   512→      return r.stdout || r.stderr
   513→   513→    })
   514→   514→  } catch (err) {
   515→   515→    return `Failed to fetch logs: ${err instanceof Error ? err.message : err}`
   516→   516→  }
   517→   517→}

   521→   521→async function pickExternalPort(serverId: string): Promise<number> {
   522→   522→  // Find the highest externalPort in use on this server + 1, starting from 30000.
   523→   523→  const projects = await db.project.findMany({
   524→   524→    where: { serverId, externalPort: { not: null } },
   525→   525→    select: { externalPort: true },
   526→   526→  })
   527→   527→  const usedPorts = new Set(projects.map((p) => p.externalPort!))
   528→   528→  for (let port = 30000; port < 40000; port++) {
   529→   529→    if (!usedPorts.has(port)) return port
   530→   530→  }
   531→   531→  return 30000 + Math.floor(Math.random() * 9999)
   532→   532→}

   534→   534→async function getProjectEnvVars(projectId: string): Promise<Record<string, string>> {
   535→   535→  const envVars = await db.environmentVariable.findMany({
   536→   536→    where: { projectId },
   537→   537→    select: { key: true, value: true, encrypted: true },
   538→   538→  })
   539→   539→  // Note: env var values are encrypted at rest; decrypt here. The actual
   540→   540→  // decryption is handled by the env var API; for compose we re-read the
   541→   541→  // raw value via the encryption module.
   542→   542→  const { decrypt } = await import('@/lib/encryption')
   543→   543→  const out: Record<string, string> = {}
   544→   544→  for (const v of envVars) {
   545→   545→    try {
   546→   546→      out[v.key] = v.encrypted ? decrypt(v.value) : v.value
   547→   547→    } catch {
   548→   548→      out[v.key] = v.value
   549→   549→    }
   550→   550→  }
   551→   551→  return out
   552→   552→}
