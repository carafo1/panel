   140→function formatDate(dateStr?: string) {
   141→  if (!dateStr) return '-'
   142→  try {
   143→    return new Date(dateStr).toLocaleDateString('en-US', {
   144→      month: 'short',
   145→      day: 'numeric',
   146→      year: 'numeric',
   147→    })
   148→  } catch {
   149→    return dateStr
   150→  }
   151→}

   153→function formatTTL(ttl: number) {
   154→  if (ttl >= 3600) return `${ttl / 3600}h`
   155→  if (ttl >= 60) return `${ttl / 60}m`
   156→  return `${ttl}s`
   157→}

   204→// Simulated SSL data per domain (deterministic based on domain name)
   205→function getSimulatedSslData(domainName: string): SslCertificate {
   206→  const hash = domainName.split('').reduce((acc, c) => acc + c.charCodeAt(0), 0)
   207→  const statuses: SslCertificate['status'][] = ['active', 'active', 'active', 'expired', 'none']
   208→  const status = statuses[hash % statuses.length]
   209→
   210→  if (status === 'none') {
   211→    return {
   212→      status: 'none',
   213→      type: "Let's Encrypt",
   214→      issuer: '-',
   215→      validFrom: '-',
   216→      validTo: '-',
   217→      autoRenew: false,
   218→      domain: domainName,
   219→    }
   220→  }
   221→
   222→  const now = new Date()
   223→  const validFrom = new Date(now)
   224→  validFrom.setMonth(validFrom.getMonth() - 3)
   225→  const validTo = new Date(now)
   226→  if (status === 'expired') {
   227→    validTo.setMonth(validTo.getMonth() - 1)
   228→  } else {
   229→    validTo.setMonth(validTo.getMonth() + 9)
   230→  }
   231→
   232→  return {
   233→    status,
   234→    type: hash % 3 === 0 ? 'Custom' : "Let's Encrypt",
   235→    issuer: hash % 3 === 0 ? 'DigiCert Inc.' : "Let's Encrypt Authority X3",
   236→    validFrom: validFrom.toISOString(),
   237→    validTo: validTo.toISOString(),
   238→    autoRenew: hash % 3 !== 0,
   239→    domain: domainName,
   240→    fingerprint: Array.from({ length: 8 }, (_, i) => ((hash * (i + 1)) % 256).toString(16).padStart(2, '0')).join(':'),
   241→  }
   242→}

   485→function DomainDetailView({
   486→  domainId,
   487→  onBack,
   488→  onRefresh,
   489→}: {
   490→  domainId: string
   491→  onBack: () => void
   492→  onRefresh: () => void
   493→}) {
   494→  const [domain, setDomain] = useState<Domain | null>(null)
   495→  const [loading, setLoading] = useState(true)
   496→  const [showAddDns, setShowAddDns] = useState(false)
   497→  const [showLinkProject, setShowLinkProject] = useState(false)
   498→
   499→  // DNS Propagation state
   500→  const [propagationServers, setPropagationServers] = useState<DnsPropagationServer[]>([])
   501→  const [checkingPropagation, setCheckingPropagation] = useState(false)
   502→  const [propagationChecked, setPropagationChecked] = useState(false)
   503→  const propagationTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   504→
   505→  // SSL state
   506→  const [sslData, setSslData] = useState<SslCertificate | null>(null)
   507→  const [renewingSsl, setRenewingSsl] = useState(false)
   508→  const [showUploadCert, setShowUploadCert] = useState(false)
   509→  const [certPem, setCertPem] = useState('')
   510→  const [certKey, setCertKey] = useState('')
   511→  const [uploadingCert, setUploadingCert] = useState(false)
   512→
   513→  const fetchDomain = useCallback(async () => {
   514→    try {
   515→      const res = await fetch(`/api/domains/${domainId}`, {
   516→        headers: getAuthHeaders(),
   517→      })
   518→      const data = await res.json()
   519→      if (!res.ok) {
   520→        toast.error(data.error || 'Failed to load domain')
   521→        onBack()
   522→        return
   523→      }
   524→      setDomain(data.domain)
   525→    } catch {
   526→      toast.error('Network error')
   527→      onBack()
   528→    } finally {
   529→      setLoading(false)
   530→    }
   531→  }, [domainId, onBack])
   532→
   533→  useEffect(() => {
   534→    setLoading(true)
   535→    fetchDomain()
   536→  }, [fetchDomain])
   537→
   538→  // Initialize SSL data when domain loads
   539→  useEffect(() => {
   540→    if (domain) {
   541→      setSslData(getSimulatedSslData(domain.name))
   542→    }
   543→  }, [domain])
   544→
   545→  // Cleanup propagation timers
   546→  useEffect(() => {
   547→    return () => {
   548→      if (propagationTimerRef.current) {
   549→        clearTimeout(propagationTimerRef.current)
   550→      }
   551→    }
   552→  }, [])
   553→
   554→  // DNS Propagation check handler
   555→  const handleCheckPropagation = useCallback(() => {
   556→    if (!domain) return
   557→    setCheckingPropagation(true)
   558→    setPropagationChecked(false)
   559→
   560→    // Reset all servers to pending
   561→    const servers: DnsPropagationServer[] = DNS_SERVERS.map((s) => ({
   562→      ...s,
   563→      status: 'pending' as const,
   564→    }))
   565→    setPropagationServers(servers)
   566→
   567→    // Generate simulated results based on domain name hash
   568→    const hash = domain.name.split('').reduce((acc, c) => acc + c.charCodeAt(0), 0)
   569→    const simulatedResults: { status: 'propagated' | 'not-found'; latency: number; ip: string }[] = DNS_SERVERS.map((_, i) => {
   570→      const seed = (hash * (i + 7)) % 100
   571→      const propagated = seed > 20 // 80% chance of propagated
   572→      return {
   573→        status: propagated ? 'propagated' : 'not-found',
   574→        latency: propagated ? Math.floor(15 + (seed % 120)) : 0,
   575→        ip: propagated ? `${192 + (seed % 4)}.${168 + (seed % 16)}.${seed % 256}.${1 + (seed % 254)}` : '',
   576→      }
   577→    })
   578→
   579→    // Animate through servers sequentially
   580→    let currentIndex = 0
   581→    const checkNext = () => {
   582→      if (currentIndex >= servers.length) {
   583→        setCheckingPropagation(false)
   584→        setPropagationChecked(true)
   585→        return
   586→      }
   587→
   588→      // Set current to "checking"
   589→      setPropagationServers((prev) =>
   590→        prev.map((s, idx) => (idx === currentIndex ? { ...s, status: 'checking' as const } : s))
   591→      )
   592→
   593→      const delay = 400 + Math.random() * 600
   594→      propagationTimerRef.current = setTimeout(() => {
   595→        const result = simulatedResults[currentIndex]
   596→        setPropagationServers((prev) =>
   597→          prev.map((s, idx) =>
   598→            idx === currentIndex
   599→              ? { ...s, status: result.status, latency: result.latency, resolvedIp: result.ip }
   600→              : s
   601→          )
   602→        )
   603→        currentIndex++
   604→        checkNext()
   605→      }, delay)
   606→    }
   607→
   608→    // Start checking after a brief delay
   609→    propagationTimerRef.current = setTimeout(checkNext, 300)
   610→  }, [domain])
   611→
   612→  // Compute propagation percentage
   613→  const propagationPercentage =
   614→    propagationServers.length > 0
   615→      ? Math.round(
   616→          (propagationServers.filter((s) => s.status === 'propagated').length / propagationServers.length) * 100
   617→        )
   618→      : 0
   619→
   620→  // SSL Renew handler
   621→  const handleRenewSsl = async () => {
   622→    setRenewingSsl(true)
   623→    // Simulate renewal
   624→    await new Promise((resolve) => setTimeout(resolve, 2000))
   625→    if (sslData) {
   626→      const now = new Date()
   627→      const validFrom = new Date(now)
   628→      const validTo = new Date(now)
   629→      validTo.setMonth(validTo.getMonth() + 3)
   630→      setSslData({
