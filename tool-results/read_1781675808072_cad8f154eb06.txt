   287→function timeAgo(dateStr: string): string {
   288→  const seconds = Math.floor((Date.now() - new Date(dateStr).getTime()) / 1000)
   289→  if (seconds < 60) return 'just now'
   290→  const minutes = Math.floor(seconds / 60)
   291→  if (minutes < 60) return `${minutes}m ago`
   292→  const hours = Math.floor(minutes / 60)
   293→  if (hours < 24) return `${hours}h ago`
   294→  const days = Math.floor(hours / 24)
   295→  return `${days}d ago`
   296→}

   298→function formatDate(dateStr: string) {
   299→  try {
   300→    return new Date(dateStr).toLocaleDateString('en-US', {
   301→      month: 'short',
   302→      day: 'numeric',
   303→      year: 'numeric',
   304→      hour: '2-digit',
   305→      minute: '2-digit',
   306→    })
   307→  } catch {
   308→    return dateStr
   309→  }
   310→}

   312→function generateSimulatedBuildLog(project: ProjectDetail, deployment: Deployment): string {
   313→  const isFailed = deployment.status === 'failed'
   314→  const fw = project.framework === 'nextjs' ? 'next' : project.framework
   315→  const now = new Date(deployment.createdAt)
   316→  const pad = (n: number) => String(n).padStart(2, '0')
   317→  const ts = `${pad(now.getHours())}:${pad(now.getMinutes())}:${pad(now.getSeconds())}`
   318→
   319→  if (isFailed) {
   320→    return `${ts} > Cloning repository...
   321→${ts} > Installing dependencies...
   322→${ts} > npm install --production
   323→${ts} > Building application...
   324→${ts} > ${fw} build
   325→${ts} ✗ Type error: Property 'fetchData' does not exist on type 'never'.
   326→${ts} ✗ Build failed with 1 error
   327→${ts} > Error: Build process exited with code 1
   328→${ts} > Deployment failed!`
   329→  }
   330→
   331→  return `${ts} > Cloning repository...
   332→${ts} ✓ Repository cloned successfully
   333→${ts} > Installing dependencies...
   334→${ts} > npm install --production
   335→${ts} ✓ Dependencies installed (1.2s)
   336→${ts} > Building application...
   337→${ts} > ${fw} build
   338→${ts} ✓ Compiled successfully
   339→${ts} ✓ Linting and checking validity of types
   340→${ts} ✓ Collecting page data
   341→${ts} ✓ Generating static pages (8/8)
   342→${ts} ✓ Finalizing page optimization
   343→${ts} > Deploying to server...
   344→${ts} ✓ Deployment successful!
   345→${ts} > Live at https://${project.subdomain}`
   346→}

   348→function colorizeLogLine(line: string): React.ReactNode {
   349→  if (line.includes('✓') || line.includes('successful') || line.includes('Live at')) {
   350→    return <span className="text-emerald-400">{line}</span>
   351→  }
   352→  if (line.includes('✗') || line.includes('Error') || line.includes('failed')) {
   353→    return <span className="text-red-400">{line}</span>
   354→  }
   355→  if (line.includes('Warning') || line.includes('warn')) {
   356→    return <span className="text-yellow-400">{line}</span>
   357→  }
   358→  if (line.startsWith('>')) {
   359→    return <span className="text-zinc-400">{line}</span>
   360→  }
   361→  return <span className="text-zinc-200">{line}</span>
   362→}

   440→function CreateProjectDialog({
   441→  open,
   442→  onClose,
   443→  onCreated,
   444→}: {
   445→  open: boolean
   446→  onClose: () => void
   447→  onCreated: () => void
   448→}) {
   449→  const [name, setName] = useState('')
   450→  const [framework, setFramework] = useState('nextjs')
   451→  const [planId, setPlanId] = useState('')
   452→  const [plans, setPlans] = useState<Plan[]>([])
   453→  const [loading, setLoading] = useState(false)
   454→  const [plansLoading, setPlansLoading] = useState(false)
   455→
   456→  useEffect(() => {
   457→    if (open) {
   458→      setPlansLoading(true)
   459→      fetch('/api/billing/plans')
   460→        .then((r) => r.json())
   461→        .then((data) => {
   462→          setPlans(data.plans || [])
   463→          if (data.plans?.length > 0 && !planId) {
   464→            setPlanId(data.plans[0].id)
   465→          }
   466→        })
   467→        .catch(() => toast.error('Failed to load plans'))
   468→        .finally(() => setPlansLoading(false))
   469→    }
   470→  }, [open, planId])
   471→
   472→  const handleCreate = async () => {
   473→    if (!name.trim()) {
   474→      toast.error('Project name is required')
   475→      return
   476→    }
   477→    setLoading(true)
   478→    try {
   479→      const res = await fetch('/api/hosting/create', {
   480→        method: 'POST',
   481→        headers: getAuthHeaders(),
   482→        body: JSON.stringify({ name: name.trim(), framework, planId: planId || undefined }),
   483→      })
   484→      const data = await res.json()
   485→      if (!res.ok) {
   486→        toast.error(data.error || 'Failed to create project')
   487→        return
   488→      }
   489→      toast.success('Project created successfully!')
   490→      setName('')
   491→      setFramework('nextjs')
   492→      onCreated()
   493→      onClose()
   494→    } catch {
   495→      toast.error('Network error. Please try again.')
   496→    } finally {
   497→      setLoading(false)
   498→    }
   499→  }
   500→
   501→  return (
   502→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   503→      <DialogContent className="sm:max-w-lg">
   504→        <DialogHeader>
   505→          <DialogTitle>Create New Project</DialogTitle>
   506→          <DialogDescription>
   507→            Set up a new hosting project with your preferred framework and plan.
   508→          </DialogDescription>
   509→        </DialogHeader>
   510→
   511→        <div className="grid gap-4 py-4">
   512→          <div className="grid gap-2">
   513→            <Label htmlFor="project-name">Project Name</Label>
   514→            <Input
   515→              id="project-name"
   516→              placeholder="my-awesome-app"
   517→              value={name}
   518→              onChange={(e) => setName(e.target.value)}
   519→            />
   520→          </div>
   521→
   522→          <div className="grid gap-2">
   523→            <Label>Framework</Label>
   524→            <Select value={framework} onValueChange={setFramework}>
   525→              <SelectTrigger className="w-full">
   526→                <SelectValue placeholder="Select framework" />
   527→              </SelectTrigger>
   528→              <SelectContent>
   529→                <SelectItem value="nextjs">Next.js</SelectItem>
   530→                <SelectItem value="react">React</SelectItem>
   531→                <SelectItem value="node">Node.js</SelectItem>
   532→                <SelectItem value="static">Static</SelectItem>
   533→              </SelectContent>
   534→            </Select>
   535→          </div>
   536→
   537→          <div className="grid gap-2">
   538→            <Label>Plan</Label>
   539→            {plansLoading ? (
   540→              <div className="grid gap-2">
   541→                <Skeleton className="h-20 w-full" />
   542→                <Skeleton className="h-20 w-full" />
   543→              </div>
   544→            ) : (
   545→              <div className="grid gap-2">
   546→                {plans.map((plan) => (
   547→                  <button
   548→                    key={plan.id}
   549→                    type="button"
   550→                    onClick={() => setPlanId(plan.id)}
   551→                    className={`relative flex items-center gap-3 rounded-lg border p-3 text-left transition-all hover:border-emerald-300 dark:hover:border-emerald-700 ${
   552→                      planId === plan.id
   553→                        ? 'border-emerald-500 bg-emerald-50 dark:bg-emerald-900/20 ring-1 ring-emerald-500'
   554→                        : 'border-border'
   555→                    }`}
   556→                  >
   557→                    {plan.popular && (
   558→                      <Badge className="absolute -top-2 right-3 bg-emerald-600 text-white text-[10px] px-1.5 py-0">
   559→                        Popular
   560→                      </Badge>
   561→                    )}
   562→                    <div className="flex-1">
   563→                      <div className="font-medium text-sm">{plan.name}</div>
   564→                      <div className="text-xs text-muted-foreground mt-0.5">
   565→                        {plan.cpu} &middot; {plan.ram} &middot; {plan.storage}
   566→                      </div>
   567→                    </div>
   568→                    <div className="text-right">
   569→                      <div className="font-semibold text-emerald-600 dark:text-emerald-400">
   570→                        ${plan.price}
   571→                      </div>
   572→                      <div className="text-[10px] text-muted-foreground">/mo</div>
   573→                    </div>
   574→                  </button>
   575→                ))}
   576→              </div>
   577→            )}
   578→          </div>
   579→        </div>
   580→
   581→        <DialogFooter>
   582→          <Button variant="outline" onClick={onClose} disabled={loading}>
   583→            Cancel
   584→          </Button>
   585→          <Button
   586→            onClick={handleCreate}
   587→            disabled={loading || !name.trim()}
   588→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   589→          >
   590→            {loading ? (
   591→              <>
   592→                <RefreshCw className="size-4 mr-2 animate-spin" />
   593→                Creating...
   594→              </>
   595→            ) : (
   596→              <>
   597→                <Rocket className="size-4 mr-2" />
   598→                Create Project
   599→              </>
   600→            )}
   601→          </Button>
   602→        </DialogFooter>
   603→      </DialogContent>
   604→    </Dialog>
   605→  )
   606→}

   610→function BuildLogsDialog({
   611→  open,
   612→  onClose,
   613→  logContent,
   614→  deployment,
   615→  project,
   616→  loading,
   617→}: {
   618→  open: boolean
   619→  onClose: () => void
   620→  logContent: string
   621→  deployment: Deployment | null
   622→  project: ProjectDetail | null
   623→  loading: boolean
   624→}) {
   625→  const scrollRef = useRef<HTMLDivElement>(null)
   626→
   627→  useEffect(() => {
   628→    if (open && scrollRef.current) {
   629→      scrollRef.current.scrollTop = scrollRef.current.scrollHeight
   630→    }
   631→  }, [open, logContent])
   632→
   633→  const displayLog = loading
   634→    ? 'Loading build logs...'
   635→    : logContent || (deployment && project ? generateSimulatedBuildLog(project, deployment) : 'No build logs available')
   636→
   637→  const isAnimating = deployment?.status === 'building' || deployment?.status === 'deploying'
   638→
   639→  return (
   640→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
