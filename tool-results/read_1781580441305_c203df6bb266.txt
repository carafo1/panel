    76→    76→  // Chart data: deployments per day for last 7 days
    77→    77→  const [chartData, setChartData] = useState<Array<{ day: string; success: number; failed: number }>>([])
    78→    78→
    79→    79→  useEffect(() => {
    80→    80→    async function loadData() {
    81→    81→      const token = localStorage.getItem('deployhub_token')
    82→    82→      if (!token) return
    83→    83→      try {
    84→    84→        const headers = { Authorization: `Bearer ${token}`, 'Content-Type': 'application/json' }
    85→    85→        const [projRes, domRes, depRes, logRes, subRes] = await Promise.all([
    86→    86→          fetch('/api/hosting', { headers }).catch(() => null),
    87→    87→          fetch('/api/domains', { headers }).catch(() => null),
    88→    88→          fetch('/api/deployments', { headers }).catch(() => null),
    89→    89→          fetch('/api/automation/logs?limit=20', { headers }).catch(() => null),

    93→    93→        const projData = projRes ? await projRes.json().catch(() => ({ projects: [] })) : { projects: [] }

    95→    95→        const depData = depRes ? await depRes.json().catch(() => ({ deployments: [] })) : { deployments: [] }
    96→    96→        const logData = logRes ? await logRes.json().catch(() => ({ logs: [] })) : { logs: [] }

    99→    99→        const projCount = Array.isArray(projData.projects) ? projData.projects.length : 0
   100→   100→        const domCount = Array.isArray(domData.domains) ? domData.domains.length : 0
   101→   101→        const depList = Array.isArray(depData.deployments) ? depData.deployments : []
   102→   102→        const depCount = depList.length
   103→   103→
   104→   104→        setProjectsCount(projCount)
   105→   105→        setDomainsCount(domCount)
   106→   106→        setDeploymentsCount(depCount)
   107→   107→        setDeployments(depList.slice(0, 20))
   108→   108→
   109→   109→        // Plan info
   110→   110→        if (subData.subscription?.plan) {
   111→   111→          setPlanInfo({
   112→   112→            name: subData.subscription.plan.name,
   113→   113→            storage: subData.subscription.plan.storage || '10GB',
   114→   114→            bandwidth: subData.subscription.plan.bandwidth || '100GB',
   115→   115→            projects: subData.subscription.plan.projects || 3,
   116→   116→          })
   117→   117→        }
   118→   118→
   119→   119→        // Build chart data: deployments per day for last 7 days

   121→   121→        const now = new Date()

   124→   124→          const d = new Date(now)
   125→   125→          d.setDate(d.getDate() - i)
   126→   126→          const dateStr = d.toISOString().split('T')[0]
   127→   127→          days.push({ date: dateStr, day: dayNames[d.getDay()], success: 0, failed: 0 })
   128→   128→        }
   129→   129→        depList.forEach((dep: { createdAt: string; status: string }) => {
   130→   130→          const depDate = new Date(dep.createdAt).toISOString().split('T')[0]
   131→   131→          const dayEntry = days.find((d) => d.date === depDate)
   132→   132→          if (dayEntry) {
   133→   133→            if (dep.status === 'failed' || dep.status === 'rolled_back') {
   134→   134→              dayEntry.failed++
   135→   135→            } else {
   136→   136→              dayEntry.success++
   137→   137→            }
   138→   138→          }
   139→   139→        })
   140→   140→        // If all zeros, add some mock data for visual appeal
   141→   141→        const hasData = days.some((d) => d.success > 0 || d.failed > 0)
   142→   142→        if (!hasData) {
   143→   143→          days.forEach((d, i) => {
   144→   144→            d.success = [2, 4, 1, 3, 5, 2, 3][i]
   145→   145→            d.failed = [0, 1, 0, 0, 1, 0, 0][i]
   146→   146→          })
   147→   147→        }
   148→   148→        setChartData(days)
   149→   149→
   150→   150→        // Build activity timeline from automation logs + deployments + domains
   151→   151→        const timeline: Array<{ id: string; type: string; message: string; time: string; status: string }> = []
   152→   152→
   153→   153→        // From automation logs
   154→   154→        const logs = Array.isArray(logData.logs) ? logData.logs : []
   155→   155→        logs.forEach((log: { id: string; eventType: string; action: string; status: string; createdAt: string; error?: string }) => {
   156→   156→          let type = 'automation'
   157→   157→          if (log.action?.includes('deploy') || log.eventType?.includes('hosting')) type = 'deployment'
   158→   158→          else if (log.action?.includes('domain') || log.action?.includes('dns')) type = 'domain'
   159→   159→          else if (log.eventType?.includes('payment')) type = 'payment'
   160→   160→          else if (log.action?.includes('ssl')) type = 'ssl'
   161→   161→
   162→   162→          const message = log.action
   163→   163→            ? log.action.replace(/_/g, ' ').replace(/\b\w/g, (c: string) => c.toUpperCase())
   164→   164→            : log.eventType?.replace(/_/g, ' ').replace(/\b\w/g, (c: string) => c.toUpperCase()) || 'Automation event'
   165→   165→
   166→   166→          timeline.push({ id: `log-${log.id}`, type, message, time: log.createdAt, status: log.status })
   167→   167→        })
   168→   168→
   169→   169→        // From deployments
   170→   170→        depList.slice(0, 5).forEach((d: { id: string; status: string; createdAt: string; project?: { name: string } }) => {
   171→   171→          const projName = d.project?.name || 'Project'
   172→   172→          const statusLabel = d.status === 'live' ? 'deployed successfully' : d.status === 'failed' ? 'deployment failed' : `deployment ${d.status}`
   173→   173→          timeline.push({
   174→   174→            id: `dep-${d.id}`,
   175→   175→            type: 'deployment',
   176→   176→            message: `${projName} ${statusLabel}`,
   177→   177→            time: d.createdAt,
   178→   178→            status: d.status,
   179→   179→          })
   180→   180→        })
   181→   181→
   182→   182→        // From domains
   183→   183→        const domList = Array.isArray(domData.domains) ? domData.domains : []
   184→   184→        domList.slice(0, 3).forEach((d: { id: string; status: string; name: string; createdAt: string }) => {
   185→   185→          timeline.push({
   186→   186→            id: `dom-${d.id}`,
   187→   187→            type: 'domain',
   188→   188→            message: `Domain ${d.name} ${d.status === 'active' ? 'registered' : d.status}`,
   189→   189→            time: d.createdAt,
   190→   190→            status: d.status,
   191→   191→          })
   192→   192→        })
   193→   193→
   194→   194→        // Sort by time descending, deduplicate
   195→   195→        timeline.sort((a, b) => new Date(b.time).getTime() - new Date(a.time).getTime())
   196→   196→        const seen = new Set<string>()

   198→   198→          if (seen.has(item.id)) return false
   199→   199→          seen.add(item.id)
   200→   200→          return true
   201→   201→        })
   202→   202→        setActivityTimeline(unique.slice(0, 10))
   203→   203→      } catch {
   204→   204→        // Silently handle
   205→   205→      } finally {
   206→   206→        setLoading(false)
   207→   207→      }
   208→   208→    }
   209→   209→    loadData()
   210→   210→  }, [])

   212→   212→  // Helpers

   216→   216→
   217→   217→  const relativeTime = (dateStr: string) => {
   218→   218→    const diff = Date.now() - new Date(dateStr).getTime()

   220→   220→    if (mins < 1) return 'Just now'

   223→   223→    if (hours < 24) return `${hours}h ago`
   224→   224→    const days = Math.floor(hours / 24)
   225→   225→    if (days < 7) return `${days}d ago`

   234→   234→  // Resource usage calculations
