    77→// ── API helper ─────────────────────────────────────────────────────────
    78→async function apiFetch<T>(path: string, options?: RequestInit): Promise<T> {
    79→  const token = localStorage.getItem('deployhub_token')
    80→  const res = await fetch(`/api${path}`, {
    81→    ...options,
    82→    headers: {
    83→      'Content-Type': 'application/json',
    84→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    85→      ...options?.headers,
    86→    },
    87→  })
    88→  if (!res.ok) {
    89→    const data = await res.json().catch(() => ({ error: 'Request failed' }))
    90→    throw new Error(data.error || `Request failed with status ${res.status}`)
    91→  }
    92→  return res.json()
    93→}

   116→// ── Time helpers ───────────────────────────────────────────────────────
   117→function timeAgo(date: string): string {
   118→  const seconds = Math.floor((Date.now() - new Date(date).getTime()) / 1000)
   119→  if (seconds < 60) return 'just now'
   120→  const minutes = Math.floor(seconds / 60)
   121→  if (minutes < 60) return `${minutes}m ago`
   122→  const hours = Math.floor(minutes / 60)
   123→  if (hours < 24) return `${hours}h ago`
   124→  const days = Math.floor(hours / 24)
   125→  return `${days}d ago`
   126→}
   127→
   128→function formatDuration(seconds: number | null): string {
   129→  if (!seconds) return '—'
   130→  if (seconds < 60) return `${seconds}s`
   131→  const m = Math.floor(seconds / 60)
   132→  const s = seconds % 60
   133→  return `${m}m ${s}s`
   134→}

   149→// ── Simulated build logs ───────────────────────────────────────────────
   150→function generateSimulatedBuildLog(deployment: DeploymentDetail): string {
   151→  const isFailed = deployment.status === 'failed'
   152→  const isBuilding = deployment.status === 'building' || deployment.status === 'deploying'
   153→  const fw = deployment.project?.framework === 'nextjs' ? 'next' : (deployment.project?.framework || 'node')
   154→  const now = new Date(deployment.createdAt)
   155→  const pad = (n: number) => String(n).padStart(2, '0')
   156→  const ts = `${pad(now.getHours())}:${pad(now.getMinutes())}:${pad(now.getSeconds())}`
   157→
   158→  if (isFailed) {
   159→    return `${ts} > Cloning repository...
   160→${ts} ✓ Repository cloned successfully
   161→${ts} > Installing dependencies...
   162→${ts} > npm install --production
   163→${ts} ✓ Dependencies installed (1.2s)
   164→${ts} > Building application...
   165→${ts} > ${fw} build
   166→${ts} ✗ Type error: Property 'fetchData' does not exist on type 'never'.
   167→${ts} ✗ Build failed with 1 error
   168→${ts} > Error: Build process exited with code 1
   169→${ts} > Deployment failed!`
   170→  }
   171→
   172→  if (isBuilding) {
   173→    return `${ts} > Cloning repository...
   174→${ts} ✓ Repository cloned successfully
   175→${ts} > Installing dependencies...
   176→${ts} > npm install --production
   177→${ts} ✓ Dependencies installed (1.2s)
   178→${ts} > Building application...
   179→${ts} > ${fw} build
   180→${ts} ⠹ Compiling...`
   181→  }
   182→
   183→  return `${ts} > Cloning repository...
   184→${ts} ✓ Repository cloned successfully
   185→${ts} > Installing dependencies...
   186→${ts} > npm install --production
   187→${ts} ✓ Dependencies installed (1.2s)
   188→${ts} > Building application...
   189→${ts} > ${fw} build
   190→${ts} ✓ Compiled successfully
   191→${ts} ✓ Linting and checking validity of types
   192→${ts} ✓ Collecting page data
   193→${ts} ✓ Generating static pages
   194→${ts} ✓ Finalizing page optimization
   195→${ts} > Deploying to server...
   196→${ts} ✓ Deployment successful!
   197→${ts} > Live at https://${deployment.project?.subdomain || 'myapp'}.deployhub.io`
   198→}

   200→function colorizeLogLine(line: string): React.ReactNode {
   201→  if (line.includes('✓') || line.includes('successful') || line.includes('Live at')) {
   202→    return <span className="text-emerald-400">{line}</span>
   203→  }
   204→  if (line.includes('✗') || line.includes('Error') || line.includes('failed')) {
   205→    return <span className="text-red-400">{line}</span>
   206→  }
   207→  if (line.includes('Warning') || line.includes('warn') || line.includes('⠹') || line.includes('⠋') || line.includes('⠼')) {
   208→    return <span className="text-yellow-400">{line}</span>
   209→  }
   210→  if (line.startsWith('>')) {
   211→    return <span className="text-zinc-400">{line}</span>
   212→  }
   213→  return <span className="text-zinc-200">{line}</span>
   214→}

   232→  const logEndRef = useRef<HTMLDivElement>(null)
   233→
   234→  const fetchDeployments = useCallback(async () => {
   235→    try {
   236→      const data = await apiFetch<{ deployments: Deployment[] }>('/deployments')
   237→      setDeployments(data.deployments)
   238→    } catch {
   239→      toast.error('Failed to load deployments')
   240→    } finally {
   241→      setLoading(false)
   242→    }
   243→  }, [])
   244→
   245→  useEffect(() => {
   246→    fetchDeployments()
   247→  }, [fetchDeployments])
   248→
   249→  // Auto-refresh when there are active deployments
   250→  useEffect(() => {
   251→    const hasActiveDeployments = deployments.some(
   252→      (d) => d.status === 'building' || d.status === 'deploying' || d.status === 'queued'
   253→    )

   255→    if (hasActiveDeployments) {
   256→      autoRefreshRef.current = setInterval(() => {
   257→        fetchDeployments()

   259→    }
   260→
   261→    return () => {
   262→      if (autoRefreshRef.current) {
   263→        clearInterval(autoRefreshRef.current)
   264→        autoRefreshRef.current = null
   265→      }
   266→    }
   267→  }, [deployments, fetchDeployments])
   268→

   271→    if (!selectedId) return
   272→
   273→    const isActive = detail?.status === 'building' || detail?.status === 'deploying' || detail?.status === 'queued' || liveStatus === 'building' || liveStatus === 'deploying' || liveStatus === 'queued'
   274→
   275→    if (!isActive) return
   276→
   277→    // Clear previous live logs when connecting to a new deployment
   278→    setLiveLogs([])
   279→    setIsLiveConnected(true)
   280→
   281→    let cleanupFn: (() => void) | null = null
   282→
   283→    // Try WebSocket first, fallback to SSE
   284→    try {
   285→      const handleLog = (data: DeploymentLogData) => {

   287→          setLiveLogs((prev) => [...prev, data.line])
   288→        }
   289→      }
   290→
   291→      const handleStatus = (data: DeploymentStatusData) => {
   292→        if (data.deploymentId === selectedId) {
   293→          setLiveStatus(data.status)
   294→
   295→          if (data.status === 'live' || data.status === 'failed' || data.status === 'rolled_back') {
   296→            setDetail((prev) => prev ? { ...prev, status: data.status, deployUrl: data.url || prev.deployUrl } : prev)
   297→            setIsLiveConnected(false)
   298→            fetchDeployments()
   299→            if (data.status === 'live') {
   300→              toast.success('Deployment is now live!')
   301→            }
   302→          }
   303→        }
   304→      }
   305→
   306→      joinDeploymentRoom(selectedId, handleLog, handleStatus)
   307→      cleanupFn = () => {
   308→        leaveDeploymentRoom(selectedId)
   309→      }
   310→    } catch {
   311→      // WebSocket not available, use SSE fallback
   312→      const token = localStorage.getItem('deployhub_token')
   313→      const eventSource = new EventSource(`/api/deployments/${selectedId}/events?token=${token}`)
   314→
   315→      eventSource.onmessage = (event) => {
   316→        try {
   317→          const data = JSON.parse(event.data)
   318→          if (data.status) {
   319→            setLiveStatus(data.status)
   320→          }
   321→          if (data.buildLog) {
   322→            const lines = data.buildLog.split('\n').filter(Boolean)
   323→            setLiveLogs(lines)
   324→          }
   325→          if (data.status === 'live' || data.status === 'failed' || data.status === 'rolled_back') {
   326→            setDetail((prev) => prev ? { ...prev, status: data.status, deployUrl: data.deployUrl || prev.deployUrl, duration: data.duration || prev.duration } : prev)
   327→            setIsLiveConnected(false)
   328→            eventSource.close()
   329→            fetchDeployments()
   330→            if (data.status === 'live') {
   331→              toast.success('Deployment is now live!')
   332→            }
   333→          }
   334→        } catch {
   335→          // Ignore parse errors
   336→        }
   337→      }
   338→
   339→      eventSource.onerror = () => {
   340→        eventSource.close()
   341→        setIsLiveConnected(false)
   342→      }
   343→
   344→      cleanupFn = () => {
   345→        eventSource.close()
   346→      }
   347→    }
   348→
   349→    return () => {
   350→      if (cleanupFn) cleanupFn()
   351→      setIsLiveConnected(false)

   362→

   365→    if (!selectedId) {
   366→      setLiveLogs([])
   367→      setLiveStatus(null)

   384→      setDetailLoading(false)
   385→    }
   386→  }, [])
   387→
   388→  useEffect(() => {
   389→    if (selectedId) {
   390→      fetchDetail(selectedId)

   392→  }, [selectedId, fetchDetail])

   408→

   411→    if (activeFilter === 'all') return true
   412→    if (activeFilter === 'live') return d.status === 'live'
   413→    if (activeFilter === 'building') return d.status === 'building' || d.status === 'deploying' || d.status === 'queued'
   414→    if (activeFilter === 'failed') return d.status === 'failed' || d.status === 'rolled_back'

   424→  }

   429→      <div className="space-y-4">
   430→        <Skeleton className="h-10 w-full rounded-lg" />
