   129→// ── Helper ───────────────────────────────────────────────────────────────
   130→
   131→function truncateUrl(url: string, maxLen = 45): string {
   132→  if (url.length <= maxLen) return url
   133→  return url.substring(0, maxLen - 3) + '...'
   134→}
   135→
   136→function formatRelativeTime(dateStr: string | null): string {
   137→  if (!dateStr) return 'Never'
   138→  const diff = Date.now() - new Date(dateStr).getTime()
   139→  const seconds = Math.floor(diff / 1000)
   140→  if (seconds < 60) return 'Just now'
   141→  const minutes = Math.floor(seconds / 60)
   142→  if (minutes < 60) return `${minutes}m ago`
   143→  const hours = Math.floor(minutes / 60)
   144→  if (hours < 24) return `${hours}h ago`
   145→  const days = Math.floor(hours / 24)
   146→  return `${days}d ago`
   147→}
   148→
   149→function getEventBadgeColor(eventId: string): string {
   150→  const prefix = eventId.split('.')[0]
   151→  switch (prefix) {
   152→    case 'deployment':
   153→      return 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300'
   154→    case 'domain':
   155→      return 'bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300'
   156→    case 'project':
   157→      return 'bg-violet-100 text-violet-700 dark:bg-violet-950 dark:text-violet-300'
   158→    case 'ssl':
   159→      return 'bg-cyan-100 text-cyan-700 dark:bg-cyan-950 dark:text-cyan-300'
   160→    case 'payment':
   161→      return 'bg-rose-100 text-rose-700 dark:bg-rose-950 dark:text-rose-300'
   162→    default:
   163→      return 'bg-gray-100 text-gray-700 dark:bg-gray-800 dark:text-gray-300'
   164→  }
   165→}
   166→
   167→function getEventLabel(eventId: string): string {
   168→  const found = WEBHOOK_EVENTS.find((e) => e.id === eventId)
   169→  return found ? found.label : eventId
   170→}

   214→      setLoading(false)
   215→    }
   216→  }, [])

   243→

   245→  const handleCreate = async () => {
   246→    if (!createForm.name.trim()) {
   247→      toast.error('Name is required')
   248→      return
   249→    }
   250→    if (!createForm.url.trim()) {
   251→      toast.error('URL is required')
   252→      return
   253→    }
   254→    try {
   255→      new URL(createForm.url)
   256→    } catch {
   257→      toast.error('Invalid URL format')
   258→      return
   259→    }
   260→    if (createForm.events.length === 0) {
   261→      toast.error('Select at least one event')
   262→      return

   266→      setCreating(true)
   267→      await apiFetch('/webhooks', {
   268→        method: 'POST',
   269→        body: JSON.stringify({
   270→          name: createForm.name.trim(),
   271→          url: createForm.url.trim(),
   272→          projectId: createForm.projectId || null,
   273→          events: createForm.events,

   278→        description: `${createForm.name} is now active.`,

   282→      fetchWebhooks()
   283→    } catch (err) {
   284→      toast.error('Failed to create webhook', {
   285→        description: err instanceof Error ? err.message : 'Unknown error',
   286→      })
   287→    } finally {
   288→      setCreating(false)
   289→    }
   290→  }
   291→

   298→        method: 'PATCH',

   306→        setSelectedWebhook(null)

   309→      fetchWebhooks()
   310→    } catch (err) {

   312→        description: err instanceof Error ? err.message : 'Unknown error',

   337→        description: err instanceof Error ? err.message : 'Unknown error',
   338→      })

   341→    }
   342→  }

   364→      toast.error('Failed to load deliveries', {
   365→        description: err instanceof Error ? err.message : 'Unknown error',
   366→      })
   367→    } finally {
   368→      setDeliveriesLoading(false)
   369→    }
   370→  }
   371→

   398→          description: data.delivery?.statusCode

   402→      }

   405→        fetchDeliveries(id)
   406→        fetchWebhookDetail(id)
   407→      }
   408→      fetchWebhooks()
   409→    } catch (err) {
   410→      toast.error('Test delivery failed', {
   411→        description: err instanceof Error ? err.message : 'Unknown error',

   413→    } finally {
   414→      setTestingWebhookId(null)
   415→    }

   451→                  <Skeleton className="h-6 w-24" />
   452→                  <Skeleton className="h-6 w-24" />
   453→                </div>
