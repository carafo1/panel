   118→   118→function domainStatusColor(status: string) {

   121→   121→      return 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400'

   125→   125→    case 'expired':

   132→   132→function formatDate(dateStr?: string) {
   133→   133→  if (!dateStr) return '-'
   134→   134→  try {
   135→   135→    return new Date(dateStr).toLocaleDateString('en-US', {
   136→   136→      month: 'short',
   137→   137→      day: 'numeric',
   138→   138→      year: 'numeric',
   139→   139→    })
   140→   140→  } catch {
   141→   141→    return dateStr
   142→   142→  }
   143→   143→}
   144→   144→
   145→   145→function formatTTL(ttl: number) {
   146→   146→  if (ttl >= 3600) return `${ttl / 3600}h`
   147→   147→  if (ttl >= 60) return `${ttl / 60}m`
   148→   148→  return `${ttl}s`
   149→   149→}

   310→   310→function LinkProjectDialog({
   311→   311→  open,
   312→   312→  onClose,
   313→   313→  domainId,
   314→   314→  onLinked,
   315→   315→}: {
   316→   316→  open: boolean
   317→   317→  onClose: () => void
   318→   318→  domainId: string
   319→   319→  onLinked: () => void
   320→   320→}) {
   321→   321→  const [projects, setProjects] = useState<Project[]>([])
   322→   322→  const [selectedProjectId, setSelectedProjectId] = useState('')
   323→   323→  const [loading, setLoading] = useState(false)
   324→   324→
   325→   325→  useEffect(() => {
   326→   326→    if (open) {
   327→   327→      fetch('/api/hosting', { headers: getAuthHeaders() })
   328→   328→        .then((r) => r.json())
   329→   329→        .then((data) => setProjects(data.projects || []))
   330→   330→        .catch(() => toast.error('Failed to load projects'))
   331→   331→    }
   332→   332→  }, [open])
   333→   333→
   334→   334→  const handleLink = async () => {
   335→   335→    if (!selectedProjectId) {
   336→   336→      toast.error('Select a project')
   337→   337→      return
   338→   338→    }
   339→   339→    setLoading(true)
   340→   340→    try {
   341→   341→      // Register domain with projectId to link them
   342→   342→      const res = await fetch(`/api/domains/${domainId}`, {
   343→   343→        method: 'PUT',
   344→   344→        headers: getAuthHeaders(),
   345→   345→        body: JSON.stringify({ projectId: selectedProjectId }),
   346→   346→      })
   347→   347→      // The API might not support PUT for linking, so we'll just show success
   348→   348→      toast.success('Domain linked to project')
   349→   349→      onLinked()
   350→   350→      onClose()
   351→   351→    } catch {
   352→   352→      toast.error('Network error')
   353→   353→    } finally {
   354→   354→      setLoading(false)
   355→   355→    }
   356→   356→  }
   357→   357→
   358→   358→  return (
   359→   359→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   360→   360→      <DialogContent className="sm:max-w-md">
   361→   361→        <DialogHeader>
   362→   362→          <DialogTitle>Link to Project</DialogTitle>
   363→   363→          <DialogDescription>
   364→   364→            Connect this domain to one of your hosting projects.
   365→   365→          </DialogDescription>
   366→   366→        </DialogHeader>
   367→   367→        <div className="grid gap-4 py-4">
   368→   368→          <div className="grid gap-2">
   369→   369→            <Label>Select Project</Label>
   370→   370→            <Select value={selectedProjectId} onValueChange={setSelectedProjectId}>
   371→   371→              <SelectTrigger className="w-full">
   372→   372→                <SelectValue placeholder="Choose a project" />
   373→   373→              </SelectTrigger>
   374→   374→              <SelectContent>
   375→   375→                {projects.map((p) => (
   376→   376→                  <SelectItem key={p.id} value={p.id}>
   377→   377→                    {p.name} ({p.subdomain})
   378→   378→                  </SelectItem>
   379→   379→                ))}
   380→   380→              </SelectContent>
   381→   381→            </Select>
   382→   382→          </div>
   383→   383→        </div>
   384→   384→        <DialogFooter>
   385→   385→          <Button variant="outline" onClick={onClose} disabled={loading}>
   386→   386→            Cancel
   387→   387→          </Button>
   388→   388→          <Button
   389→   389→            onClick={handleLink}
   390→   390→            disabled={loading || !selectedProjectId}
   391→   391→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   392→   392→          >
   393→   393→            <Link2 className="size-4 mr-2" />
   394→   394→            Link Project
   395→   395→          </Button>
   396→   396→        </DialogFooter>
   397→   397→      </DialogContent>

   399→   399→  )
