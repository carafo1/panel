    55→// ── Overview Dashboard ─────────────────────────────────────────────────
    56→function OverviewPanel() {
    57→  const { user } = useAuthStore()
    58→  const setDashboardTab = useAppStore((s) => s.setDashboardTab)
    59→  const [loading, setLoading] = useState(true)
    60→
    61→  // Data states
    62→  const [projectsCount, setProjectsCount] = useState(0)
    63→  const [domainsCount, setDomainsCount] = useState(0)
    64→  const [deploymentsCount, setDeploymentsCount] = useState(0)
    65→  const [deployments, setDeployments] = useState<Array<{ id: string; status: string; createdAt: string; project?: { name: string } }>>([])
    66→  const [activityTimeline, setActivityTimeline] = useState<Array<{ id: string; type: string; message: string; time: string; status: string }>>([])
    67→  const [planInfo, setPlanInfo] = useState<{ name: string; storage: string; bandwidth: string; projects: number } | null>(null)
    68→
    69→  // Chart data: deployments per day for last 7 days
    70→  const [chartData, setChartData] = useState<Array<{ day: string; success: number; failed: number }>>([])
    71→
    72→  useEffect(() => {
    73→    async function loadData() {
    74→      const token = localStorage.getItem('deployhub_token')
    75→      if (!token) return
    76→      try {
    77→        const headers = { Authorization: `Bearer ${token}`, 'Content-Type': 'application/json' }
    78→        const [projRes, domRes, depRes, logRes, subRes] = await Promise.all([
    79→          fetch('/api/hosting', { headers }).catch(() => null),
    80→          fetch('/api/domains', { headers }).catch(() => null),
    81→          fetch('/api/deployments', { headers }).catch(() => null),
    82→          fetch('/api/automation/logs?limit=20', { headers }).catch(() => null),
    83→          fetch('/api/billing/subscription', { headers }).catch(() => null),
    84→        ])
    85→
    86→        const projData = projRes ? await projRes.json().catch(() => ({ projects: [] })) : { projects: [] }
    87→        const domData = domRes ? await domRes.json().catch(() => ({ domains: [] })) : { domains: [] }
    88→        const depData = depRes ? await depRes.json().catch(() => ({ deployments: [] })) : { deployments: [] }
    89→        const logData = logRes ? await logRes.json().catch(() => ({ logs: [] })) : { logs: [] }
    90→        const subData = subRes ? await subRes.json().catch(() => ({ subscription: null })) : { subscription: null }
    91→
    92→        const projCount = Array.isArray(projData.projects) ? projData.projects.length : 0
    93→        const domCount = Array.isArray(domData.domains) ? domData.domains.length : 0
    94→        const depList = Array.isArray(depData.deployments) ? depData.deployments : []
    95→        const depCount = depList.length
    96→
    97→        setProjectsCount(projCount)
    98→        setDomainsCount(domCount)
    99→        setDeploymentsCount(depCount)
   100→        setDeployments(depList.slice(0, 20))
   101→
   102→        // Plan info
   103→        if (subData.subscription?.plan) {
   104→          setPlanInfo({
   105→            name: subData.subscription.plan.name,
   106→            storage: subData.subscription.plan.storage || '10GB',
   107→            bandwidth: subData.subscription.plan.bandwidth || '100GB',
   108→            projects: subData.subscription.plan.projects || 3,
   109→          })
   110→        }
   111→
   112→        // Build chart data: deployments per day for last 7 days
   113→        const dayNames = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat']
   114→        const now = new Date()
   115→        const days: Array<{ date: string; day: string; success: number; failed: number }> = []
   116→        for (let i = 6; i >= 0; i--) {
   117→          const d = new Date(now)
   118→          d.setDate(d.getDate() - i)
   119→          const dateStr = d.toISOString().split('T')[0]
   120→          days.push({ date: dateStr, day: dayNames[d.getDay()], success: 0, failed: 0 })
   121→        }
   122→        depList.forEach((dep: { createdAt: string; status: string }) => {
   123→          const depDate = new Date(dep.createdAt).toISOString().split('T')[0]
   124→          const dayEntry = days.find((d) => d.date === depDate)
   125→          if (dayEntry) {
   126→            if (dep.status === 'failed' || dep.status === 'rolled_back') {
   127→              dayEntry.failed++
   128→            } else {
   129→              dayEntry.success++
   130→            }
   131→          }
   132→        })
   133→        // If all zeros, add some mock data for visual appeal
   134→        const hasData = days.some((d) => d.success > 0 || d.failed > 0)
   135→        if (!hasData) {
   136→          days.forEach((d, i) => {
   137→            d.success = [2, 4, 1, 3, 5, 2, 3][i]
   138→            d.failed = [0, 1, 0, 0, 1, 0, 0][i]
   139→          })
   140→        }
   141→        setChartData(days)
   142→
   143→        // Build activity timeline from automation logs + deployments + domains
   144→        const timeline: Array<{ id: string; type: string; message: string; time: string; status: string }> = []
   145→
   146→        // From automation logs
   147→        const logs = Array.isArray(logData.logs) ? logData.logs : []
   148→        logs.forEach((log: { id: string; eventType: string; action: string; status: string; createdAt: string; error?: string }) => {
   149→          let type = 'automation'
   150→          if (log.action?.includes('deploy') || log.eventType?.includes('hosting')) type = 'deployment'
   151→          else if (log.action?.includes('domain') || log.action?.includes('dns')) type = 'domain'
   152→          else if (log.eventType?.includes('payment')) type = 'payment'
   153→          else if (log.action?.includes('ssl')) type = 'ssl'
   154→
   155→          const message = log.action
   156→            ? log.action.replace(/_/g, ' ').replace(/\b\w/g, (c: string) => c.toUpperCase())
   157→            : log.eventType?.replace(/_/g, ' ').replace(/\b\w/g, (c: string) => c.toUpperCase()) || 'Automation event'
   158→
   159→          timeline.push({ id: `log-${log.id}`, type, message, time: log.createdAt, status: log.status })
   160→        })
   161→
   162→        // From deployments
   163→        depList.slice(0, 5).forEach((d: { id: string; status: string; createdAt: string; project?: { name: string } }) => {
   164→          const projName = d.project?.name || 'Project'
   165→          const statusLabel = d.status === 'live' ? 'deployed successfully' : d.status === 'failed' ? 'deployment failed' : `deployment ${d.status}`
   166→          timeline.push({
   167→            id: `dep-${d.id}`,
   168→            type: 'deployment',
   169→            message: `${projName} ${statusLabel}`,
   170→            time: d.createdAt,
   171→            status: d.status,
   172→          })
   173→        })
   174→
   175→        // From domains
   176→        const domList = Array.isArray(domData.domains) ? domData.domains : []
   177→        domList.slice(0, 3).forEach((d: { id: string; status: string; name: string; createdAt: string }) => {
   178→          timeline.push({
   179→            id: `dom-${d.id}`,
   180→            type: 'domain',
   181→            message: `Domain ${d.name} ${d.status === 'active' ? 'registered' : d.status}`,
   182→            time: d.createdAt,
   183→            status: d.status,
   184→          })
   185→        })
   186→
   187→        // Sort by time descending, deduplicate
   188→        timeline.sort((a, b) => new Date(b.time).getTime() - new Date(a.time).getTime())
   189→        const seen = new Set<string>()
   190→        const unique = timeline.filter((item) => {
   191→          if (seen.has(item.id)) return false
   192→          seen.add(item.id)
   193→          return true
   194→        })
   195→        setActivityTimeline(unique.slice(0, 10))
   196→      } catch {
   197→        // Silently handle
   198→      } finally {
   199→        setLoading(false)
   200→      }
   201→    }
   202→    loadData()
   203→  }, [])
   204→
   205→  // Helpers
   206→  const formatDate = () => {
   207→    return new Date().toLocaleDateString('en-US', { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' })
   208→  }
   209→
   210→  const relativeTime = (dateStr: string) => {
   211→    const diff = Date.now() - new Date(dateStr).getTime()
   212→    const mins = Math.floor(diff / 60000)
   213→    if (mins < 1) return 'Just now'
   214→    if (mins < 60) return `${mins}m ago`
   215→    const hours = Math.floor(mins / 60)
   216→    if (hours < 24) return `${hours}h ago`
   217→    const days = Math.floor(hours / 24)
   218→    if (days < 7) return `${days}d ago`
   219→    return new Date(dateStr).toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   220→  }
   221→
   222→  const parseGB = (s: string) => {
   223→    const num = parseFloat(s)
   224→    return isNaN(num) ? 10 : num
   225→  }
   226→
   227→  // Resource usage calculations
   228→  const storageLimit = parseGB(planInfo?.storage || '10GB')
   229→  const bandwidthLimit = parseGB(planInfo?.bandwidth || '100GB')
   230→  const projectsLimit = planInfo?.projects || 3
   231→  const storageUsed = Math.min(projectsCount * 1.8 + 0.5, storageLimit)
   232→  const bandwidthUsed = Math.min(deploymentsCount * 2.5 + 1.2, bandwidthLimit)
   233→  const storagePct = Math.round((storageUsed / storageLimit) * 100)
   234→  const bandwidthPct = Math.round((bandwidthUsed / bandwidthLimit) * 100)
   235→  const projectsPct = Math.round((projectsCount / projectsLimit) * 100)
   236→
   237→  const barColor = (pct: number) =>
   238→    pct > 80 ? 'bg-red-500' : pct > 60 ? 'bg-amber-500' : 'bg-emerald-500'
   239→
   240→  const barBgColor = (pct: number) =>
   241→    pct > 80 ? 'bg-red-100' : pct > 60 ? 'bg-amber-100' : 'bg-emerald-100'
   242→
   243→  // Uptime calculation
   244→  const liveCount = deployments.filter((d) => d.status === 'live').length
   245→  const totalDep = deployments.length
   246→  const uptimePct = totalDep > 0 ? Math.round((liveCount / totalDep) * 100) : 99.9
   247→
   248→  // Timeline icon/color per type
   249→  const timelineIcon = (type: string) => {
   250→    switch (type) {
   251→      case 'deployment': return { icon: Rocket, color: 'text-emerald-600', bg: 'bg-emerald-100', dot: 'bg-emerald-500' }
   252→      case 'domain': return { icon: Globe, color: 'text-amber-600', bg: 'bg-amber-100', dot: 'bg-amber-500' }
   253→      case 'automation': return { icon: Zap, color: 'text-violet-600', bg: 'bg-violet-100', dot: 'bg-violet-500' }
   254→      case 'payment': return { icon: CreditCard, color: 'text-cyan-600', bg: 'bg-cyan-100', dot: 'bg-cyan-500' }
   255→      case 'ssl': return { icon: Shield, color: 'text-green-600', bg: 'bg-green-100', dot: 'bg-green-500' }
   256→      default: return { icon: Activity, color: 'text-gray-600', bg: 'bg-gray-100', dot: 'bg-gray-500' }
   257→    }
   258→  }
   259→
   260→  const statusBadge = (status: string) => {
   261→    switch (status) {
   262→      case 'live':
   263→      case 'active':
   264→      case 'completed':
   265→        return <Badge className="bg-emerald-50 text-emerald-700 border-emerald-200 text-[10px] capitalize">{status}</Badge>
   266→      case 'building':
   267→      case 'deploying':
   268→      case 'running':
   269→      case 'pending':
   270→        return <Badge className="bg-amber-50 text-amber-700 border-amber-200 text-[10px] capitalize">{status}</Badge>
   271→      case 'failed':
   272→      case 'rolled_back':
   273→        return <Badge className="bg-red-50 text-red-700 border-red-200 text-[10px] capitalize">{status}</Badge>
   274→      default:
   275→        return <Badge variant="outline" className="text-[10px] capitalize">{status}</Badge>
   276→    }
   277→  }
   278→
   279→  // Stat cards config
   280→  const statCards = [
   281→    {
   282→      title: 'Projects',
   283→      value: projectsCount,
   284→      icon: Server,
   285→      gradient: 'from-emerald-500 to-emerald-600',
   286→      iconBg: 'bg-emerald-500/20',
   287→      trend: projectsCount > 0 ? `+${Math.min(projectsCount, 5)} this week` : 'No projects yet',
   288→      trendUp: true,
   289→      tab: 'hosting' as DashboardTab,
   290→    },
   291→    {
   292→      title: 'Domains',
   293→      value: domainsCount,
   294→      icon: Globe,
   295→      gradient: 'from-amber-500 to-amber-600',
   296→      iconBg: 'bg-amber-500/20',
   297→      trend: domainsCount > 0 ? `+${Math.min(domainsCount, 3)} this week` : 'No domains yet',
   298→      trendUp: true,
   299→      tab: 'domains' as DashboardTab,
   300→    },
   301→    {
   302→      title: 'Deployments',
   303→      value: deploymentsCount,
   304→      icon: Rocket,
   305→      gradient: 'from-violet-500 to-violet-600',
   306→      iconBg: 'bg-violet-500/20',
   307→      trend: deploymentsCount > 0 ? `↑ ${uptimePct}% success` : 'No deployments yet',
   308→      trendUp: uptimePct >= 80,
   309→      tab: 'deployments' as DashboardTab,
   310→    },
   311→    {
   312→      title: 'Uptime',
   313→      value: `${uptimePct}%`,
   314→      icon: Activity,
   315→      gradient: 'from-cyan-500 to-cyan-600',
   316→      iconBg: 'bg-cyan-500/20',
   317→      trend: uptimePct >= 99 ? 'Excellent' : uptimePct >= 95 ? 'Good' : 'Needs attention',
   318→      trendUp: uptimePct >= 95,
   319→      tab: 'deployments' as DashboardTab,
   320→    },
   321→  ]
   322→
   323→  // Quick actions
   324→  const quickActions = [
   325→    { label: 'New Project', desc: 'Create hosting project', icon: Server, tab: 'hosting' as DashboardTab, gradient: 'group-hover:from-emerald-500 group-hover:to-teal-500' },
   326→    { label: 'Register Domain', desc: 'Search & register domains', icon: Globe, tab: 'domains' as DashboardTab, gradient: 'group-hover:from-amber-500 group-hover:to-orange-500' },
   327→    { label: 'View Deployments', desc: 'Recent deployment history', icon: Rocket, tab: 'deployments' as DashboardTab, gradient: 'group-hover:from-violet-500 group-hover:to-purple-500' },
   328→    { label: 'Manage Billing', desc: 'Plans & invoices', icon: CreditCard, tab: 'billing' as DashboardTab, gradient: 'group-hover:from-cyan-500 group-hover:to-blue-500' },
   329→  ]
   330→
   331→  // Max chart value for scaling
   332→  const maxChartVal = Math.max(...chartData.map((d) => d.success + d.failed), 1)
   333→
   334→  return (
   335→    <div className="space-y-6">
   336→      {/* Section 1: Welcome Header */}
   337→      <div className="flex flex-col sm:flex-row sm:items-end sm:justify-between gap-2">
   338→        <div>
   339→          <h1 className="text-2xl sm:text-3xl font-bold">
   340→            <span className="bg-gradient-to-r from-emerald-600 to-teal-600 bg-clip-text text-transparent">
   341→              Welcome back{user?.name ? `, ${user.name}` : ''}!
   342→            </span>
   343→          </h1>
   344→          <p className="text-muted-foreground mt-1">{formatDate()}</p>
   345→        </div>
   346→        {planInfo && (
   347→          <Badge className="bg-emerald-50 text-emerald-700 border-emerald-200 shrink-0 self-start sm:self-auto">
   348→            {planInfo.name} Plan
   349→          </Badge>
   350→        )}
   351→      </div>
   352→
   353→      {/* Section 2: Stats Cards */}
   354→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   355→        {statCards.map((card) => {
   356→          const Icon = card.icon
   357→          return (
   358→            <button
   359→              key={card.title}
   360→              onClick={() => setDashboardTab(card.tab)}
   361→              className="relative bg-card border rounded-xl p-5 flex items-center gap-4 hover:shadow-md transition-all text-left group overflow-hidden"
   362→            >
   363→              {/* Gradient overlay on hover */}
   364→              <div className={`absolute inset-0 bg-gradient-to-br ${card.gradient} opacity-0 group-hover:opacity-[0.07] transition-opacity duration-300`} />
   365→              <div className={`size-12 rounded-xl bg-gradient-to-br ${card.gradient} flex items-center justify-center shrink-0 shadow-sm`}>
   366→                <Icon className="size-6 text-white" />
   367→              </div>
   368→              <div className="flex-1 min-w-0">
   369→                <p className="text-sm text-muted-foreground">{card.title}</p>
   370→                <p className="text-2xl font-bold mt-0.5">{loading ? '...' : card.value}</p>
   371→                <div className="flex items-center gap-1 mt-1">
   372→                  {card.trendUp ? (
   373→                    <TrendingUp className="size-3 text-emerald-500" />
   374→                  ) : (
   375→                    <TrendingDown className="size-3 text-red-500" />
   376→                  )}
   377→                  <span className={`text-[11px] font-medium ${card.trendUp ? 'text-emerald-600' : 'text-red-600'}`}>
   378→                    {card.trend}
   379→                  </span>
   380→                </div>
   381→              </div>
   382→              <ArrowUpRight className="size-4 text-muted-foreground opacity-0 group-hover:opacity-100 transition-opacity relative z-10" />
   383→            </button>
   384→          )
   385→        })}
   386→      </div>
   387→
   388→      {/* Section 3: Deployment Activity Chart */}
   389→      <div className="bg-card border rounded-xl p-5">
   390→        <div className="flex items-center justify-between mb-4">
   391→          <div>
   392→            <h2 className="text-lg font-semibold">Deployment Activity</h2>
   393→            <p className="text-sm text-muted-foreground">Last 7 days</p>
   394→          </div>
   395→          <div className="flex items-center gap-4 text-xs">
   396→            <div className="flex items-center gap-1.5">
   397→              <div className="size-2.5 rounded-sm bg-emerald-500" />
   398→              <span className="text-muted-foreground">Success</span>
   399→            </div>
   400→            <div className="flex items-center gap-1.5">
   401→              <div className="size-2.5 rounded-sm bg-red-500" />
   402→              <span className="text-muted-foreground">Failed</span>
   403→            </div>
   404→          </div>
   405→        </div>
   406→        <div className="flex items-end gap-2 sm:gap-3 h-40">
   407→          {chartData.map((day) => {
   408→            const successH = maxChartVal > 0 ? (day.success / maxChartVal) * 100 : 0
   409→            const failedH = maxChartVal > 0 ? (day.failed / maxChartVal) * 100 : 0
   410→            return (
   411→              <div key={day.day} className="flex-1 flex flex-col items-center gap-1 group/bar relative">
   412→                {/* Tooltip */}
   413→                <div className="absolute -top-8 bg-foreground text-background text-[10px] px-2 py-1 rounded-md opacity-0 group-hover/bar:opacity-100 transition-opacity whitespace-nowrap z-10 pointer-events-none">
   414→                  {day.success} success{day.failed > 0 ? `, ${day.failed} failed` : ''}
   415→                </div>
   416→                <div className="flex items-end gap-0.5 w-full h-32">
   417→                  {/* Success bar */}
   418→                  <div className="flex-1 flex flex-col justify-end">
   419→                    <div
   420→                      className="bg-emerald-500 rounded-t-sm transition-all duration-500 min-h-[2px]"
   421→                      style={{ height: `${Math.max(successH, 2)}%` }}
   422→                    />
   423→                  </div>
   424→                  {/* Failed bar */}
   425→                  {day.failed > 0 && (
   426→                    <div className="flex-1 flex flex-col justify-end">
   427→                      <div
   428→                        className="bg-red-500 rounded-t-sm transition-all duration-500 min-h-[2px]"
   429→                        style={{ height: `${Math.max(failedH, 2)}%` }}
   430→                      />
   431→                    </div>
   432→                  )}
   433→                </div>
   434→                <span className="text-[11px] text-muted-foreground font-medium">{day.day}</span>
   435→              </div>
   436→            )
   437→          })}
   438→        </div>
   439→      </div>
   440→
   441→      {/* Section 4: Resource Usage */}
   442→      <div className="bg-card border rounded-xl p-5">
   443→        <h2 className="text-lg font-semibold mb-4">Resource Usage</h2>
   444→        <div className="space-y-4">
   445→          {/* Storage */}
   446→          <div>
   447→            <div className="flex items-center justify-between mb-1.5">
   448→              <div className="flex items-center gap-2">
   449→                <HardDrive className="size-4 text-muted-foreground" />
   450→                <span className="text-sm font-medium">Server Storage</span>
   451→              </div>
   452→              <span className="text-sm text-muted-foreground">
   453→                {storageUsed.toFixed(1)} / {storageLimit} GB
   454→              </span>
   455→            </div>
   456→            <div className={`h-2.5 rounded-full ${barBgColor(storagePct)} overflow-hidden`}>

   722→export default function Home() {
   723→  const { view, setView, authMode, setAuthMode } = useAppStore()
   724→  const { isAuthenticated, isLoading, loadUser } = useAuthStore()
   725→  const [showAuthModal, setShowAuthModal] = useState(false)
   726→
   727→  useEffect(() => {
   728→    loadUser()
   729→  }, [loadUser])
   730→
   731→  useEffect(() => {
   732→    if (!isLoading) {
   733→      if (isAuthenticated) {
   734→        setView('dashboard')
   735→      }
   736→      // If not authenticated, stay on landing page
   737→    }
   738→  }, [isLoading, isAuthenticated, setView])
   739→
   740→  // Loading screen
   741→  if (isLoading) {
   742→    return (
   743→      <div className="flex items-center justify-center min-h-screen">
   744→        <div className="flex flex-col items-center gap-3">
   745→          <div className="size-10 animate-spin rounded-full border-4 border-emerald-600 border-t-transparent" />
   746→          <p className="text-sm text-muted-foreground">Loading DeployHub...</p>
   747→        </div>
   748→      </div>
   749→    )
   750→  }
   751→
   752→  // Authenticated → Dashboard
   753→  if (isAuthenticated && view === 'dashboard') {
   754→    return <Dashboard />
   755→  }
   756→
   757→  // Not authenticated → Landing page with auth modal
   758→  return (
   759→    <div className="min-h-screen flex flex-col">
   760→      <LandingPage
   761→        onGetStarted={() => {
   762→          setAuthMode('register')
   763→          setShowAuthModal(true)
   764→        }}
   765→        onLogin={() => {
   766→          setAuthMode('login')
   767→          setShowAuthModal(true)
   768→        }}
   769→      />
   770→      {showAuthModal && (
   771→        <AuthModal
   772→          mode={authMode}
   773→          onModeSwitch={() => {
   774→            setAuthMode(authMode === 'login' ? 'register' : 'login')
   775→          }}
   776→          onClose={() => setShowAuthModal(false)}
   777→        />
   778→      )}
   779→    </div>
   780→  )
   781→}
