    62→// ── Overview Dashboard ─────────────────────────────────────────────────
    63→function OverviewPanel() {
    64→  const { user } = useAuthStore()
    65→  const setDashboardTab = useAppStore((s) => s.setDashboardTab)
    66→  const [loading, setLoading] = useState(true)
    67→
    68→  // Data states
    69→  const [projectsCount, setProjectsCount] = useState(0)
    70→  const [domainsCount, setDomainsCount] = useState(0)
    71→  const [deploymentsCount, setDeploymentsCount] = useState(0)
    72→  const [deployments, setDeployments] = useState<Array<{ id: string; status: string; createdAt: string; project?: { name: string } }>>([])
    73→  const [activityTimeline, setActivityTimeline] = useState<Array<{ id: string; type: string; message: string; time: string; status: string }>>([])
    74→  const [planInfo, setPlanInfo] = useState<{ name: string; storage: string; bandwidth: string; projects: number } | null>(null)
    75→
    76→  // Chart data: deployments per day for last 7 days
    77→  const [chartData, setChartData] = useState<Array<{ day: string; success: number; failed: number }>>([])
    78→
    79→  useEffect(() => {
    80→    async function loadData() {
    81→      const token = localStorage.getItem('deployhub_token')
    82→      if (!token) return
    83→      try {
    84→        const headers = { Authorization: `Bearer ${token}`, 'Content-Type': 'application/json' }
    85→        const [projRes, domRes, depRes, logRes, subRes] = await Promise.all([
    86→          fetch('/api/hosting', { headers }).catch(() => null),
    87→          fetch('/api/domains', { headers }).catch(() => null),
    88→          fetch('/api/deployments', { headers }).catch(() => null),
    89→          fetch('/api/automation/logs?limit=20', { headers }).catch(() => null),
    90→          fetch('/api/billing/subscription', { headers }).catch(() => null),
    91→        ])
    92→
    93→        const projData = projRes ? await projRes.json().catch(() => ({ projects: [] })) : { projects: [] }
    94→        const domData = domRes ? await domRes.json().catch(() => ({ domains: [] })) : { domains: [] }
    95→        const depData = depRes ? await depRes.json().catch(() => ({ deployments: [] })) : { deployments: [] }
    96→        const logData = logRes ? await logRes.json().catch(() => ({ logs: [] })) : { logs: [] }
    97→        const subData = subRes ? await subRes.json().catch(() => ({ subscription: null })) : { subscription: null }
    98→
    99→        const projCount = Array.isArray(projData.projects) ? projData.projects.length : 0
   100→        const domCount = Array.isArray(domData.domains) ? domData.domains.length : 0
   101→        const depList = Array.isArray(depData.deployments) ? depData.deployments : []
   102→        const depCount = depList.length
   103→
   104→        setProjectsCount(projCount)
   105→        setDomainsCount(domCount)
   106→        setDeploymentsCount(depCount)
   107→        setDeployments(depList.slice(0, 20))
   108→
   109→        // Plan info
   110→        if (subData.subscription?.plan) {
   111→          setPlanInfo({
   112→            name: subData.subscription.plan.name,
   113→            storage: subData.subscription.plan.storage || '10GB',
   114→            bandwidth: subData.subscription.plan.bandwidth || '100GB',
   115→            projects: subData.subscription.plan.projects || 3,
   116→          })
   117→        }
   118→
   119→        // Build chart data: deployments per day for last 7 days
   120→        const dayNames = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat']
   121→        const now = new Date()
   122→        const days: Array<{ date: string; day: string; success: number; failed: number }> = []
   123→        for (let i = 6; i >= 0; i--) {
   124→          const d = new Date(now)
   125→          d.setDate(d.getDate() - i)
   126→          const dateStr = d.toISOString().split('T')[0]
   127→          days.push({ date: dateStr, day: dayNames[d.getDay()], success: 0, failed: 0 })
   128→        }
   129→        depList.forEach((dep: { createdAt: string; status: string }) => {
   130→          const depDate = new Date(dep.createdAt).toISOString().split('T')[0]
   131→          const dayEntry = days.find((d) => d.date === depDate)
   132→          if (dayEntry) {
   133→            if (dep.status === 'failed' || dep.status === 'rolled_back') {
   134→              dayEntry.failed++
   135→            } else {
   136→              dayEntry.success++
   137→            }
   138→          }
   139→        })
   140→        // If all zeros, add some mock data for visual appeal
   141→        const hasData = days.some((d) => d.success > 0 || d.failed > 0)
   142→        if (!hasData) {
   143→          days.forEach((d, i) => {
   144→            d.success = [2, 4, 1, 3, 5, 2, 3][i]
   145→            d.failed = [0, 1, 0, 0, 1, 0, 0][i]
   146→          })
   147→        }
   148→        setChartData(days)
   149→
   150→        // Build activity timeline from automation logs + deployments + domains
   151→        const timeline: Array<{ id: string; type: string; message: string; time: string; status: string }> = []
   152→
   153→        // From automation logs
   154→        const logs = Array.isArray(logData.logs) ? logData.logs : []
   155→        logs.forEach((log: { id: string; eventType: string; action: string; status: string; createdAt: string; error?: string }) => {
   156→          let type = 'automation'
   157→          if (log.action?.includes('deploy') || log.eventType?.includes('hosting')) type = 'deployment'
   158→          else if (log.action?.includes('domain') || log.action?.includes('dns')) type = 'domain'
   159→          else if (log.eventType?.includes('payment')) type = 'payment'
   160→          else if (log.action?.includes('ssl')) type = 'ssl'
   161→
   162→          const message = log.action
   163→            ? log.action.replace(/_/g, ' ').replace(/\b\w/g, (c: string) => c.toUpperCase())
   164→            : log.eventType?.replace(/_/g, ' ').replace(/\b\w/g, (c: string) => c.toUpperCase()) || 'Automation event'
   165→
   166→          timeline.push({ id: `log-${log.id}`, type, message, time: log.createdAt, status: log.status })
   167→        })
   168→
   169→        // From deployments
   170→        depList.slice(0, 5).forEach((d: { id: string; status: string; createdAt: string; project?: { name: string } }) => {
   171→          const projName = d.project?.name || 'Project'
   172→          const statusLabel = d.status === 'live' ? 'deployed successfully' : d.status === 'failed' ? 'deployment failed' : `deployment ${d.status}`
   173→          timeline.push({
   174→            id: `dep-${d.id}`,
   175→            type: 'deployment',
   176→            message: `${projName} ${statusLabel}`,
   177→            time: d.createdAt,
   178→            status: d.status,
   179→          })
   180→        })
   181→
   182→        // From domains
   183→        const domList = Array.isArray(domData.domains) ? domData.domains : []
   184→        domList.slice(0, 3).forEach((d: { id: string; status: string; name: string; createdAt: string }) => {
   185→          timeline.push({
   186→            id: `dom-${d.id}`,
   187→            type: 'domain',
   188→            message: `Domain ${d.name} ${d.status === 'active' ? 'registered' : d.status}`,
   189→            time: d.createdAt,
   190→            status: d.status,
   191→          })
   192→        })
   193→
   194→        // Sort by time descending, deduplicate
   195→        timeline.sort((a, b) => new Date(b.time).getTime() - new Date(a.time).getTime())
   196→        const seen = new Set<string>()
   197→        const unique = timeline.filter((item) => {
   198→          if (seen.has(item.id)) return false
   199→          seen.add(item.id)
   200→          return true
   201→        })
   202→        setActivityTimeline(unique.slice(0, 10))
   203→      } catch {
   204→        // Silently handle
   205→      } finally {
   206→        setLoading(false)
   207→      }
   208→    }
   209→    loadData()
   210→  }, [])
   211→
   212→  // Helpers
   213→  const formatDate = () => {
   214→    return new Date().toLocaleDateString('en-US', { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' })
   215→  }
   216→
   217→  const relativeTime = (dateStr: string) => {
   218→    const diff = Date.now() - new Date(dateStr).getTime()
   219→    const mins = Math.floor(diff / 60000)
   220→    if (mins < 1) return 'Just now'
   221→    if (mins < 60) return `${mins}m ago`
   222→    const hours = Math.floor(mins / 60)
   223→    if (hours < 24) return `${hours}h ago`
   224→    const days = Math.floor(hours / 24)
   225→    if (days < 7) return `${days}d ago`
   226→    return new Date(dateStr).toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   227→  }
   228→
   229→  const parseGB = (s: string) => {
   230→    const num = parseFloat(s)
   231→    return isNaN(num) ? 10 : num
   232→  }
   233→
   234→  // Resource usage calculations
   235→  const storageLimit = parseGB(planInfo?.storage || '10GB')
   236→  const bandwidthLimit = parseGB(planInfo?.bandwidth || '100GB')
   237→  const projectsLimit = planInfo?.projects || 3
   238→  const storageUsed = Math.min(projectsCount * 1.8 + 0.5, storageLimit)
   239→  const bandwidthUsed = Math.min(deploymentsCount * 2.5 + 1.2, bandwidthLimit)
   240→  const storagePct = Math.round((storageUsed / storageLimit) * 100)
   241→  const bandwidthPct = Math.round((bandwidthUsed / bandwidthLimit) * 100)
   242→  const projectsPct = Math.round((projectsCount / projectsLimit) * 100)
   243→
   244→  const barColor = (pct: number) =>
   245→    pct > 80 ? 'bg-red-500' : pct > 60 ? 'bg-amber-500' : 'bg-emerald-500'
   246→
   247→  const barBgColor = (pct: number) =>
   248→    pct > 80 ? 'bg-red-100' : pct > 60 ? 'bg-amber-100' : 'bg-emerald-100'
   249→
   250→  // Uptime calculation
   251→  const liveCount = deployments.filter((d) => d.status === 'live').length
   252→  const totalDep = deployments.length
   253→  const uptimePct = totalDep > 0 ? Math.round((liveCount / totalDep) * 100) : 99.9
   254→
   255→  // Timeline icon/color per type
   256→  const timelineIcon = (type: string) => {
   257→    switch (type) {
   258→      case 'deployment': return { icon: Rocket, color: 'text-emerald-600', bg: 'bg-emerald-100', dot: 'bg-emerald-500' }
   259→      case 'domain': return { icon: Globe, color: 'text-amber-600', bg: 'bg-amber-100', dot: 'bg-amber-500' }
   260→      case 'automation': return { icon: Zap, color: 'text-violet-600', bg: 'bg-violet-100', dot: 'bg-violet-500' }
   261→      case 'payment': return { icon: CreditCard, color: 'text-cyan-600', bg: 'bg-cyan-100', dot: 'bg-cyan-500' }
   262→      case 'ssl': return { icon: Shield, color: 'text-green-600', bg: 'bg-green-100', dot: 'bg-green-500' }
   263→      default: return { icon: Activity, color: 'text-gray-600', bg: 'bg-gray-100', dot: 'bg-gray-500' }
   264→    }
   265→  }
   266→
   267→  const statusBadge = (status: string) => {
   268→    switch (status) {
   269→      case 'live':
   270→      case 'active':
   271→      case 'completed':
   272→        return <Badge className="bg-emerald-50 text-emerald-700 border-emerald-200 text-[10px] capitalize">{status}</Badge>
   273→      case 'building':
   274→      case 'deploying':
   275→      case 'running':
   276→      case 'pending':
   277→        return <Badge className="bg-amber-50 text-amber-700 border-amber-200 text-[10px] capitalize">{status}</Badge>
   278→      case 'failed':
   279→      case 'rolled_back':
   280→        return <Badge className="bg-red-50 text-red-700 border-red-200 text-[10px] capitalize">{status}</Badge>
   281→      default:
   282→        return <Badge variant="outline" className="text-[10px] capitalize">{status}</Badge>
   283→    }
   284→  }
   285→
   286→  // Stat cards config
   287→  const statCards = [
   288→    {
   289→      title: 'Projects',
   290→      value: projectsCount,
   291→      icon: Server,
   292→      gradient: 'from-emerald-500 to-emerald-600',
   293→      iconBg: 'bg-emerald-500/20',
   294→      trend: projectsCount > 0 ? `+${Math.min(projectsCount, 5)} this week` : 'No projects yet',
   295→      trendUp: true,
   296→      tab: 'hosting' as DashboardTab,
   297→    },
   298→    {
   299→      title: 'Domains',
   300→      value: domainsCount,
   301→      icon: Globe,
   302→      gradient: 'from-amber-500 to-amber-600',
   303→      iconBg: 'bg-amber-500/20',
   304→      trend: domainsCount > 0 ? `+${Math.min(domainsCount, 3)} this week` : 'No domains yet',
   305→      trendUp: true,
   306→      tab: 'domains' as DashboardTab,
   307→    },
   308→    {
   309→      title: 'Deployments',
   310→      value: deploymentsCount,
   311→      icon: Rocket,
   312→      gradient: 'from-violet-500 to-violet-600',
   313→      iconBg: 'bg-violet-500/20',
   314→      trend: deploymentsCount > 0 ? `↑ ${uptimePct}% success` : 'No deployments yet',
   315→      trendUp: uptimePct >= 80,
   316→      tab: 'deployments' as DashboardTab,
   317→    },
   318→    {
   319→      title: 'Uptime',
   320→      value: `${uptimePct}%`,
   321→      icon: Activity,
   322→      gradient: 'from-cyan-500 to-cyan-600',
   323→      iconBg: 'bg-cyan-500/20',
   324→      trend: uptimePct >= 99 ? 'Excellent' : uptimePct >= 95 ? 'Good' : 'Needs attention',
   325→      trendUp: uptimePct >= 95,
   326→      tab: 'deployments' as DashboardTab,
   327→    },
   328→  ]
   329→
   330→  // Quick actions
   331→  const quickActions = [
   332→    { label: 'New Project', desc: 'Create hosting project', icon: Server, tab: 'hosting' as DashboardTab, gradient: 'group-hover:from-emerald-500 group-hover:to-teal-500' },
   333→    { label: 'Register Domain', desc: 'Search & register domains', icon: Globe, tab: 'domains' as DashboardTab, gradient: 'group-hover:from-amber-500 group-hover:to-orange-500' },
   334→    { label: 'View Deployments', desc: 'Recent deployment history', icon: Rocket, tab: 'deployments' as DashboardTab, gradient: 'group-hover:from-violet-500 group-hover:to-purple-500' },
   335→    { label: 'Manage Billing', desc: 'Plans & invoices', icon: CreditCard, tab: 'billing' as DashboardTab, gradient: 'group-hover:from-cyan-500 group-hover:to-blue-500' },
   336→  ]
   337→
   338→  // Max chart value for scaling
   339→  const maxChartVal = Math.max(...chartData.map((d) => d.success + d.failed), 1)
   340→
   341→  return (
   342→    <div className="space-y-6">
   343→      {/* Section 1: Welcome Header */}
   344→      <div className="flex flex-col sm:flex-row sm:items-end sm:justify-between gap-2">
   345→        <div>
   346→          <h1 className="text-2xl sm:text-3xl font-bold">
   347→            <span className="bg-gradient-to-r from-emerald-600 to-teal-600 bg-clip-text text-transparent">
   348→              Welcome back{user?.name ? `, ${user.name}` : ''}!
   349→            </span>
   350→          </h1>
   351→          <p className="text-muted-foreground mt-1">{formatDate()}</p>
   352→        </div>
   353→        {planInfo && (
   354→          <Badge className="bg-emerald-50 text-emerald-700 border-emerald-200 shrink-0 self-start sm:self-auto">
   355→            {planInfo.name} Plan
   356→          </Badge>
   357→        )}
   358→      </div>
   359→
   360→      {/* Section 2: Stats Cards */}
   361→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   362→        {statCards.map((card) => {
   363→          const Icon = card.icon
   364→          return (
   365→            <button
   366→              key={card.title}
   367→              onClick={() => setDashboardTab(card.tab)}
   368→              className="relative bg-card border rounded-xl p-5 flex items-center gap-4 hover:shadow-md transition-all text-left group overflow-hidden"
   369→            >
   370→              {/* Gradient overlay on hover */}
   371→              <div className={`absolute inset-0 bg-gradient-to-br ${card.gradient} opacity-0 group-hover:opacity-[0.07] transition-opacity duration-300`} />
   372→              <div className={`size-12 rounded-xl bg-gradient-to-br ${card.gradient} flex items-center justify-center shrink-0 shadow-sm`}>
   373→                <Icon className="size-6 text-white" />
   374→              </div>
   375→              <div className="flex-1 min-w-0">
   376→                <p className="text-sm text-muted-foreground">{card.title}</p>
   377→                <p className="text-2xl font-bold mt-0.5">{loading ? '...' : card.value}</p>
   378→                <div className="flex items-center gap-1 mt-1">
   379→                  {card.trendUp ? (
   380→                    <TrendingUp className="size-3 text-emerald-500" />
   381→                  ) : (
   382→                    <TrendingDown className="size-3 text-red-500" />
   383→                  )}
   384→                  <span className={`text-[11px] font-medium ${card.trendUp ? 'text-emerald-600' : 'text-red-600'}`}>
   385→                    {card.trend}
   386→                  </span>
   387→                </div>
   388→              </div>
   389→              <ArrowUpRight className="size-4 text-muted-foreground opacity-0 group-hover:opacity-100 transition-opacity relative z-10" />
   390→            </button>
   391→          )
   392→        })}
   393→      </div>
   394→
   395→      {/* Section 3: Deployment Activity Chart */}
   396→      <div className="bg-card border rounded-xl p-5">
   397→        <div className="flex items-center justify-between mb-4">
   398→          <div>
   399→            <h2 className="text-lg font-semibold">Deployment Activity</h2>
   400→            <p className="text-sm text-muted-foreground">Last 7 days</p>
   401→          </div>
   402→          <div className="flex items-center gap-4 text-xs">
   403→            <div className="flex items-center gap-1.5">
   404→              <div className="size-2.5 rounded-sm bg-emerald-500" />
   405→              <span className="text-muted-foreground">Success</span>
   406→            </div>
   407→            <div className="flex items-center gap-1.5">
   408→              <div className="size-2.5 rounded-sm bg-red-500" />
   409→              <span className="text-muted-foreground">Failed</span>
   410→            </div>
   411→          </div>
   412→        </div>
   413→        <div className="flex items-end gap-2 sm:gap-3 h-40">
   414→          {chartData.map((day) => {
   415→            const successH = maxChartVal > 0 ? (day.success / maxChartVal) * 100 : 0
   416→            const failedH = maxChartVal > 0 ? (day.failed / maxChartVal) * 100 : 0
   417→            return (
   418→              <div key={day.day} className="flex-1 flex flex-col items-center gap-1 group/bar relative">
   419→                {/* Tooltip */}
   420→                <div className="absolute -top-8 bg-foreground text-background text-[10px] px-2 py-1 rounded-md opacity-0 group-hover/bar:opacity-100 transition-opacity whitespace-nowrap z-10 pointer-events-none">
   421→                  {day.success} success{day.failed > 0 ? `, ${day.failed} failed` : ''}
   422→                </div>
   423→                <div className="flex items-end gap-0.5 w-full h-32">
   424→                  {/* Success bar */}
   425→                  <div className="flex-1 flex flex-col justify-end">
   426→                    <div
   427→                      className="bg-emerald-500 rounded-t-sm transition-all duration-500 min-h-[2px]"
   428→                      style={{ height: `${Math.max(successH, 2)}%` }}
   429→                    />
   430→                  </div>
   431→                  {/* Failed bar */}
   432→                  {day.failed > 0 && (
   433→                    <div className="flex-1 flex flex-col justify-end">
   434→                      <div
   435→                        className="bg-red-500 rounded-t-sm transition-all duration-500 min-h-[2px]"
   436→                        style={{ height: `${Math.max(failedH, 2)}%` }}
   437→                      />
   438→                    </div>
   439→                  )}
   440→                </div>
   441→                <span className="text-[11px] text-muted-foreground font-medium">{day.day}</span>
   442→              </div>
   443→            )
   444→          })}
   445→        </div>
   446→      </div>
   447→
   448→      {/* Section 4: Resource Usage */}
   449→      <div className="bg-card border rounded-xl p-5">
   450→        <h2 className="text-lg font-semibold mb-4">Resource Usage</h2>
   451→        <div className="space-y-4">
   452→          {/* Storage */}
   453→          <div>
   454→            <div className="flex items-center justify-between mb-1.5">
   455→              <div className="flex items-center gap-2">
   456→                <HardDrive className="size-4 text-muted-foreground" />

   732→// ── Root page ──────────────────────────────────────────────────────────
   733→export default function Home() {
   734→  const { view, setView, authMode, setAuthMode } = useAppStore()
   735→  const { isAuthenticated, isLoading, loadUser } = useAuthStore()
   736→  const [showAuthModal, setShowAuthModal] = useState(false)
   737→  const [showOnboarding, setShowOnboarding] = useState(() => {
   738→    if (typeof window !== 'undefined') {
   739→      return localStorage.getItem('deployhub_onboarding_complete') !== 'true'
   740→    }
   741→    return false
   742→  })
   743→
   744→  useEffect(() => {
   745→    loadUser()
   746→  }, [loadUser])
   747→
   748→  useEffect(() => {
   749→    if (!isLoading) {
   750→      if (isAuthenticated) {
   751→        setView('dashboard')
   752→      }
   753→      // If not authenticated, stay on landing page
   754→    }
   755→  }, [isLoading, isAuthenticated, setView])
   756→
   757→  const handleOnboardingComplete = () => {
   758→    localStorage.setItem('deployhub_onboarding_complete', 'true')
   759→    setShowOnboarding(false)
   760→  }
   761→
   762→  const handleOnboardingSkip = () => {
   763→    localStorage.setItem('deployhub_onboarding_complete', 'true')
   764→    setShowOnboarding(false)
   765→  }
   766→
   767→  // Loading screen
   768→  if (isLoading) {
   769→    return (
   770→      <div className="flex items-center justify-center min-h-screen">
   771→        <div className="flex flex-col items-center gap-3">
   772→          <div className="size-10 animate-spin rounded-full border-4 border-emerald-600 border-t-transparent" />
   773→          <p className="text-sm text-muted-foreground">Loading DeployHub...</p>
   774→        </div>
   775→      </div>
   776→    )
   777→  }
   778→
   779→  // Authenticated + Onboarding
   780→  if (isAuthenticated && showOnboarding) {
   781→    return (
   782→      <OnboardingWizard
   783→        onComplete={handleOnboardingComplete}
   784→        onSkip={handleOnboardingSkip}
   785→      />
   786→    )
   787→  }
   788→
   789→  // Authenticated → Dashboard
   790→  if (isAuthenticated && view === 'dashboard') {
   791→    return <Dashboard />
   792→  }
   793→
   794→  // Not authenticated → Landing page with auth modal
   795→  return (
   796→    <div className="min-h-screen flex flex-col">
   797→      <LandingPage
   798→        onGetStarted={() => {
   799→          setAuthMode('register')
   800→          setShowAuthModal(true)
   801→        }}
   802→        onLogin={() => {
   803→          setAuthMode('login')
   804→          setShowAuthModal(true)
   805→        }}
   806→      />
   807→      {showAuthModal && (
   808→        <AuthModal
   809→          mode={authMode}
   810→          onModeSwitch={() => {
   811→            setAuthMode(authMode === 'login' ? 'register' : 'login')
   812→          }}
   813→          onClose={() => setShowAuthModal(false)}
   814→        />
   815→      )}
   816→    </div>
   817→  )
   818→}
