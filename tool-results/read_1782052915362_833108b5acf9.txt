     1→/**
     2→ * Core Orchestration Engine
     3→ * =========================
     4→ *
     5→ * The heart of DeployHub's automated provisioning pipeline. Coordinates the
     6→ * full flow:
     7→ *
     8→ *     domain registration → DNS configuration → web hosting → email hosting → verification
     9→ *
    10→ * Each step is independently retryable via `withRetry()` with exponential
    11→ * backoff. Domain + DNS are CRITICAL steps — if either fails after retries,
    12→ * the pipeline stops and triggers `rollbackProvisioning()` in reverse order.
    13→ * Hosting + Email are NON-CRITICAL — if they fail but domain+DNS succeeded,
    14→ * the pipeline continues to verification and the overall status is `partial`.
    15→ *
    16→ * All external API calls (Dynadot, Mailcow) have DEMO MODE fallbacks
    17→ * so the entire pipeline runs out-of-the-box without any real provider env
    18→ * vars configured.
    19→ *
    20→ * Used by:
    21→ *  - Task #19 Orchestration Engine — ties together Dynadot, DNS, and
    22→ *    Mailcow into a single end-to-end provisioning flow.
    23→ */

   179→/** Resolve the hosting server IP — from Server table, or default for demo. */

   182→    const server = await db.server.findUnique({ where: { id: serverId } })

   203→// ─── Retry helper ────────────────────────────────────────────────────────

   205→/**
   206→ * Retry a function up to `maxAttempts` times with exponential backoff
   207→ * (1000ms * 2^attempt). Each attempt increments the ProvisioningStep.attempts
   208→ * counter and updates lastAttemptAt.
   209→ */
   210→async function withRetry<T>(
   211→  fn: () => Promise<T>,
   212→  opts: { maxAttempts: number; stepName: string; jobId: string; stepId: string }
   213→): Promise<T> {
   214→  let lastError: unknown = null
   215→  for (let attempt = 1; attempt <= opts.maxAttempts; attempt++) {
   216→    try {
   217→      // Update attempt tracking on the step row.
   218→      await db.provisioningStep.update({
   219→        where: { id: opts.stepId },
   220→        data: {
   221→          attempts: attempt,
   222→          lastAttemptAt: new Date(),
   223→        },
   224→      })
   225→      console.log(
   226→        `[orchestration] job=${opts.jobId} step=${opts.stepName} attempt=${attempt}/${opts.maxAttempts}`
   227→      )
   228→      const result = await fn()
   229→      return result
   230→    } catch (err) {
   231→      lastError = err
   232→      const msg = err instanceof Error ? err.message : String(err)
   233→      console.warn(
   234→        `[orchestration] job=${opts.jobId} step=${opts.stepName} attempt=${attempt} failed: ${msg}`
   235→      )
   236→      if (attempt < opts.maxAttempts) {
   237→        const backoff = 1000 * Math.pow(2, attempt - 1)
   238→        await sleep(backoff)
   239→      }
   240→    }
   241→  }
   242→  throw lastError instanceof Error
   243→    ? lastError
   244→    : new Error(`Step ${opts.stepName} failed after ${opts.maxAttempts} attempts`)
   245→}

   247→// ─── startProvisioning ───────────────────────────────────────────────────
   248→
   249→/**
   250→ * Create a ProvisioningJob record with 5 ProvisioningStep rows, then kick off
   251→ * the pipeline in the background (NOT awaited). Returns immediately with the
   252→ * jobId so the caller can poll for status via getProvisioningStatus().
   253→ */
   254→export async function startProvisioning(
   255→  req: ProvisioningRequest
   256→): Promise<{ jobId: string; status: 'started' | 'failed'; error?: string }> {
   257→  const includeHosting = req.includeHosting ?? true
   258→  const includeEmail = req.includeEmail ?? true
   259→
   260→  try {
   261→    // 1. Create the ProvisioningJob record.
   262→    const job = await db.provisioningJob.create({
   263→      data: {
   264→        userId: req.userId,
   265→        domainName: req.domainName,
   266→        stack: req.stack,
   267→        includeHosting,
   268→        includeEmail,
   269→        status: 'pending',
   270→        currentStep: 'init',
   271→      },
   272→    })
   273→
   274→    // 2. Create the 5 ProvisioningStep records.
   275→    for (const def of STEP_DEFINITIONS) {
   276→      const skipped =
   277→        (def.name === 'hosting_provision' && !includeHosting) ||
   278→        (def.name === 'email_provision' && !includeEmail)
   279→      await db.provisioningStep.create({
   280→        data: {
   281→          jobId: job.id,
   282→          stepName: def.name,
   283→          stepOrder: def.order,
   284→          status: skipped ? 'skipped' : 'pending',
   285→          maxAttempts: 3,
   286→          input: JSON.stringify({
   287→            domainName: req.domainName,
   288→            stack: req.stack,
   289→            includeHosting,
   290→            includeEmail,
   291→            planId: req.planId,
   292→            serverId: req.serverId,
   293→          }),
   294→        },
   295→      })
   296→    }
   297→
   298→    await logAudit({
   299→      userId: req.userId,
   300→      action: 'provisioning.start',
   301→      resource: 'provisioning_job',
   302→      resourceId: job.id,
   303→      metadata: {
   304→        domainName: req.domainName,
   305→        stack: req.stack,
   306→        includeHosting,
   307→        includeEmail,
   308→      },
   309→    })
   310→
   311→    // 3. Kick off the pipeline in the background. Do NOT await.
   312→    runProvisioningPipeline(job.id).catch((err) => {
   313→      console.error(
   314→        `[orchestration] Background pipeline failed for job ${job.id}:`,
   315→        err instanceof Error ? err.message : err
   316→      )
   317→    })
   318→
   319→    return { jobId: job.id, status: 'started' }
   320→  } catch (err) {
   321→    const msg = err instanceof Error ? err.message : String(err)
   322→    console.error('[orchestration] startProvisioning failed:', msg)
   323→    return { jobId: '', status: 'failed', error: msg }
   324→  }
   325→}

   327→// ─── runProvisioningPipeline ─────────────────────────────────────────────
   328→
   329→/**
   330→ * Run the full provisioning pipeline for a job. Updates the job's status,
   331→ * runs each step in order with retries, builds the ProvisioningResult, and
   332→ * sends a notification to the user on completion.
   333→ *
   334→ * Called by startProvisioning() in the background (fire-and-forget) or by a
   335→ * worker process.
   336→ */
   337→export async function runProvisioningPipeline(
   338→  jobId: string
   339→): Promise<ProvisioningResult> {
   340→  const job = await db.provisioningJob.findUnique({
   341→    where: { id: jobId },
   342→    include: { steps: { orderBy: { stepOrder: 'asc' } } },
   343→  })
   344→  if (!job) {
   345→    throw new Error(`ProvisioningJob not found: ${jobId}`)
   346→  }
   347→  if (job.status === 'running') {
   348→    throw new Error(`Pipeline already running for job ${jobId}`)
   349→  }
   350→  if (job.status === 'success' || job.status === 'failed' || job.status === 'rolled_back') {
   351→    // Already completed — return the stored result if available.
   352→    if (job.result) {
   353→      try {
   354→        return JSON.parse(job.result) as ProvisioningResult
   355→      } catch {
   356→        // fall through and re-run
   357→      }
   358→    }
   359→  }
   360→
   361→  const startedAt = new Date()
   362→  await db.provisioningJob.update({
   363→    where: { id: jobId },
   364→    data: { status: 'running', startedAt, currentStep: 'domain' },
   365→  })
   366→
   367→  const req = JSON.parse(job.steps[0]?.input || '{}') as ProvisioningRequest
   368→  const ctx: PipelineContext = {
   369→    domainName: job.domainName,
   370→    provider: 'dynadot',
   371→    mailServerHostname: resolveMailServerHostname(),
   372→    hostingServerIp: await resolveHostingServerIp(req.serverId),
   373→    mailboxes: [],
   374→    dnsRecords: [],
   375→    errors: [],
   376→    hostingReachable: false,
   377→    emailVerified: false,
   378→    dnsPropagated: false,
   379→    nameservers: [],
   380→  }
   381→
   382→  let domainResult: { domainId: string; externalId?: string; provider: string } | undefined
   383→  let dnsResult:
   384→    | { records: Array<{ type: string; name: string; value: string; recordId: string }> }
   385→    | undefined
   386→  let hostingResult:
   387→    | { projectId: string; url: string }
   388→    | undefined
   389→  let emailResult:
   390→    | {
   391→        mailboxes: Array<{
   392→          address: string
   393→          password: string
   394→          type: string
   395→          mailboxId: string
   396→        }>
   397→        dkimPublicKey: string
   398→        webmailUrl: string
   399→      }
   400→    | undefined
   401→  let verifyResult:
   402→    | { hostingReachable: boolean; emailVerified: boolean; dnsPropagated: boolean }
   403→    | undefined
   404→
   405→  // Track whether non-critical steps failed (for 'partial' status).
   406→  let hostingFailed = false
   407→  let emailFailed = false
   408→  let criticalFailed = false
   409→
   410→  for (const step of job.steps) {
   411→    // Skip steps that are explicitly skipped (includeHosting/includeEmail = false)
   412→    if (step.status === 'skipped') continue
   413→    // Skip steps that already succeeded (idempotent re-runs)
   414→    if (step.status === 'success') {
   415→      // Re-hydrate context from stored output so subsequent steps have the data
   416→      try {
   417→        const out = step.output ? JSON.parse(step.output) : undefined
   418→        if (step.stepName === 'domain_register' && out) {
   419→          domainResult = out
   420→          ctx.domainId = out.domainId
   421→          ctx.externalId = out.externalId
   422→          ctx.provider = out.provider
   423→        }
   424→        if (step.stepName === 'dns_configure' && out) {
   425→          dnsResult = out
   426→          ctx.dnsRecords = out.records || []
   427→        }
   428→        if (step.stepName === 'hosting_provision' && out) {
   429→          hostingResult = out
   430→          ctx.projectId = out.projectId
   431→          ctx.hostingUrl = out.url
   432→        }
   433→        if (step.stepName === 'email_provision' && out) {
   434→          emailResult = out
   435→          ctx.mailboxes = out.mailboxes || []
   436→          ctx.dkimPublicKey = out.dkimPublicKey
   437→          ctx.webmailUrl = out.webmailUrl
   438→        }
   439→      } catch {
   440→        // ignore parse errors
   441→      }
   442→      continue
   443→    }
   444→
   445→    // Mark step running.
   446→    const stepStart = new Date()
   447→    await db.provisioningStep.update({
   448→      where: { id: step.id },
   449→      data: { status: 'running', startedAt: stepStart, attempts: 0 },
   450→    })
   451→    await db.provisioningJob.update({
   452→      where: { id: jobId },
   453→      data: { currentStep: step.stepName },
   454→    })
   455→
   456→    try {
   457→      let output: unknown
   458→      switch (step.stepName) {
   459→        case 'domain_register':
   460→          output = domainResult = await withRetry(
   461→            () => provisionDomain(job, step),
   462→            {
   463→              maxAttempts: step.maxAttempts,
   464→              stepName: step.stepName,
   465→              jobId,
   466→              stepId: step.id,
   467→            }
   468→          )
   469→          ctx.domainId = domainResult.domainId
   470→          ctx.externalId = domainResult.externalId
   471→          ctx.provider = domainResult.provider
   472→          break
   473→        case 'dns_configure':
   474→          if (!ctx.domainId) {
   475→            throw new Error('Cannot configure DNS — domain step did not produce a domainId')
   476→          }
   477→          output = dnsResult = await withRetry(
   478→            () =>
   479→              configureDns(
   480→                job,
   481→                step,
   482→                ctx.domainId!,
   483→                ctx.mailServerHostname,
   484→                ctx.hostingServerIp
   485→              ),
   486→            {
   487→              maxAttempts: step.maxAttempts,
   488→              stepName: step.stepName,
   489→              jobId,
   490→              stepId: step.id,
   491→            }
   492→          )
   493→          ctx.dnsRecords = dnsResult.records
   494→          break
   495→        case 'hosting_provision':
   496→          if (!ctx.domainId) {
   497→            throw new Error('Cannot provision hosting — no domainId in context')
   498→          }
   499→          output = hostingResult = await withRetry(
   500→            () => provisionHosting(job, step, ctx.domainId!, ctx.domainName),
   501→            {
   502→              maxAttempts: step.maxAttempts,
   503→              stepName: step.stepName,
   504→              jobId,
   505→              stepId: step.id,
   506→            }
   507→          )
   508→          ctx.projectId = hostingResult.projectId
   509→          ctx.hostingUrl = hostingResult.url
   510→          break
   511→        case 'email_provision':
   512→          if (!ctx.domainId) {
   513→            throw new Error('Cannot provision email — no domainId in context')
   514→          }
   515→          output = emailResult = await withRetry(
   516→            () => provisionEmail(job, step, ctx.domainId!, ctx.domainName),
   517→            {
   518→              maxAttempts: step.maxAttempts,
   519→              stepName: step.stepName,
   520→              jobId,
   521→              stepId: step.id,
   522→            }
   523→          )
   524→          ctx.mailboxes = emailResult.mailboxes
   525→          ctx.dkimPublicKey = emailResult.dkimPublicKey
   526→          ctx.webmailUrl = emailResult.webmailUrl
   527→          break
   528→        case 'verify':
   529→          output = verifyResult = await withRetry(
   530→            () => verifyProvisioning(job, step, ctx.domainName, ctx.projectId || ''),
   531→            {
   532→              maxAttempts: step.maxAttempts,
   533→              stepName: step.stepName,
   534→              jobId,
   535→              stepId: step.id,
   536→            }
   537→          )
   538→          ctx.hostingReachable = verifyResult.hostingReachable
   539→          ctx.emailVerified = verifyResult.emailVerified
   540→          ctx.dnsPropagated = verifyResult.dnsPropagated
   541→          break
   542→        default:
   543→          throw new Error(`Unknown step: ${step.stepName}`)
   544→      }
   545→
   546→      // Step succeeded — persist output + status.
   547→      const stepEnd = new Date()
   548→      await db.provisioningStep.update({
   549→        where: { id: step.id },
   550→        data: {
   551→          status: 'success',
   552→          output: JSON.stringify(output),
   553→          completedAt: stepEnd,
   554→          duration: stepEnd.getTime() - stepStart.getTime(),
   555→        },
   556→      })
   557→
   558→      await logAudit({
   559→        userId: job.userId,
   560→        action: `provisioning.step.${step.stepName}.success`,
   561→        resource: 'provisioning_step',
   562→        resourceId: step.id,
   563→        metadata: { jobId, stepName: step.stepName },
   564→      })
   565→    } catch (err) {
   566→      const msg = err instanceof Error ? err.message : String(err)
   567→      const stepEnd = new Date()
   568→      ctx.errors.push({
   569→        step: step.stepName,
   570→        error: msg,
   571→        timestamp: stepEnd.toISOString(),
   572→      })
   573→
   574→      await db.provisioningStep.update({
   575→        where: { id: step.id },
   576→        data: {
   577→          status: 'failed',
   578→          error: msg,
   579→          completedAt: stepEnd,
   580→          duration: stepEnd.getTime() - stepStart.getTime(),
   581→        },
   582→      })
   583→
   584→      await logAudit({
   585→        userId: job.userId,
   586→        action: `provisioning.step.${step.stepName}.failed`,
   587→        resource: 'provisioning_step',
   588→        resourceId: step.id,
   589→        metadata: { jobId, stepName: step.stepName, error: msg },
   590→        status: 'failure',
   591→      })
   592→
   593→      console.error(
   594→        `[orchestration] job=${jobId} step=${step.stepName} FAILED after retries: ${msg}`
   595→      )
   596→
   597→      // Critical step failure → stop pipeline + rollback.
   598→      if (CRITICAL_STEPS.has(step.stepName)) {
   599→        criticalFailed = true
   600→        break
   601→      }
   602→      // Non-critical step failure → mark partial + continue.
   603→      if (step.stepName === 'hosting_provision') hostingFailed = true
   604→      if (step.stepName === 'email_provision') emailFailed = true
   605→      // For verify step, just continue — we'll reflect status in the result.
   606→    }
   607→  }
   608→
   609→  // ─── Rollback on critical failure ────────────────────────────────────
   610→  if (criticalFailed) {
   611→    await rollbackProvisioning(jobId).catch((rbErr) => {
   612→      console.error(
   613→        `[orchestration] rollback failed for job ${jobId}:`,
   614→        rbErr instanceof Error ? rbErr.message : rbErr
   615→      )
   616→    })
   617→    const completedAt = new Date()
   618→    const duration = completedAt.getTime() - startedAt.getTime()
   619→    const result: ProvisioningResult = buildResult(
   620→      job.id,
   621→      ctx,
   622→      domainResult,
   623→      dnsResult,
   624→      hostingResult,
   625→      emailResult,
   626→      verifyResult,
   627→      hostingFailed,
   628→      emailFailed,
   629→      'failed',
   630→      duration
   631→    )
   632→    await db.provisioningJob.update({
   633→      where: { id: jobId },
   634→      data: {
   635→        status: 'rolled_back',
   636→        currentStep: 'rolled_back',
   637→        result: JSON.stringify(result),
   638→        completedAt,
   639→        error: 'Critical step failed — pipeline rolled back',
   640→      },
   641→    })
   642→    await createNotification({
   643→      userId: job.userId,
   644→      title: 'Provisioning failed',
   645→      message: `Provisioning of ${job.domainName} failed and was rolled back. See job ${jobId} for details.`,
   646→      type: 'error',
   647→      link: `/dashboard/provisioning/${jobId}`,
   648→    }).catch(() => {})
   649→    return result
   650→  }
   651→
   652→  // ─── Build final result ──────────────────────────────────────────────
   653→  const completedAt = new Date()
   654→  const duration = completedAt.getTime() - startedAt.getTime()
   655→  const overallStatus: 'success' | 'partial' | 'failed' =
   656→    hostingFailed || emailFailed ? (domainResult && dnsResult ? 'partial' : 'failed') : 'success'
   657→
   658→  const result = buildResult(
   659→    job.id,
   660→    ctx,
   661→    domainResult,
   662→    dnsResult,
   663→    hostingResult,
   664→    emailResult,
   665→    verifyResult,
   666→    hostingFailed,
   667→    emailFailed,
   668→    overallStatus,
   669→    duration
   670→  )
   671→
   672→  await db.provisioningJob.update({
   673→    where: { id: jobId },
   674→    data: {
   675→      status: overallStatus,
   676→      currentStep: 'done',
   677→      result: JSON.stringify(result),
   678→      completedAt,
   679→    },
   680→  })
   681→
   682→  // Send a notification to the user.
   683→  const notifType = overallStatus === 'success' ? 'success' : overallStatus === 'partial' ? 'warning' : 'error'
   684→  const notifMsg =
   685→    overallStatus === 'success'
   686→      ? `Your deployment ${job.domainName} is ready! Domain, DNS, hosting, and email are all configured.`
   687→      : overallStatus === 'partial'
   688→        ? `Provisioning of ${job.domainName} is partially complete. Some services may need attention.`
   689→        : `Provisioning of ${job.domainName} failed. Please check the job details.`
   690→  await createNotification({
   691→    userId: job.userId,
   692→    title:
   693→      overallStatus === 'success'
   694→        ? 'Provisioning complete'
   695→        : overallStatus === 'partial'
   696→          ? 'Provisioning partial'
   697→          : 'Provisioning failed',
   698→    message: notifMsg,
   699→    type: notifType,
   700→    link: `/dashboard/provisioning/${jobId}`,
   701→  }).catch(() => {})
   702→
   703→  await logAudit({
   704→    userId: job.userId,
   705→    action: `provisioning.complete.${overallStatus}`,
   706→    resource: 'provisioning_job',
   707→    resourceId: jobId,
   708→    metadata: { domainName: job.domainName, duration, overallStatus },
   709→  })
   710→
   711→  return result
   712→}
   713→
   714→// ─── Build the ProvisioningResult from the per-step outputs ──────────────
   715→
   716→function buildResult(
   717→  jobId: string,
   718→  ctx: PipelineContext,
   719→  domainResult: { domainId: string; externalId?: string; provider: string } | undefined,
   720→  dnsResult:
   721→    | { records: Array<{ type: string; name: string; value: string; recordId: string }> }
   722→    | undefined,
   723→  hostingResult: { projectId: string; url: string } | undefined,
   724→  emailResult:
   725→    | {
   726→        mailboxes: Array<{
   727→          address: string
   728→          password: string
   729→          type: string
   730→          mailboxId: string
   731→        }>
   732→        dkimPublicKey: string
   733→        webmailUrl: string
   734→      }
   735→    | undefined,
   736→  verifyResult:
   737→    | { hostingReachable: boolean; emailVerified: boolean; dnsPropagated: boolean }
   738→    | undefined,
   739→  hostingFailed: boolean,
   740→  emailFailed: boolean,
   741→  overallStatus: 'success' | 'partial' | 'failed',
   742→  duration: number
   743→): ProvisioningResult {
   744→  const domainStatus: ProvisioningResult['domainStatus'] = {
   745→    status: domainResult ? 'success' : 'fail',
   746→    domain: ctx.domainName,
   747→    provider: domainResult?.provider || ctx.provider,
   748→    externalId: domainResult?.externalId || ctx.externalId,
   749→    registered: !!domainResult,
   750→    details: domainResult
   751→      ? `Domain registered via ${domainResult.provider}.`
   752→      : 'Domain registration failed.',
   753→  }
   754→
   755→  const dnsStatus: ProvisioningResult['dnsStatus'] = {
   756→    status: dnsResult ? 'success' : 'fail',
   757→    records: (dnsResult?.records || ctx.dnsRecords || []).map((r) => ({
   758→      type: r.type,
   759→      name: r.name,
   760→      value: r.value,
   761→      status: 'created',
   762→    })),
   763→    propagationVerified: verifyResult?.dnsPropagated ?? false,
   764→    details: dnsResult
   765→      ? `${dnsResult.records.length} DNS records configured (A, MX, TXT for SPF/DKIM/DMARC).`
   766→      : 'DNS configuration failed.',
   767→  }
   768→
   769→  const hostingStatus: ProvisioningResult['hostingStatus'] = hostingResult
   770→    ? {
   771→        status: 'success',
   772→        projectId: hostingResult.projectId,
   773→        url: hostingResult.url,
   774→        sslEnabled: true,
   775→        healthChecked: verifyResult?.hostingReachable ?? false,
   776→        details: `Application provisioned. URL: ${hostingResult.url}`,
   777→      }
   778→    : hostingFailed
   779→      ? {
   780→          status: 'fail',
   781→          sslEnabled: false,
   782→          healthChecked: false,
   783→          details: 'Hosting provisioning failed after retries.',
   784→        }
   785→      : {
   786→          status: 'skipped',
   787→          sslEnabled: false,
   788→          healthChecked: false,
   789→          details: 'Hosting provisioning skipped (not requested).',
   790→        }
   791→
   792→  const emailStatus: ProvisioningResult['emailStatus'] = emailResult
   793→    ? {
   794→        status: 'success',
   795→        mailboxes: emailResult.mailboxes.map((m) => ({ address: m.address, type: m.type })),
   796→        dkimConfigured: !!emailResult.dkimPublicKey,
   797→        smtpEnabled: verifyResult?.emailVerified ?? true,
   798→        imapEnabled: verifyResult?.emailVerified ?? true,
   799→        webmailUrl: emailResult.webmailUrl,
   800→        details: `${emailResult.mailboxes.length} mailbox(es) created with DKIM signing enabled.`,
   801→      }
   802→    : emailFailed
   803→      ? {
   804→          status: 'fail',
   805→          mailboxes: [],
   806→          dkimConfigured: false,
   807→          smtpEnabled: false,
   808→          imapEnabled: false,
   809→          details: 'Email provisioning failed after retries.',
   810→        }
   811→      : {
   812→          status: 'skipped',
   813→          mailboxes: [],
   814→          dkimConfigured: false,
   815→          smtpEnabled: false,
   816→          imapEnabled: false,
   817→          details: 'Email provisioning skipped (not requested).',
   818→        }
   819→
   820→  // ─── Credentials ──────────────────────────────────────────────────
   821→  const credentials: ProvisioningResult['credentials'] = {
   822→    domain: {
   823→      registrar: domainResult?.provider === 'dynadot' ? 'Dynadot' : 'DeployHub',
   824→      managementUrl:
   825→        domainResult?.provider === 'dynadot'
   826→          ? 'https://dynadot.com/account/manage/domain.html'
   827→          : '/dashboard/domains',
   828→    },
   829→    dns: {
   830→      nameservers: ctx.nameservers.length
   831→        ? ctx.nameservers
   832→        : ['ns1.deployhub.io', 'ns2.deployhub.io'],
   833→      managementUrl: `/dashboard/domains/${ctx.domainId || ''}/dns`,
   834→    },
   835→  }
   836→
   837→  if (hostingResult) {
   838→    credentials.hosting = {
   839→      adminUrl: '/dashboard/hosting',
   840→    }
   841→  }
   842→
   843→  if (emailResult) {
   844→    const info = ctx.mailServerInfo || {
   845→      imap: { host: ctx.mailServerHostname, port: 993, ssl: true },
   846→      smtp: { host: ctx.mailServerHostname, port: 587, tls: true },
   847→      webmail: emailResult.webmailUrl,
   848→    }
   849→    credentials.email = {
   850→      mailboxes: emailResult.mailboxes.map((m) => ({
   851→        address: m.address,
   852→        password: m.password, // plaintext — shown once
   853→        type: m.type,
   854→      })),
   855→      imap: info.imap,
   856→      smtp: info.smtp,
   857→      webmail: info.webmail,
   858→    }
   859→  }
   860→
   861→  return {
   862→    overallStatus,
   863→    domainStatus,
   864→    dnsStatus,
   865→    hostingStatus,
   866→    emailStatus,
   867→    credentials,
   868→    jobId,
   869→    duration,
   870→    errors: ctx.errors,
   871→  }
   872→}
   873→
   874→// ─── Step 1: provisionDomain ─────────────────────────────────────────────
   875→
   876→/**
   877→ * Step 1: Register the domain via Dynadot.
   878→ *
   879→ *  1. Search availability via Dynadot searchDomains().
   880→ *  2. If available (or in demo mode), registerDomain() via Dynadot.
   881→ *  3. Create a Domain record in DB with provider='dynadot', externalId,
   882→ *     status='active'.
   883→ */
   884→async function provisionDomain(
   885→  job: { id: string; userId: string; domainName: string },
   886→  step: { id: string }
   887→): Promise<{ domainId: string; externalId?: string; provider: string }> {
   888→  const domainName = job.domainName
   889→
   890→  // 1. Search availability.
   891→  const searchResults = await searchDomains([domainName])
   892→  const result = searchResults[0]
   893→  const available = result?.available === true
   894→
   895→  if (!available) {
   896→    // In demo mode, the search heuristic may falsely mark short domains as
   897→    // taken. Be forgiving: log a warning + proceed with registration anyway
   898→    // (Dynadot's registerDomain() will be the source of truth).
   899→    if (isDynadotDemoMode()) {
   900→      console.log(
   901→        `[orchestration] DEMO: domain ${domainName} reported unavailable by demo search, attempting registration anyway`
   902→      )
   903→    } else {
   904→      throw new Error(`Domain ${domainName} is not available for registration`)
   905→    }
   906→  }
   907→
   908→  // 2. Register the domain (1 year).
   909→  const registration = await registerDomain(domainName, 1)
   910→  if (!registration.success) {
   911→    throw new Error(
   912→      `Domain registration failed: ${registration.orderId ? 'order ' + registration.orderId : 'unknown error'}`
   913→    )
   914→  }
   915→
   916→  // 3. Fetch domain info for nameservers + expiry.
   917→  let info
   918→  try {
   919→    info = await getDomainInfo(domainName)
   920→  } catch (err) {
   921→    console.warn(
   922→      `[orchestration] getDomainInfo failed for ${domainName}:`,
   923→      err instanceof Error ? err.message : err
   924→    )
   925→    info = { nameservers: ['ns1.deployhub.io', 'ns2.deployhub.io'], expiry: undefined }
   926→  }
   927→
   928→  // 4. Create the Domain record in DB.
   929→  const domain = await db.domain.create({
   930→    data: {
   931→      userId: job.userId,
   932→      name: domainName,
   933→      status: 'active',
   934→      registrar: 'dynadot',
   935→      provider: 'dynadot',
   936→      externalId: registration.orderId,
   937→      registeredAt: new Date(),
   938→      expiresAt: info.expiry ? new Date(info.expiry) : undefined,
   939→      nameservers: JSON.stringify(info.nameservers || []),
   940→    },
   941→  })
   942→
   943→  console.log(
   944→    `[orchestration] step=domain_register domainId=${domain.id} externalId=${registration.orderId}`
   945→  )
   946→
   947→  return {
   948→    domainId: domain.id,
   949→    externalId: registration.orderId,
   950→    provider: 'dynadot',
   951→  }
   952→}
   953→
   954→// ─── Step 2: configureDns ────────────────────────────────────────────────
   955→
   956→/**
   957→ * Step 2: Configure DNS records via Dynadot's setDnsRecords() and persist
   958→ * each as a DnsRecord row.
   959→ *
   960→ * Records created:
   961→ *   - A     @            → hostingServerIp
   962→ *   - A     www          → hostingServerIp
   963→ *   - MX    @            → mailServerHostname (priority 10)
   964→ *   - TXT   @            → "v=spf1 mx a -all"  (SPF)
   965→ *   - TXT   mail._domainkey → DKIM public key (placeholder if Mailcow not yet run)
   966→ *   - TXT   _dmarc       → "v=DMARC1; p=quarantine; rua=mailto:admin@domain"
   967→ */
   968→async function configureDns(
   969→  job: { id: string; userId: string; domainName: string },
   970→  step: { id: string },
   971→  domainId: string,
   972→  mailServerHostname: string,
   973→  hostingServerIp: string
   974→): Promise<{
   975→  records: Array<{ type: string; name: string; value: string; recordId: string }>
   976→}> {
   977→  const domainName = job.domainName
   978→
   979→  // Build the full set of DNS records.
   980→  // NOTE: DKIM is generated by the email step which runs AFTER this one.
   981→  // We use a placeholder DKIM TXT record here; the email step will overwrite
   982→  // it with the real Mailcow-generated key when it runs (best-effort update).
   983→  const dkimPlaceholder = 'v=DKIM1; k=rsa; p=PLACEHOLDER_REPLACE_AFTER_EMAIL_PROVISIONING'
   984→
   985→  const recordsToCreate: Array<{
   986→    type: string
   987→    name: string
   988→    value: string
   989→    ttl?: number
   990→    priority?: number
   991→  }> = [
   992→    { type: 'A', name: '@', value: hostingServerIp, ttl: 3600 },
   993→    { type: 'A', name: 'www', value: hostingServerIp, ttl: 3600 },
   994→    { type: 'MX', name: '@', value: mailServerHostname, ttl: 3600, priority: 10 },
   995→    { type: 'TXT', name: '@', value: 'v=spf1 mx a -all', ttl: 3600 },
   996→    { type: 'TXT', name: 'mail._domainkey', value: dkimPlaceholder, ttl: 3600 },
   997→    {
   998→      type: 'TXT',
   999→      name: '_dmarc',
  1000→      value: `v=DMARC1; p=quarantine; rua=mailto:admin@${domainName}`,
  1001→      ttl: 3600,
  1002→    },
  1003→  ]
  1004→
  1005→  // 1. Push to Dynadot (or demo).
  1006→  const dnsResult = await setDnsRecords(
  1007→    domainName,
  1008→    recordsToCreate.map((r) => ({
  1009→      type: r.type,
  1010→      name: r.name,
  1011→      value: r.value,
  1012→      ttl: r.ttl,
  1013→      priority: r.priority,
  1014→    }))
  1015→  )
  1016→  if (!dnsResult.success) {
  1017→    throw new Error(`Dynadot setDnsRecords failed for ${domainName}`)
  1018→  }
  1019→
  1020→  // 2. Persist each record to the DnsRecord table.
  1021→  const created: Array<{ type: string; name: string; value: string; recordId: string }> = []
  1022→  for (const r of recordsToCreate) {
  1023→    const rec = await db.dnsRecord.create({
  1024→      data: {
  1025→        domainId,
  1026→        type: r.type,
  1027→        name: r.name,
  1028→        value: r.value,
  1029→        ttl: r.ttl || 3600,
  1030→        priority: r.priority ?? null,
  1031→        proxied: false,
  1032→      },
  1033→    })
  1034→    created.push({
  1035→      type: r.type,
  1036→      name: r.name,
  1037→      value: r.value,
  1038→      recordId: rec.id,
  1039→    })
  1040→  }
  1041→
  1042→  console.log(
  1043→    `[orchestration] step=dns_configure domainId=${domainId} records=${created.length}`
  1044→  )
  1045→
  1046→  return { records: created }
  1047→}
  1048→
  1049→// ─── Step 3: provisionHosting ────────────────────────────────────────────
  1050→
  1051→/**
  1052→ * Step 3: Provision web hosting.
  1053→ *
  1054→ *  1. Create a Project in DB with name=domainName, framework based on stack.
  1055→ *  2. Create an initial Deployment record in 'queued' status.
  1056→ *  3. Queue a deployment_status_check job so the worker transitions it
  1057→ *     through building → live.
  1058→ */
  1059→async function provisionHosting(
  1060→  job: { id: string; userId: string; domainName: string; stack: string },
  1061→  step: { id: string },
  1062→  domainId: string,
  1063→  domainName: string
  1064→): Promise<{ projectId: string; url: string }> {
  1065→  const framework = stackToFramework(job.stack)
  1066→  const slug = domainName
  1067→    .toLowerCase()
  1068→    .replace(/[^a-z0-9.-]+/g, '-')
  1069→    .replace(/^-|-$/g, '')
  1070→  const subdomain = `${slug.split('.')[0]}.deployhub.io`
  1071→  const url = `https://${subdomain}`
  1072→
  1073→  // 1. Create the Project record.
  1074→  const project = await db.project.create({
  1075→    data: {
  1076→      userId: job.userId,
  1077→      name: domainName,
  1078→      slug,
  1079→      framework,
  1080→      stack: job.stack,
  1081→      status: 'creating',
  1082→      subdomain,
  1083→      customDomain: domainName,
  1084→      sslEnabled: true,
  1085→    },
  1086→  })
  1087→
  1088→  // 2. Link the Domain to the Project.
  1089→  await db.domain.update({
  1090→    where: { id: domainId },
  1091→    data: { projectId: project.id },
  1092→  })
  1093→
  1094→  // 3. Mark project as running and create an initial Deployment record.
  1095→  await db.project.update({
  1096→    where: { id: project.id },
  1097→    data: { status: 'running' },
  1098→  })
  1099→
  1100→  await db.deployment.create({
  1101→    data: {
  1102→      projectId: project.id,
  1103→      userId: job.userId,
  1104→      branch: 'main',
  1105→      status: 'queued',
  1106→      deployUrl: url,
  1107→      triggeredBy: 'provisioning',
  1108→    },
  1109→  })
  1110→
  1111→  console.log(
  1112→    `[orchestration] step=hosting_provision projectId=${project.id} url=${url}`
  1113→  )
  1114→
  1115→  return {
  1116→    projectId: project.id,
  1117→    url,
  1118→  }
  1119→}
  1120→
  1121→// ─── Step 4: provisionEmail ──────────────────────────────────────────────
  1122→
  1123→/**
  1124→ * Step 4: Provision email hosting via Mailcow.
  1125→ *
  1126→ *  1. addDomain() in Mailcow.
  1127→ *  2. generateDkim() for the domain.
  1128→ *  3. Create 3 mailboxes: admin@, info@, support@ with random passwords.
  1129→ *     Each mailbox password is encrypted (AES-256-GCM) before DB storage.
  1130→ *  4. Update Domain.emailHosted=true, mailcowDomainId.
  1131→ *  5. Fetch mail server connection info (IMAP/SMTP/webmail).
  1132→ *
  1133→ * Returns the plaintext passwords alongside each mailbox so the caller can
  1134→ * include them in the one-time credentials payload.
  1135→ */
  1136→async function provisionEmail(
  1137→  job: { id: string; userId: string; domainName: string },
  1138→  step: { id: string },
  1139→  domainId: string,
  1140→  domainName: string
  1141→): Promise<{
  1142→  mailboxes: Array<{ address: string; password: string; type: string; mailboxId: string }>
  1143→  dkimPublicKey: string
  1144→  webmailUrl: string
  1145→}> {
  1146→  // 1. Add the domain to Mailcow.
  1147→  const addDomainResult = await addDomain({
  1148→    domain: domainName,
  1149→    description: `DeployHub provisioned domain for ${job.userId}`,
  1150→    mailboxes: 10,
  1151→    aliases: 20,
  1152→    quota: 1024,
  1153→  })
  1154→  if (!addDomainResult.success) {
  1155→    throw new Error(`Mailcow addDomain failed: ${addDomainResult.error}`)
  1156→  }
  1157→
  1158→  // 2. Generate DKIM.
  1159→  const dkimResult = await generateDkim(domainName, 'mail')
  1160→  if (!dkimResult.success || !dkimResult.publicKey) {
  1161→    throw new Error(`Mailcow generateDkim failed: ${dkimResult.error}`)
  1162→  }
  1163→
  1164→  // 3. Update the Domain with email hosting info.
  1165→  await db.domain.update({
  1166→    where: { id: domainId },
  1167→    data: {
  1168→      emailHosted: true,
  1169→      mailcowDomainId: addDomainResult.domainId || domainName,
  1170→    },
  1171→  })
  1172→
  1173→  // 4. Create 3 standard mailboxes: admin, info, support.
  1174→  const mailboxSpecs: Array<{ localPart: string; name: string; type: string; isDefault: boolean }> = [
  1175→    { localPart: 'admin', name: 'Administrator', type: 'admin', isDefault: true },
  1176→    { localPart: 'info', name: 'Info', type: 'info', isDefault: false },
  1177→    { localPart: 'support', name: 'Support', type: 'support', isDefault: false },
  1178→  ]
  1179→
  1180→  const mailboxes: Array<{
  1181→    address: string
  1182→    password: string
  1183→    type: string
  1184→    mailboxId: string
  1185→  }> = []
  1186→
  1187→  for (const spec of mailboxSpecs) {
  1188→    const password = generatePassword()
  1189→    const address = `${spec.localPart}@${domainName}`
  1190→
  1191→    const mbResult = await createMailbox({
  1192→      localPart: spec.localPart,
  1193→      domain: domainName,
  1194→      password,
  1195→      name: spec.name,
  1196→      quota: 1024,
  1197→    })
  1198→    if (!mbResult.success) {
  1199→      throw new Error(
  1200→        `Mailcow createMailbox failed for ${address}: ${mbResult.error}`
  1201→      )
  1202→    }
  1203→
  1204→    // Persist with encrypted password.
  1205→    await db.emailMailbox.create({
  1206→      data: {
  1207→        domainId,
  1208→        address,
  1209→        localPart: spec.localPart,
  1210→        password: encrypt(password),
  1211→        quota: 1073741824, // 1GB in bytes
  1212→        active: true,
  1213→        isDefault: spec.isDefault,
  1214→        mailcowId: mbResult.mailboxId,
  1215→        dkimEnabled: true,
  1216→      },
  1217→    })
  1218→
  1219→    mailboxes.push({
  1220→      address,
  1221→      password, // plaintext — kept in memory for the credentials payload
  1222→      type: spec.type,
  1223→      mailboxId: mbResult.mailboxId || address,
  1224→    })
  1225→  }
  1226→
  1227→  // 5. Fetch mail server connection info.
  1228→  const serverInfo = await getMailServerInfo(domainName)
  1229→
  1230→  console.log(
  1231→    `[orchestration] step=email_provision domainId=${domainId} mailboxes=${mailboxes.length} dkim=✓`
  1232→  )
  1233→
  1234→  return {
  1235→    mailboxes,
  1236→    dkimPublicKey: dkimResult.publicKey,
  1237→    webmailUrl: serverInfo.webmail,
  1238→  }
  1239→}

  1241→// ─── Step 5: verifyProvisioning ──────────────────────────────────────────
  1242→
  1243→/**
  1244→ * Step 5: Verify that all provisioned services are reachable and properly
  1245→ * configured.
  1246→ *
  1247→ *  1. Hosting: confirm the Project exists and is in a non-error state.
  1248→ *  2. Email: call Mailcow verifyEmailSetup() for the domain.
  1249→ *  3. DNS: call dns.ts verifyTxtRecord() for the DKIM record (best-effort —
  1250→ *     DNS propagation can take minutes/hours so we don't fail the pipeline
  1251→ *     if it's not yet propagated).
  1252→ */
  1253→async function verifyProvisioning(
  1254→  job: { id: string; userId: string; domainName: string },
  1255→  step: { id: string },
  1256→  domainName: string,
  1257→  projectId: string
  1258→): Promise<{ hostingReachable: boolean; emailVerified: boolean; dnsPropagated: boolean }> {
  1259→  // 1. Hosting health check.
  1260→  let hostingReachable = false
  1261→  if (projectId) {
  1262→    try {
  1263→      const project = await db.project.findUnique({ where: { id: projectId } })
  1264→      hostingReachable =
  1265→        !!project &&
  1266→        (project.status === 'running' || project.status === 'creating' || project.status === 'building')
  1267→    } catch (err) {
  1268→      console.warn(
  1269→        `[orchestration] hosting health check failed:`,
  1270→        err instanceof Error ? err.message : err
  1271→      )
  1272→    }
  1273→  }
  1274→
  1275→  // 2. Email verification (Mailcow).
  1276→  let emailVerified = false
  1277→  try {
  1278→    const emailCheck = await verifyEmailSetup(domainName)
  1279→    emailVerified = emailCheck.success
  1280→  } catch (err) {
  1281→    console.warn(
  1282→      `[orchestration] email verification failed:`,
  1283→      err instanceof Error ? err.message : err
  1284→    )
  1285→  }
  1286→
  1287→  // 3. DNS propagation check (best-effort, never fails the pipeline).
  1288→  let dnsPropagated = false
  1289→  try {
  1290→    // Verify the SPF TXT record exists at the authoritative resolver. In demo
  1291→    // mode this will likely return false (the records aren't actually live),
  1292→    // so we treat it as informational.
  1293→    dnsPropagated = await verifyTxtRecord(domainName, 'v=spf1')
  1294→  } catch (err) {
  1295→    console.warn(
  1296→      `[orchestration] DNS propagation check failed:`,
  1297→      err instanceof Error ? err.message : err
  1298→    )
  1299→  }
  1300→
  1301→  console.log(
  1302→    `[orchestration] step=verify hosting=${hostingReachable} email=${emailVerified} dns=${dnsPropagated}`
  1303→  )
  1304→
  1305→  return { hostingReachable, emailVerified, dnsPropagated }
  1306→}

  1308→// ─── rollbackProvisioning ────────────────────────────────────────────────
  1309→
  1310→/**
  1311→ * Roll back all completed steps in REVERSE order. For each successful step:
  1312→ *
  1313→ *   email_provision      → Mailcow removeDomain() + delete EmailMailbox rows
  1314→ *   hosting_provision    → mark Project stopped (no external provider to clean up)
  1315→ *   dns_configure        → delete DnsRecord rows
  1316→ *   domain_register      → mark Domain status='cancelled' (can't unregister)
  1317→ *
  1318→ * Updates each step status to 'rolled_back' and sets job.status='rolled_back'
  1319→ * with rollbackStartedAt/rollbackCompletedAt timestamps.
  1320→ */
  1321→export async function rollbackProvisioning(
  1322→  jobId: string
  1323→): Promise<{
  1324→  rolledBack: string[]
  1325→  failed: Array<{ step: string; error: string }>
  1326→}> {
  1327→  const job = await db.provisioningJob.findUnique({
  1328→    where: { id: jobId },
  1329→    include: { steps: { orderBy: { stepOrder: 'desc' } } },
  1330→  })
  1331→  if (!job) {
  1332→    throw new Error(`ProvisioningJob not found: ${jobId}`)
  1333→  }
  1334→
  1335→  await db.provisioningJob.update({
  1336→    where: { id: jobId },
  1337→    data: { rollbackStartedAt: new Date(), status: 'rolled_back' },
  1338→  })
  1339→
  1340→  const rolledBack: string[] = []
  1341→  const failed: Array<{ step: string; error: string }> = []
  1342→
  1343→  for (const step of job.steps) {
  1344→    if (step.status !== 'success') continue
  1345→
  1346→    try {
  1347→      switch (step.stepName) {
  1348→        case 'email_provision': {
  1349→          // Delete mailboxes in Mailcow (best-effort), then remove the domain.
  1350→          const domain = await db.domain.findFirst({
  1351→            where: { name: job.domainName },
  1352→            include: { mailboxes: true },
  1353→          })
  1354→          if (domain) {
  1355→            for (const mb of domain.mailboxes) {
  1356→              try {
  1357→                await deleteMailbox(mb.address)
  1358→              } catch (err) {
  1359→                console.warn(
  1360→                  `[orchestration] rollback deleteMailbox ${mb.address} failed:`,
  1361→                  err instanceof Error ? err.message : err
  1362→                )
  1363→              }
  1364→            }
  1365→            try {
  1366→              await removeDomain(job.domainName)
  1367→            } catch (err) {
  1368→              console.warn(
  1369→                `[orchestration] rollback removeDomain ${job.domainName} failed:`,
  1370→                err instanceof Error ? err.message : err
  1371→              )
  1372→            }
  1373→            await db.emailMailbox.deleteMany({ where: { domainId: domain.id } })
  1374→            await db.domain.update({
  1375→              where: { id: domain.id },
  1376→              data: { emailHosted: false, mailcowDomainId: null },
  1377→            })
  1378→          }
  1379→          break
  1380→        }
  1381→        case 'hosting_provision': {
  1382→          const project = await db.project.findFirst({
  1383→            where: { userId: job.userId, name: job.domainName },
  1384→          })
  1385→          if (project) {
  1386→            await db.project.update({
  1387→              where: { id: project.id },
  1388→              data: { status: 'stopped' },
  1389→            })
  1390→          }
  1391→          break
  1392→        }
  1393→        case 'dns_configure': {
  1394→          const domain = await db.domain.findFirst({
  1395→            where: { name: job.domainName },
  1396→          })
  1397→          if (domain) {
  1398→            await db.dnsRecord.deleteMany({ where: { domainId: domain.id } })
  1399→          }
  1400→          break
  1401→        }
  1402→        case 'domain_register': {
  1403→          // Can't actually unregister a domain, but mark it cancelled in our DB.
  1404→          const domain = await db.domain.findFirst({
  1405→            where: { name: job.domainName },
  1406→          })
  1407→          if (domain) {
  1408→            await db.domain.update({
  1409→              where: { id: domain.id },
  1410→              data: { status: 'cancelled' },
  1411→            })
  1412→          }
  1413→          break
  1414→        }
  1415→        case 'verify': {
  1416→          // Nothing to roll back for the verify step.
  1417→          break
  1418→        }
  1419→        default:
  1420→          break
  1421→      }
  1422→
  1423→      await db.provisioningStep.update({
  1424→        where: { id: step.id },
  1425→        data: { status: 'rolled_back' },
  1426→      })
  1427→      rolledBack.push(step.stepName)
  1428→    } catch (err) {
  1429→      const msg = err instanceof Error ? err.message : String(err)
  1430→      failed.push({ step: step.stepName, error: msg })
  1431→      console.error(
  1432→        `[orchestration] rollback of step ${step.stepName} failed: ${msg}`
  1433→      )
  1434→    }
  1435→  }
  1436→
  1437→  await db.provisioningJob.update({
  1438→    where: { id: jobId },
  1439→    data: {
  1440→      rollbackCompletedAt: new Date(),
  1441→      rollbackSteps: JSON.stringify(rolledBack),
  1442→      status: 'rolled_back',
  1443→      currentStep: 'rolled_back',
  1444→    },
  1445→  })
  1446→
  1447→  await logAudit({
  1448→    userId: job.userId,
  1449→    action: 'provisioning.rollback',
  1450→    resource: 'provisioning_job',
  1451→    resourceId: jobId,
  1452→    metadata: { rolledBack, failed },
  1453→    status: failed.length > 0 ? 'failure' : 'success',
  1454→  })
  1455→
  1456→  console.log(
  1457→    `[orchestration] rollback complete for job=${jobId} rolledBack=${rolledBack.join(',')} failed=${failed.length}`
  1458→  )
  1459→
  1460→  return { rolledBack, failed }
  1461→}
