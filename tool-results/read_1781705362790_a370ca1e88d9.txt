   231→function formatDate(dateStr?: string) {
   232→  if (!dateStr) return '-'
   233→  try {
   234→    return new Date(dateStr).toLocaleDateString('en-US', {
   235→      month: 'short',
   236→      day: 'numeric',
   237→      year: 'numeric',
   238→    })
   239→  } catch {
   240→    return dateStr
   241→  }
   242→}

   244→function formatTTL(ttl: number) {
   245→  if (ttl >= 3600) return `${ttl / 3600}h`
   246→  if (ttl >= 60) return `${ttl / 60}m`
   247→  return `${ttl}s`
   248→}

   262→function PropagationRing({ percentage, size = 80 }: { percentage: number; size?: number }) {
   263→  const strokeWidth = 6
   264→  const radius = (size - strokeWidth) / 2
   265→  const circumference = radius * 2 * Math.PI
   266→  const offset = circumference - (percentage / 100) * circumference
   267→
   268→  const color =
   269→    percentage >= 80
   270→      ? 'text-emerald-500'
   271→      : percentage >= 50
   272→      ? 'text-yellow-500'
   273→      : 'text-red-500'
   274→
   275→  const strokeColor =
   276→    percentage >= 80
   277→      ? 'stroke-emerald-500'
   278→      : percentage >= 50
   279→      ? 'stroke-yellow-500'
   280→      : 'stroke-red-500'
   281→
   282→  return (
   283→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   284→      <svg className="transform -rotate-90" width={size} height={size}>
   285→        <circle
   286→          className="stroke-muted"
   287→          strokeWidth={strokeWidth}
   288→          fill="transparent"
   289→          r={radius}
   290→          cx={size / 2}
   291→          cy={size / 2}
   292→        />
   293→        <circle
   294→          className={strokeColor}
   295→          strokeWidth={strokeWidth}
   296→          strokeDasharray={circumference}
   297→          strokeDashoffset={offset}
   298→          strokeLinecap="round"
   299→          fill="transparent"
   300→          r={radius}
   301→          cx={size / 2}
   302→          cy={size / 2}
   303→          style={{ transition: 'stroke-dashoffset 0.8s ease-in-out' }}
   304→        />
   305→      </svg>
   306→      <div className={`absolute text-sm font-bold ${color}`}>
   307→        {percentage}%
   308→      </div>
   309→    </div>
   310→  )
   311→}

   462→function LinkProjectDialog({
   463→  open,
   464→  onClose,
   465→  domainId,
   466→  onLinked,
   467→}: {
   468→  open: boolean
   469→  onClose: () => void
   470→  domainId: string
   471→  onLinked: () => void
   472→}) {
   473→  const [projects, setProjects] = useState<Project[]>([])
   474→  const [selectedProjectId, setSelectedProjectId] = useState('')
   475→  const [loading, setLoading] = useState(false)
   476→
   477→  useEffect(() => {
   478→    if (open) {
   479→      fetch('/api/hosting', { headers: getAuthHeaders() })
   480→        .then((r) => r.json())
   481→        .then((data) => setProjects(data.projects || []))
   482→        .catch(() => toast.error('Failed to load projects'))
   483→    }
   484→  }, [open])
   485→
   486→  const handleLink = async () => {
   487→    if (!selectedProjectId) {
   488→      toast.error('Select a project')
   489→      return
   490→    }
   491→    setLoading(true)
   492→    try {
   493→      await fetch(`/api/domains/${domainId}`, {
   494→        method: 'PUT',
   495→        headers: getAuthHeaders(),
   496→        body: JSON.stringify({ projectId: selectedProjectId }),
   497→      })
   498→      toast.success('Domain linked to project')
   499→      onLinked()
   500→      onClose()
   501→    } catch {
   502→      toast.error('Network error')
   503→    } finally {
   504→      setLoading(false)
   505→    }
   506→  }
   507→
   508→  return (
   509→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   510→      <DialogContent className="sm:max-w-md">
   511→        <DialogHeader>
   512→          <DialogTitle>Link to Project</DialogTitle>

   581→  const [expandedRecords, setExpandedRecords] = useState<Set<string>>(new Set())

   583→  // Verification state (for custom / pending_verification domains)

   585→  const [verifyInstructions, setVerifyInstructions] = useState<string>('')
   586→  const [verifyLoading, setVerifyLoading] = useState(false)

   647→  const fetchVerifyCode = useCallback(async () => {

   656→        setVerifyInstructions(data.instructions || '')
   657→      }
   658→    } catch {
   659→      // Silent - verification is optional
   660→    } finally {
   661→      setVerifyLoading(false)
   662→    }
   663→  }, [domainId])
   664→
