   105→   105→// ── API helper ─────────────────────────────────────────────────────────
   106→   106→async function apiFetch<T>(path: string, options?: RequestInit): Promise<T> {
   107→   107→  const token = localStorage.getItem('deployhub_token')
   108→   108→  const res = await fetch(`/api${path}`, {
   109→   109→    ...options,
   110→   110→    headers: {
   111→   111→      'Content-Type': 'application/json',
   112→   112→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
   113→   113→      ...options?.headers,
   114→   114→    },
   115→   115→  })
   116→   116→  if (!res.ok) {
   117→   117→    const data = await res.json().catch(() => ({ error: 'Request failed' }))
   118→   118→    throw new Error(data.error || `Request failed with status ${res.status}`)
   119→   119→  }
   120→   120→  return res.json()
   121→   121→}
