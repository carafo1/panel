   282→   282→function CreateProjectDialog({
   283→   283→  open,
   284→   284→  onClose,
   285→   285→  onCreated,
   286→   286→}: {
   287→   287→  open: boolean
   288→   288→  onClose: () => void
   289→   289→  onCreated: () => void
   290→   290→}) {
   291→   291→  const [name, setName] = useState('')
   292→   292→  const [framework, setFramework] = useState('nextjs')
   293→   293→  const [planId, setPlanId] = useState('')
   294→   294→  const [plans, setPlans] = useState<Plan[]>([])
   295→   295→  const [loading, setLoading] = useState(false)
   296→   296→  const [plansLoading, setPlansLoading] = useState(false)
   297→   297→
   298→   298→  useEffect(() => {
   299→   299→    if (open) {
   300→   300→      setPlansLoading(true)
   301→   301→      fetch('/api/billing/plans')
   302→   302→        .then((r) => r.json())
   303→   303→        .then((data) => {
   304→   304→          setPlans(data.plans || [])
   305→   305→          if (data.plans?.length > 0 && !planId) {
   306→   306→            setPlanId(data.plans[0].id)
   307→   307→          }
   308→   308→        })
   309→   309→        .catch(() => toast.error('Failed to load plans'))
   310→   310→        .finally(() => setPlansLoading(false))
   311→   311→    }
   312→   312→  }, [open, planId])
   313→   313→
   314→   314→  const handleCreate = async () => {
   315→   315→    if (!name.trim()) {
   316→   316→      toast.error('Project name is required')
   317→   317→      return
   318→   318→    }
   319→   319→    setLoading(true)
   320→   320→    try {
   321→   321→      const res = await fetch('/api/hosting/create', {
   322→   322→        method: 'POST',
   323→   323→        headers: getAuthHeaders(),
   324→   324→        body: JSON.stringify({ name: name.trim(), framework, planId: planId || undefined }),
   325→   325→      })
   326→   326→      const data = await res.json()
   327→   327→      if (!res.ok) {
   328→   328→        toast.error(data.error || 'Failed to create project')
   329→   329→        return
   330→   330→      }
   331→   331→      toast.success('Project created successfully!')
   332→   332→      setName('')
   333→   333→      setFramework('nextjs')
   334→   334→      onCreated()
   335→   335→      onClose()
   336→   336→    } catch {
   337→   337→      toast.error('Network error. Please try again.')
   338→   338→    } finally {
   339→   339→      setLoading(false)
   340→   340→    }
   341→   341→  }
   342→   342→
   343→   343→  return (
   344→   344→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   345→   345→      <DialogContent className="sm:max-w-lg">
   346→   346→        <DialogHeader>
   347→   347→          <DialogTitle>Create New Project</DialogTitle>
   348→   348→          <DialogDescription>
   349→   349→            Set up a new hosting project with your preferred framework and plan.
   350→   350→          </DialogDescription>
   351→   351→        </DialogHeader>
   352→   352→
   353→   353→        <div className="grid gap-4 py-4">
   354→   354→          <div className="grid gap-2">
   355→   355→            <Label htmlFor="project-name">Project Name</Label>
   356→   356→            <Input
   357→   357→              id="project-name"
   358→   358→              placeholder="my-awesome-app"
   359→   359→              value={name}
   360→   360→              onChange={(e) => setName(e.target.value)}
   361→   361→            />
   362→   362→          </div>
   363→   363→
   364→   364→          <div className="grid gap-2">
   365→   365→            <Label>Framework</Label>
   366→   366→            <Select value={framework} onValueChange={setFramework}>
   367→   367→              <SelectTrigger className="w-full">
   368→   368→                <SelectValue placeholder="Select framework" />
   369→   369→              </SelectTrigger>
   370→   370→              <SelectContent>
   371→   371→                <SelectItem value="nextjs">Next.js</SelectItem>
   372→   372→                <SelectItem value="react">React</SelectItem>
   373→   373→                <SelectItem value="node">Node.js</SelectItem>
   374→   374→                <SelectItem value="static">Static</SelectItem>
   375→   375→              </SelectContent>
   376→   376→            </Select>
   377→   377→          </div>
   378→   378→
   379→   379→          <div className="grid gap-2">
   380→   380→            <Label>Plan</Label>
   381→   381→            {plansLoading ? (
   382→   382→              <div className="grid gap-2">
   383→   383→                <Skeleton className="h-20 w-full" />
   384→   384→                <Skeleton className="h-20 w-full" />
   385→   385→              </div>
   386→   386→            ) : (
   387→   387→              <div className="grid gap-2">
   388→   388→                {plans.map((plan) => (
   389→   389→                  <button
   390→   390→                    key={plan.id}
   391→   391→                    type="button"
   392→   392→                    onClick={() => setPlanId(plan.id)}
   393→   393→                    className={`relative flex items-center gap-3 rounded-lg border p-3 text-left transition-all hover:border-emerald-300 dark:hover:border-emerald-700 ${
   394→   394→                      planId === plan.id
   395→   395→                        ? 'border-emerald-500 bg-emerald-50 dark:bg-emerald-900/20 ring-1 ring-emerald-500'
   396→   396→                        : 'border-border'
   397→   397→                    }`}
   398→   398→                  >
   399→   399→                    {plan.popular && (
   400→   400→                      <Badge className="absolute -top-2 right-3 bg-emerald-600 text-white text-[10px] px-1.5 py-0">
   401→   401→                        Popular
   402→   402→                      </Badge>
   403→   403→                    )}
   404→   404→                    <div className="flex-1">
   405→   405→                      <div className="font-medium text-sm">{plan.name}</div>
   406→   406→                      <div className="text-xs text-muted-foreground mt-0.5">
   407→   407→                        {plan.cpu} &middot; {plan.ram} &middot; {plan.storage}
   408→   408→                      </div>
   409→   409→                    </div>
   410→   410→                    <div className="text-right">
   411→   411→                      <div className="font-semibold text-emerald-600 dark:text-emerald-400">
   412→   412→                        ${plan.price}
   413→   413→                      </div>
   414→   414→                      <div className="text-[10px] text-muted-foreground">/mo</div>
   415→   415→                    </div>
   416→   416→                  </button>
   417→   417→                ))}
   418→   418→              </div>
   419→   419→            )}
   420→   420→          </div>
   421→   421→        </div>
   422→   422→
   423→   423→        <DialogFooter>
   424→   424→          <Button variant="outline" onClick={onClose} disabled={loading}>
   425→   425→            Cancel
   426→   426→          </Button>
   427→   427→          <Button
   428→   428→            onClick={handleCreate}
   429→   429→            disabled={loading || !name.trim()}
   430→   430→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   431→   431→          >
   432→   432→            {loading ? (
   433→   433→              <>
   434→   434→                <RefreshCw className="size-4 mr-2 animate-spin" />
   435→   435→                Creating...
   436→   436→              </>
   437→   437→            ) : (
   438→   438→              <>
   439→   439→                <Rocket className="size-4 mr-2" />
   440→   440→                Create Project
   441→   441→              </>
   442→   442→            )}
   443→   443→          </Button>
   444→   444→        </DialogFooter>
   445→   445→      </DialogContent>
   446→   446→    </Dialog>
   447→   447→  )
   448→   448→}

   450→   450→function ProjectDetailView({
   451→   451→  projectId,
   452→   452→  onBack,
   453→   453→  onRefresh,
   454→   454→}: {
   455→   455→  projectId: string
   456→   456→  onBack: () => void
   457→   457→  onRefresh: () => void
   458→   458→}) {
   459→   459→  const [project, setProject] = useState<ProjectDetail | null>(null)
   460→   460→  const [loading, setLoading] = useState(true)
   461→   461→  const [deploying, setDeploying] = useState(false)
   462→   462→
   463→   463→  const fetchProject = useCallback(async () => {
   464→   464→    try {
   465→   465→      const res = await fetch(`/api/hosting/${projectId}`, {
   466→   466→        headers: getAuthHeaders(),
   467→   467→      })
   468→   468→      const data = await res.json()
   469→   469→      if (!res.ok) {
   470→   470→        toast.error(data.error || 'Failed to load project')
   471→   471→        onBack()
   472→   472→        return
   473→   473→      }
   474→   474→      setProject(data.project)
   475→   475→    } catch {
   476→   476→      toast.error('Network error')
   477→   477→      onBack()
   478→   478→    } finally {
   479→   479→      setLoading(false)
   480→   480→    }
   481→   481→  }, [projectId, onBack])
   482→   482→
   483→   483→  useEffect(() => {
   484→   484→    setLoading(true)
   485→   485→    fetchProject()
   486→   486→  }, [fetchProject])
   487→   487→
   488→   488→  const handleDeploy = async () => {
   489→   489→    setDeploying(true)
   490→   490→    try {
   491→   491→      const res = await fetch(`/api/hosting/${projectId}/deploy`, {
   492→   492→        method: 'POST',
   493→   493→        headers: getAuthHeaders(),
   494→   494→      })
   495→   495→      const data = await res.json()
   496→   496→      if (!res.ok) {
   497→   497→        toast.error(data.error || 'Deploy failed')
   498→   498→        return
   499→   499→      }
   500→   500→      toast.success('Deployment triggered!')
   501→   501→      fetchProject()
   502→   502→      onRefresh()
   503→   503→    } catch {
   504→   504→      toast.error('Network error')
   505→   505→    } finally {
   506→   506→      setDeploying(false)
   507→   507→    }
   508→   508→  }
   509→   509→
   510→   510→  const handleStop = async () => {
   511→   511→    try {
   512→   512→      const res = await fetch(`/api/hosting/${projectId}`, {
   513→   513→        method: 'DELETE',
   514→   514→        headers: getAuthHeaders(),
   515→   515→      })
   516→   516→      const data = await res.json()
   517→   517→      if (!res.ok) {
   518→   518→        toast.error(data.error || 'Failed to stop project')
   519→   519→        return
   520→   520→      }
   521→   521→      toast.success('Project stopped')
   522→   522→      fetchProject()
   523→   523→      onRefresh()
   524→   524→    } catch {
   525→   525→      toast.error('Network error')
   526→   526→    }
   527→   527→  }
   528→   528→
   529→   529→  const handleDelete = async () => {
   530→   530→    if (!confirm('Are you sure you want to delete this project? This action cannot be undone.')) return
   531→   531→    try {
   532→   532→      const res = await fetch(`/api/hosting/${projectId}`, {
   533→   533→        method: 'DELETE',
