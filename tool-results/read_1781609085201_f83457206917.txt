   286→function generateStageId(): string {
   287→  return `stage_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`
   288→}

   290→function createDefaultStage(type: PipelineStage['type'], order: number): PipelineStage {
   291→  const def = getStageTypeDef(type)
   292→  return {
   293→    id: generateStageId(),
   294→    name: def.label,
   295→    type,
   296→    command: def.defaultCommand,
   297→    workingDir: '/',
   298→    env: {},
   299→    timeout: 300,
   300→    continueOnError: false,
   301→    order,
   302→  }
   303→}

   305→/** Migrate old-style stages (no id, command as config.command) to new format */
   306→function migrateStages(raw: unknown[]): PipelineStage[] {
   307→  return raw.map((s: Record<string, unknown>, i: number) => ({
   308→    id: (s.id as string) || generateStageId(),
   309→    name: (s.name as string) || `Stage ${i + 1}`,
   310→    type: (s.type as PipelineStage['type']) || 'custom',
   311→    command: (s.command as string) || (s.config && typeof s.config === 'object' ? String((s.config as Record<string, unknown>).command || '') : '') || '',
   312→    workingDir: (s.workingDir as string) || '/',
   313→    env: (s.env as Record<string, string>) || {},
   314→    timeout: (s.timeout as number) ?? 300,
   315→    continueOnError: (s.continueOnError as boolean) ?? false,
   316→    order: (s.order as number) ?? i,
   317→  }))
   318→}
   319→
   320→function parseStages(raw: string): PipelineStage[] {
   321→  try {
   322→    const parsed = JSON.parse(raw)
   323→    if (!Array.isArray(parsed)) return []
   324→    return migrateStages(parsed)
   325→  } catch {
   326→    return []
   327→  }
   328→}
   329→
   330→function parseTriggerEvents(raw: string): string[] {
   331→  try {
   332→    const parsed = JSON.parse(raw)
   333→    return Array.isArray(parsed) ? parsed : []
   334→  } catch {
   335→    return []
   336→  }
   337→}

   339→function formatDate(dateStr: string): string {
   340→  const d = new Date(dateStr)
   341→  return d.toLocaleDateString('en-US', {
   342→    month: 'short',
   343→    day: 'numeric',
   344→    year: 'numeric',
   345→  })
   346→}
   347→
   348→function formatTime(dateStr: string): string {
   349→  const d = new Date(dateStr)
   350→  return d.toLocaleTimeString('en-US', {
   351→    hour: '2-digit',
   352→    minute: '2-digit',
   353→  })
   354→}
   355→
   356→function formatDuration(seconds: number | null | undefined): string {
   357→  if (seconds == null) return '—'
   358→  if (seconds < 60) return `${seconds}s`
   359→  const mins = Math.floor(seconds / 60)
   360→  const secs = seconds % 60
   361→  return `${mins}m ${secs}s`
   362→}

   364→function getStatusColor(status: string): string {
   365→  switch (status) {
   366→    case 'success':
   367→      return 'bg-emerald-500'
   368→    case 'failed':
   369→      return 'bg-red-500'
   370→    case 'running':
   371→      return 'bg-amber-500'
   372→    case 'cancelled':
   373→      return 'bg-gray-400'
   374→    case 'active':
   375→      return 'bg-emerald-500'
   376→    case 'paused':
   377→      return 'bg-amber-500'
   378→    case 'draft':
   379→      return 'bg-gray-400'
   380→    default:
   381→      return 'bg-gray-400'
   382→  }
   383→}

   840→function VisualStageEditor({
   841→  stages,
   842→  onUpdateStages,
   843→  readOnly,
   844→}: {
   845→  stages: PipelineStage[]
   846→  onUpdateStages: (stages: PipelineStage[]) => void
   847→  readOnly?: boolean
   848→}) {
   849→  const [pickerOpen, setPickerOpen] = useState(false)
   850→  const [configOpen, setConfigOpen] = useState(false)
   851→  const [editingStage, setEditingStage] = useState<PipelineStage | null>(null)
   852→  const [activeId, setActiveId] = useState<string | null>(null)
   853→
   854→  const sensors = useSensors(
   855→    useSensor(PointerSensor, {
   856→      activationConstraint: { distance: 5 },
   857→    })
   858→  )
   859→
   860→  const handleDragStart = (event: DragStartEvent) => {
   861→    setActiveId(String(event.active.id))
   862→  }
   863→
   864→  const handleDragEnd = (event: DragEndEvent) => {
   865→    setActiveId(null)
   866→    const { active, over } = event
   867→    if (!over || active.id === over.id) return
   868→
   869→    const oldIndex = stages.findIndex((s) => s.id === active.id)
   870→    const newIndex = stages.findIndex((s) => s.id === over.id)
   871→    if (oldIndex === -1 || newIndex === -1) return
   872→
   873→    const reordered = arrayMove(stages, oldIndex, newIndex).map((s, i) => ({
   874→      ...s,
   875→      order: i,
   876→    }))
   877→    onUpdateStages(reordered)
   878→  }
   879→
   880→  const handleAddStage = (type: PipelineStage['type']) => {
   881→    const newStage = createDefaultStage(type, stages.length)
   882→    onUpdateStages([...stages, newStage])
   883→  }
   884→
   885→  const removeStage = (id: string) => {
   886→    const updated = stages.filter((s) => s.id !== id).map((s, i) => ({ ...s, order: i }))
   887→    onUpdateStages(updated)
   888→  }
   889→
   890→  const updateStage = (id: string, updates: Partial<PipelineStage>) => {
   891→    const updated = stages.map((s) => (s.id === id ? { ...s, ...updates } : s))
   892→    onUpdateStages(updated)
   893→  }
   894→
   895→  const openConfig = (stage: PipelineStage) => {
   896→    setEditingStage(stage)
   897→    setConfigOpen(true)
   898→  }
   899→
   900→  const handleConfigSave = (updated: PipelineStage) => {
   901→    updateStage(updated.id, updated)
   902→  }
   903→
   904→  const activeStage = activeId ? stages.find((s) => s.id === activeId) : null
   905→
   906→  return (
   907→    <div className="space-y-4">
   908→      <div className="flex items-center justify-between">
   909→        <Label className="text-sm font-medium">Pipeline Stages</Label>
   910→        {!readOnly && (
   911→          <Button variant="outline" size="sm" onClick={() => setPickerOpen(true)} className="h-7 text-xs">
   912→            <Plus className="size-3 mr-1" />
   913→            Add Stage
   914→          </Button>
   915→        )}
   916→      </div>
   917→
   918→      {stages.length === 0 ? (
   919→        <div className="flex flex-col items-center justify-center py-12 border-2 border-dashed rounded-xl">
   920→          <GitMerge className="size-10 text-muted-foreground/40 mb-3" />
   921→          <p className="text-sm text-muted-foreground mb-1">No stages configured</p>
   922→          <p className="text-xs text-muted-foreground/60 mb-4">
   923→            Add stages to define your pipeline workflow
   924→          </p>
   925→          {!readOnly && (
   926→            <Button variant="outline" size="sm" onClick={() => setPickerOpen(true)}>
   927→              <Plus className="size-3.5 mr-1" />
   928→              Add First Stage
   929→            </Button>
   930→          )}
   931→        </div>
   932→      ) : (
   933→        <div className="overflow-x-auto pb-2">
   934→          <DndContext
   935→            sensors={sensors}
   936→            collisionDetection={closestCenter}
   937→            onDragStart={handleDragStart}
   938→            onDragEnd={handleDragEnd}
   939→          >
   940→            <SortableContext

   942→              strategy={horizontalListSortingStrategy}

   944→              <div className="flex items-start gap-0 min-w-max">
   945→                {stages.map((stage, index) => (
   946→                  <div key={stage.id} className="flex items-start">
   947→                    <SortableStageCard
   948→                      stage={stage}
   949→                      index={index}
   950→                      readOnly={!!readOnly}
   951→                      onRemove={() => removeStage(stage.id)}
   952→                      onEdit={() => openConfig(stage)}
   953→                    />
   954→                    {/* Connector Arrow */}
