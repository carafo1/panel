    77→// ── API helper ─────────────────────────────────────────────────────────
    78→async function apiFetch<T>(path: string, options?: RequestInit): Promise<T> {
    79→  const token = localStorage.getItem('deployhub_token')
    80→  const res = await fetch(`/api${path}`, {
    81→    ...options,
    82→    headers: {
    83→      'Content-Type': 'application/json',
    84→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    85→      ...options?.headers,
    86→    },
    87→  })
    88→  if (!res.ok) {
    89→    const data = await res.json().catch(() => ({ error: 'Request failed' }))
    90→    throw new Error(data.error || `Request failed with status ${res.status}`)
    91→  }
    92→  return res.json()
    93→}

   116→// ── Time helpers ───────────────────────────────────────────────────────
   117→function timeAgo(date: string): string {
   118→  const seconds = Math.floor((Date.now() - new Date(date).getTime()) / 1000)
   119→  if (seconds < 60) return 'just now'
   120→  const minutes = Math.floor(seconds / 60)
   121→  if (minutes < 60) return `${minutes}m ago`
   122→  const hours = Math.floor(minutes / 60)
   123→  if (hours < 24) return `${hours}h ago`
   124→  const days = Math.floor(hours / 24)
   125→  return `${days}d ago`
   126→}
   127→
   128→function formatDuration(seconds: number | null): string {
   129→  if (!seconds) return '—'
   130→  if (seconds < 60) return `${seconds}s`
   131→  const m = Math.floor(seconds / 60)
   132→  const s = seconds % 60
   133→  return `${m}m ${s}s`
   134→}

   149→// ── Simulated build logs ───────────────────────────────────────────────
   150→function generateSimulatedBuildLog(deployment: DeploymentDetail): string {
   151→  const isFailed = deployment.status === 'failed'
   152→  const isBuilding = deployment.status === 'building' || deployment.status === 'deploying'
   153→  const fw = deployment.project?.framework === 'nextjs' ? 'next' : (deployment.project?.framework || 'node')
   154→  const now = new Date(deployment.createdAt)
   155→  const pad = (n: number) => String(n).padStart(2, '0')
   156→  const ts = `${pad(now.getHours())}:${pad(now.getMinutes())}:${pad(now.getSeconds())}`
   157→
   158→  if (isFailed) {
   159→    return `${ts} > Cloning repository...
   160→${ts} ✓ Repository cloned successfully
   161→${ts} > Installing dependencies...
   162→${ts} > npm install --production
   163→${ts} ✓ Dependencies installed (1.2s)
   164→${ts} > Building application...
   165→${ts} > ${fw} build
   166→${ts} ✗ Type error: Property 'fetchData' does not exist on type 'never'.
   167→${ts} ✗ Build failed with 1 error
   168→${ts} > Error: Build process exited with code 1
   169→${ts} > Deployment failed!`
   170→  }
   171→
   172→  if (isBuilding) {
   173→    return `${ts} > Cloning repository...
   174→${ts} ✓ Repository cloned successfully
   175→${ts} > Installing dependencies...
   176→${ts} > npm install --production
   177→${ts} ✓ Dependencies installed (1.2s)
   178→${ts} > Building application...
   179→${ts} > ${fw} build
   180→${ts} ⠹ Compiling...`
   181→  }
   182→
   183→  return `${ts} > Cloning repository...
   184→${ts} ✓ Repository cloned successfully
   185→${ts} > Installing dependencies...
   186→${ts} > npm install --production
   187→${ts} ✓ Dependencies installed (1.2s)
   188→${ts} > Building application...
   189→${ts} > ${fw} build
   190→${ts} ✓ Compiled successfully
   191→${ts} ✓ Linting and checking validity of types
   192→${ts} ✓ Collecting page data
   193→${ts} ✓ Generating static pages
   194→${ts} ✓ Finalizing page optimization
   195→${ts} > Deploying to server...
   196→${ts} ✓ Deployment successful!
   197→${ts} > Live at https://${deployment.project?.subdomain || 'myapp'}.deployhub.io`
   198→}

   200→function colorizeLogLine(line: string): React.ReactNode {
   201→  if (line.includes('✓') || line.includes('successful') || line.includes('Live at')) {
   202→    return <span className="text-emerald-400">{line}</span>
   203→  }
   204→  if (line.includes('✗') || line.includes('Error') || line.includes('failed')) {
   205→    return <span className="text-red-400">{line}</span>
   206→  }
   207→  if (line.includes('Warning') || line.includes('warn') || line.includes('⠹') || line.includes('⠋') || line.includes('⠼')) {
   208→    return <span className="text-yellow-400">{line}</span>
   209→  }
   210→  if (line.startsWith('>')) {
   211→    return <span className="text-zinc-400">{line}</span>
   212→  }
   213→  return <span className="text-zinc-200">{line}</span>
   214→}

   241→      setLoading(false)

   246→    fetchDeployments()

   294→          setIsLiveConnected(false)
   295→
   296→          // Refresh the full deployment list
   297→          fetchDeployments()

   299→          if (data.status === 'live') {

   302→        }
   303→      }
   304→    }
   305→

   307→

   309→      leaveDeploymentRoom(selectedId)
   310→      setIsLiveConnected(false)

   321→

   383→  }

   388→      <div className="space-y-4">
