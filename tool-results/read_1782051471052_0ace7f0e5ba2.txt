   295→function timeAgo(dateStr: string): string {
   296→  const seconds = Math.floor((Date.now() - new Date(dateStr).getTime()) / 1000)
   297→  if (seconds < 60) return 'just now'
   298→  const minutes = Math.floor(seconds / 60)
   299→  if (minutes < 60) return `${minutes}m ago`
   300→  const hours = Math.floor(minutes / 60)
   301→  if (hours < 24) return `${hours}h ago`
   302→  const days = Math.floor(hours / 24)
   303→  return `${days}d ago`
   304→}
   305→
   306→function formatDate(dateStr: string) {
   307→  try {
   308→    return new Date(dateStr).toLocaleDateString('en-US', {
   309→      month: 'short',
   310→      day: 'numeric',
   311→      year: 'numeric',
   312→      hour: '2-digit',
   313→      minute: '2-digit',
   314→    })
   315→  } catch {
   316→    return dateStr
   317→  }
   318→}

   320→function generateSimulatedBuildLog(project: ProjectDetail, deployment: Deployment): string {
   321→  const isFailed = deployment.status === 'failed'
   322→  const fw = project.framework === 'nextjs' ? 'next' : project.framework
   323→  const now = new Date(deployment.createdAt)
   324→  const pad = (n: number) => String(n).padStart(2, '0')
   325→  const ts = `${pad(now.getHours())}:${pad(now.getMinutes())}:${pad(now.getSeconds())}`
   326→
   327→  if (isFailed) {
   328→    return `${ts} > Cloning repository...
   329→${ts} > Installing dependencies...
   330→${ts} > npm install --production
   331→${ts} > Building application...
   332→${ts} > ${fw} build
   333→${ts} ✗ Type error: Property 'fetchData' does not exist on type 'never'.
   334→${ts} ✗ Build failed with 1 error
   335→${ts} > Error: Build process exited with code 1
   336→${ts} > Deployment failed!`
   337→  }
   338→
   339→  return `${ts} > Cloning repository...
   340→${ts} ✓ Repository cloned successfully
   341→${ts} > Installing dependencies...
   342→${ts} > npm install --production
   343→${ts} ✓ Dependencies installed (1.2s)
   344→${ts} > Building application...
   345→${ts} > ${fw} build
   346→${ts} ✓ Compiled successfully
   347→${ts} ✓ Linting and checking validity of types
   348→${ts} ✓ Collecting page data
   349→${ts} ✓ Generating static pages (8/8)
   350→${ts} ✓ Finalizing page optimization
   351→${ts} > Deploying to server...
   352→${ts} ✓ Deployment successful!
   353→${ts} > Live at https://${project.subdomain}`
   354→}

   356→function colorizeLogLine(line: string): React.ReactNode {
   357→  if (line.includes('✓') || line.includes('successful') || line.includes('Live at')) {
   358→    return <span className="text-emerald-400">{line}</span>
   359→  }
   360→  if (line.includes('✗') || line.includes('Error') || line.includes('failed')) {
   361→    return <span className="text-red-400">{line}</span>
   362→  }
   363→  if (line.includes('Warning') || line.includes('warn')) {
   364→    return <span className="text-yellow-400">{line}</span>
   365→  }
   366→  if (line.startsWith('>')) {
   367→    return <span className="text-zinc-400">{line}</span>
   368→  }
   369→  return <span className="text-zinc-200">{line}</span>
   370→}

   487→function CreateProjectDialog({
   488→  open,
   489→  onClose,
   490→  onCreated,
   491→}: {
   492→  open: boolean
   493→  onClose: () => void
   494→  onCreated: () => void
   495→}) {
   496→  const [name, setName] = useState('')
   497→  const [framework, setFramework] = useState('nextjs')
   498→  const [planId, setPlanId] = useState('')
   499→  const [plans, setPlans] = useState<Plan[]>([])
   500→  const [loading, setLoading] = useState(false)
   501→  const [plansLoading, setPlansLoading] = useState(false)
   502→
   503→  useEffect(() => {
   504→    if (open) {
   505→      setPlansLoading(true)
   506→      fetch('/api/billing/plans')
   507→        .then((r) => r.json())
   508→        .then((data) => {
   509→          setPlans(data.plans || [])
   510→          if (data.plans?.length > 0 && !planId) {
   511→            setPlanId(data.plans[0].id)
   512→          }
   513→        })
   514→        .catch(() => toast.error('Failed to load plans'))
   515→        .finally(() => setPlansLoading(false))
   516→    }
   517→  }, [open, planId])
   518→
   519→  const handleCreate = async () => {
   520→    if (!name.trim()) {
   521→      toast.error('Project name is required')
   522→      return
   523→    }
   524→    setLoading(true)
   525→    try {
   526→      const res = await fetch('/api/hosting/create', {
   527→        method: 'POST',
   528→        headers: getAuthHeaders(),
   529→        body: JSON.stringify({ name: name.trim(), framework, planId: planId || undefined }),
   530→      })
   531→      const data = await res.json()
   532→      if (!res.ok) {
   533→        toast.error(data.error || 'Failed to create project')
   534→        return
   535→      }
   536→      toast.success('Project created successfully!')
   537→      setName('')
   538→      setFramework('nextjs')
   539→      onCreated()
   540→      onClose()
   541→    } catch {
   542→      toast.error('Network error. Please try again.')
   543→    } finally {
   544→      setLoading(false)
   545→    }
   546→  }
   547→
   548→  return (
   549→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   550→      <DialogContent className="sm:max-w-lg">
   551→        <DialogHeader>
   552→          <DialogTitle>Create New Project</DialogTitle>
   553→          <DialogDescription>
   554→            Set up a new hosting project with your preferred framework and plan.
   555→          </DialogDescription>
   556→        </DialogHeader>
   557→
   558→        <div className="grid gap-4 py-4">
   559→          <div className="grid gap-2">
   560→            <Label htmlFor="project-name">Project Name</Label>
   561→            <Input
   562→              id="project-name"
   563→              placeholder="my-awesome-app"
   564→              value={name}
   565→              onChange={(e) => setName(e.target.value)}
   566→            />
   567→          </div>
   568→
   569→          <div className="grid gap-2">
   570→            <Label>Framework</Label>
   571→            <Select value={framework} onValueChange={setFramework}>
   572→              <SelectTrigger className="w-full">
   573→                <SelectValue placeholder="Select framework" />
   574→              </SelectTrigger>
   575→              <SelectContent>
   576→                <SelectItem value="nextjs">Next.js</SelectItem>
   577→                <SelectItem value="react">React</SelectItem>
   578→                <SelectItem value="node">Node.js</SelectItem>
   579→                <SelectItem value="static">Static</SelectItem>
   580→              </SelectContent>
   581→            </Select>
   582→          </div>
   583→
   584→          <div className="grid gap-2">
   585→            <Label>Plan</Label>
   586→            {plansLoading ? (
   587→              <div className="grid gap-2">
   588→                <Skeleton className="h-20 w-full" />
   589→                <Skeleton className="h-20 w-full" />
   590→              </div>
   591→            ) : (
   592→              <div className="grid gap-2">
   593→                {plans.map((plan) => (
   594→                  <button
   595→                    key={plan.id}
   596→                    type="button"
   597→                    onClick={() => setPlanId(plan.id)}
   598→                    className={`relative flex items-center gap-3 rounded-lg border p-3 text-left transition-all hover:border-emerald-300 dark:hover:border-emerald-700 ${
   599→                      planId === plan.id
   600→                        ? 'border-emerald-500 bg-emerald-50 dark:bg-emerald-900/20 ring-1 ring-emerald-500'
   601→                        : 'border-border'
   602→                    }`}
   603→                  >
   604→                    {plan.popular && (
   605→                      <Badge className="absolute -top-2 right-3 bg-emerald-600 text-white text-[10px] px-1.5 py-0">
   606→                        Popular
   607→                      </Badge>
   608→                    )}
   609→                    <div className="flex-1">
   610→                      <div className="font-medium text-sm">{plan.name}</div>
   611→                      <div className="text-xs text-muted-foreground mt-0.5">
   612→                        {plan.cpu} &middot; {plan.ram} &middot; {plan.storage}
   613→                      </div>
   614→                    </div>
   615→                    <div className="text-right">
   616→                      <div className="font-semibold text-emerald-600 dark:text-emerald-400">
   617→                        ${plan.price}
   618→                      </div>
   619→                      <div className="text-[10px] text-muted-foreground">/mo</div>
   620→                    </div>
   621→                  </button>
   622→                ))}
   623→              </div>
   624→            )}
   625→          </div>
   626→        </div>
   627→
   628→        <DialogFooter>
   629→          <Button variant="outline" onClick={onClose} disabled={loading}>
   630→            Cancel
   631→          </Button>
   632→          <Button
   633→            onClick={handleCreate}
   634→            disabled={loading || !name.trim()}
   635→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   636→          >
   637→            {loading ? (
   638→              <>
