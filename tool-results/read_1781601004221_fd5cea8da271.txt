    64→/* ── Confetti particles ── */
    65→function ConfettiEffect() {
    66→  const particles = Array.from({ length: 30 }, (_, i) => ({
    67→    id: i,
    68→    left: `${Math.random() * 100}%`,
    69→    delay: `${Math.random() * 0.8}s`,
    70→    color: ['#10b981', '#34d399', '#6ee7b7', '#fbbf24', '#f59e0b', '#8b5cf6', '#ec4899'][i % 7],
    71→    size: Math.random() * 8 + 4,
    72→  }))
    73→
    74→  return (
    75→    <div className="absolute inset-0 overflow-hidden pointer-events-none">
    76→      {particles.map((p) => (
    77→        <div
    78→          key={p.id}
    79→          className="absolute animate-confetti"
    80→          style={{
    81→            left: p.left,
    82→            top: '30%',
    83→            animationDelay: p.delay,
    84→            width: p.size,
    85→            height: p.size,
    86→            backgroundColor: p.color,
    87→            borderRadius: Math.random() > 0.5 ? '50%' : '2px',
    88→          }}
    89→        />
    90→      ))}
    91→    </div>
    92→  )
    93→}
