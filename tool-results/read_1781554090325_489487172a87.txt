   246→function timeAgo(dateStr: string): string {
   247→  const seconds = Math.floor((Date.now() - new Date(dateStr).getTime()) / 1000)
   248→  if (seconds < 60) return 'just now'
   249→  const minutes = Math.floor(seconds / 60)
   250→  if (minutes < 60) return `${minutes}m ago`
   251→  const hours = Math.floor(minutes / 60)
   252→  if (hours < 24) return `${hours}h ago`
   253→  const days = Math.floor(hours / 24)
   254→  return `${days}d ago`
   255→}

   257→function formatDate(dateStr: string) {
   258→  try {
   259→    return new Date(dateStr).toLocaleDateString('en-US', {
   260→      month: 'short',
   261→      day: 'numeric',
   262→      year: 'numeric',
   263→      hour: '2-digit',
   264→      minute: '2-digit',
   265→    })
   266→  } catch {
   267→    return dateStr
   268→  }
   269→}

   271→function generateSimulatedBuildLog(project: ProjectDetail, deployment: Deployment): string {
   272→  const isFailed = deployment.status === 'failed'
   273→  const fw = project.framework === 'nextjs' ? 'next' : project.framework
   274→  const now = new Date(deployment.createdAt)
   275→  const pad = (n: number) => String(n).padStart(2, '0')
   276→  const ts = `${pad(now.getHours())}:${pad(now.getMinutes())}:${pad(now.getSeconds())}`
   277→
   278→  if (isFailed) {
   279→    return `${ts} > Cloning repository...
   280→${ts} > Installing dependencies...
   281→${ts} > npm install --production
   282→${ts} > Building application...
   283→${ts} > ${fw} build
   284→${ts} ✗ Type error: Property 'fetchData' does not exist on type 'never'.
   285→${ts} ✗ Build failed with 1 error
   286→${ts} > Error: Build process exited with code 1
   287→${ts} > Deployment failed!`
   288→  }
   289→
   290→  return `${ts} > Cloning repository...
   291→${ts} ✓ Repository cloned successfully
   292→${ts} > Installing dependencies...
   293→${ts} > npm install --production
   294→${ts} ✓ Dependencies installed (1.2s)
   295→${ts} > Building application...
   296→${ts} > ${fw} build
   297→${ts} ✓ Compiled successfully
   298→${ts} ✓ Linting and checking validity of types
   299→${ts} ✓ Collecting page data
   300→${ts} ✓ Generating static pages (8/8)
   301→${ts} ✓ Finalizing page optimization
   302→${ts} > Deploying to server...
   303→${ts} ✓ Deployment successful!
   304→${ts} > Live at https://${project.subdomain}`
   305→}

   307→function colorizeLogLine(line: string): React.ReactNode {
   308→  if (line.includes('✓') || line.includes('successful') || line.includes('Live at')) {
   309→    return <span className="text-emerald-400">{line}</span>
   310→  }
   311→  if (line.includes('✗') || line.includes('Error') || line.includes('failed')) {
   312→    return <span className="text-red-400">{line}</span>
   313→  }
   314→  if (line.includes('Warning') || line.includes('warn')) {
   315→    return <span className="text-yellow-400">{line}</span>
   316→  }
   317→  if (line.startsWith('>')) {
   318→    return <span className="text-zinc-400">{line}</span>
   319→  }
   320→  return <span className="text-zinc-200">{line}</span>
   321→}

   399→function CreateProjectDialog({
   400→  open,
   401→  onClose,
   402→  onCreated,
   403→}: {
   404→  open: boolean
   405→  onClose: () => void
   406→  onCreated: () => void
   407→}) {
   408→  const [name, setName] = useState('')
   409→  const [framework, setFramework] = useState('nextjs')
   410→  const [planId, setPlanId] = useState('')
   411→  const [plans, setPlans] = useState<Plan[]>([])
   412→  const [loading, setLoading] = useState(false)
   413→  const [plansLoading, setPlansLoading] = useState(false)
   414→
   415→  useEffect(() => {
   416→    if (open) {
   417→      setPlansLoading(true)
   418→      fetch('/api/billing/plans')
   419→        .then((r) => r.json())
   420→        .then((data) => {
   421→          setPlans(data.plans || [])
   422→          if (data.plans?.length > 0 && !planId) {
   423→            setPlanId(data.plans[0].id)
   424→          }
   425→        })
   426→        .catch(() => toast.error('Failed to load plans'))
   427→        .finally(() => setPlansLoading(false))
   428→    }
   429→  }, [open, planId])
   430→
   431→  const handleCreate = async () => {
   432→    if (!name.trim()) {
   433→      toast.error('Project name is required')
   434→      return
   435→    }
   436→    setLoading(true)
   437→    try {
   438→      const res = await fetch('/api/hosting/create', {
   439→        method: 'POST',
   440→        headers: getAuthHeaders(),
   441→        body: JSON.stringify({ name: name.trim(), framework, planId: planId || undefined }),
   442→      })
   443→      const data = await res.json()
   444→      if (!res.ok) {
   445→        toast.error(data.error || 'Failed to create project')
   446→        return
   447→      }
   448→      toast.success('Project created successfully!')
   449→      setName('')
   450→      setFramework('nextjs')
   451→      onCreated()
   452→      onClose()
   453→    } catch {
   454→      toast.error('Network error. Please try again.')
   455→    } finally {
   456→      setLoading(false)
   457→    }
   458→  }
   459→
   460→  return (
   461→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   462→      <DialogContent className="sm:max-w-lg">
   463→        <DialogHeader>
   464→          <DialogTitle>Create New Project</DialogTitle>
   465→          <DialogDescription>
   466→            Set up a new hosting project with your preferred framework and plan.
   467→          </DialogDescription>
   468→        </DialogHeader>
   469→
   470→        <div className="grid gap-4 py-4">
   471→          <div className="grid gap-2">
   472→            <Label htmlFor="project-name">Project Name</Label>
   473→            <Input
   474→              id="project-name"
   475→              placeholder="my-awesome-app"
   476→              value={name}
   477→              onChange={(e) => setName(e.target.value)}
   478→            />
   479→          </div>
   480→
   481→          <div className="grid gap-2">
   482→            <Label>Framework</Label>
   483→            <Select value={framework} onValueChange={setFramework}>
   484→              <SelectTrigger className="w-full">
   485→                <SelectValue placeholder="Select framework" />
   486→              </SelectTrigger>
   487→              <SelectContent>
   488→                <SelectItem value="nextjs">Next.js</SelectItem>
   489→                <SelectItem value="react">React</SelectItem>
   490→                <SelectItem value="node">Node.js</SelectItem>
   491→                <SelectItem value="static">Static</SelectItem>
   492→              </SelectContent>
   493→            </Select>
   494→          </div>
   495→
   496→          <div className="grid gap-2">
   497→            <Label>Plan</Label>
   498→            {plansLoading ? (
   499→              <div className="grid gap-2">
   500→                <Skeleton className="h-20 w-full" />
   501→                <Skeleton className="h-20 w-full" />
   502→              </div>
   503→            ) : (
   504→              <div className="grid gap-2">
   505→                {plans.map((plan) => (
   506→                  <button
   507→                    key={plan.id}
   508→                    type="button"
   509→                    onClick={() => setPlanId(plan.id)}
   510→                    className={`relative flex items-center gap-3 rounded-lg border p-3 text-left transition-all hover:border-emerald-300 dark:hover:border-emerald-700 ${
   511→                      planId === plan.id
   512→                        ? 'border-emerald-500 bg-emerald-50 dark:bg-emerald-900/20 ring-1 ring-emerald-500'
   513→                        : 'border-border'
   514→                    }`}
   515→                  >
   516→                    {plan.popular && (
   517→                      <Badge className="absolute -top-2 right-3 bg-emerald-600 text-white text-[10px] px-1.5 py-0">
   518→                        Popular
   519→                      </Badge>
   520→                    )}
   521→                    <div className="flex-1">
   522→                      <div className="font-medium text-sm">{plan.name}</div>
   523→                      <div className="text-xs text-muted-foreground mt-0.5">
   524→                        {plan.cpu} &middot; {plan.ram} &middot; {plan.storage}
   525→                      </div>
   526→                    </div>
   527→                    <div className="text-right">
   528→                      <div className="font-semibold text-emerald-600 dark:text-emerald-400">
   529→                        ${plan.price}
   530→                      </div>
   531→                      <div className="text-[10px] text-muted-foreground">/mo</div>
   532→                    </div>
   533→                  </button>
   534→                ))}
   535→              </div>
   536→            )}
   537→          </div>
   538→        </div>
   539→
   540→        <DialogFooter>
   541→          <Button variant="outline" onClick={onClose} disabled={loading}>
   542→            Cancel
   543→          </Button>
   544→          <Button
   545→            onClick={handleCreate}
   546→            disabled={loading || !name.trim()}
   547→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   548→          >
   549→            {loading ? (
   550→              <>
   551→                <RefreshCw className="size-4 mr-2 animate-spin" />
   552→                Creating...
   553→              </>
   554→            ) : (
   555→              <>
   556→                <Rocket className="size-4 mr-2" />
   557→                Create Project
   558→              </>
   559→            )}
   560→          </Button>
   561→        </DialogFooter>
   562→      </DialogContent>
   563→    </Dialog>
   564→  )
   565→}

   569→function BuildLogsDialog({

   595→

   597→

   609→                  <span className="relative inline-flex size-2 rounded-full bg-emerald-500" />
   610→                </span>
   611→                Live
   612→              </span>
   613→            )}
   614→          </DialogTitle>
   615→          <DialogDescription>
   616→            {deployment ? (
   617→              <>
   618→                <span className="font-mono">{deployment.commit?.slice(0, 8) || 'Initial deploy'}</span>
   619→                {' on '}
   620→                <span className="font-mono">{deployment.branch}</span>
   621→              </>
   622→            ) : (
   623→              'Build output'
   624→            )}
   625→          </DialogDescription>
