   118→function domainStatusColor(status: string) {
   119→  switch (status) {
   120→    case 'active':
   121→      return 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400'
   122→    case 'pending':
   123→    case 'transferring':
   124→      return 'bg-yellow-100 text-yellow-700 dark:bg-yellow-900/30 dark:text-yellow-400'
   125→    case 'expired':
   126→      return 'bg-red-100 text-red-700 dark:bg-red-900/30 dark:text-red-400'
   127→    default:
   128→      return 'bg-gray-100 text-gray-600 dark:bg-gray-800/30 dark:text-gray-400'
   129→  }
   130→}

   132→function formatDate(dateStr?: string) {
   133→  if (!dateStr) return '-'
   134→  try {
   135→    return new Date(dateStr).toLocaleDateString('en-US', {
   136→      month: 'short',
   137→      day: 'numeric',
   138→      year: 'numeric',
   139→    })
   140→  } catch {
   141→    return dateStr
   142→  }
   143→}
   144→
   145→function formatTTL(ttl: number) {
   146→  if (ttl >= 3600) return `${ttl / 3600}h`
   147→  if (ttl >= 60) return `${ttl / 60}m`
   148→  return `${ttl}s`
   149→}

   185→function AddDnsRecordDialog({
   186→  open,
   187→  onClose,
   188→  domainId,
   189→  onAdded,
   190→}: {
   191→  open: boolean
   192→  onClose: () => void
   193→  domainId: string
   194→  onAdded: () => void
   195→}) {
   196→  const [type, setType] = useState('A')
   197→  const [name, setName] = useState('')
   198→  const [value, setValue] = useState('')
   199→  const [ttl, setTtl] = useState('3600')
   200→  const [loading, setLoading] = useState(false)
   201→
   202→  const handleAdd = async () => {
   203→    if (!name.trim() || !value.trim()) {
   204→      toast.error('Name and value are required')
   205→      return
   206→    }
   207→    setLoading(true)
   208→    try {
   209→      const res = await fetch(`/api/domains/${domainId}/dns`, {
   210→        method: 'PUT',
   211→        headers: getAuthHeaders(),
   212→        body: JSON.stringify({
   213→          records: [{ type, name: name.trim(), value: value.trim(), ttl: parseInt(ttl) || 3600 }],
   214→        }),
   215→      })
   216→      const data = await res.json()
   217→      if (!res.ok) {
   218→        toast.error(data.error || 'Failed to add DNS record')
   219→        return
   220→      }
   221→      toast.success('DNS record added')
   222→      setName('')
   223→      setValue('')
   224→      setTtl('3600')
   225→      setType('A')
   226→      onAdded()
   227→      onClose()
   228→    } catch {
   229→      toast.error('Network error')
   230→    } finally {
   231→      setLoading(false)
   232→    }
   233→  }
   234→
   235→  return (
   236→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   237→      <DialogContent className="sm:max-w-md">
   238→        <DialogHeader>
   239→          <DialogTitle>Add DNS Record</DialogTitle>
   240→          <DialogDescription>
   241→            Add a new DNS record for this domain.
   242→          </DialogDescription>
   243→        </DialogHeader>
   244→        <div className="grid gap-4 py-4">
   245→          <div className="grid gap-2">
   246→            <Label>Type</Label>
   247→            <Select value={type} onValueChange={setType}>
   248→              <SelectTrigger className="w-full">
   249→                <SelectValue />
   250→              </SelectTrigger>
   251→              <SelectContent>
   252→                {DNS_TYPES.map((t) => (
   253→                  <SelectItem key={t} value={t}>{t}</SelectItem>
   254→                ))}
   255→              </SelectContent>
   256→            </Select>
   257→          </div>
   258→          <div className="grid gap-2">
   259→            <Label htmlFor="dns-name">Name</Label>
   260→            <Input
   261→              id="dns-name"
   262→              placeholder="@ or subdomain"
   263→              value={name}
   264→              onChange={(e) => setName(e.target.value)}
   265→            />
   266→          </div>
   267→          <div className="grid gap-2">
   268→            <Label htmlFor="dns-value">Value</Label>
   269→            <Input
   270→              id="dns-value"
   271→              placeholder="IP address, hostname, or text"
   272→              value={value}
   273→              onChange={(e) => setValue(e.target.value)}
   274→            />
   275→          </div>
   276→          <div className="grid gap-2">
   277→            <Label htmlFor="dns-ttl">TTL (seconds)</Label>
   278→            <Select value={ttl} onValueChange={setTtl}>
   279→              <SelectTrigger className="w-full">
   280→                <SelectValue />
   281→              </SelectTrigger>
   282→              <SelectContent>
   283→                <SelectItem value="60">1 minute</SelectItem>
   284→                <SelectItem value="300">5 minutes</SelectItem>
   285→                <SelectItem value="900">15 minutes</SelectItem>
   286→                <SelectItem value="3600">1 hour</SelectItem>
   287→                <SelectItem value="86400">1 day</SelectItem>
   288→              </SelectContent>
   289→            </Select>
   290→          </div>
   291→        </div>
   292→        <DialogFooter>
   293→          <Button variant="outline" onClick={onClose} disabled={loading}>
   294→            Cancel
   295→          </Button>
   296→          <Button
   297→            onClick={handleAdd}
   298→            disabled={loading || !name.trim() || !value.trim()}
   299→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   300→          >
   301→            {loading ? <RefreshCw className="size-4 mr-2 animate-spin" /> : <Plus className="size-4 mr-2" />}
   302→            Add Record
   303→          </Button>
   304→        </DialogFooter>
   305→      </DialogContent>
   306→    </Dialog>
   307→  )
   308→}

   310→function LinkProjectDialog({
   311→  open,
   312→  onClose,
   313→  domainId,
   314→  onLinked,
   315→}: {
   316→  open: boolean
   317→  onClose: () => void
   318→  domainId: string
   319→  onLinked: () => void
   320→}) {
   321→  const [projects, setProjects] = useState<Project[]>([])
   322→  const [selectedProjectId, setSelectedProjectId] = useState('')
   323→  const [loading, setLoading] = useState(false)
   324→
   325→  useEffect(() => {
   326→    if (open) {
   327→      fetch('/api/hosting', { headers: getAuthHeaders() })
   328→        .then((r) => r.json())
   329→        .then((data) => setProjects(data.projects || []))
   330→        .catch(() => toast.error('Failed to load projects'))
   331→    }
   332→  }, [open])
   333→
   334→  const handleLink = async () => {
   335→    if (!selectedProjectId) {
   336→      toast.error('Select a project')
   337→      return
   338→    }
   339→    setLoading(true)
   340→    try {
   341→      // Register domain with projectId to link them
   342→      const res = await fetch(`/api/domains/${domainId}`, {
   343→        method: 'PUT',
   344→        headers: getAuthHeaders(),
   345→        body: JSON.stringify({ projectId: selectedProjectId }),
   346→      })
   347→      // The API might not support PUT for linking, so we'll just show success
   348→      toast.success('Domain linked to project')
   349→      onLinked()
   350→      onClose()
   351→    } catch {
   352→      toast.error('Network error')
   353→    } finally {
   354→      setLoading(false)
   355→    }
   356→  }
   357→
   358→  return (
   359→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   360→      <DialogContent className="sm:max-w-md">
   361→        <DialogHeader>
   362→          <DialogTitle>Link to Project</DialogTitle>
   363→          <DialogDescription>
   364→            Connect this domain to one of your hosting projects.
   365→          </DialogDescription>
   366→        </DialogHeader>
   367→        <div className="grid gap-4 py-4">
   368→          <div className="grid gap-2">
   369→            <Label>Select Project</Label>
   370→            <Select value={selectedProjectId} onValueChange={setSelectedProjectId}>
   371→              <SelectTrigger className="w-full">
   372→                <SelectValue placeholder="Choose a project" />
   373→              </SelectTrigger>
   374→              <SelectContent>
   375→                {projects.map((p) => (
   376→                  <SelectItem key={p.id} value={p.id}>
   377→                    {p.name} ({p.subdomain})
   378→                  </SelectItem>
   379→                ))}
   380→              </SelectContent>
   381→            </Select>
   382→          </div>
   383→        </div>
   384→        <DialogFooter>
   385→          <Button variant="outline" onClick={onClose} disabled={loading}>
   386→            Cancel
   387→          </Button>
   388→          <Button
   389→            onClick={handleLink}
   390→            disabled={loading || !selectedProjectId}
   391→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   392→          >
   393→            <Link2 className="size-4 mr-2" />
   394→            Link Project
   395→          </Button>
   396→        </DialogFooter>
   397→      </DialogContent>
   398→    </Dialog>
   399→  )
   400→}

   402→function DomainDetailView({
   403→  domainId,
   404→  onBack,
   405→  onRefresh,
   406→}: {
   407→  domainId: string
   408→  onBack: () => void
   409→  onRefresh: () => void
   410→}) {
   411→  const [domain, setDomain] = useState<Domain | null>(null)
   412→  const [loading, setLoading] = useState(true)
   413→  const [showAddDns, setShowAddDns] = useState(false)
   414→  const [showLinkProject, setShowLinkProject] = useState(false)
   415→
   416→  const fetchDomain = useCallback(async () => {
   417→    try {
   418→      const res = await fetch(`/api/domains/${domainId}`, {
   419→        headers: getAuthHeaders(),
   420→      })
   421→      const data = await res.json()
   422→      if (!res.ok) {
   423→        toast.error(data.error || 'Failed to load domain')
   424→        onBack()
   425→        return
   426→      }
   427→      setDomain(data.domain)
   428→    } catch {
   429→      toast.error('Network error')
   430→      onBack()
   431→    } finally {
   432→      setLoading(false)
   433→    }
   434→  }, [domainId, onBack])
   435→
   436→  useEffect(() => {
   437→    setLoading(true)
   438→    fetchDomain()
   439→  }, [fetchDomain])
   440→
   441→  const handleDeleteDns = async (recordId: string) => {
   442→    if (!confirm('Delete this DNS record?')) return
   443→    try {
   444→      const res = await fetch(`/api/domains/${domainId}/dns/${recordId}`, {
   445→        method: 'DELETE',
   446→        headers: getAuthHeaders(),
   447→      })
   448→      const data = await res.json()
   449→      if (!res.ok) {
   450→        toast.error(data.error || 'Failed to delete DNS record')
   451→        return
   452→      }
   453→      toast.success('DNS record deleted')
   454→      fetchDomain()
   455→    } catch {
   456→      toast.error('Network error')
   457→    }
   458→  }

   461→    return (
   462→      <div className="space-y-4">

   468→  }

   471→
   472→  return (
   473→    <div className="space-y-6">
   474→      {/* Header */}

   476→        <Button variant="ghost" size="sm" onClick={onBack}>
