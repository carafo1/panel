   123→// ── API helper ─────────────────────────────────────────────────────────
   124→async function apiFetch<T>(path: string, options?: RequestInit): Promise<T> {
   125→  const token = localStorage.getItem('deployhub_token')
   126→  const res = await fetch(`/api${path}`, {
   127→    ...options,
   128→    headers: {
   129→      'Content-Type': 'application/json',
   130→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
   131→      ...options?.headers,
   132→    },
   133→  })
   134→  if (!res.ok) {
   135→    const data = await res.json().catch(() => ({ error: 'Request failed' }))
   136→    throw new Error(data.error || `Request failed with status ${res.status}`)
   137→  }
   138→  return res.json()
   139→}

   141→// ── Badge helpers ──────────────────────────────────────────────────────
   142→//
   143→// Roles: super_admin (highest), admin, client, user. Each gets a distinct
   144→// colour so admins can spot privileged users at a glance.
   145→const ROLE_STYLES: Record<
   146→  string,
   147→  { className: string; icon?: React.ElementType }
   148→> = {
   149→  super_admin: {
   150→    className: 'bg-purple-100 text-purple-700 border-purple-200',
   151→    icon: Crown,
   152→  },
   153→  admin: {
   154→    className: 'bg-emerald-100 text-emerald-700 border-emerald-200',
   155→    icon: ShieldCheck,
   156→  },
   157→  client: {
   158→    className: 'bg-blue-100 text-blue-700 border-blue-200',
   159→    icon: Building2,
   160→  },
   161→  user: {
   162→    className: 'bg-gray-100 text-gray-700 border-gray-200',
   163→    icon: UserIcon,
   164→  },
   165→}
   166→
   167→function roleLabel(role: string): string {
   168→  switch (role) {
   169→    case 'super_admin':
   170→      return 'Super Admin'
   171→    case 'admin':
   172→      return 'Admin'
   173→    case 'client':
   174→      return 'Client'
   175→    case 'user':
   176→      return 'User'
   177→    default:
   178→      return role.charAt(0).toUpperCase() + role.slice(1)
   179→  }
   180→}
   181→
   182→function RoleBadge({ role }: { role: string }) {
   183→  const cfg = ROLE_STYLES[role] || ROLE_STYLES.user
   184→  const Icon = cfg.icon
   185→  return (
   186→    <Badge variant="outline" className={cfg.className}>
   187→      {Icon && <Icon className="size-3 mr-1" />}
   188→      {roleLabel(role)}
   189→    </Badge>
   190→  )
   191→}

   193→function JobStatusBadge({ status }: { status: string }) {
   194→  const map: Record<string, { className: string; icon: React.ElementType }> = {
   195→    queued: { className: 'bg-yellow-100 text-yellow-700 border-yellow-200', icon: Clock },
   196→    processing: { className: 'bg-blue-100 text-blue-700 border-blue-200', icon: Loader2 },
   197→    completed: { className: 'bg-emerald-100 text-emerald-700 border-emerald-200', icon: CheckCircle2 },
   198→    failed: { className: 'bg-red-100 text-red-700 border-red-200', icon: XCircle },
   199→  }
   200→  const cfg = map[status] || map.queued
   201→  const Icon = cfg.icon
   202→  return (
   203→    <Badge variant="outline" className={cfg.className}>
   204→      <Icon className={`size-3 mr-1 ${status === 'processing' ? 'animate-spin' : ''}`} />
   205→      {status.charAt(0).toUpperCase() + status.slice(1)}
   206→    </Badge>
   207→  )
   208→}
   209→
   210→function ServerStatusBadge({ status }: { status: string }) {
   211→  const map: Record<string, { className: string }> = {
   212→    active: { className: 'bg-emerald-100 text-emerald-700 border-emerald-200' },
   213→    maintenance: { className: 'bg-yellow-100 text-yellow-700 border-yellow-200' },
   214→    offline: { className: 'bg-red-100 text-red-700 border-red-200' },
   215→  }
   216→  const cfg = map[status] || map.offline
   217→  return (
   218→    <Badge variant="outline" className={cfg.className}>
   219→      {status === 'active' && <CheckCircle2 className="size-3 mr-1" />}
   220→      {status === 'maintenance' && <AlertTriangle className="size-3 mr-1" />}
   221→      {status === 'offline' && <XCircle className="size-3 mr-1" />}
   222→      {status.charAt(0).toUpperCase() + status.slice(1)}
   223→    </Badge>
   224→  )
   225→}

   303→        prev.map((u) =>
