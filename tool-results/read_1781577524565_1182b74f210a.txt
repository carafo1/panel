   260→function timeAgo(dateStr: string): string {
   261→  const seconds = Math.floor((Date.now() - new Date(dateStr).getTime()) / 1000)
   262→  if (seconds < 60) return 'just now'
   263→  const minutes = Math.floor(seconds / 60)
   264→  if (minutes < 60) return `${minutes}m ago`
   265→  const hours = Math.floor(minutes / 60)
   266→  if (hours < 24) return `${hours}h ago`
   267→  const days = Math.floor(hours / 24)
   268→  return `${days}d ago`
   269→}
   270→
   271→function formatDate(dateStr: string) {
   272→  try {
   273→    return new Date(dateStr).toLocaleDateString('en-US', {
   274→      month: 'short',
   275→      day: 'numeric',
   276→      year: 'numeric',
   277→      hour: '2-digit',
   278→      minute: '2-digit',
   279→    })
   280→  } catch {
   281→    return dateStr
   282→  }
   283→}

   285→function generateSimulatedBuildLog(project: ProjectDetail, deployment: Deployment): string {
   286→  const isFailed = deployment.status === 'failed'
   287→  const fw = project.framework === 'nextjs' ? 'next' : project.framework
   288→  const now = new Date(deployment.createdAt)
   289→  const pad = (n: number) => String(n).padStart(2, '0')
   290→  const ts = `${pad(now.getHours())}:${pad(now.getMinutes())}:${pad(now.getSeconds())}`
   291→
   292→  if (isFailed) {
   293→    return `${ts} > Cloning repository...
   294→${ts} > Installing dependencies...
   295→${ts} > npm install --production
   296→${ts} > Building application...
   297→${ts} > ${fw} build
   298→${ts} ✗ Type error: Property 'fetchData' does not exist on type 'never'.
   299→${ts} ✗ Build failed with 1 error
   300→${ts} > Error: Build process exited with code 1
   301→${ts} > Deployment failed!`
   302→  }
   303→
   304→  return `${ts} > Cloning repository...
   305→${ts} ✓ Repository cloned successfully
   306→${ts} > Installing dependencies...
   307→${ts} > npm install --production
   308→${ts} ✓ Dependencies installed (1.2s)
   309→${ts} > Building application...
   310→${ts} > ${fw} build
   311→${ts} ✓ Compiled successfully
   312→${ts} ✓ Linting and checking validity of types
   313→${ts} ✓ Collecting page data
   314→${ts} ✓ Generating static pages (8/8)
   315→${ts} ✓ Finalizing page optimization
   316→${ts} > Deploying to server...
   317→${ts} ✓ Deployment successful!
   318→${ts} > Live at https://${project.subdomain}`
   319→}

   321→function colorizeLogLine(line: string): React.ReactNode {
   322→  if (line.includes('✓') || line.includes('successful') || line.includes('Live at')) {
   323→    return <span className="text-emerald-400">{line}</span>
   324→  }
   325→  if (line.includes('✗') || line.includes('Error') || line.includes('failed')) {
   326→    return <span className="text-red-400">{line}</span>
   327→  }
   328→  if (line.includes('Warning') || line.includes('warn')) {
   329→    return <span className="text-yellow-400">{line}</span>
   330→  }
   331→  if (line.startsWith('>')) {
   332→    return <span className="text-zinc-400">{line}</span>
   333→  }
   334→  return <span className="text-zinc-200">{line}</span>
   335→}

   413→function CreateProjectDialog({
   414→  open,
   415→  onClose,
   416→  onCreated,
   417→}: {
   418→  open: boolean
   419→  onClose: () => void
   420→  onCreated: () => void
   421→}) {
   422→  const [name, setName] = useState('')
   423→  const [framework, setFramework] = useState('nextjs')
   424→  const [planId, setPlanId] = useState('')
   425→  const [plans, setPlans] = useState<Plan[]>([])
   426→  const [loading, setLoading] = useState(false)
   427→  const [plansLoading, setPlansLoading] = useState(false)
   428→
   429→  useEffect(() => {
   430→    if (open) {
   431→      setPlansLoading(true)
   432→      fetch('/api/billing/plans')
   433→        .then((r) => r.json())
   434→        .then((data) => {
   435→          setPlans(data.plans || [])
   436→          if (data.plans?.length > 0 && !planId) {
   437→            setPlanId(data.plans[0].id)
   438→          }
   439→        })
   440→        .catch(() => toast.error('Failed to load plans'))
   441→        .finally(() => setPlansLoading(false))
   442→    }
   443→  }, [open, planId])
   444→
   445→  const handleCreate = async () => {
   446→    if (!name.trim()) {
   447→      toast.error('Project name is required')
   448→      return
   449→    }
   450→    setLoading(true)
   451→    try {
   452→      const res = await fetch('/api/hosting/create', {
   453→        method: 'POST',
   454→        headers: getAuthHeaders(),
   455→        body: JSON.stringify({ name: name.trim(), framework, planId: planId || undefined }),
   456→      })
   457→      const data = await res.json()
   458→      if (!res.ok) {
   459→        toast.error(data.error || 'Failed to create project')
   460→        return
   461→      }
   462→      toast.success('Project created successfully!')
   463→      setName('')
   464→      setFramework('nextjs')
   465→      onCreated()
   466→      onClose()
   467→    } catch {
   468→      toast.error('Network error. Please try again.')
   469→    } finally {
   470→      setLoading(false)
   471→    }
   472→  }
   473→
   474→  return (
   475→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   476→      <DialogContent className="sm:max-w-lg">
   477→        <DialogHeader>
   478→          <DialogTitle>Create New Project</DialogTitle>
   479→          <DialogDescription>
   480→            Set up a new hosting project with your preferred framework and plan.
   481→          </DialogDescription>
   482→        </DialogHeader>
   483→
   484→        <div className="grid gap-4 py-4">
   485→          <div className="grid gap-2">
   486→            <Label htmlFor="project-name">Project Name</Label>
   487→            <Input
   488→              id="project-name"
   489→              placeholder="my-awesome-app"
   490→              value={name}
   491→              onChange={(e) => setName(e.target.value)}
   492→            />
   493→          </div>
   494→
   495→          <div className="grid gap-2">
   496→            <Label>Framework</Label>
   497→            <Select value={framework} onValueChange={setFramework}>
   498→              <SelectTrigger className="w-full">
   499→                <SelectValue placeholder="Select framework" />
   500→              </SelectTrigger>
   501→              <SelectContent>
   502→                <SelectItem value="nextjs">Next.js</SelectItem>
   503→                <SelectItem value="react">React</SelectItem>
   504→                <SelectItem value="node">Node.js</SelectItem>
   505→                <SelectItem value="static">Static</SelectItem>
   506→              </SelectContent>
   507→            </Select>
   508→          </div>
   509→
   510→          <div className="grid gap-2">
   511→            <Label>Plan</Label>
   512→            {plansLoading ? (
   513→              <div className="grid gap-2">
   514→                <Skeleton className="h-20 w-full" />
   515→                <Skeleton className="h-20 w-full" />
   516→              </div>
   517→            ) : (
   518→              <div className="grid gap-2">
   519→                {plans.map((plan) => (
   520→                  <button
   521→                    key={plan.id}
   522→                    type="button"
   523→                    onClick={() => setPlanId(plan.id)}
   524→                    className={`relative flex items-center gap-3 rounded-lg border p-3 text-left transition-all hover:border-emerald-300 dark:hover:border-emerald-700 ${
   525→                      planId === plan.id
   526→                        ? 'border-emerald-500 bg-emerald-50 dark:bg-emerald-900/20 ring-1 ring-emerald-500'
   527→                        : 'border-border'
   528→                    }`}
   529→                  >
   530→                    {plan.popular && (
   531→                      <Badge className="absolute -top-2 right-3 bg-emerald-600 text-white text-[10px] px-1.5 py-0">
   532→                        Popular
   533→                      </Badge>
   534→                    )}
   535→                    <div className="flex-1">
   536→                      <div className="font-medium text-sm">{plan.name}</div>
   537→                      <div className="text-xs text-muted-foreground mt-0.5">
   538→                        {plan.cpu} &middot; {plan.ram} &middot; {plan.storage}
   539→                      </div>
   540→                    </div>
   541→                    <div className="text-right">
   542→                      <div className="font-semibold text-emerald-600 dark:text-emerald-400">
   543→                        ${plan.price}
   544→                      </div>
   545→                      <div className="text-[10px] text-muted-foreground">/mo</div>
   546→                    </div>
   547→                  </button>
   548→                ))}
   549→              </div>
   550→            )}
   551→          </div>
   552→        </div>
   553→
   554→        <DialogFooter>
   555→          <Button variant="outline" onClick={onClose} disabled={loading}>
   556→            Cancel
   557→          </Button>
   558→          <Button
   559→            onClick={handleCreate}
   560→            disabled={loading || !name.trim()}
   561→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   562→          >
   563→            {loading ? (
   564→              <>
   565→                <RefreshCw className="size-4 mr-2 animate-spin" />
   566→                Creating...
   567→              </>
   568→            ) : (
   569→              <>
   570→                <Rocket className="size-4 mr-2" />
   571→                Create Project
   572→              </>
   573→            )}
   574→          </Button>
   575→        </DialogFooter>
   576→      </DialogContent>
   577→    </Dialog>
   578→  )
   579→}

   676→function ProjectDetailView({

   683→  onRefresh: () => void

   685→  const [project, setProject] = useState<ProjectDetail | null>(null)
   686→  const [loading, setLoading] = useState(true)

   688→  const [logsDialogOpen, setLogsDialogOpen] = useState(false)
   689→  const [selectedDeployment, setSelectedDeployment] = useState<Deployment | null>(null)

   691→  const [logLoading, setLogLoading] = useState(false)

   693→  const [deleting, setDeleting] = useState(false)
   694→

   704→  const [envLoading, setEnvLoading] = useState(true)

   710→  const [envValue, setEnvValue] = useState('')
   711→  const [envEncrypted, setEnvEncrypted] = useState(false)

   754→      toast.error('Failed to load env vars')

   756→      setEnvLoading(false)

   785→      toast.error('Network error')
   786→    } finally {
   787→      setDeploying(false)
   788→    }

   793→      const res = await fetch(`/api/hosting/${projectId}`, {
   794→        method: 'DELETE',
   795→        headers: getAuthHeaders(),
   796→      })
   797→      const data = await res.json()

   799→        toast.error(data.error || 'Failed to stop project')
   800→        return
   801→      }

   804→      onRefresh()

   806→      toast.error('Network error')
   807→    }
   808→  }
   809→

   815→      const res = await fetch(`/api/deployments/${dep.id}/logs`, {
   816→        headers: getAuthHeaders(),
   817→      })

   822→        setLogContent('')

   905→          method: 'PATCH',
   906→          headers: getAuthHeaders(),
   907→          body: JSON.stringify(body),

   984→    <div className="space-y-6">
   985→      {/* Header */}

   995→          <div className="flex items-center gap-3">
   996→            {frameworkIcon(project.framework)}

  1028→            disabled={deploying || project.status !== 'running'}
  1029→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
  1030→          >
  1031→            {deploying ? (
  1032→              <RefreshCw className="size-4 mr-2 animate-spin" />
  1033→            ) : (
  1034→              <Rocket className="size-4 mr-2" />

  1050→                <Trash2 className="size-4 mr-2" />
  1051→                Delete Project
  1052→              </DropdownMenuItem>
