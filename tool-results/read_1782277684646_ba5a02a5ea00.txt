   140→function demoIsAvailable(domain: string): boolean {
   141→  const lower = domain.toLowerCase()
   142→  if (lower.includes('taken') || lower.includes('unavailable')) return false
   143→  if (lower.includes('premium')) return true
   144→  // Deterministic pseudo-randomness based on string hash
   145→  const tld = '.' + lower.split('.').slice(1).join('.')
   146→  if (!DEMO_TLD_PRICES[tld]) return false
   147→  let hash = 0
   148→  for (let i = 0; i < lower.length; i++) {
   149→    hash = (hash << 5) - hash + lower.charCodeAt(i)
   150→    hash |= 0
   151→  }
   152→  // ~60% available
   153→  return Math.abs(hash) % 10 < 6
   154→}
   155→
   156→function demoIsPremium(domain: string): boolean {
   157→  const lower = domain.toLowerCase()
   158→  if (lower.includes('premium')) return true
   159→  // Short single-word .com / .ai treated as premium
   160→  const base = lower.split('.')[0]
   161→  return (base.length <= 4 && (lower.endsWith('.com') || lower.endsWith('.ai')))
   162→}

   166→class DynadotService {
   167→  private apiKey: string | null

   177→  /**
   178→   * Low-level call to Dynadot with retry + backoff + rate-limit handling.
   179→   */
   180→  private async call(
   181→    params: Record<string, string>,
   182→    attempt = 0
   183→  ): Promise<Record<string, unknown>> {
   184→    if (!this.apiKey) {
   185→      throw new Error('DYNADOT_API_KEY not configured')
   186→    }
   187→    const url = new URL(API_BASE)
   188→    url.searchParams.set('key', this.apiKey)
   189→    for (const [k, v] of Object.entries(params)) {
   190→      url.searchParams.set(k, v)
   191→    }
   192→
   193→    try {
   194→      const res = await fetch(url.toString(), {
   195→        method: 'GET',
   196→        headers: { Accept: 'application/json' },
   197→      })
   198→
   199→      // Rate limited
   200→      if (res.status === 429) {
   201→        const retryAfter = parseInt(res.headers.get('retry-after') || '2', 10)
   202→        if (attempt < MAX_RETRIES) {
   203→          await this.sleep(retryAfter * 1000)
   204→          return this.call(params, attempt + 1)
   205→        }
   206→        throw new Error('Dynadot rate limit exceeded')
   207→      }
   208→
   209→      // 5xx → retry
   210→      if (res.status >= 500 && attempt < MAX_RETRIES) {
   211→        await this.sleep(BASE_DELAY_MS * Math.pow(2, attempt))
   212→        return this.call(params, attempt + 1)
   213→      }
   214→
   215→      if (!res.ok) {
   216→        throw new Error(`Dynadot API HTTP ${res.status}: ${await res.text()}`)
   217→      }
   218→
   219→      const json = (await res.json()) as Record<string, unknown>
   220→
   221→      // Dynadot returns HTTP 200 but embeds errors in the body.
   222→      // Errors can appear in ANY top-level *Response envelope, not just "Response"
   223→      // (e.g. domain_info → DomainInfoResponse, get_ns → GetNsResponse).
   224→      // Detect IP-authorization errors specifically — these are not
   225→      // retryable (retrying just hits another unauthorized IP).
   226→      const envelopes = [
   227→        json.Response,
   228→        json.DomainInfoResponse,
   229→        json.GetNsResponse,
   230→        json.SetNsResponse,
   231→        json.SearchResponse,
   232→      ]
   233→        .filter(Boolean) as Array<Record<string, unknown>>
   234→      for (const env of envelopes) {
   235→        if (env.ResponseCode === '-1' || env.ResponseCode === -1 || env.Status === 'error') {
   236→          const errMsg = String(env.Error || 'Unknown Dynadot error')
   237→          if (errMsg.includes('unauthorized ip address')) {
   238→            // Non-retryable — throw immediately, bypass the catch retry
   239→            const ipErr = new Error(`Dynadot IP not authorized: ${errMsg}`)
   240→            ;(ipErr as Error & { noRetry?: boolean }).noRetry = true
   241→            throw ipErr
   242→          }
   243→          // Search errors with "invalid characters" are also non-retryable
   244→          if (errMsg.includes('invalid characters')) {
   245→            const charErr = new Error(`Dynadot API error: ${errMsg}`)
   246→            ;(charErr as Error & { noRetry?: boolean }).noRetry = true
   247→            throw charErr
   248→          }
   249→          throw new Error(`Dynadot API error: ${errMsg}`)
   250→        }
   251→      }
   252→
   253→      return json
   254→    } catch (err) {
   255→      // Network failure → retry (but never retry IP-auth or non-retryable errors)
   256→      const noRetry = err instanceof Error && (err as Error & { noRetry?: boolean }).noRetry
   257→      if (noRetry) throw err
   258→      if (attempt < MAX_RETRIES && !(err instanceof Error && err.message.includes('rate limit'))) {
   259→        await this.sleep(BASE_DELAY_MS * Math.pow(2, attempt))
   260→        return this.call(params, attempt + 1)
   261→      }
   262→      throw err
   263→    }
   264→  }
   265→
   266→  private sleep(ms: number): Promise<void> {
   267→    return new Promise((r) => setTimeout(r, ms))
   268→  }

   270→  private tldOf(domain: string): string {
   271→    const parts = domain.toLowerCase().split('.')
   272→    if (parts.length >= 3 && ['co', 'org', 'com'].includes(parts[parts.length - 2])) {
   273→      return '.' + parts.slice(-2).join('.')
   274→    }
   275→    return '.' + parts[parts.length - 1]
   276→  }

   288→  /**
   289→   * Bulk search across many domains at once.
   290→   *
   291→   * Uses `show_price=1` so Dynadot returns real Registration + Renewal prices
   292→   * in the `Price` field (a human-readable string we parse).
   293→   */
   294→  async bulkSearch(domains: string[]): Promise<DomainSearchResult[]> {
   295→    const cleaned = domains.map((d) => d.toLowerCase().trim()).filter(Boolean)
   296→
   297→    if (this.demoMode) {
   298→      return cleaned.map((domain) => {
   299→        const tld = this.tldOf(domain)
   300→        const prices = DEMO_TLD_PRICES[tld]
   301→        const available = prices ? demoIsAvailable(domain) : false
   302→        const premium = demoIsPremium(domain)
   303→        return {
   304→          domain,
   305→          available,
   306→          premium,
   307→          registrationPrice: available ? prices?.reg : undefined,
   308→          renewalPrice: available ? prices?.renew : undefined,
   309→          transferPrice: available ? prices?.transfer : undefined,
   310→          currency: 'USD',
   311→        }
   312→      })
   313→    }
   314→
   315→    // Dynadot's search API requires separate domain0, domain1, domain2... params.
   316→    // A single comma-separated domain0 value returns "invalid characters in domain".
   317→    // show_price=1 makes Dynadot return real Registration + Renewal prices.
   318→    const searchParams: Record<string, string> = {
   319→      command: 'search',
   320→      show_price: '1',
   321→      currency: 'USD',
   322→    }
   323→    cleaned.forEach((d, i) => {
   324→      searchParams[`domain${i}`] = d
   325→    })
   326→    const json = await this.call(searchParams)
   327→    const out: DomainSearchResult[] = []
   328→
   329→    // Dynadot search API returns:
   330→    //   { SearchResponse: { ResponseCode: "0", SearchResults: [{ DomainName, Status, Available: "yes"|"no", Price?, Premium? }] } }
   331→    // Price (when show_price=1) is a string like:
   332→    //   "Registration Price: 10.88 in USD and Renewal price: 10.88 in USD and Domain is not a Premium Domain"
   333→    const searchResp = json.SearchResponse as
   334→      | { SearchResults?: Array<Record<string, unknown>> }
   335→      | undefined
   336→
   337→    if (searchResp?.SearchResults && Array.isArray(searchResp.SearchResults)) {
   338→      for (const item of searchResp.SearchResults) {
   339→        const name = String(item.DomainName || '')
   340→        // Dynadot returns Available as "yes"/"no" (string)
   341→        const available = String(item.Available || '').toLowerCase() === 'yes'
   342→        const priceRaw = item.Price ? String(item.Price) : ''
   343→        const parsed = this.parsePriceString(priceRaw)
   344→        const premium = parsed?.premium ?? (String(item.Premium || '').toLowerCase() === 'yes')
   345→        out.push({
   346→          domain: name,
   347→          available,
   348→          premium,
   349→          registrationPrice: parsed?.registration,
   350→          renewalPrice: parsed?.renewal,
   351→          transferPrice: parsed?.renewal, // Dynadot transfer = renewal
   352→          currency: 'USD',
   353→        })
   354→      }
   355→    } else {
   356→      // Fallback: old Dynadot format { Response: { domain0: {...}, domain1: {...} } }
   357→      const response = (json.Response as Record<string, unknown>) || {}
   358→      for (const [key, value] of Object.entries(response)) {
   359→        if (key.startsWith('domain') && typeof value === 'object' && value !== null) {
   360→          const dv = value as { Name?: string; Available?: string; Price?: string; Premium?: string }
   361→          const name = dv.Name || ''
   362→          const parsed = this.parsePriceString(dv.Price)
   363→          out.push({
   364→            domain: name,
   365→            available: String(dv.Available).toLowerCase() === 'yes',
   366→            premium: parsed?.premium ?? (String(dv.Premium || '').toLowerCase() === 'yes'),
   367→            registrationPrice: parsed?.registration,
   368→            renewalPrice: parsed?.renewal,
   369→            transferPrice: parsed?.renewal,
   370→            currency: 'USD',
   371→          })
   372→        }
   373→      }
   374→    }
   375→
   376→    // Ensure all requested domains have an entry
   377→    return cleaned.map((d) => {
   378→      const found = out.find((r) => r.domain.toLowerCase() === d)
   379→      return (
   380→        found || {
   381→          domain: d,
   382→          available: false,
   383→          premium: false,
   384→        }
   385→      )
   386→    })
   387→  }
   388→
   389→  /**
   390→   * Parse Dynadot's `show_price=1` Price string.
   391→   * Example: "Registration Price: 10.88 in USD and Renewal price: 10.88 in USD and Domain is not a Premium Domain"
   392→   */
   393→  private parsePriceString(
   394→    raw: string | undefined
   395→  ): { registration: number; renewal: number; premium: boolean } | null {
   396→    if (!raw || typeof raw !== 'string') return null
   397→    const matches = raw.match(/([\d.]+)\s+in\s+USD/gi)
   398→    if (!matches || matches.length < 1) return null
   399→    const nums = matches.map((m) => parseFloat(m.replace(/[^\d.]/g, '')))
   400→    const registration = nums[0] || 0
   401→    const renewal = nums[1] || registration
   402→    const premium = /is\s+a\s+Premium/i.test(raw)
   403→    return { registration, renewal, premium }
   404→  }

   511→  async getDomainInfo(domain: string): Promise<DomainInfo> {
   512→    if (this.demoMode) {
   513→      const expiry = new Date()
   514→      expiry.setFullYear(expiry.getFullYear() + 1)
   515→      return {
   516→        domain,
   517→        status: 'active',
   518→        expiry: expiry.toISOString(),
   519→        registeredAt: new Date(Date.now() - 30 * 24 * 60 * 60 * 1000).toISOString(),
   520→        nameservers: ['ns1.deployhub.io', 'ns2.deployhub.io'],
   521→        locked: true,
   522→        whoisPrivacy: true,
   523→        autoRenew: true,
   524→        premium: demoIsPremium(domain),
   525→      }
   526→    }
   527→
   528→    const json = await this.call({ command: 'domain_info', domain })
   529→    const resp = (json.DomainInfoResponse as Record<string, unknown>) || {}
   530→    const info = (resp.DomainInfo || {}) as Record<string, unknown>
   531→
   532→    // Expiration / Registration are Unix timestamps in MILLISECONDS
   533→    const expiryRaw = info.Expiration ? String(info.Expiration) : undefined
   534→    const expiry = expiryRaw && !isNaN(Number(expiryRaw))
   535→      ? new Date(Number(expiryRaw)).toISOString()
   536→      : undefined
   537→    const regRaw = info.Registration ? String(info.Registration) : undefined
   538→    const registeredAt = regRaw && !isNaN(Number(regRaw))
   539→      ? new Date(Number(regRaw)).toISOString()
   540→      : undefined
   541→
   542→    // domain_info returns NameServerSettings (a mode object), NOT hostnames.
   543→    // Fetch the real hostnames via the separate get_ns command.
   544→    let nameservers: string[] = []
   545→    try {
   546→      const nsJson = await this.call({ command: 'get_ns', domain })
   547→      const nsResp = (nsJson.GetNsResponse as Record<string, unknown>) || {}
   548→      const nsHosts = nsResp.NameServer || nsResp.Ns
   549→      if (Array.isArray(nsHosts)) {
   550→        nameservers = nsHosts.map((s) => String(s).trim()).filter(Boolean)
   551→      } else if (nsHosts) {
   552→        nameservers = String(nsHosts).split(',').map((s) => s.trim()).filter(Boolean)
   553→      }
   554→    } catch {
   555→      // best-effort — other fields are still useful
   556→    }
   557→
   558→    const renewOption = String(info.RenewOption || '').toLowerCase()
   559→    return {
   560→      domain,
   561→      status: String(info.Status || 'unknown'),
   562→      expiry,
   563→      registeredAt,
   564→      nameservers,
   565→      locked: String(info.Locked || 'no').toLowerCase() === 'yes',
   566→      whoisPrivacy: String(info.Privacy || 'none').toLowerCase() !== 'none',
   567→      autoRenew: renewOption === 'auto' || renewOption === 'renew',
   568→      premium: String(info.isForSale || 'no').toLowerCase() === 'yes',
   569→    }
   570→  }

   731→  async getExpiration(domain: string): Promise<{ expiry?: string; daysUntilExpiry?: number }> {
   732→    const info = await this.getDomainInfo(domain)
   733→    if (!info.expiry) return {}
   734→    const expiryDate = new Date(info.expiry)
   735→    const days = Math.ceil((expiryDate.getTime() - Date.now()) / (24 * 60 * 60 * 1000))
   736→    return { expiry: info.expiry, daysUntilExpiry: days }
