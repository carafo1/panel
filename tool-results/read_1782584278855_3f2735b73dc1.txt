   263→function timeAgo(dateStr: string | null): string {
   264→  if (!dateStr) return 'never'
   265→  const seconds = Math.floor((Date.now() - new Date(dateStr).getTime()) / 1000)
   266→  if (seconds < 0) return 'just now'
   267→  if (seconds < 60) return `${seconds}s ago`
   268→  const minutes = Math.floor(seconds / 60)
   269→  if (minutes < 60) return `${minutes}m ago`
   270→  const hours = Math.floor(minutes / 60)
   271→  if (hours < 24) return `${hours}h ago`
   272→  const days = Math.floor(hours / 24)
   273→  if (days < 30) return `${days}d ago`
   274→  return new Date(dateStr).toLocaleDateString()
   275→}
   276→
   277→function formatDateTime(dateStr: string | null): string {
   278→  if (!dateStr) return '—'
   279→  try {
   280→    return new Date(dateStr).toLocaleString('en-US', {
   281→      month: 'short',
   282→      day: 'numeric',
   283→      year: 'numeric',
   284→      hour: '2-digit',
   285→      minute: '2-digit',
   286→    })
   287→  } catch {
   288→    return dateStr
   289→  }
   290→}

   294→function ServerStatusBadge({ status }: { status: string }) {
   295→  const map: Record<string, { className: string; dot?: string }> = {
   296→    active: {
   297→      className: 'bg-emerald-100 text-emerald-700 border-emerald-200 dark:bg-emerald-900/30 dark:text-emerald-400',
   298→    },
   299→    provisioning: {

   301→      dot: 'animate-pulse',
   302→    },
   303→    maintenance: {
   304→      className: 'bg-amber-100 text-amber-700 border-amber-200 dark:bg-amber-900/30 dark:text-amber-400',
   305→    },
   306→    offline: {
   307→      className: 'bg-red-100 text-red-700 border-red-200 dark:bg-red-900/30 dark:text-red-400',
   308→    },
   309→  }
   310→  const cfg = map[status] || {
   311→    className: 'bg-gray-100 text-gray-700 border-gray-200 dark:bg-gray-800/30 dark:text-gray-400',
   312→  }

   315→      {cfg.dot && (
   316→        <span className={`size-1.5 rounded-full bg-current mr-1.5 ${cfg.dot}`} />
   317→      )}
   318→      {status.charAt(0).toUpperCase() + status.slice(1)}
   319→    </Badge>
   320→  )

   440→  const [refreshing, setRefreshing] = useState(false)
   441→

   443→  const [editOpen, setEditOpen] = useState(false)
   444→  const [editingServer, setEditingServer] = useState<Server | null>(null)

   452→    try {
   453→      const res = await fetch('/api/admin/servers', { headers: authHeaders() })
   454→      const data = await res.json()
   455→      if (!res.ok) {
   456→        toast.error(data.error || 'Failed to load servers')
   457→        return
   458→      }
   459→      setServers(data.servers || [])
   460→    } catch {
   461→      toast.error('Network error')
   462→    } finally {
   463→      setLoading(false)
   464→      setRefreshing(false)
   465→    }
   466→  }, [])
   467→
   468→  useEffect(() => {
   469→    fetchServers()
   470→  }, [fetchServers])
   471→
   472→  const handleRefresh = () => {
   473→    setRefreshing(true)
   474→    fetchServers()
   475→  }
   476→
   477→  return (
   478→    <div className="space-y-4">
   479→      {/* Header */}

   502→          </Button>
   503→          <Button
   504→            size="sm"
   505→            onClick={() => {
   506→              setEditingServer(null)
   507→              setEditOpen(true)
   508→            }}
   509→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   510→          >
   511→            <Plus className="size-4 mr-1.5" />
   512→            Add Server
   513→          </Button>
   514→        </div>

   535→            <Button
   536→              size="sm"
   537→              onClick={() => {
   538→                setEditingServer(null)
   539→                setEditOpen(true)
   540→              }}
   541→              className="mt-4 bg-emerald-600 hover:bg-emerald-700 text-white"

   822→function EditServerDialog({
   823→  open,
   824→  server,
   825→  onClose,
   826→  onSaved,
   827→}: {
   828→  open: boolean
   829→  server: Server | null
   830→  onClose: () => void
   831→  onSaved: () => void
   832→}) {
   833→  const isEdit = !!server
   834→  const [saving, setSaving] = useState(false)
   835→
   836→  // Form state
   837→  const [name, setName] = useState('')
   838→  const [ip, setIp] = useState('')
   839→  const [provider, setProvider] = useState<string>('hetzner')
   840→  const [region, setRegion] = useState('')
   841→  const [cpuCores, setCpuCores] = useState('2')
   842→  const [ramGB, setRamGB] = useState('4')
   843→  const [storageGB, setStorageGB] = useState('80')
   844→  const [bandwidthGB, setBandwidthGB] = useState('1000')
   845→  const [sshHost, setSshHost] = useState('')
   846→  const [sshPort, setSshPort] = useState('22')
   847→  const [sshUser, setSshUser] = useState('root')
   848→  const [sshPassword, setSshPassword] = useState('')
   849→  const [sshPrivateKey, setSshPrivateKey] = useState('')
   850→  const [showPassword, setShowPassword] = useState(false)
   851→  const [showKey, setShowKey] = useState(false)
   852→
   853→  // Sync form whenever dialog opens or target server changes
   854→  useEffect(() => {
   855→    if (open) {
   856→      if (server) {
   857→        setName(server.name)
   858→        setIp(server.ip)
   859→        setProvider(server.provider || 'hetzner')
   860→        setRegion(server.region || '')
   861→        setCpuCores(String(server.cpuCores))
   862→        setRamGB(String(server.ramGB))
   863→        setStorageGB(String(server.storageGB))
   864→        setBandwidthGB(String(server.bandwidthGB))
   865→        setSshHost(server.sshHost || server.ip)
   866→        setSshPort(String(server.sshPort || 22))
   867→        setSshUser(server.sshUser || 'root')
   868→      } else {
   869→        // Reset for "create"
   870→        setName('')
   871→        setIp('')
   872→        setProvider('hetzner')
   873→        setRegion('')
   874→        setCpuCores('2')
   875→        setRamGB('4')
   876→        setStorageGB('80')
   877→        setBandwidthGB('1000')
   878→        setSshHost('')
   879→        setSshPort('22')
   880→        setSshUser('root')
   881→      }
   882→      setSshPassword('')
   883→      setSshPrivateKey('')

   888→

   890→  useEffect(() => {
   891→    if (!isEdit && !sshHost && ip) {
   892→      setSshHost(ip)
   893→    }
   894→  }, [ip, sshHost, isEdit])
   895→
   896→  const handleSave = async () => {
   897→    // Validation per spec: name + ip + cpuCores + ramGB + storageGB required
   898→    if (!name.trim()) return toast.error('Server name is required')
   899→    if (!ip.trim()) return toast.error('IP address is required')
   900→    const cpu = parseInt(cpuCores, 10)
   901→    const ram = parseFloat(ramGB)
   902→    const storage = parseFloat(storageGB)
   903→    if (!Number.isFinite(cpu) || cpu <= 0) return toast.error('CPU cores must be a positive number')
   904→    if (!Number.isFinite(ram) || ram <= 0) return toast.error('RAM (GB) must be a positive number')
   905→    if (!Number.isFinite(storage) || storage <= 0) return toast.error('Storage (GB) must be a positive number')
   906→
   907→    setSaving(true)
   908→    try {
   909→      const body: Record<string, unknown> = {
   910→        name: name.trim(),
   911→        ip: ip.trim(),
   912→        provider,
   913→        region: region.trim() || 'unknown',
   914→        cpuCores: cpu,
   915→        ramGB: ram,
   916→        storageGB: storage,
   917→        bandwidthGB: parseFloat(bandwidthGB) || 1000,
   918→        sshHost: sshHost.trim() || ip.trim(),
   919→        sshPort: parseInt(sshPort, 10) || 22,
   920→        sshUser: sshUser.trim() || 'root',
   921→      }
   922→      if (sshPassword) body.sshPassword = sshPassword
   923→      if (sshPrivateKey) body.sshPrivateKey = sshPrivateKey
   924→
   925→      const url = isEdit
   926→        ? `/api/admin/servers/${server!.id}`
   927→        : '/api/admin/servers'
   928→      const method = isEdit ? 'PATCH' : 'POST'
   929→
   930→      const res = await fetch(url, {
   931→        method,
   932→        headers: authHeaders(),
   933→        body: JSON.stringify(body),
   934→      })
   935→      const data = await res.json()
   936→      if (!res.ok) {
   937→        toast.error(data.error || `Failed to ${isEdit ? 'update' : 'create'} server`)
   938→        return
   939→      }
   940→      toast.success(isEdit ? 'Server updated' : 'Server created successfully')
   941→      onSaved()
   942→    } catch {
   943→      toast.error('Network error')
   944→    } finally {
   945→      setSaving(false)
   946→    }
   947→  }

   949→  return (
   950→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   951→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto [&::-webkit-scrollbar]:w-2 [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-muted-foreground/20">

   956→          </DialogTitle>

   958→            {isEdit
   959→              ? `Update configuration for ${server?.name}`

   962→        </DialogHeader>
   963→
   964→        <div className="space-y-6 py-2">
   965→          {/* Basic Info */}

   967→            <div className="flex items-center gap-2 text-sm font-medium">
   968→              <Globe className="size-4 text-emerald-600" />
   969→              Basic Information
   970→            </div>
   971→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   972→              <div className="space-y-1.5">
   973→                <Label htmlFor="srv-name">Name *</Label>
   974→                <Input
   975→                  id="srv-name"
   976→                  placeholder="prod-web-01"
   977→                  value={name}
   978→                  onChange={(e) => setName(e.target.value)}
   979→                />
   980→              </div>
   981→              <div className="space-y-1.5">
   982→                <Label htmlFor="srv-ip">IP Address *</Label>
   983→                <Input
   984→                  id="srv-ip"
   985→                  placeholder="203.0.113.10"
   986→                  value={ip}

   988→                  className="font-mono text-sm"
   989→                />
   990→              </div>
   991→              <div className="space-y-1.5">
   992→                <Label htmlFor="srv-provider">Provider</Label>
   993→                <Select value={provider} onValueChange={setProvider}>
   994→                  <SelectTrigger id="srv-provider">
