    77→// ── API helper ─────────────────────────────────────────────────────────
    78→async function apiFetch<T>(path: string, options?: RequestInit): Promise<T> {
    79→  const token = localStorage.getItem('deployhub_token')
    80→  const res = await fetch(`/api${path}`, {
    81→    ...options,
    82→    headers: {
    83→      'Content-Type': 'application/json',
    84→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    85→      ...options?.headers,
    86→    },
    87→  })
    88→  if (!res.ok) {
    89→    const data = await res.json().catch(() => ({ error: 'Request failed' }))
    90→    throw new Error(data.error || `Request failed with status ${res.status}`)
    91→  }
    92→  return res.json()
    93→}

   486→    setCustomOpen(true)

   490→    setContent((c) => {
   491→      const next = { ...c }

   495→  }

   592→          <div className="flex items-center gap-2 shrink-0">
   593→            {customKeys.length > 0 && <Badge variant="outline" className="text-xs">{customKeys.length}</Badge>}
   594→            <ChevronDown className={cn('size-4 text-muted-foreground transition-transform', customOpen && 'rotate-180')} />
   595→          </div>
   596→        </button>
   597→        {customOpen && (
   598→          <CardContent className="pt-0 border-t space-y-3">
   599→            <div className="pt-4 space-y-2">
   600→              {customKeys.length === 0 ? (
   601→                <p className="text-sm text-muted-foreground py-2">No custom keys. Add one below if you need a value not covered by the groups above.</p>
   602→              ) : (
   603→                customKeys.map((key) => (
   604→                  <div key={key} className="grid grid-cols-1 md:grid-cols-[12rem_1fr_auto] gap-2 items-start">
   605→                    <div className="flex items-center gap-1 pt-2">
   606→                      <code className="text-xs rounded bg-muted px-1.5 py-0.5 truncate">{key}</code>
   607→                      <button onClick={() => removeKey(key)} className="text-red-400 hover:text-red-600" aria-label="Remove key">
   608→                        <X className="size-3.5" />
   609→                      </button>
   610→                    </div>
   611→                    <Textarea
   612→                      value={content[key] ?? ''}
   613→                      onChange={(e) => setField(key, e.target.value)}
   614→                      rows={2}
   615→                      className="text-sm"
   616→                    />
   617→                  </div>
   618→                ))
   619→              )}
   620→            </div>
   621→            <div className="flex gap-2 pt-2 border-t">
   622→              <Input
   623→                placeholder="new.key (e.g. hero.promoBanner)"
   624→                value={newKey}
   625→                onChange={(e) => setNewKey(e.target.value)}
   626→                onKeyDown={(e) => { if (e.key === 'Enter') { e.preventDefault(); addCustomKey() } }}
   627→                className="text-sm max-w-xs"
   628→              />
   629→              <Button variant="outline" size="sm" onClick={addCustomKey}>
   630→                <Plus className="size-4 mr-1" /> Add Key
   631→              </Button>
   632→            </div>
   633→          </CardContent>
   634→        )}
   635→      </Card>
   636→    </div>
   637→  )
   638→}
