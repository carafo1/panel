   136→function statusColor(status: string) {
   137→  switch (status) {
   138→    case 'running':
   139→    case 'live':
   140→      return 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400'
   141→    case 'creating':
   142→    case 'building':
   143→    case 'deploying':
   144→    case 'queued':
   145→      return 'bg-yellow-100 text-yellow-700 dark:bg-yellow-900/30 dark:text-yellow-400'
   146→    case 'stopped':
   147→    case 'rolled_back':
   148→      return 'bg-gray-100 text-gray-600 dark:bg-gray-800/30 dark:text-gray-400'
   149→    case 'error':
   150→    case 'failed':
   151→      return 'bg-red-100 text-red-700 dark:bg-red-900/30 dark:text-red-400'
   152→    default:
   153→      return 'bg-gray-100 text-gray-600 dark:bg-gray-800/30 dark:text-gray-400'
   154→  }
   155→}

   282→function CreateProjectDialog({
   283→  open,
   284→  onClose,
   285→  onCreated,
   286→}: {
   287→  open: boolean
   288→  onClose: () => void
   289→  onCreated: () => void
   290→}) {
   291→  const [name, setName] = useState('')
   292→  const [framework, setFramework] = useState('nextjs')
   293→  const [planId, setPlanId] = useState('')
   294→  const [plans, setPlans] = useState<Plan[]>([])
   295→  const [loading, setLoading] = useState(false)
   296→  const [plansLoading, setPlansLoading] = useState(false)
   297→
   298→  useEffect(() => {
   299→    if (open) {
   300→      setPlansLoading(true)
   301→      fetch('/api/billing/plans')
   302→        .then((r) => r.json())
   303→        .then((data) => {
   304→          setPlans(data.plans || [])
   305→          if (data.plans?.length > 0 && !planId) {
   306→            setPlanId(data.plans[0].id)
   307→          }
   308→        })
   309→        .catch(() => toast.error('Failed to load plans'))
   310→        .finally(() => setPlansLoading(false))
   311→    }
   312→  }, [open, planId])
   313→
   314→  const handleCreate = async () => {
   315→    if (!name.trim()) {
   316→      toast.error('Project name is required')
   317→      return
   318→    }
   319→    setLoading(true)
   320→    try {
   321→      const res = await fetch('/api/hosting/create', {
   322→        method: 'POST',
   323→        headers: getAuthHeaders(),
   324→        body: JSON.stringify({ name: name.trim(), framework, planId: planId || undefined }),
   325→      })
   326→      const data = await res.json()
   327→      if (!res.ok) {
   328→        toast.error(data.error || 'Failed to create project')
   329→        return
   330→      }
   331→      toast.success('Project created successfully!')
   332→      setName('')
   333→      setFramework('nextjs')
   334→      onCreated()
   335→      onClose()
   336→    } catch {
   337→      toast.error('Network error. Please try again.')
   338→    } finally {
   339→      setLoading(false)
   340→    }
   341→  }
   342→
   343→  return (
   344→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   345→      <DialogContent className="sm:max-w-lg">
   346→        <DialogHeader>
   347→          <DialogTitle>Create New Project</DialogTitle>
   348→          <DialogDescription>
   349→            Set up a new hosting project with your preferred framework and plan.
   350→          </DialogDescription>
   351→        </DialogHeader>
   352→
   353→        <div className="grid gap-4 py-4">
   354→          <div className="grid gap-2">
   355→            <Label htmlFor="project-name">Project Name</Label>
   356→            <Input
   357→              id="project-name"
   358→              placeholder="my-awesome-app"
   359→              value={name}
   360→              onChange={(e) => setName(e.target.value)}
   361→            />
   362→          </div>
   363→
   364→          <div className="grid gap-2">
   365→            <Label>Framework</Label>
   366→            <Select value={framework} onValueChange={setFramework}>
   367→              <SelectTrigger className="w-full">
   368→                <SelectValue placeholder="Select framework" />
   369→              </SelectTrigger>
   370→              <SelectContent>
   371→                <SelectItem value="nextjs">Next.js</SelectItem>
   372→                <SelectItem value="react">React</SelectItem>
   373→                <SelectItem value="node">Node.js</SelectItem>
   374→                <SelectItem value="static">Static</SelectItem>
   375→              </SelectContent>
   376→            </Select>
   377→          </div>
   378→
   379→          <div className="grid gap-2">
   380→            <Label>Plan</Label>
   381→            {plansLoading ? (
   382→              <div className="grid gap-2">
   383→                <Skeleton className="h-20 w-full" />
   384→                <Skeleton className="h-20 w-full" />
   385→              </div>
   386→            ) : (
   387→              <div className="grid gap-2">
   388→                {plans.map((plan) => (
   389→                  <button
   390→                    key={plan.id}
   391→                    type="button"
   392→                    onClick={() => setPlanId(plan.id)}
   393→                    className={`relative flex items-center gap-3 rounded-lg border p-3 text-left transition-all hover:border-emerald-300 dark:hover:border-emerald-700 ${
   394→                      planId === plan.id
   395→                        ? 'border-emerald-500 bg-emerald-50 dark:bg-emerald-900/20 ring-1 ring-emerald-500'
   396→                        : 'border-border'
   397→                    }`}
   398→                  >
   399→                    {plan.popular && (
   400→                      <Badge className="absolute -top-2 right-3 bg-emerald-600 text-white text-[10px] px-1.5 py-0">
   401→                        Popular
   402→                      </Badge>
   403→                    )}
   404→                    <div className="flex-1">
   405→                      <div className="font-medium text-sm">{plan.name}</div>
   406→                      <div className="text-xs text-muted-foreground mt-0.5">
   407→                        {plan.cpu} &middot; {plan.ram} &middot; {plan.storage}
   408→                      </div>
   409→                    </div>
   410→                    <div className="text-right">
   411→                      <div className="font-semibold text-emerald-600 dark:text-emerald-400">
   412→                        ${plan.price}
   413→                      </div>
   414→                      <div className="text-[10px] text-muted-foreground">/mo</div>
   415→                    </div>
   416→                  </button>
   417→                ))}
   418→              </div>
   419→            )}
   420→          </div>
   421→        </div>
   422→
   423→        <DialogFooter>
   424→          <Button variant="outline" onClick={onClose} disabled={loading}>
   425→            Cancel
   426→          </Button>
   427→          <Button
   428→            onClick={handleCreate}
   429→            disabled={loading || !name.trim()}
   430→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   431→          >
   432→            {loading ? (
   433→              <>
   434→                <RefreshCw className="size-4 mr-2 animate-spin" />
   435→                Creating...
   436→              </>
   437→            ) : (
   438→              <>
   439→                <Rocket className="size-4 mr-2" />
   440→                Create Project
   441→              </>
   442→            )}
   443→          </Button>
   444→        </DialogFooter>
   445→      </DialogContent>
   446→    </Dialog>
   447→  )
   448→}

   450→function ProjectDetailView({
   451→  projectId,
   452→  onBack,
   453→  onRefresh,
   454→}: {
   455→  projectId: string
   456→  onBack: () => void
   457→  onRefresh: () => void
   458→}) {
   459→  const [project, setProject] = useState<ProjectDetail | null>(null)
   460→  const [loading, setLoading] = useState(true)
   461→  const [deploying, setDeploying] = useState(false)
   462→
   463→  const fetchProject = useCallback(async () => {
   464→    try {
   465→      const res = await fetch(`/api/hosting/${projectId}`, {
   466→        headers: getAuthHeaders(),
   467→      })
   468→      const data = await res.json()
   469→      if (!res.ok) {
   470→        toast.error(data.error || 'Failed to load project')
   471→        onBack()
   472→        return
   473→      }
   474→      setProject(data.project)
   475→    } catch {
   476→      toast.error('Network error')
   477→      onBack()
   478→    } finally {
   479→      setLoading(false)
   480→    }
   481→  }, [projectId, onBack])
   482→
   483→  useEffect(() => {
   484→    setLoading(true)
   485→    fetchProject()
   486→  }, [fetchProject])
   487→
   488→  const handleDeploy = async () => {
   489→    setDeploying(true)
   490→    try {
   491→      const res = await fetch(`/api/hosting/${projectId}/deploy`, {
   492→        method: 'POST',
   493→        headers: getAuthHeaders(),
   494→      })
   495→      const data = await res.json()
   496→      if (!res.ok) {
   497→        toast.error(data.error || 'Deploy failed')
   498→        return
   499→      }
   500→      toast.success('Deployment triggered!')
   501→      fetchProject()
   502→      onRefresh()
   503→    } catch {
   504→      toast.error('Network error')
   505→    } finally {
   506→      setDeploying(false)
   507→    }
   508→  }
   509→
   510→  const handleStop = async () => {
   511→    try {
   512→      const res = await fetch(`/api/hosting/${projectId}`, {
   513→        method: 'DELETE',
   514→        headers: getAuthHeaders(),
   515→      })
   516→      const data = await res.json()
   517→      if (!res.ok) {
   518→        toast.error(data.error || 'Failed to stop project')
   519→        return
   520→      }
   521→      toast.success('Project stopped')
   522→      fetchProject()
   523→      onRefresh()
   524→    } catch {
   525→      toast.error('Network error')
   526→    }
   527→  }
   528→
   529→  const handleDelete = async () => {
   530→    if (!confirm('Are you sure you want to delete this project? This action cannot be undone.')) return
   531→    try {
   532→      const res = await fetch(`/api/hosting/${projectId}`, {
   533→        method: 'DELETE',
   534→        headers: getAuthHeaders(),
   535→      })
   536→      const data = await res.json()
   537→      if (!res.ok) {
   538→        toast.error(data.error || 'Failed to delete project')
   539→        return
   540→      }
   541→      toast.success('Project deleted')
   542→      onRefresh()
   543→      onBack()
   544→    } catch {
   545→      toast.error('Network error')
   546→    }
   547→  }
   548→
   549→  if (loading) {
   550→    return (
   551→      <div className="space-y-4">
   552→        <Skeleton className="h-8 w-32" />
   553→        <Skeleton className="h-40 w-full" />
   554→        <Skeleton className="h-60 w-full" />
   555→      </div>
   556→    )
   557→  }
   558→
   559→  if (!project) return null
   560→
   561→  return (
   562→    <div className="space-y-6">
   563→      {/* Header */}
   564→      <div className="flex items-center gap-3">
   565→        <Button variant="ghost" size="sm" onClick={onBack}>
   566→          <ArrowLeft className="size-4 mr-1" />
   567→          Back
   568→        </Button>
   569→      </div>
   570→
   571→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   572→        <div>
   573→          <div className="flex items-center gap-3">
   574→            <h2 className="text-2xl font-bold">{project.name}</h2>
   575→            <Badge variant="outline" className={statusColor(project.status)}>
   576→              {project.status}
   577→            </Badge>
   578→            <Badge variant="outline" className={frameworkColor(project.framework)}>
   579→              {frameworkLabel(project.framework)}
   580→            </Badge>
   581→          </div>
   582→          <div className="flex items-center gap-2 mt-1 text-sm text-muted-foreground">
   583→            <Globe className="size-3.5" />
   584→            <a
   585→              href={`https://${project.subdomain}`}
   586→              target="_blank"
   587→              rel="noopener noreferrer"
   588→              className="hover:text-emerald-600 transition-colors font-mono"
   589→            >
   590→              {project.subdomain}
   591→              <ExternalLink className="size-3 inline ml-1" />
   592→            </a>
   593→          </div>
   594→          {project.customDomain && (
   595→            <div className="flex items-center gap-2 mt-1 text-sm text-muted-foreground">
   596→              <Globe className="size-3.5" />
   597→              <span className="font-mono">{project.customDomain}</span>
   598→            </div>
   599→          )}
   600→        </div>
   601→
   602→        <div className="flex items-center gap-2 flex-wrap">
   603→          <Button
   604→            onClick={handleDeploy}
   605→            disabled={deploying || project.status !== 'running'}
   606→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   607→          >
   608→            {deploying ? (
   609→              <RefreshCw className="size-4 mr-2 animate-spin" />
   610→            ) : (
   611→              <Rocket className="size-4 mr-2" />
   612→            )}
   613→            Deploy
   614→          </Button>
   615→          <Button variant="outline" onClick={handleStop} disabled={project.status !== 'running'}>
   616→            <StopCircle className="size-4 mr-2" />
   617→            Stop
   618→          </Button>
   619→          <DropdownMenu>
   620→            <DropdownMenuTrigger asChild>
   621→              <Button variant="outline" size="icon">
   622→                <MoreVertical className="size-4" />
   623→              </Button>
   624→            </DropdownMenuTrigger>
   625→            <DropdownMenuContent align="end">
   626→              <DropdownMenuItem className="text-red-600 focus:text-red-600" onClick={handleDelete}>
   627→                <Trash2 className="size-4 mr-2" />
   628→                Delete Project
   629→              </DropdownMenuItem>
   630→            </DropdownMenuContent>
   631→          </DropdownMenu>
   632→        </div>
   633→      </div>
   634→
   635→      <Separator />
   636→
   637→      {/* Info Cards */}
   638→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   639→        <Card>
   640→          <CardHeader className="pb-2">
   641→            <CardDescription className="flex items-center gap-1.5">
   642→              <Server className="size-3.5" />
   643→              Server
   644→            </CardDescription>
   645→          </CardHeader>
   646→          <CardContent>
   647→            {project.server ? (
   648→              <div>
   649→                <div className="font-medium text-sm">{project.server.name}</div>
   650→                <div className="text-xs text-muted-foreground font-mono">{project.server.ip}</div>
