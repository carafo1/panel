   225→// ─── Helpers ─────────────────────────────────────────────────────────────────
   226→
   227→function formatNumber(n: number): string {
   228→  if (n >= 1000) {
   229→    return `${(n / 1000).toFixed(1).replace(/\.0$/, '')}k`
   230→  }
   231→  return n.toString()
   232→}
   233→
   234→function slugify(text: string): string {
   235→  return text
   236→    .toLowerCase()
   237→    .replace(/[^a-z0-9]+/g, '-')
   238→    .replace(/^-|-$/g, '')
   239→}

   270→      if (activeCategory !== 'All' && t.category !== activeCategory) {

   277→        const matchesDesc = t.description.toLowerCase().includes(q)
   278→        const matchesTags = t.tags.some((tag) => tag.toLowerCase().includes(q))

   285→  // ── Derived: auto-generated subdomain ───────────────────────────────────

   290→  }, [projectName])
   291→
   292→  // ── Handlers ────────────────────────────────────────────────────────────
   293→

   295→    setSelectedTemplate(template)

   297→  }, [])

   299→  const handleOpenDeploy = useCallback(
   300→    (template: Template) => {

   307→    []

   317→        body: JSON.stringify({
   318→          name: projectName.trim(),
   319→          framework: deployingTemplate.framework,
   320→          repoUrl: deployingTemplate.repoUrl,

   322→        }),
   323→      })

   325→      setDeploySuccess(true)

   336→      const message =
   337→        err instanceof Error ? err.message : 'Failed to deploy template'

   340→      setIsDeploying(false)
   341→    }
   342→  }, [deployingTemplate, projectName, subdomain, setDashboardTab])
   343→
   344→  // ── Render ──────────────────────────────────────────────────────────────
   345→
   346→  return (

   353→            Template Marketplace

   356→        <p className="text-sm text-muted-foreground">

   359→        </p>
   360→      </div>
   361→
   362→      {/* ── Search & Filter Bar ─────────────────────────────────────────── */}

   366→          <Input
   367→            placeholder="Search templates by name, description, or tag…"

   369→            onChange={(e) => setSearchQuery(e.target.value)}

   371→          />
   372→        </div>

   546→              </DialogHeader>
   547→
   548→              {deploySuccess ? (
   549→                <div className="flex flex-col items-center justify-center py-8 gap-3">
   550→                  <div className="flex h-14 w-14 items-center justify-center rounded-full bg-emerald-100 dark:bg-emerald-900/40">

   553→                  <h3 className="text-lg font-semibold">
   554→                    Deployment Started!
   555→                  </h3>
   556→                  <p className="text-sm text-muted-foreground text-center">

   567→              ) : (
   568→                <div className="flex flex-col gap-4 mt-2">

   573→                    >
   574→                      {deployingTemplate.icon}
   575→                    </div>
   576→                    <div className="min-w-0">
   577→                      <p className="text-sm font-medium">
   578→                        {deployingTemplate.name}
   579→                      </p>
