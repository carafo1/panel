   246→   246→function timeAgo(dateStr: string): string {
   247→   247→  const seconds = Math.floor((Date.now() - new Date(dateStr).getTime()) / 1000)
   248→   248→  if (seconds < 60) return 'just now'
   249→   249→  const minutes = Math.floor(seconds / 60)
   250→   250→  if (minutes < 60) return `${minutes}m ago`
   251→   251→  const hours = Math.floor(minutes / 60)
   252→   252→  if (hours < 24) return `${hours}h ago`
   253→   253→  const days = Math.floor(hours / 24)
   254→   254→  return `${days}d ago`
   255→   255→}
   256→   256→
   257→   257→function formatDate(dateStr: string) {
   258→   258→  try {
   259→   259→    return new Date(dateStr).toLocaleDateString('en-US', {
   260→   260→      month: 'short',
   261→   261→      day: 'numeric',
   262→   262→      year: 'numeric',
   263→   263→      hour: '2-digit',
   264→   264→      minute: '2-digit',
   265→   265→    })
   266→   266→  } catch {
   267→   267→    return dateStr
   268→   268→  }
   269→   269→}

   271→   271→function generateSimulatedBuildLog(project: ProjectDetail, deployment: Deployment): string {
   272→   272→  const isFailed = deployment.status === 'failed'
   273→   273→  const fw = project.framework === 'nextjs' ? 'next' : project.framework
   274→   274→  const now = new Date(deployment.createdAt)
   275→   275→  const pad = (n: number) => String(n).padStart(2, '0')
   276→   276→  const ts = `${pad(now.getHours())}:${pad(now.getMinutes())}:${pad(now.getSeconds())}`
   277→   277→
   278→   278→  if (isFailed) {
   279→   279→    return `${ts} > Cloning repository...
   280→   280→${ts} > Installing dependencies...
   281→   281→${ts} > npm install --production
   282→   282→${ts} > Building application...
   283→   283→${ts} > ${fw} build

   307→   307→function colorizeLogLine(line: string): React.ReactNode {
   308→   308→  if (line.includes('✓') || line.includes('successful') || line.includes('Live at')) {
   309→   309→    return <span className="text-emerald-400">{line}</span>
   310→   310→  }
   311→   311→  if (line.includes('✗') || line.includes('Error') || line.includes('failed')) {
   312→   312→    return <span className="text-red-400">{line}</span>
   313→   313→  }
   314→   314→  if (line.includes('Warning') || line.includes('warn')) {
   315→   315→    return <span className="text-yellow-400">{line}</span>
   316→   316→  }
   317→   317→  if (line.startsWith('>')) {
   318→   318→    return <span className="text-zinc-400">{line}</span>
   319→   319→  }
   320→   320→  return <span className="text-zinc-200">{line}</span>
   321→   321→}

   399→   399→function CreateProjectDialog({
   400→   400→  open,
   401→   401→  onClose,
   402→   402→  onCreated,
   403→   403→}: {
   404→   404→  open: boolean
   405→   405→  onClose: () => void
   406→   406→  onCreated: () => void
   407→   407→}) {
   408→   408→  const [name, setName] = useState('')
   409→   409→  const [framework, setFramework] = useState('nextjs')
   410→   410→  const [planId, setPlanId] = useState('')
   411→   411→  const [plans, setPlans] = useState<Plan[]>([])
   412→   412→  const [loading, setLoading] = useState(false)
   413→   413→  const [plansLoading, setPlansLoading] = useState(false)
   414→   414→
   415→   415→  useEffect(() => {
   416→   416→    if (open) {
   417→   417→      setPlansLoading(true)
   418→   418→      fetch('/api/billing/plans')
   419→   419→        .then((r) => r.json())
   420→   420→        .then((data) => {
   421→   421→          setPlans(data.plans || [])
   422→   422→          if (data.plans?.length > 0 && !planId) {
   423→   423→            setPlanId(data.plans[0].id)
   424→   424→          }
   425→   425→        })
   426→   426→        .catch(() => toast.error('Failed to load plans'))
   427→   427→        .finally(() => setPlansLoading(false))
   428→   428→    }
   429→   429→  }, [open, planId])
   430→   430→
   431→   431→  const handleCreate = async () => {
   432→   432→    if (!name.trim()) {
   433→   433→      toast.error('Project name is required')
   434→   434→      return
   435→   435→    }
   436→   436→    setLoading(true)
   437→   437→    try {
   438→   438→      const res = await fetch('/api/hosting/create', {
   439→   439→        method: 'POST',
   440→   440→        headers: getAuthHeaders(),
   441→   441→        body: JSON.stringify({ name: name.trim(), framework, planId: planId || undefined }),
   442→   442→      })
   443→   443→      const data = await res.json()
   444→   444→      if (!res.ok) {
   445→   445→        toast.error(data.error || 'Failed to create project')
   446→   446→        return
   447→   447→      }
   448→   448→      toast.success('Project created successfully!')
   449→   449→      setName('')
   450→   450→      setFramework('nextjs')
   451→   451→      onCreated()
   452→   452→      onClose()
   453→   453→    } catch {
   454→   454→      toast.error('Network error. Please try again.')
   455→   455→    } finally {
   456→   456→      setLoading(false)
   457→   457→    }
   458→   458→  }
   459→   459→
   460→   460→  return (
   461→   461→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   462→   462→      <DialogContent className="sm:max-w-lg">
   463→   463→        <DialogHeader>
   464→   464→          <DialogTitle>Create New Project</DialogTitle>
   465→   465→          <DialogDescription>
   466→   466→            Set up a new hosting project with your preferred framework and plan.
   467→   467→          </DialogDescription>
   468→   468→        </DialogHeader>
   469→   469→
   470→   470→        <div className="grid gap-4 py-4">
   471→   471→          <div className="grid gap-2">
   472→   472→            <Label htmlFor="project-name">Project Name</Label>
   473→   473→            <Input
   474→   474→              id="project-name"
   475→   475→              placeholder="my-awesome-app"
   476→   476→              value={name}
   477→   477→              onChange={(e) => setName(e.target.value)}
   478→   478→            />
   479→   479→          </div>
   480→   480→
   481→   481→          <div className="grid gap-2">
   482→   482→            <Label>Framework</Label>
   483→   483→            <Select value={framework} onValueChange={setFramework}>
   484→   484→              <SelectTrigger className="w-full">
   485→   485→                <SelectValue placeholder="Select framework" />
   486→   486→              </SelectTrigger>
   487→   487→              <SelectContent>
   488→   488→                <SelectItem value="nextjs">Next.js</SelectItem>
   489→   489→                <SelectItem value="react">React</SelectItem>
   490→   490→                <SelectItem value="node">Node.js</SelectItem>
   491→   491→                <SelectItem value="static">Static</SelectItem>
   492→   492→              </SelectContent>
   493→   493→            </Select>
   494→   494→          </div>
   495→   495→
   496→   496→          <div className="grid gap-2">
   497→   497→            <Label>Plan</Label>
   498→   498→            {plansLoading ? (
   499→   499→              <div className="grid gap-2">
   500→   500→                <Skeleton className="h-20 w-full" />
   501→   501→                <Skeleton className="h-20 w-full" />
   502→   502→              </div>
   503→   503→            ) : (
   504→   504→              <div className="grid gap-2">
   505→   505→                {plans.map((plan) => (
   506→   506→                  <button
   507→   507→                    key={plan.id}
   508→   508→                    type="button"

   667→   667→  projectId: string
   668→   668→  onBack: () => void
   669→   669→  onRefresh: () => void

   671→   671→  const [project, setProject] = useState<ProjectDetail | null>(null)
   672→   672→  const [loading, setLoading] = useState(true)
   673→   673→  const [deploying, setDeploying] = useState(false)
   674→   674→  const [logsDialogOpen, setLogsDialogOpen] = useState(false)
   675→   675→  const [selectedDeployment, setSelectedDeployment] = useState<Deployment | null>(null)

   691→   691→        headers: getAuthHeaders(),
